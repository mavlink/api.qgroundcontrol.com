130std::pair<QString, QString> lookUpProxy(GProxyResolver* resolver, const char* uri, std::chrono::milliseconds timeout)
131{
132 // A private context bounds the wait; Qt's event loop doesn't run during the self-test anyway.
133 GMainContext* const context = g_main_context_new();
134 g_main_context_push_thread_default(context);
135
136 bool timedOut = false;
137 GSource* const timer = g_timeout_source_new(static_cast<guint>(std::max(timeout, 0ms).count()));
138 g_source_set_callback(timer, markTimedOut, &timedOut, nullptr);
139 (void) g_source_attach(timer, context);
140
141 auto lookup = std::make_unique<ProxyLookup>();
142 g_proxy_resolver_lookup_async(resolver, uri, nullptr, onProxyLookedUp, lookup.get());
143 while (!lookup->done && !timedOut) {
144 (void) g_main_context_iteration(context, TRUE);
145 }
146
147 g_source_destroy(timer);
148 g_source_unref(timer);
149 g_main_context_pop_thread_default(context);
150 g_main_context_unref(context);
151
152 if (!lookup->done) {
153 // Nothing iterates the context again, so the callback never runs: leave the lookup to it.
154 (void) lookup.release();
155 return {QStringLiteral("no answer within %1 ms").arg(timeout.count()), QString()};
156 }
157 return {lookup->failure, lookup->proxy};
158}

187bool Checker::run()
188{
189 QElapsedTimer elapsed;
190 elapsed.start();
191
192 const bool bundled = !_runtime.pluginDirs.isEmpty();
193 if (bundled) {
194 _checkBundledPlugins();
195 _checkPluginScanner();
196 }
197 _checkFeatures();
198 if (bundled || kStaticRuntime) {
199 _checkTls();
200 }
201 // libproxy loads its backend with dlopen(), so only a lookup shows it was bundled with what it needs.
202 if (hasGioModule(_runtime.gioModuleDir, QStringLiteral("giolibproxy"))) {
203 _checkProxy();
204 }
205
206 const QByteArray h264 = _readSample(kH264Sample);
207 if (!h264.isEmpty()) {
208 _checkDecoding(kH264Sample, h264);
209 _checkRecordings(h264);
210 }
211 // Decoder availability differs per platform; H.264 is the codec QGC must always play.
212 if (hasDecoderFor(kH265Sample)) {
213 const QByteArray h265 = _readSample(kH265Sample);
214 if (!h265.isEmpty()) {
215 _checkDecoding(kH265Sample, h265);
216 }
217 } else {
218 _notes.append(QStringLiteral("no H.265 decoder to test"));
219 }
220
221 if (_failures == 0) {
222 qCInfo(GStreamerSelfTestLog).noquote() << QStringLiteral("GStreamer self-test passed in %1 ms: %2")
223 .arg(elapsed.elapsed())
224 .arg(_notes.join(QStringLiteral(", ")));
225 return true;
226 }
227 qCCritical(GStreamerSelfTestLog).noquote()
228 << QStringLiteral("GStreamer self-test failed %1 check(s) in %2 ms").arg(_failures).arg(elapsed.elapsed());
229 return false;
230}

356void Checker::_checkDecoding(const Sample& sample, const QByteArray& data)
357{
358 DecodeResult result = decodeSample(sample, data, _timeout());
359 if (!result.failure.isEmpty() && result.hardware) {
360 // decodebin3 doesn't fall back when the decoder it plugged fails, as a hardware decoder may in a VM: retry once
361 // the way the Software decoder setting does.
362 const ScopedDecoderOption forceSoftware(ForceVideoDecoderSoftware);
363 const DecodeResult software = decodeSample(sample, data, _timeout());
364 if (software.failure.isEmpty()) {
365 qCWarning(GStreamerSelfTestLog).noquote()
366 << QStringLiteral("%1 hardware decoding with %2 failed (%3); software decoding with %4 works")
367 .arg(QLatin1String(sample.codec), result.decoders.join(QLatin1Char('+')), result.failure,
368 software.decoders.join(QLatin1Char('+')));
369 result = software;
370 } else {
371 result.failure += QStringLiteral("; with software decoders forced: %1").arg(software.failure);
372 result.decoders += software.decoders;
373 }
374 }
375
376 if (result.failure.isEmpty()) {
377 // decodebin3 replaces a decoder that fails to start, so the last one decoded.
378 const QString decoder = result.decoders.isEmpty() ? QStringLiteral("?") : result.decoders.takeLast();
379 QString note = QStringLiteral("%1 decoded by %2").arg(QLatin1String(sample.codec), decoder);
380 if (!result.decoders.isEmpty()) {
381 note += QStringLiteral(" (%1 failed to start)").arg(result.decoders.join(QLatin1Char('+')));
382 }
383 _notes.append(note);
384 return;
385 }
386 _fail(QStringLiteral("%1 decoding failed with decoders [%2]: %3")
387 .arg(QLatin1String(sample.codec), result.decoders.join(QStringLiteral(", ")), result.failure));
388}

391{
392 const QTemporaryDir dir;
393 if (!dir.isValid()) {

421QStringList bundledPluginIssues(const QList<RegisteredPlugin>& registered, const QStringList& pluginDirs,
422 const QStringList& pluginFiles)
423{
424 QStringList dirs;
425 for (const QString& dir : pluginDirs) {
426 dirs.append(normalizedPath(dir) + QLatin1Char('/'));
427 }
428 const auto inBundle = [&dirs](const QString& file) {
429 return std::ranges::any_of(dirs, [&file](const QString& dir) { return file.startsWith(dir, kPathCase); });
430 };
431
432 QSet<QString> shipped;
433 for (const QString& file : pluginFiles) {
434 shipped.insert(pathKey(file));
435 }
436
437 QStringList issues;
438 QSet<QString> registeredFiles;
439 for (const RegisteredPlugin& plugin : registered) {
440 if (plugin.filename.isEmpty()) {
441 continue;
442 }
443 const QString file = normalizedPath(plugin.filename);
444 registeredFiles.insert(pathKey(file));
445 if (!inBundle(file)) {
446 issues.append(QStringLiteral("Plugin %1 was loaded from %2, outside the bundled plugin directory %3: a "
447 "GStreamer installed outside the app leaks into it")
448 .arg(plugin.name, file, pluginDirs.join(QStringLiteral(", "))));
449 } else if (!shipped.contains(pathKey(file))) {
450 issues.append(QStringLiteral("Plugin %1 is registered from %2, which the bundle does not ship: the plugin "
451 "registry is stale")
452 .arg(plugin.name, file));
453 } else if (plugin.blacklisted) {
454 issues.append(
455 QStringLiteral("Plugin %1 (%2) is blacklisted: it failed to load, typically because a library "
456 "it links is missing from the bundle")
457 .arg(plugin.name, file));
458 }
459 }
460
461 for (const QString& file : pluginFiles) {
462 if (!registeredFiles.contains(pathKey(file))) {
463 issues.append(QStringLiteral("Plugin file %1 is not in GStreamer's registry: GStreamer failed to load it")
464 .arg(normalizedPath(file)));
465 }
466 }
467 return issues;
468}
