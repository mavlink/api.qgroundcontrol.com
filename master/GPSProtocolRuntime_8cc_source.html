59bool GPSProtocolRuntime::configure(const GPSConfig& config, unsigned& baud)
60{
61 _channel._resetError();
62 if (!_protocol) {
63 return false;
64 }
65 _channel._detectedBaud = config.detectedBaud;
66 const bool configured = _protocol->configure(_channel, config, baud);
67 _channel.finishCommand(GPSCommandOutcome::Written);
68 if (configured) {
69 (void) consume({});
70 } else {
71 // Every failed configuration reports a failure, keeping the detail the family left.
