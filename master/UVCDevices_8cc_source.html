59QStringList UVCDevices::formatNames(const QCameraDevice& device)
60{
61 QList<QCameraFormat> formats = device.videoFormats();
62 std::stable_sort(formats.begin(), formats.end(), [](const QCameraFormat& a, const QCameraFormat& b) {
63 const qint64 pixelsA = qint64(a.resolution().width()) * a.resolution().height();
64 const qint64 pixelsB = qint64(b.resolution().width()) * b.resolution().height();
65 return (pixelsA != pixelsB) ? (pixelsA > pixelsB) : (a.maxFrameRate() > b.maxFrameRate());
66 });
67 QStringList names;
68 for (const QCameraFormat& format : std::as_const(formats)) {
69 const QString name = formatName(format);
70 if (!names.contains(name)) {
71 names.append(name);

73 }
