100 function _updateTabs() {
101 let tabs = []
102 for (const child of _contentLayout.children) {
103 if (child._isQGCGroupBoxTab === true) {
104 child._groupBox = control
105 tabs.push(child)
106 }

108 // Reassigning the model rebuilds every header button, so only do it when the tabs actually changed
