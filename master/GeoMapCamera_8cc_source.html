22// Camera-to-world rotation for the pose convention R = Rz(heading) * Rx(tilt),
23// in double precision (QQuaternion/QVector3D floats lose meters at mercator world scale).
24Vec3 rotateCameraToWorld(const Vec3& v, qreal headingDeg, qreal tiltDeg)
25{
26 const double t = qDegreesToRadians(tiltDeg);
27 const double h = qDegreesToRadians(headingDeg);
28
29 // Rx(tilt)
30 const Vec3 a{v.x, (v.y * std::cos(t)) - (v.z * std::sin(t)), (v.y * std::sin(t)) + (v.z * std::cos(t))};
31 // Rz(heading)
32 return Vec3{(a.x * std::cos(h)) - (a.y * std::sin(h)), (a.x * std::sin(h)) + (a.y * std::cos(h)), a.z};
33}

35QPointF rotated(const QPointF& vec, qreal degrees)
36{
37 const double rad = qDegreesToRadians(degrees);
38 const double c = std::cos(rad);
39 const double s = std::sin(rad);
40 return QPointF((vec.x() * c) - (vec.y() * s), (vec.x() * s) + (vec.y() * c));
41}

49// Camera offset from the look-at center for the pose convention: the camera
50// sits distance meters along the back-rotated view axis (top-down at tilt 0,
51// moving south and down-range as tilt grows, swung around by heading)
52Vec3 cameraOffset(qreal headingDeg, qreal tiltDeg, qreal distance)
53{
54 const double tiltRad = qDegreesToRadians(tiltDeg);
55 const double headingRad = qDegreesToRadians(headingDeg);
56 const double sinTilt = std::sin(tiltRad);
57 return Vec3{distance * sinTilt * std::sin(headingRad), -distance * sinTilt * std::cos(headingRad),
58 distance * std::cos(tiltRad)};
59}

61Ray pickRay(const QPointF& centerWorld, qreal heading, qreal tilt, qreal distance, qreal centerElevation,
62 const QSizeF& viewport, qreal fov, const QPointF& screenPos)
63{
64 const double aspect = viewport.width() / viewport.height();
65 const double tanHalfFov = std::tan(qDegreesToRadians(fov) / 2.0);
66 const double ndcX = ((2.0 * screenPos.x()) / viewport.width()) - 1.0;
67 const double ndcY = 1.0 - ((2.0 * screenPos.y()) / viewport.height());
68
69 const Vec3 dir = rotateCameraToWorld(Vec3{ndcX * tanHalfFov * aspect, ndcY * tanHalfFov, -1.0}, heading, tilt);
70
71 const Vec3 offset = cameraOffset(heading, tilt, distance);
72 return Ray{Vec3{centerWorld.x() + offset.x, centerWorld.y() + offset.y, centerElevation + offset.z}, dir};
73}

266std::optional<QPointF> GeoMapCamera::screenToGround(const QPointF& screenPos, double planeZ) const
267{
268 if (_viewportSize.isEmpty()) {
269 return std::nullopt;
270 }
271
272 const Ray ray =
273 pickRay(_centerWorld, _heading, _tilt, _distance, _centerElevation, _viewportSize, verticalFieldOfView(), screenPos);
274 if (ray.dir.z >= 0.0) {
275 return std::nullopt; // at or above the horizon
276 }
277

279 if (s <= 0.0) {

294std::optional<QPointF> GeoMapCamera::groundPointCapped(const QPointF& screenPos, double maxRange) const
295{
296 if (_viewportSize.isEmpty()) {
297 return std::nullopt;
298 }
299
300 const Ray ray =
301 pickRay(_centerWorld, _heading, _tilt, _distance, _centerElevation, _viewportSize, verticalFieldOfView(), screenPos);
302 const QPointF cameraGround(ray.origin.x, ray.origin.y);
303
304 if (ray.dir.z < 0.0) {
305 const double s = -ray.origin.z / ray.dir.z;
306 const QPointF hit(ray.origin.x + (s * ray.dir.x), ray.origin.y + (s * ray.dir.y));
307 const QPointF offset = hit - cameraGround;
308 const double range = std::hypot(offset.x(), offset.y());
309 if (range <= maxRange) {
310 return hit;
311 }
312 return cameraGround + (offset * (maxRange / range));
313 }
314
315 // Above the horizon: cap along the ray's horizontal direction
316 const double horizontal = std::hypot(ray.dir.x, ray.dir.y);
317 if (horizontal < 1e-12) {
318 return cameraGround; // straight up
319 }
320 return cameraGround + (QPointF(ray.dir.x, ray.dir.y) * (maxRange / horizontal));
321}

323std::optional<QPointF> GeoMapCamera::worldToScreen(const QPointF& worldGround, double worldZ) const
324{
325 if (_viewportSize.isEmpty()) {
326 return std::nullopt;
327 }
328
329 const Vec3 offset = cameraOffset(_heading, _tilt, _distance);
330 const Vec3 d{worldGround.x() - (_centerWorld.x() + offset.x), worldGround.y() - (_centerWorld.y() + offset.y),
331 worldZ - (_centerElevation + offset.z)};
332
333 // World-to-camera: inverse of the pose rotation, R^T = Rx(-tilt) * Rz(-heading)
334 const double h = qDegreesToRadians(_heading);
335 const double t = qDegreesToRadians(_tilt);
336 const Vec3 a{(d.x * std::cos(h)) + (d.y * std::sin(h)), (-d.x * std::sin(h)) + (d.y * std::cos(h)), d.z};
337 const Vec3 c{a.x, (a.y * std::cos(t)) + (a.z * std::sin(t)), (-a.y * std::sin(t)) + (a.z * std::cos(t))};
338
339 // Camera looks along -z; the epsilon guards the projection divide (not a near plane)
340 const double depth = -c.z;
341 if (depth <= 1e-9) {
342 return std::nullopt;
343 }
344
345 const double aspect = _viewportSize.width() / _viewportSize.height();
346 const double tanHalfFov = std::tan(qDegreesToRadians(verticalFieldOfView()) / 2.0);
347 const double ndcX = c.x / (depth * tanHalfFov * aspect);
348 const double ndcY = c.y / (depth * tanHalfFov);
349 return QPointF(((ndcX + 1.0) / 2.0) * _viewportSize.width(), ((1.0 - ndcY) / 2.0) * _viewportSize.height());
350}

390QGeoCoordinate GeoMapCamera::centerForCoordinateAtScreenPoint(const QGeoCoordinate& coordinate,
391 const QPointF& screenPos, double worldZ,
392 double centerElevation) const
393{
394 if (_viewportSize.isEmpty()) {
395 return center();
396 }
397
398 // Intersect the pick ray with the horizontal plane at worldZ: solving on the
399 // ground plane instead would center the coordinate's ground footprint, leaving
400 // an elevated point (e.g. a flying vehicle) high on screen under a tilted camera.
401 const double pivotElevation = std::isfinite(centerElevation) ? centerElevation : _centerElevation;
402 const Ray ray =
403 pickRay(_centerWorld, _heading, _tilt, _distance, pivotElevation, _viewportSize, verticalFieldOfView(), screenPos);
404 if (ray.dir.z >= 0.0) {
405 return center(); // at or above the horizon
406 }
407 const double s = (worldZ - ray.origin.z) / ray.dir.z;
408 if (s <= 0.0) {
409 return center(); // plane is behind the camera
410 }
411
412 const QPointF hit(ray.origin.x + (s * ray.dir.x), ray.origin.y + (s * ray.dir.y));
413 const QPointF delta = TileMath::geoToWorld(coordinate) - hit;
414 return TileMath::worldToGeo(_centerWorld + delta);
415}

441void GeoMapCamera::orbitTo(const QPointF& screenPos)
442{
443 if (!_orbitAnchorWorld || _viewportSize.isEmpty()) {
444 return;
445 }
446
447 const QPointF delta = screenPos - _orbitStartScreen;
448 const qreal headingDelta = (delta.x() / _viewportSize.width()) * 360.0;
449 const qreal tiltDelta = (-delta.y() / _viewportSize.height()) * 180.0;
450
451 // Heading change is a rigid rotation of the camera about the vertical axis
452 // through the anchor: rotate the center around the anchor by the same angle.
453 const qreal newHeading = _normalizedHeading(_orbitStartHeading + headingDelta);
454 const qreal appliedHeadingDelta = std::remainder(newHeading - _heading, 360.0);
455 _setCenterWorld(*_orbitAnchorWorld + rotated(_centerWorld - *_orbitAnchorWorld, appliedHeadingDelta));
456 setHeading(newHeading);
457 if (_mode == Mode::Mode3D) {
458 setTilt(_orbitStartTilt + tiltDelta);
459 }
460
461 // Keep the anchor pinned to its on-screen position from gesture start
462 _anchorToScreen(*_orbitAnchorWorld, _orbitAnchorScreen, _orbitAnchorZ);
463}

475void GeoMapCamera::lookTo(const QPointF& screenPos)
476{
477 if (!_lookCameraGround || _viewportSize.isEmpty()) {
478 return;
479 }
480
481 const QPointF delta = screenPos - _lookStartScreen;
482 const qreal headingDelta = (delta.x() / _viewportSize.width()) * 360.0;
483 const qreal tiltDelta = (-delta.y() / _viewportSize.height()) * 180.0;
484
485 const qreal newHeading = _normalizedHeading(_lookStartHeading + headingDelta);
486 const qreal newTilt = (_mode == Mode::Mode3D) ? std::clamp(_lookStartTilt + tiltDelta, kMinTilt, kMaxTilt) : _tilt;
487
488 // The new center is the fixed camera's forward ray intersected with the
489 // pivot-elevation plane; solved via the pose convention so camera z tracks
490 // _lookCameraZ exactly even when centerElevation moves mid-gesture.
491 // When the consumer re-samples centerElevation on centerChanged (GeoMap
492 // terrain-following), that update lands after this solve: camera z is off
493 // by the per-event elevation delta until the next lookTo re-solves — a
494 // one-event lag, self-correcting while the drag continues.
495 const double height = _lookCameraZ - _centerElevation;
496 if (height <= 0.0) {
497 return; // camera at/below the pivot plane: no forward ground intersection
498 }
499 // Range-safety clamp: when it engages, the fixed-camera invariant yields
500 // and the camera slides along the view axis to stay within distance limits
501 const qreal newDistance = std::clamp(height / std::cos(qDegreesToRadians(newTilt)), kMinDistance, kMaxDistance);
502
503 const Vec3 offset = cameraOffset(newHeading, newTilt, newDistance);
504 _setCenterWorld(QPointF(_lookCameraGround->x() - offset.x, _lookCameraGround->y() - offset.y));
505 setHeading(newHeading);
506 setTilt(newTilt);
507 setDistance(newDistance);
508}
