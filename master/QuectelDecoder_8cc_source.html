17{
18 const auto body = QuectelCodec::checkedBody(line);

20 return 0;
21 }

24 }

35 _sawBoot = true;

37 _surveyPhase = SurveyPhase::Verifying;
38 }

40 _configured = false;
41 _surveyPhase = SurveyPhase::Off;
42 _revokeSurvey();

93bool GPSNativeQuectel::_handleSurvey(std::string_view body)
94{
95 const Fields reply(body);
96 unsigned version = 0;
97 unsigned tow = 0;
98 unsigned validity = 0;
99 unsigned observations = 0;
100 unsigned configuredCount = 0;
101 double accuracy = 0;
102 EcefMeters ecef;
103 if (reply.size() != 12 || !number(reply[1], version) || version != 1 || !number(reply[2], tow) ||
104 tow >= GPS_WEEK_MS) {
105 return false;
106 }
107 if (_lastSurveyTow) {
108 // TOW is a modular measurement clock, not a receipt timestamp. A late packet
109 // from before Sunday's rollover must not look newer than the current week.
110 const unsigned advance = (tow + GPS_WEEK_MS - *_lastSurveyTow) % GPS_WEEK_MS;
111 if (advance == 0 || advance >= GPS_WEEK_MS / 2) {
112 return false;
113 }
114 }
115 _lastSurveyTow = tow;
116 if ((_surveyPhase != SurveyPhase::Verifying && _surveyPhase != SurveyPhase::Monitoring) ||
117 _outputMode != OutputMode::RTCM) {
118 return false;
119 }
120 if (!number(reply[3], validity) || validity > 2 || !reply[4].empty() || !number(reply[6], observations) ||
121 observations > 86400 || !number(reply[7], configuredCount) || configuredCount > 86400 ||
122 !number(reply[8], ecef.x) || !number(reply[9], ecef.y) || !number(reply[10], ecef.z) ||
123 !number(reply[11], accuracy) || accuracy < 0 || accuracy > std::numeric_limits<uint32_t>::max() / 1000.0) {
124 _revokeSurvey();
125 return false;
126 }
127 const bool matches = _baseConfig.useFixedBase ? configuredCount == 0 && observations == 0
128 : configuredCount == _baseConfig.surveyInDurationSecs;
129 const double radius = std::hypot(ecef.x, ecef.y, ecef.z);
130 const bool coordinatesKnown = radius >= 6000000 && radius <= 7000000;
131 if (!matches || (!coordinatesKnown && (validity == 2 || (validity == 1 && observations != 0)))) {
132 _revokeSurvey();
133 return false;
134 }
135 if (_baseConfig.useFixedBase && validity == 2 &&
136 (std::abs(ecef.x - _fixedECEF.x) > 0.001 || std::abs(ecef.y - _fixedECEF.y) > 0.001 ||
137 std::abs(ecef.z - _fixedECEF.z) > 0.001)) {
138 _revokeSurvey();
139 return false;
140 }
141 // BOOT + identity + saved-role/base readback establish this survey's session.
142 // A one-observation survey can finish before any progress notification is sent.
143 const bool valid = validity == 2 && (_baseConfig.useFixedBase || observations >= configuredCount);
144 GPSNativeSurveyReport report{};
145 report.latitude = std::numeric_limits<double>::quiet_NaN();
146 report.longitude = std::numeric_limits<double>::quiet_NaN();
147 report.altitude = std::numeric_limits<float>::quiet_NaN();
148 report.altitudeDatum = GPSNativeSurveyReport::AltitudeDatum::Ellipsoid;
149 // Fixed mode's MeanAcc=0 describes supplied coordinates, not measured position uncertainty.
150 report.accuracyKnown = !_baseConfig.useFixedBase && validity != 0 && observations != 0 && coordinatesKnown;
151 report.mean_accuracy = report.accuracyKnown ? static_cast<uint32_t>(std::llround(accuracy * 1000)) : 0;
152 // Base positioning is fixed at 1 Hz. Gaps do not count as accepted observation seconds.
153 report.duration = observations;
154 report.flags = valid ? 1 : validity == 1 ? 2 : 0;
155 if (coordinatesKnown && validity != 0) {
156 const auto position = fromEcef(ecef);
157 report.latitude = position.latitudeDegrees;
158 report.longitude = position.longitudeDegrees;
159 report.altitude = position.altitudeMeters;
160 }
161 _lastSurveyUs = nowUs();
162 report.timestamp = _lastSurveyUs;
163 _surveyReport = report;
164 _publishSurvey();
165 return true;
166}
