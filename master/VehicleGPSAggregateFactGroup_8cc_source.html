47void VehicleGPSAggregateFactGroup::bindToGps(VehicleGPSFactGroup* gps1, VehicleGPSFactGroup* gps2)
48{
49 _clearConnections();
50 _gps1 = gps1;
51 _gps2 = gps2;
52
53 const auto update = [this, revision = _bindingRevision]() {
54 if (revision == _bindingRevision) {
55 _updateAggregates();
56 }
57 };
58 for (auto* gps : {gps1, gps2}) {
59 if (gps) {
60 _connections << connect(gps, &VehicleGPSFactGroup::gnssIntegrityReceived, this, update);
61 _connections << connect(gps, &QObject::destroyed, this, update);
62 }
63 }
64 _updateAggregates();
65}

104int VehicleGPSAggregateFactGroup::_mergeAuthentication(int a, int b)
105{
106 // Priority: Unknown < Disabled < Initializing < OK < Error
107 auto getWeight = [](int val) {
108 switch (val) {
109 case AUTH_INVALID: return -1;
110 case AUTH_UNKNOWN: return 0; // lowest priority)
111 case AUTH_DISABLED: return 1;
112 case AUTH_INITIALIZING: return 2;
113 case AUTH_OK: return 3;
114 case AUTH_ERROR: return 4; // highest priority
115 default: return -1;
116 }
117 };
118
119 return (getWeight(a) >= getWeight(b)) ? a : b;
120}

122void VehicleGPSAggregateFactGroup::_updateAggregates()
123{
124 const QPointer<VehicleGPSAggregateFactGroup> guard(this);
125 const quint64 revision = ++_updateRevision;
126 _expiryTask.cancel();
127 const quint64 nowUs = _scheduler ? _scheduler->nowUs() : 0;
128 const auto remaining1 =
129 MonotonicClock::remaining(_gps1 ? _gps1->gnssIntegrityTimestampUs() : 0, nowUs, GNSS_INTEGRITY_STALE_TIMEOUT);
130 const auto remaining2 =
131 MonotonicClock::remaining(_gps2 ? _gps2->gnssIntegrityTimestampUs() : 0, nowUs, GNSS_INTEGRITY_STALE_TIMEOUT);
132 auto* gps1 = remaining1.count() > 0 ? _gps1.data() : nullptr;
133 auto* gps2 = remaining2.count() > 0 ? _gps2.data() : nullptr;
134 const bool available = gps1 || gps2;
135 if (available) {
136 const auto nextExpiry = gps1 && gps2 ? std::min(remaining1, remaining2) : std::max(remaining1, remaining2);
137 _expiryTask.schedule(nextExpiry, [this]() { _updateAggregates(); });
138 }
139
140 const int spoof1 = _valueOrInvalid(gps1 ? gps1->spoofingState() : nullptr);
141 const int spoof2 = _valueOrInvalid(gps2 ? gps2->spoofingState() : nullptr);
142 const int jam1 = _valueOrInvalid(gps1 ? gps1->jammingState() : nullptr);
143 const int jam2 = _valueOrInvalid(gps2 ? gps2->jammingState() : nullptr);
144 const int auth1 = _valueOrInvalid(gps1 ? gps1->authenticationState() : nullptr);
145 const int auth2 = _valueOrInvalid(gps2 ? gps2->authenticationState() : nullptr);
146
147 const int spoofMerged = _mergeWorst(spoof1, spoof2);
148 const int jamMerged = _mergeWorst(jam1, jam2);
149 const int authMerged = _mergeAuthentication(auth1, auth2);
150
151 _spoofingStateFact.setRawValue(spoofMerged == -1 ? 255 : spoofMerged);
152 if (!guard || revision != _updateRevision) {
153 return;
154 }
155 _jammingStateFact.setRawValue(jamMerged == -1 ? 255 : jamMerged);
156 if (!guard || revision != _updateRevision) {
157 return;
158 }
159 _authenticationStateFact.setRawValue(authMerged == -1 ? 255 : authMerged);
160 if (!guard || revision != _updateRevision) {
161 return;
162 }
163 _isStaleFact.setRawValue(!available);
164 if (!guard || revision != _updateRevision) {
165 return;
166 }
167 _setTelemetryAvailable(available);
168}
