40 const QMetaObject& enumsMetaObject = MAVLinkEnums::staticMetaObject;

98void FailureInjection::resolveResult(int ackResult, int failureCode)
99{
100 // ACKs for MAV_CMD_INJECT_FAILURE arrive in send order; resolve the oldest pending row (highest index, since newest
101 // is prepended).
102 int pendingIndex = -1;
103 for (int i = _activity.size() - 1; i >= 0; --i) {
104 if (_activity.at(i).toMap().value(QStringLiteral("result")).toString() == QStringLiteral("pending")) {
105 pendingIndex = i;
106 break;
107 }
108 }
109 if (pendingIndex < 0) {
110 return;
111 }
112
113 QString reason = failureCodeReason(failureCode);
114 if (reason.isEmpty()) {
115 switch (ackResult) {
116 // Not tr()'d: FailureInjectionComponent.qml matches this exact literal to style the row green.
117 case MAV_RESULT_ACCEPTED:
118 reason = QStringLiteral("accepted");
119 break;
120 case MAV_RESULT_TEMPORARILY_REJECTED:
121 reason = tr("Temporarily rejected");
122 break;
123 case MAV_RESULT_DENIED:
124 reason = tr("Denied");
125 break;
126 case MAV_RESULT_UNSUPPORTED:
127 reason = tr("Unsupported");
128 break;
129 case MAV_RESULT_FAILED:
130 reason = tr("Failed");
131 break;
132 case MAV_RESULT_CANCELLED:
133 reason = tr("Cancelled");
134 break;
135 default:
136 reason = QGCMAVLink::mavResultToString(static_cast<uint8_t>(ackResult));
137 break;
138 }
139 }
140
141 QVariantMap row = _activity.at(pendingIndex).toMap();
142 row[QStringLiteral("result")] = reason;
143 _activity[pendingIndex] = row;
144 emit activityChanged();
145}
