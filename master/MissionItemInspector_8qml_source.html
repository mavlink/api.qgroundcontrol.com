27 function _step(delta) {
28 if (!mainWindow.allowViewSwitch()) {
29 return
30 }
31 const index = Logic.steppedIndex(_currentIndex, delta, _itemCount)
32 if (index >= 0) {
33 missionController.setCurrentPlanViewSeqNum(missionController.visualItems.get(index).sequenceNumber, false)
34 }
35 }

37 function _remove() {
38 if (!mainWindow.allowViewSwitch()) {
39 return
40 }
41 if (_item && _deletable) {
42 const viIndex = missionController.visualItemIndexForObject(_item)
43 if (viIndex > 0) {
44 missionController.removeVisualItem(viIndex)
45 }
46 }
