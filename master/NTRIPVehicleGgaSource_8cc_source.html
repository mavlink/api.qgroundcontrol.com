67void NTRIPVehicleGgaSource::attach(NTRIPManager* ntrip)
68{
69 using Source = NTRIPGgaReporter::PositionSource;
70 using Observation = std::optional<GPSObservation>;
71 // The tracked reports are the active vehicle's; the gate still requires its link.
72 const auto gated = [source = QPointer(this)](Observation (NTRIPVehicleGgaSource::*observe)() const) {
73 return [source, observe]() -> Observation {
74 if (!source || !activeVehicleConnected()) {
75 return std::nullopt;
76 }
77 return (source->*observe)();
78 };
79 };
80 ntrip->setGgaPositionProvider(Source::VehicleGPS, gated(&NTRIPVehicleGgaSource::gpsObservation));
81 ntrip->setGgaPositionProvider(Source::VehicleEKF, gated(&NTRIPVehicleGgaSource::estimateObservation));
82
83 if (auto* vehicles = MultiVehicleManager::instance()) {
84 (void) connect(vehicles, &MultiVehicleManager::activeVehicleChanged, this, &NTRIPVehicleGgaSource::setVehicle);
85 setVehicle(vehicles->activeVehicle());
86 }
87}

89void NTRIPVehicleGgaSource::setVehicle(Vehicle* vehicle)
90{
91 QObject::disconnect(_gpsConnection);
92 QObject::disconnect(_messageConnection);
93 _gps.reset();
94 _estimate.reset();
95 if (!vehicle) {
96 return;
97 }
98 if (auto* gps = vehicle->gpsFactGroup()) {
99 _gpsConnection = connect(gps, &VehicleGPSFactGroup::positionReported, this,
100 [this, gps](const QGeoPositionInfo& position, int fixType) {
101 _gps = GpsReport{.position = position,
102 .fixType = fixType,
103 .horizontalDop = gps->hdop()->rawValue().toDouble(),
104 .receivedAtUs = _scheduler->nowUs()};
105 });
106 }
107 _messageConnection =
108 connect(vehicle, &Vehicle::mavlinkMessageReceived, this, [this](const mavlink_message_t& message) {
109 if (message.msgid != MAVLINK_MSG_ID_GLOBAL_POSITION_INT) {
110 return;
111 }
112 mavlink_global_position_int_t position;
113 mavlink_msg_global_position_int_decode(&message, &position);
114 // ArduPilot reports 0/0 without a position, to carry the relative altitude.
115 if (position.lat == 0 && position.lon == 0) {
116 _estimate.reset();
117 return;
118 }
119 _estimate =
120 EstimateReport{.coordinate = QGeoCoordinate(GPSProtocolMath::degreesFromE7(position.lat),
121 GPSProtocolMath::degreesFromE7(position.lon),
122 GPSProtocolMath::metersFromMillimeters(position.alt)),
123 .receivedAtUs = _scheduler->nowUs()};
124 });
125}

132std::optional<GPSObservation> NTRIPVehicleGgaSource::gpsObservation() const
133{
134 if (!_gps || !_fresh(_gps->receivedAtUs)) {
135 return std::nullopt;
136 }
137 GPSObservation observation = vehicleObservation(_gps->position, fixQuality(_gps->fixType), _gps->receivedAtUs);
138 observation.receiverFixValid = observation.fixQuality != GPSObservation::FixQuality::NoFix;
139 if (qIsFinite(_gps->horizontalDop)) {
140 observation.horizontalDop = _gps->horizontalDop;
141 }
142 return observation.projected(GPSObservation::PositionUse::Gga);
143}

145std::optional<GPSObservation> NTRIPVehicleGgaSource::estimateObservation() const
