123NTRIPHttpTransport::HttpRequest NTRIPHttpTransport::buildHttpRequest(const NTRIPConnectionConfig& config)
124{
125 HttpRequest result;
126 result.error = config.streamValidationError();
127 if (!result.error.isEmpty()) {
128 return result;
129 }
130
131 using Header = QHttpHeaders::WellKnownHeader;
132 QHttpHeaders headers;
133 const QByteArray host = config.host.toUtf8();
134 if (!headers.append(Header::Host, QLatin1StringView(host.constData(), host.size())) ||
135 !headers.append("Ntrip-Version", "Ntrip/2.0") ||
136 !headers.append(Header::UserAgent, "NTRIP QGroundControl/1.0")) {
137 result.error = tr("Invalid NTRIP request header");
138 return result;
139 }
140
141 const bool hasCredentials = !config.username.isEmpty() || !config.password.isEmpty();
142 if (hasCredentials) {
143 const QString authorization =
144 QStringLiteral("Basic ") + QGCNetworkHelper::createBasicAuthCredentials(config.username, config.password);
145 if (!headers.append(Header::Authorization, authorization)) {
146 result.error = tr("Invalid NTRIP authorization header");
147 return result;
148 }
149 }
150
151 result.bytes = "GET /" + config.mountpoint.toUtf8() + " HTTP/1.1\r\n";
152 // Some legacy casters match these spellings case-sensitively.
153 for (const char* name : {"Host", "Ntrip-Version", "User-Agent", "Authorization"}) {
154 if (headers.contains(QLatin1StringView(name))) {
155 const auto value = headers.value(QLatin1StringView(name));
156 result.bytes += name;
157 result.bytes += ": ";
158 result.bytes.append(value.data(), value.size());
159 result.bytes += "\r\n";
160 }
161 }
162 result.bytes += "\r\n";
163 result.credentialsInClear = hasCredentials && !config.useTls;
164 return result;
165}

330void NTRIPHttpTransport::_parseRtcm(const QByteArray& buffer, qint64 receivedAtMs)
331{
332 const QPointer<NTRIPHttpTransport> guard(this);
333 const quint64 attempt = _attempt;
334 for (char ch : buffer) {
335 if (_stopped) {
336 return;
337 }
338 const uint8_t byte = static_cast<uint8_t>(static_cast<unsigned char>(ch));
339 auto result = _rtcmDecoder.addByte(byte, receivedAtMs);
340 while (result) {
341 if (_stopped) {
342 return;
343 }
344 emit correctionFrameReceived(*result);
345 if (!guard || _stopped || _attempt != attempt) {
346 return;
347 }
348 if (!result->valid) {
349 qCWarning(NTRIPHttpTransportLog) << "Invalid RTCM frame, dropping message id" << result->messageId;
350 } else if (!result->filtered) {
351 qCDebug(NTRIPHttpTransportLog) << "RTCM packet id" << result->messageId << "len" << result->data.size();
352 } else {
353 qCDebug(NTRIPHttpTransportLog) << "Ignoring RTCM" << result->messageId;
354 }
355 result = _rtcmDecoder.nextFrame();
356 }
357 }

360void NTRIPHttpTransport::_readBytes()
361{
362 if (_stopped || !_socket || _reading) {
363 return;
364 }
365 const QPointer<NTRIPHttpTransport> guard(this);
366 const auto socket = _socket;
367 const quint64 attempt = _attempt;
368 const auto current = [this, guard, socket, attempt]() {
369 return guard && !_stopped && _attempt == attempt && socket && _socket == socket;
370 };
371 _reading = true;
372 while (current() && socket->bytesAvailable() > 0) {
373 const qint64 receivedAtMs = static_cast<qint64>(MonotonicClock::nowUs() / 1000);
374 const QByteArray bytes = socket->read(16384);
375 if (!current() || bytes.isEmpty()) {
376 break;
377 }
378 _processHttpBytes(bytes, receivedAtMs);
379 }
380 if (guard && _attempt == attempt) {
381 _reading = false;
382 if (current() && socket->state() == QAbstractSocket::UnconnectedState) {
383 _publishHttpResult(_httpDecoder.finish(), static_cast<qint64>(MonotonicClock::nowUs() / 1000));
384 }

395void NTRIPHttpTransport::_publishHttpResult(const NTRIPHttpDecoder::Result& result, qint64 receivedAtMs)
396{
397 const QPointer<NTRIPHttpTransport> guard(this);
398 const quint64 attempt = _attempt;
399 const auto current = [this, guard, attempt]() { return guard && !_stopped && _attempt == attempt; };
400 if (result.connected) {
401 _connectTimeoutTimer.stop();
402 emit connected();
403 if (!current()) {
404 return;
405 }
406 _dataWatchdogTimer.start();
407 }
408 if (!result.body.isEmpty()) {
409 _dataWatchdogTimer.start();
410 _parseRtcm(result.body, receivedAtMs);
411 if (!current()) {
412 return;
413 }
414 }
415 if (result.failure) {
416 _fail(result.failure->code, result.failure->detail, result.failure->retryAfter);
417 } else if (result.complete) {
418 _fail(NTRIPError::ServerDisconnected, tr("NTRIP correction stream ended"));
419 } else if (result.awaitingErrorBody && !_errorBodyTimer.isActive()) {
420 _connectTimeoutTimer.stop();
421 _errorBodyTimer.start();
