160void NTRIPHttpTransport::_parseRtcm(const QByteArray& buffer, qint64 receivedAtMs)
161{
162 if (_stopped) {
163 return;
164 }
165 const NTRIPHttpSession* const session = _session;
166 _rtcmDecoder.feed(buffer, receivedAtMs, [this, session](const RTCMDecodedFrame& frame) {
167 if (!frame.valid) {
168 qCWarning(NTRIPHttpTransportLog) << "Invalid RTCM frame, dropping message id" << frame.messageId;
169 return true;
170 }
171 // Whitelisting is a routing policy, not evidence of a broken caster stream.
172 _lastValidFrameMs = _nowMs();

176 return true;
177 }

179 emit correctionFrameReceived(frame);
