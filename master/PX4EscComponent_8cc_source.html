51{
52 // Output protocol values of -1 and above are PWM or OneShot, below -1 are DShot
53 bool pwm = false;
54 bool dshot = false;
55 for (int i = 0; i < kTimerGroupCount; i++) {
56 for (const QString& prefix : {QStringLiteral("PWM_MAIN_TIM"), QStringLiteral("PWM_AUX_TIM")}) {
57 const Fact* const protocol = _parameterFact(prefix + QString::number(i));
58 if (!protocol) {
59 continue;
60 }
61 if (protocol->rawValue().toInt() < -1) {
62 dshot = true;
63 } else {
64 pwm = true;
65 }
66 }
67 }
68
69 QList<VehicleSummaryItem> items;
70 if (dshot && pwm) {
71 items.append({tr("Output"), tr("DShot and PWM")});
72 } else if (dshot) {
73 items.append({tr("Output"), tr("DShot")});
74 } else if (pwm) {
75 items.append({tr("Output"), tr("PWM")});
76 }
77
78 if (dshot) {
79 Fact* const bidirectional = _parameterFact(QStringLiteral("DSHOT_BIDIR_EN"));
80 if (bidirectional) {
81 items.append({tr("Bidirectional DShot"), bidirectional->enumStringValue(),
82 bidirectional->rawValue().toInt() == 0 ? VehicleSummaryItem::Muted
83 : VehicleSummaryItem::Normal});
84 }
85 const Fact* const poles = _parameterFact(QStringLiteral("MOT_POLE_COUNT"));
86 if (poles) {
