12{
13 if (const auto* averaging = std::get_if<GPSBaseStationConfig::ReceiverAveraging>(&config.mode)) {
14 if (averaging->maximumDuration < std::chrono::seconds{1} ||
15 averaging->maximumDuration > std::chrono::hours{1}) {
16 return GPSReceiverConfigError::InvalidReceiverAveraging;

18 return GPSReceiverConfigError::None;
19 }
20 if (const auto* fixed = std::get_if<GPSBaseStationConfig::Fixed>(&config.mode)) {
21 const auto& position = fixed->position;
22 const double altitudeCm = position.altitudeMeters * 100.0;
23 if (!std::isfinite(position.latitudeDegrees) || std::abs(position.latitudeDegrees) > 90.0 ||
24 !std::isfinite(position.longitudeDegrees) || std::abs(position.longitudeDegrees) > 180.0 ||
25 !std::isfinite(altitudeCm) || altitudeCm < (std::numeric_limits<int32_t>::min)() ||
26 altitudeCm > (std::numeric_limits<int32_t>::max)() ||
27 !GPSProtocolMath::tenthMillimeters(fixed->accuracyMeters)) {
28 return GPSReceiverConfigError::InvalidFixedBase;
29 }
30 } else {
31 const auto& survey = std::get<GPSBaseStationConfig::SurveyIn>(config.mode);
32 const auto accuracyUnits = GPSProtocolMath::tenthMillimeters(survey.accuracyMeters);
33 if (!accuracyUnits || *accuracyUnits < 1 || survey.duration < std::chrono::seconds{1} ||
34 survey.duration > std::chrono::seconds((std::numeric_limits<uint32_t>::max)())) {
35 return GPSReceiverConfigError::InvalidSurveyIn;
36 }
37 }
38 return GPSReceiverConfigError::None;

44GPSReceiverConfigError validateOptions(const GPSReceiverConfig& config, const GPSReceiverCapabilities& capabilities,
45 bool passive)
46{
47 if (config.allowPersistentChanges && !capabilities.persistentConfiguration) {
48 return GPSReceiverConfigError::UnsupportedPersistentConfiguration;
49 }
50 if (!passive) {
51 if ((std::holds_alternative<GPSBaseStationConfig::ReceiverAveraging>(config.base.mode) &&
52 !capabilities.receiverAveraging) ||
53 (std::holds_alternative<GPSBaseStationConfig::SurveyIn>(config.base.mode) && !capabilities.surveyIn)) {
54 return GPSReceiverConfigError::UnsupportedBaseMode;
55 }
56 if (config.base.compactObservations && !capabilities.compactObservations) {
57 return GPSReceiverConfigError::UnsupportedCompactObservations;
58 }
59 const GPSReceiverConfigError error = gpsValidateBaseStationConfig(config.base);
60 if (error != GPSReceiverConfigError::None) {
61 return error;
62 }
63 }
64 if (passive && config.base != GPSBaseStationConfig{}) {
65 return GPSReceiverConfigError::UnsupportedBaseMode;
66 }
67 if (!gpsValidBaudRate(config.baudRate, passive)) {
68 return GPSReceiverConfigError::InvalidBaudRate;
69 }
70 return GPSReceiverConfigError::None;
71}

75GPSReceiverConfigError gpsValidateReceiverConfig(GPSType type, const GPSReceiverConfig& config)
76{
77 const GPSReceiverCapabilities capabilities = gpsReceiverCapabilities(type);
78 if (!capabilities.recognized) {
79 return GPSReceiverConfigError::UnknownReceiver;
80 }
81 return validateOptions(config, capabilities, type == GPSType::passive);
82}

146 : QCoreApplication::translate("GPSReceiverConfig", "a fixed base position");
147 //: %1 is a receiver manufacturer, such as Septentrio; %2 is a base mode, such as survey-in
148 return QCoreApplication::translate("GPSReceiverConfig", "Detected %1 receiver does not support %2")
149 .arg(receiver, mode);
150 }
151 //: %1 is a receiver manufacturer; %2 explains the unsupported request
152 return QCoreApplication::translate("GPSReceiverConfig", "Detected %1 receiver: %2")
153 .arg(receiver, gpsReceiverConfigErrorText(error));
