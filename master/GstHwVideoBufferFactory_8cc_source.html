54 if (!context.gpuEnabled || !buffer) {
55 return nullptr;
56 }
57
58 // Plane 0's allocator is the dispatch key; gst-video buffers from a single decoder use one allocator type across
59 // all planes.
60 GstMemory* mem0 = gst_buffer_peek_memory(buffer, 0);

62 return nullptr;
63 }

65 if (cache && (allocatorType != G_TYPE_INVALID) && (allocatorType == cache->unmatchedAllocator)) {
66 return nullptr;
67 }

69 // Validate before commit — failure resets matchedPath so per-path counters don't double-count.
70 auto buildOrFallback = [&matchedPath, &info, cache](auto&& buf) -> std::unique_ptr<QHwVideoBuffer> {
71 if (!buf || !buf->validatePlaneHandles()) {
72 GstHwPathTelemetry::recordFallbackReason(matchedPath, GstHwPathTelemetry::HwFallbackReason::ValidateFailed);
73 // Some imports fail on every frame of a stream (e.g. multi-planar DMABuf under a Vulkan QRhi): warn once
74 // per caps epoch, not per frame.
75 if (!cache || !std::exchange(cache->validationFailureLogged, true)) {
76 qCWarning(GstHwBufFactoryLog)
77 << GstHw::toString(matchedPath) << "buffer failed plane-handle validation — CPU fallback";
78 }
79 if (cache) {
80 cache->validated = false;
81 }
82 matchedPath = HwVideoBufferPath::None;
83 return nullptr;
84 }
85 if (cache && !cache->validated) {

88 << "format=" << gst_video_format_to_string(GST_VIDEO_INFO_FORMAT(&info)) << GST_VIDEO_INFO_WIDTH(&info)
89 << "x" << GST_VIDEO_INFO_HEIGHT(&info);
90 }
91 if (cache) {

93 cache->validated = true;

96 };

98 // A matched path owns the frame: when it declines or fails, the frame goes to the CPU without trying later paths.
99 auto importThrough = [&](const HwPathDescriptor& path) -> std::unique_ptr<QHwVideoBuffer> {
100 if (path.admits && !path.admits()) {
101 if (cache) {
102 cache->validated = false;
103 }
104 return nullptr;
105 }
106 matchedPath = path.path;
107 return buildOrFallback(path.construct(sample.get(), info, format, context));
108 };
109
110 const auto paths = GstHwPathRegistry::paths();
111
112 // Fast path: only the paths behind a sample validated earlier this epoch are tried. Allocator and handle checks
113 // stay per-buffer: decoders can swap memory or recycle bad handles under stable caps.
114 if (cache && cache->validated && cache->path != HwVideoBufferPath::None) {
115 for (const HwPathDescriptor* path : paths) {
116 if ((path->path == cache->path) && path->matches(mem0, context)) {
117 return importThrough(*path);
118 }
119 }
120 cache->validated = false;
121 }
122
123 for (const HwPathDescriptor* path : paths) {
124 if (path->matches(mem0, context)) {
125 return importThrough(*path);
126 }
127 }
128
129 // No compiled path imports this allocator's memory; that holds for the whole caps epoch.
130 if (cache) {
131 cache->unmatchedAllocator = allocatorType;
132 }
133 qCDebug(GstHwBufFactoryLog) << "no zero-copy path for memory type"
134 << (mem0->allocator ? mem0->allocator->mem_type : "<no-allocator>")
135 << "— falling back to CPU";
136 GstHwPathTelemetry::recordFallbackReason(HwVideoBufferPath::None,
137 GstHwPathTelemetry::HwFallbackReason::UnknownMemType);
138 return nullptr;
