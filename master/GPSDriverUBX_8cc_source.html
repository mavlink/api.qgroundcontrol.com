57int GPSNativeUBX::receiveInternal(unsigned timeout, bool& read_error)
58{
59 const Operation operation(*this, timeout);
60 read_error = false;
61 if (ioError()) {
62 read_error = true;
63 return ioError();
64 }
65 uint8_t buf[GPS_READ_BUFFER_SIZE];
66
67 /* timeout additional to poll */
68 uint64_t time_started = nowUs();
69
70 int handled = 0;
71
72 while (true) {
73 bool ready_to_return =
74 (_timeModeReadbackPending && _timeModeReadbackReady) ||
75 (_controller.readbackPending() && _controller.readbackReady()) ||
76 (_configured ? (_assembleEpochs ? (handled & 1) : (_got_posllh && _got_velned)) : handled);
77
78 /* return success if ready */
79 if (ready_to_return) {
80 _got_posllh = false;
81 _got_velned = false;
82 return handled;
83 }
84
85 /* Wait for only UBX_PACKET_TIMEOUT if something already received. */
86 int ret =
87 read(buf, sizeof(buf),
88 std::min<int>((_got_posllh || _got_velned) ? UBX_PACKET_TIMEOUT
89 : (_assembleEpochs ? std::min(timeout, 200U) : timeout),
90 remainingMilliseconds(time_started + uint64_t(timeout) * 1000)));
91
92 if (ret < 0) {
93 read_error = true;
94 return -1;
95
96 } else if (ret > 0) {
97 //
98
99 /* pass received bytes to the packet decoder */
100 handled |= consume({buf, static_cast<size_t>(ret)});
101 } else {
102 handled |= consume({});
103 }
104
105 /* abort after timeout if no useful packets received */
106 if (nowUs() >= _operationDeadline.untilUs) {
107 return handled ? handled : -1;
108 }
109 }
110}

112void GPSNativeUBX::servicePendingCommands()
113{
114 if (_comms_request_pending) {
115 _comms_request_pending = false;
116 requestCommsDiagnostics();
117 }
118 if (_rtcmActivationPending) {
119 _rtcmActivationPending = false;
120 if (activateRTCMOutput() < 0 && !ioError()) {
121 _io_error = -EPROTO;
122 }
123 }
124 if (_pendingDisableMessage) {
125 const auto message = _pendingDisableMessage;
126 _pendingDisableMessage = 0;
127 if (_proto_ver_27_or_higher) {
128 uint32_t key_id = 0;
129
130 switch (message) { // we cannot infer the config Key ID from message for protocol version 27+
131 case UBX_MSG_RXM_RAWX:
132 key_id = UBX_CFG_KEY_MSGOUT_UBX_RXM_RAWX_I2C;
133 break;
134
135 case UBX_MSG_RXM_SFRBX:
136 key_id = UBX_CFG_KEY_MSGOUT_UBX_RXM_SFRBX_I2C;
137 break;
138
139 case UBX_MSG_NAV_TIMEGPS:
140 key_id = UBX_CFG_KEY_MSGOUT_UBX_NAV_TIMEGPS_I2C;
141 break;
142 }
143
144 if (key_id != 0) {
145 uint64_t t = nowUs();
146
147 if (t > _disable_cmd_last + DISABLE_MSG_INTERVAL && _configured) {
148 /* don't attempt for every message to disable, some might not be disabled */
149 _disable_cmd_last = t;
150
151 initCfgValset();
152 cfgValsetPort(key_id, 0);
153 sendCfgValset();
154 }
155 }
156
157 } else {
158 uint64_t t = nowUs();
159
160 if (t > _disable_cmd_last + DISABLE_MSG_INTERVAL) {
161 /* don't attempt for every message to disable, some might not be disabled */
162 _disable_cmd_last = t;
163
164 configureMessageRate(message, 0);
165 }
166 }
167 }
