247void UVCCameraController::_applyControls()
248{
249 if (!_camera) {
250 return;
251 }
252 // QCamera leaves a white balance or focus mode the camera lacks as it is.
253 _camera->setWhiteBalanceMode(_controls.whiteBalance);
254 if (maximumZoom() > minimumZoom()) {
255 _camera->setZoomFactor(std::clamp(static_cast<qreal>(_controls.zoom), minimumZoom(), maximumZoom()));
256 }
257 if (exposureAdjustable()) {
258 _camera->setExposureCompensation(_controls.exposureCompensation);
259 }
260 if (focusAdjustable() && _camera->isFocusModeSupported(_controls.focusMode)) {

315void UVCCameraController::_createCamera()
316{
317 const QString deviceName = active() ? device() : QString();
318 const QCameraDevice device = UVCDevices::device(deviceName);
319 const QStringList formats = UVCDevices::formatNames(device);
320 if (formats != _formats) {
321 _formats = formats;
322 emit formatsChanged();
323 }
324 if (deviceName.isEmpty()) {
325 return;
326 }
327 if (device.isNull()) {
328 qCWarning(UVCCameraControllerLog) << "No camera named" << deviceName;
329 return;
330 }
331
332 // Until the user grants the permission, Qt's Android and Apple cameras refuse to start, without an error: the
333 // camera starts once it is granted.
334 UVCDevices::requestPermission(this, [this] { _connectVideoOutput(); });
335 _cameraId = device.id();
336 _camera = new QCamera(device, this);
337 if (const QCameraFormat format = UVCDevices::format(device, _formatName); !format.isNull()) {
338 qCDebug(UVCCameraControllerLog) << "camera format" << _formatName;
339 _camera->setCameraFormat(format);
340 }
341 (void) connect(_camera, &QCamera::errorOccurred, this,
342 [this](QCamera::Error, const QString& errorString) { _onCameraError(errorString); });
343 // Some platforms know what a camera supports, or take its settings, only once it runs.
344 (void) connect(_camera, &QCamera::supportedFeaturesChanged, this, &UVCCameraController::controlsChanged);
345 (void) connect(_camera, &QCamera::minimumZoomFactorChanged, this, &UVCCameraController::controlsChanged);
346 (void) connect(_camera, &QCamera::maximumZoomFactorChanged, this, &UVCCameraController::controlsChanged);
347 (void) connect(_camera, &QCamera::activeChanged, this, [this](bool active) {
348 if (active) {
349 _applyControls();
350 emit controlsChanged();
351 }
352 });
353 _applyControls();
354 _captureSession = new QMediaCaptureSession(this);
355 _captureSession->setCamera(_camera);
356 _captureSession->setRecorder(_recorder->mediaRecorder());
357 if (kStillPhotos) {
358 _captureSession->setImageCapture(_imageCapture);
359 }
360}

362void UVCCameraController::_releaseCamera()
363{
364 // The capture session's photos still pending won't come: save the frame on display instead.
365 for (const PendingPhoto& photo : std::exchange(_photoFiles, {})) {
366 _savePhotoOnDisplay(photo);
367 }
368 _monitor->detach();
369 delete _captureSession;
370 _captureSession = nullptr;
371 delete _camera;
372 _camera = nullptr;
373 _cameraId.clear();
374 _framesArriving = false;
375 _gaveUp = false;
376}

386void UVCCameraController::_connectVideoOutput()
387{
388 QVideoSink* const sink = videoSink();
389 (void) disconnect(_frameConnection);
390 if (sink) {
391 _frameConnection = connect(sink, &QVideoSink::videoFrameChanged, this, &UVCCameraController::_onVideoFrame);
392 } else {

434void UVCCameraController::_onCameraError(const QString& errorString)
435{
436 qCWarning(UVCCameraControllerLog) << "Camera error:" << device() << errorString << "retries" << _retries;
437 _framesArriving = false;
438 if (_retries >= kMaxRetries) {
439 _gaveUp = true;
440 QGC::showAppMessage(tr("USB camera %1: %2").arg(device(), errorString));
441 } else {
442 if (_retries == 0) {
443 QGC::showAppMessage(tr("USB camera %1: %2. Retrying.").arg(device(), errorString));
444 }
445 ++_retries;
446 // Deferred: the camera emitting this error can't be deleted from its own signal.
447 _retryTimer->start();
448 }
449 _updateState();
450}

452void UVCCameraController::_onVideoInputsChanged()
453{
454 if (!_cameraId.isEmpty() && !_cameraAttached()) {
455 // Qt stops an unplugged camera without an error.
456 qCWarning(UVCCameraControllerLog) << "Camera unplugged:" << device() << _cameraId;
457 QGC::showAppMessage(tr("USB camera %1 was disconnected").arg(device()));
458 // Runs another camera of the same name, if one is attached.
459 _recreateCamera();
460 } else if (active() && !_camera && _cameraAttached()) {
461 // Plugged back in.
462 _recreateCamera();
463 }
464 emit camerasChanged();
465}
