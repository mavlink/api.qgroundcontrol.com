44double heightAtPixel(const QImage& image, int px, int py)
45{
46 const QRgb rgb = image.pixel(px, py);
47 return (qRed(rgb) * 256.0) + qGreen(rgb) + (qBlue(rgb) / 256.0) - 32768.0;
48}

52double heightAtUV(const QImage& image, double u, double v)

60QList<float> sampleGrid(const QImage& image, const QRectF& subWindow, int gridSize)
61{
62 QList<float> heights;
63 heights.reserve(qsizetype(gridSize + 1) * (gridSize + 1));
64 for (int row = 0; row <= gridSize; row++) {
65 const double v = subWindow.y() + (subWindow.height() * row / gridSize);
66 for (int col = 0; col <= gridSize; col++) {
67 const double u = subWindow.x() + (subWindow.width() * col / gridSize);
68 heights.append(static_cast<float>(heightAtUV(image, u, v)));
69 }
70 }
71 return heights;
72}

89ElevationTilePyramid::Grid gridFromImage(const QImage& image)
90{
91 ElevationTilePyramid::Grid grid;
92 grid.width = image.width();
93 grid.height = image.height();
94 grid.heights.reserve(qsizetype(grid.width) * grid.height);
95 for (int py = 0; py < grid.height; py++) {
96 for (int px = 0; px < grid.width; px++) {
97 grid.heights.append(static_cast<float>(heightAtPixel(image, px, py)));
98 }
99 }
100 return grid;
101}

105TileMath::TileKey fetchKeyFor(const TileMath::TileKey& key)

122int TerrariumTileFetcher::requestPatchHeights(const TileMath::TileKey& key, int gridSize)
123{
124 const int requestId = _nextRequestId();
125
126 if ((gridSize <= 0) || (gridSize > kMaxGridSize) || !TileMath::isValidKey(key)) {
127 qCWarning(GeoMapTerrariumTileFetcherLog)
128 << "requestPatchHeights rejected: key" << key << "gridSize" << gridSize;
129 _pending.insert(requestId, PendingRequest{});
130 _failAsync(requestId);
131 return requestId;
132 }
133 if (_mapId < 0) {
134 qCDebug(GeoMapTerrariumTileFetcherLog) << "requestPatchHeights: elevation provider not registered";
135 _pending.insert(requestId, PendingRequest{});
136 _failAsync(requestId);
137 return requestId;
138 }
139
140 // Deeper than the dataset: fetch the top-zoom ancestor and sample the
141 // patch's sub-window of it
142 PendingRequest pending;
143 pending.gridSize = gridSize;
144 pending.fetchKey = fetchKeyFor(key);
145 if (key.zoom <= kMaxTileZoom) {
146 pending.subWindow = QRectF(0, 0, 1, 1);
147 } else {
148 const int shift = key.zoom - kMaxTileZoom;
149 const double scale = 1.0 / (1LL << shift);
150 pending.subWindow = QRectF((key.x - (qint64(pending.fetchKey.x) << shift)) * scale,
151 (key.y - (qint64(pending.fetchKey.y) << shift)) * scale, scale, scale);
152 }
153 _pending.insert(requestId, pending);
154 _lastFetchKey = pending.fetchKey;
155
156 const bool inFlight = _fetchInFlight(pending.fetchKey);
157 _waiters[pending.fetchKey].append(requestId);
158 qCDebug(GeoMapTerrariumTileFetcherVerboseLog)
159 << "requestPatchHeights: key" << key << "requestId" << requestId << "fetchKey" << pending.fetchKey
160 << (inFlight ? "(joining in-flight fetch)" : "(starting fetch)");
161 if (inFlight) {
162 return requestId; // a fetch for this tile is already in flight; it serves all waiters
163 }
164
165 if (!_startFetch(pending.fetchKey)) {
166 qCDebug(GeoMapTerrariumTileFetcherLog) << "requestPatchHeights: could not start fetch for" << pending.fetchKey;
167 _waiters.remove(pending.fetchKey);
168 _failAsync(requestId);
169 }
170 return requestId;
171}

173bool TerrariumTileFetcher::requestTile(const TileMath::TileKey& key)
174{
175 if (!_heightField || !TileMath::isValidKey(key)) {
176 qCWarning(GeoMapTerrariumTileFetcherLog)
177 << "requestTile rejected: key" << key << "heightFieldSet" << (_heightField != nullptr);
178 return false;
179 }
180 if (_mapId < 0) {
181 qCDebug(GeoMapTerrariumTileFetcherLog) << "requestTile: elevation provider not registered";
182 return false;
183 }
184
185 const TileMath::TileKey fetchKey = fetchKeyFor(key);
186 if (_heightField->hasTile(fetchKey) || _fieldRequests.contains(fetchKey)) {
187 return true;
188 }
189
190 const bool inFlight = _fetchInFlight(fetchKey);
191 _fieldRequests.insert(fetchKey);
192 _lastFetchKey = fetchKey;
193 qCDebug(GeoMapTerrariumTileFetcherVerboseLog)
194 << "requestTile: fetchKey" << fetchKey << (inFlight ? "(joining in-flight fetch)" : "(starting fetch)");
195 if (inFlight) {
196 return true; // piggyback on the fetch already serving patch waiters
197 }
198
199 if (!_startFetch(fetchKey)) {
200 qCDebug(GeoMapTerrariumTileFetcherLog) << "requestTile: could not start fetch for" << fetchKey;
201 _fieldRequests.remove(fetchKey);
202 return false;
203 }
204 return true;
205}

212bool TerrariumTileFetcher::_startFetch(const TileMath::TileKey& fetchKey)
213{
214 const QString providerType = terrariumProviderType();
215 QGCFetchTileTask* const task =
216 QGeoFileTileCacheQGC::createFetchTileTask(providerType, fetchKey.x, fetchKey.y, fetchKey.zoom);
217 connect(task, &QGCFetchTileTask::tileFetched, this, [this, fetchKey](QGCCacheTile* tile) {
218 const std::unique_ptr<QGCCacheTile> guard(tile); // caller-owned per task contract
219 if (!_fetchInFlight(fetchKey)) {
220 return; // all interest cancelled while the lookup was in flight
221 }
222 // An unusable cached body (empty or corrupt) is a miss, not a
223 // failure: failing here would re-read the same bytes on every retry,
224 // blocking the network fallback until cache eviction
225 const QImage image = tile ? decodeTile(tile->img) : QImage();
226 if (image.isNull()) {
227 qCDebug(GeoMapTerrariumTileFetcherLog)
228 << "tile" << fetchKey << "cached entry unusable, fetching from network";
229 _fetchFromNetwork(fetchKey);
230 return;
231 }
232 qCDebug(GeoMapTerrariumTileFetcherVerboseLog) << "tile" << fetchKey << "served from cache";
233 _deliverAll(fetchKey, image);
234 });
235 connect(task, &QGCMapTask::error, this, [this, fetchKey](QGCMapTask::TaskType, const QString&) {
236 if (!_fetchInFlight(fetchKey)) {
237 return;
238 }
239 qCDebug(GeoMapTerrariumTileFetcherVerboseLog) << "tile" << fetchKey << "not cached, fetching from network";
240 _fetchFromNetwork(fetchKey);
241 });
242
243 if (!getQGCMapEngine()->addTask(task)) {
244 qCDebug(GeoMapTerrariumTileFetcherLog) << "could not queue cache lookup for" << fetchKey;
245 task->deleteLater(); // never enqueued: the worker will not clean it up
246 return false;
247 }
248 return true;
249}

251void TerrariumTileFetcher::cancelRequest(int requestId)
252{
253 const auto pendingIt = _pending.constFind(requestId);
254 if (pendingIt == _pending.cend()) {
255 return;
256 }
257 const TileMath::TileKey fetchKey = pendingIt->fetchKey;
258 _pending.erase(pendingIt);
259
260 const auto waitersIt = _waiters.find(fetchKey);
261 if (waitersIt == _waiters.end()) {
262 return;
263 }
264 waitersIt->removeOne(requestId);
265 if (!waitersIt->isEmpty()) {
266 return; // other requests still wait on this tile's fetch
267 }
268 _waiters.erase(waitersIt);
269 if (_fieldRequests.contains(fetchKey)) {
270 return; // the field still wants this tile: keep the fetch alive
271 }
272 QNetworkReply* const reply = _activeReplies.take(fetchKey);
273 if (reply) {
274 qCDebug(GeoMapTerrariumTileFetcherVerboseLog) << "cancelRequest: aborting network fetch for" << fetchKey;
275 reply->abort(); // finished handler is a no-op once no waiters remain
276 reply->deleteLater(); // don't rely on abort emitting finished for cleanup
277 }
278}

293void TerrariumTileFetcher::_fetchFromNetwork(const TileMath::TileKey& fetchKey)
294{
295 if (_activeReplies.contains(fetchKey)) {
296 return; // a reply is already in flight for this tile
297 }
298 const QNetworkRequest request =
299 QGeoTileFetcherQGC::getNetworkRequest(_mapId, fetchKey.x, fetchKey.y, fetchKey.zoom);
300 QNetworkReply* const reply = _networkManager->get(request);
301 _activeReplies.insert(fetchKey, reply);
302
303 connect(reply, &QNetworkReply::finished, this, [this, fetchKey, reply] {
304 reply->deleteLater();
305 if (_activeReplies.value(fetchKey) != reply) {
306 return; // stale: cancelled (and possibly superseded by a newer fetch)
307 }
308 _activeReplies.remove(fetchKey);
309 if (!_fetchInFlight(fetchKey)) {
310 return; // all interest cancelled (abort also lands here)
311 }
312 if (reply->error() != QNetworkReply::NoError) {
313 _failAll(fetchKey, QStringLiteral("network error: %1 (http %2)")
314 .arg(reply->errorString())
315 .arg(reply->attribute(QNetworkRequest::HttpStatusCodeAttribute).toInt()));
316 return;
317 }
318 const QByteArray data = reply->readAll();
319 if (data.isEmpty()) {
320 _failAll(fetchKey, QStringLiteral("network fetch returned empty body"));
321 return;
322 }
323 // Never cache bodies that delivery would reject (e.g. HTTP-200 error
324 // pages): a cached invalid tile would fail every retry from then on
325 const QImage image = decodeTile(data);
326 if (image.isNull()) {
327 _failAll(fetchKey, QStringLiteral("network body is not a terrarium tile, %1 bytes").arg(data.size()));
328 return;
329 }
330 qCDebug(GeoMapTerrariumTileFetcherVerboseLog)
331 << "tile" << fetchKey << "fetched from network," << data.size() << "bytes, caching";
332 // Store back so the shared cache serves this tile from now on
333 const QString providerType = terrariumProviderType();
334 QGeoFileTileCacheQGC::cacheTile(providerType, fetchKey.x, fetchKey.y, fetchKey.zoom, data,
335 UrlFactory::getImageFormat(providerType, data));
336 _deliverAll(fetchKey, image);
337 });
338}

340void TerrariumTileFetcher::_deliverAll(const TileMath::TileKey& fetchKey, const QImage& image)
341{
342 const bool forField = _fieldRequests.remove(fetchKey) && _heightField;
343 if (forField) {
344 _heightField->insertTile(fetchKey, gridFromImage(image));
345 }
346
347 const QList<int> requestIds = _waiters.take(fetchKey);
348 qCDebug(GeoMapTerrariumTileFetcherVerboseLog)
349 << "delivering tile" << fetchKey << "to" << requestIds.count() << "waiters, field insert:" << forField;
350 for (int requestId : requestIds) {
351 if (_pending.contains(requestId)) {
352 _deliver(requestId, image);
353 }
354 }
355}

357void TerrariumTileFetcher::_failAll(const TileMath::TileKey& fetchKey, const QString& reason)
358{
359 // A failed tile inserts nothing: the field keeps its current estimate, and
360 // clearing the in-flight key lets a later request retry
361 _fieldRequests.remove(fetchKey);
362
363 const QList<int> requestIds = _waiters.take(fetchKey);
364 if (_shouldWarnFailure()) {
365 qCWarning(GeoMapTerrariumTileFetcherLog)
366 << "tile" << fetchKey << "failed:" << reason << "- failing" << requestIds.count() << "waiters";
367 } else {
368 qCDebug(GeoMapTerrariumTileFetcherLog) << "tile" << fetchKey << "failed:" << reason << "- failing"
369 << requestIds.count() << "waiters (warning suppressed)";
370 }
371 for (int requestId : requestIds) {
372 if (_pending.contains(requestId)) {
373 _finishFailed(requestId);
374 }
375 }
376}
