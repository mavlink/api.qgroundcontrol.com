44const QRegularExpression* usualFunctionsRegex(int portId)
45{
46 static const QRegularExpression gpsRegex(QStringLiteral("^(GPS_\\d_CONFIG|SEP_PORT\\d_CFG)$"));
47 static const QRegularExpression telemetryRegex(QStringLiteral("^(MAV_\\d_CONFIG|UXRCE_DDS_CFG)$"));
48 static const QRegularExpression rcRegex(QStringLiteral("^RC_"));
49
50 const QString tag = portTags().value(portId);
51 if (tag.startsWith(QStringLiteral("TEL")) || (tag == QStringLiteral("WIFI")) || (portId == kEthernetPortId)) {
52 return &telemetryRegex;
53 }
54 if (tag.startsWith(QStringLiteral("GPS"))) {
55 return &gpsRegex;

65bool isFunctionParam(const QVariantList& enumValues)
66{
67 bool hasDisabled = false;
68 bool hasNonUartPort = false;
69
70 for (const QVariant& enumValue : enumValues) {
71 bool ok = false;
72 const int value = enumValue.toInt(&ok);
73 if (!ok) {
74 return false;
75 }
76 if (value == kDisabledPortId) {
77 hasDisabled = true;
78 } else if (portTags().contains(value)) {
79 // UART indices 1-9 overlap ordinary small enums, so they don't identify a port selection on their own
80 hasNonUartPort = hasNonUartPort || (value >= kFirstNonUartPortId);
81 } else if (value != kEthernetPortId) {
82 return false;
83 }
84 }
85
86 return hasDisabled && hasNonUartPort;
87}

183QList<ParamWrite> assignPort(const QList<Function>& functions, int portId, const QString& functionParamName)
184{
185 QList<ParamWrite> writes;
186
187 if (!functionParamName.isEmpty()) {
188 const Function* const function = findFunction(functions, functionParamName);
189 if (!function || !allowsPort(*function, portId)) {
190 return writes;
191 }
192 }
193
194 if (!isSharedPort(portId)) {
195 for (const Function& function : functions) {
196 if ((function.currentPortId == portId) && (function.paramName != functionParamName) &&
197 (function.defaultPortId != portId)) {
198 writes.append({.paramName = function.paramName, .value = kDisabledPortId});
199 }
200 }
201 }
202
203 if (!functionParamName.isEmpty() && (findFunction(functions, functionParamName)->currentPortId != portId)) {
204 writes.append({.paramName = functionParamName, .value = portId});
205 }
206
207 return writes;
