12// Compares digit runs by numeric value so "control[2]" sorts before "control[10]"
13bool _naturalLess(const QString& a, const QString& b)
14{
15 const auto isDigit = [](QChar c) { return (c >= QLatin1Char('0')) && (c <= QLatin1Char('9')); };
16
17 qsizetype i = 0;
18 qsizetype j = 0;
19 while ((i < a.size()) && (j < b.size())) {
20 if (isDigit(a[i]) && isDigit(b[j])) {
21 const qsizetype aStart = i;
22 const qsizetype bStart = j;
23 while ((i < a.size()) && isDigit(a[i])) {
24 i++;
25 }
26 while ((j < b.size()) && isDigit(b[j])) {
27 j++;
28 }
29 const QStringView aNumber = QStringView(a).sliced(aStart, i - aStart);
30 const QStringView bNumber = QStringView(b).sliced(bStart, j - bStart);
31 if (aNumber.size() != bNumber.size()) {
32 return aNumber.size() < bNumber.size();
33 }
34 const int numberCompare = aNumber.compare(bNumber);
35 if (numberCompare != 0) {
36 return numberCompare < 0;
37 }
38 continue;
39 }
40 if (a[i] != b[j]) {
41 return a[i] < b[j];
42 }
43 i++;
44 j++;
45 }
46
47 return (a.size() - i) < (b.size() - j);
48}

185void LogViewerController::_rebuildFieldRows()
186{
187 QHash<QString, QStringList> groupedMap;
188 QStringList groups;
189
190 for (const QString &field : _plottableFields) {
191 const int splitIndex = field.indexOf('.');
192 const QString groupName = (splitIndex > 0) ? field.left(splitIndex) : tr("Other");
193 const QString shortName = (splitIndex > 0) ? field.mid(splitIndex + 1) : field;
194 if (!groupedMap.contains(groupName)) {
195 groups.append(groupName);
196 }
197 groupedMap[groupName].append(shortName);
198 }
199
200 std::sort(groups.begin(), groups.end(), _naturalLess);
201
202 QVariantList rows;
203 for (const QString &groupName : groups) {
204 QVariantMap groupRow;
205 groupRow[QStringLiteral("rowType")] = QStringLiteral("group");
206 groupRow[QStringLiteral("group")] = groupName;
207 rows.append(groupRow);
208
209 if (!_expandedGroups.contains(groupName)) {
210 continue;
211 }
212
213 QStringList fieldNames = groupedMap.value(groupName);
214 std::sort(fieldNames.begin(), fieldNames.end(), _naturalLess);
215 for (const QString &shortName : fieldNames) {
216 QVariantMap fieldRow;
217 fieldRow[QStringLiteral("rowType")] = QStringLiteral("field");
218 fieldRow[QStringLiteral("group")] = groupName;
219 fieldRow[QStringLiteral("shortName")] = shortName;
220 fieldRow[QStringLiteral("fullName")] = QStringLiteral("%1.%2").arg(groupName, shortName);
221 rows.append(fieldRow);
222 }
223 }
224
225 _fieldRows = rows;
