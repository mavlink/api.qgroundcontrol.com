167{
168 if (failed() || !buffer.data() || buffer.empty()) {

170 }
171 const auto result =

200bool GPSCommandChannel::write(QByteArrayView bytes)
201{
202 if (failed()) {
203 return false;
204 }
205 if (!bytes.data()) {
206 _errorDetail.clear();
207 _error = GPSProtocolError::InvalidArgument;
208 return false;
209 }
210 const uint64_t deadline =
211 std::min(_operationDeadline.untilUs, _commandCompleted ? UINT64_MAX : _commandDeadline.untilUs);
212 // A late command that wrote nothing just times out. A late later part goes to the transport, whose sticky failure
213 // stops the session rather than leave a torn frame on the link.
214 if (nowUs() >= deadline && (_commandCompleted || _command.acceptedBytes == 0)) {
215 finishCommand(GPSCommandOutcome::TimedOut);
216 return false;
217 }
218 const auto result =
219 _runtime._io.cancelToken.isCancelled()
220 ? GPSWriteResult{GPSWriteStatus::Cancelled}
221 : _runtime._io.write({reinterpret_cast<const uint8_t*>(bytes.data()), static_cast<size_t>(bytes.size())},
222 {deadline});
223 const auto length = bytes.size();
224 _errorDetail = result.detail;
225 // A write outside an attempt belongs to no reported evidence.
226 if (!_commandCompleted) {
227 _command.acceptedBytes += result.acceptedBytes;
228 _command.writtenBytes += result.writtenBytes;
229 }
230 if (result.status == GPSWriteStatus::Completed && result.acceptedBytes == length && result.writtenBytes == length &&
231 result.uncertainBytes() == 0) {
232 return true;
233 }
234 if (result.status == GPSWriteStatus::Unsupported && result.acceptedBytes == 0 && result.writtenBytes == 0) {
235 finishCommand(GPSCommandOutcome::TransportError);
236 return false;
237 }
238 // The transport found the command late before taking any of its bytes, as the check above can just miss.
239 if (result.status == GPSWriteStatus::TimedOut && result.acceptedBytes == 0 && result.writtenBytes == 0 &&
240 (_commandCompleted || _command.acceptedBytes == 0)) {
241 finishCommand(GPSCommandOutcome::TimedOut);
242 return false;
243 }
244 _error = result.status == GPSWriteStatus::Cancelled ? GPSProtocolError::Cancelled : GPSProtocolError::Transport;
245 if (_error == GPSProtocolError::Transport && _errorDetail.isEmpty()) {
246 const QString receiver = gpsReceiverName(_runtime.family().type);
247 const QString command = _commandCompleted ? QString() : QString::fromUtf8(_command.command);
248 if (command.isEmpty()) {
249 _errorDetail = QStringLiteral("Could not write to the %1 receiver").arg(receiver);
250 } else if (result.acceptedBytes > 0) {
251 _errorDetail = QStringLiteral("%1 receiver did not accept the full '%2' command").arg(receiver, command);
252 } else {
253 _errorDetail = QStringLiteral("Could not write '%2' to the %1 receiver").arg(receiver, command);
254 }
255 }
256 finishCommand(result.status == GPSWriteStatus::Cancelled ? GPSCommandOutcome::Cancelled
257 : GPSCommandOutcome::TransportError);
258 return false;
259}

304GPSConfigurationEvidence GPSCommandChannel::awaitReply(GPSReplyPoll reply)
305{
306 if (_commandCompleted) {
307 return _command;
308 }
309 const auto scope = deadlineScope(_commandDeadline);
310 const uint64_t deadline = _operationDeadline.untilUs;
311 GPSCommandOutcome outcome = GPSCommandOutcome::Pending;
312 for (;;) {
313 if (const auto failure = failureOutcome(); failure != GPSCommandOutcome::Pending) {
314 outcome = failure;

318 outcome = response;

384GPSCommandSequence::Result GPSCommandChannel::runSequence(const GPSCommandSequence& sequence)
385{
386 for (size_t index = 0; index < sequence.steps.size(); ++index) {
387 const auto& command = sequence.steps[index];
388 GPSConfigurationEvidence result;
389 for (unsigned attempt = 0; attempt < (std::max) (command.attempts, 1U); ++attempt) {
390 result = transact(command);
391 if (result.succeeded() || failed()) {
392 break;
393 }
394 }
395 if (!result.succeeded() && (command.step.required || failed())) {
396 return GPSCommandSequence::Result{
397 .failedStep = index, .failedLabel = command.step.command, .outcome = result.outcome};
398 }
399 }
400 return GPSCommandSequence::Result{};

412GPSBaudDetection GPSCommandChannel::detectBaud(std::span<const unsigned> candidates, unsigned baud,
413 GPSBaudProbeFunction probe)
414{
415 std::vector<unsigned> order;
416 if (baud) {
417 order.push_back(baud);
418 } else {
419 order.assign(candidates.begin(), candidates.end());
420 if (_detectedBaud != 0) {
421 // Detection may recognise a receiver by its traffic at a rate the family does not list.
422 std::erase(order, _detectedBaud);
423 order.insert(order.begin(), _detectedBaud);
424 }
425 }
426 GPSBaudDetection result{.baud = baud};
427 for (const unsigned candidate : order) {
428 result.baud = candidate;
429 const GPSBaudStatus status = _setBaudrate(candidate);
430 if (status == GPSBaudStatus::Unsupported && !baud) {
431 continue;
432 }
433 if (status != GPSBaudStatus::Configured) {
434 result.linkFailed = true;
435 return result;
436 }
437 result.probed.push_back(candidate);
438 const GPSBaudProbe outcome = probe();
439 if (outcome == GPSBaudProbe::Found) {
440 result.found = true;
441 return result;
442 }
443 if (outcome == GPSBaudProbe::Stop || failed()) {
444 return result;
445 }
446 }
447 return result;
448}

450GPSReceiveUpdates GPSCommandChannel::receiveCycle(std::chrono::milliseconds timeout)
451{
452 const auto scope = deadlineScope(timeout);
453 if (failed()) {
454 return GPSReceiveUpdates{};
455 }
456 GPSFamilyProtocol& protocol = _runtime.protocol();
457 GPSReceiveUpdates handled;
458 std::array<uint8_t, READ_CHUNK_SIZE> buffer{};
459 for (;;) {

464 if (count < 0) {
