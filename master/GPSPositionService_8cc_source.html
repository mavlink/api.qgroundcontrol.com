199{

201 return;
202 }

205 return;

240GPSPositionService::SelectedSource GPSPositionService::_choosePositionSource()
241{
242 const SelectedSource internal =
243 _sourceFor(SelectedSource::Internal)
244 ? SelectedSource::Internal
245 : (_sourceFor(SelectedSource::Simulated) ? SelectedSource::Simulated : SelectedSource::Internal);
246 switch (_sourceMode) {
247 case SourceMode::ReceiverOnly:
248 return SelectedSource::Receiver;
249 case SourceMode::NmeaOnly:
250 return SelectedSource::Nmea;
251 case SourceMode::InternalOnly:
252 return internal;
253 case SourceMode::LegacyPriority:
254 return _sourceFor(SelectedSource::Receiver)
255 ? SelectedSource::Receiver
256 : (_sourceFor(SelectedSource::Nmea) ? SelectedSource::Nmea : internal);
257 case SourceMode::Automatic:
258 break;
259 }
260 const std::array priorityKinds = {SelectedSource::Receiver, SelectedSource::Nmea, internal};
261 std::array<GPSPositionSourceSelector::Candidate, 3> priority{};
262 for (size_t index = 0; index < priority.size(); ++index) {
263 const auto kind = priorityKinds[index];
264 auto* health = _binding(kind).adapter->health();
265 priority[index] = {static_cast<int>(kind), _sourceFor(kind) != nullptr,
266 health && _acceptedSourceObservation(kind).has_value()};
267 }
268 const qint64 nowMs = _scheduler ? _scheduler->nowMs() : 0;
269 const auto selected = static_cast<SelectedSource>(_selector.select(
270 priority, _currentSource ? static_cast<int>(_selectedKind) : -1, nowMs, RECOVERY_DELAY.count()));
271 if (_selector.recovering() && _scheduler) {
272 _recoveryTask.schedule(std::chrono::milliseconds(_selector.remainingRecoveryMs(nowMs, RECOVERY_DELAY.count())),
273 [this]() { _selectPositionSource(); });
274 } else {
275 _recoveryTask.cancel();
276 }
277 return selected;

372void GPSPositionService::_updateSelectionStatus()
373{
374 const auto selected = _currentSource ? _selectedKind : SelectedSource::None;
375 SourceStatus status = SourceStatus::NoSource;
376 if (_currentHealth) {
377 switch (_currentHealth->state()) {
378 case GPSSourceHealth::State::NoData:
379 status = SourceStatus::WaitingForFix;
380 break;
381 case GPSSourceHealth::State::Usable:
382 status = !_acceptedSourceObservation(_selectedKind) ? SourceStatus::Stale
383 : _gcsPosition.isValid() ? SourceStatus::Active
384 : SourceStatus::WaitingForFix;
385 break;
386 case GPSSourceHealth::State::Stale:
387 status = SourceStatus::Stale;
388 break;
389 case GPSSourceHealth::State::Invalid:
390 status = SourceStatus::InvalidFix;
391 break;
392 }
393 } else if (_sourceMode != SourceMode::ReceiverOnly && _sourceMode != SourceMode::NmeaOnly) {
394 status = _platformStatus;
395 }
396 if (selected == SelectedSource::Internal &&
397 (_platformStatus == SourceStatus::PermissionDenied || _platformStatus == SourceStatus::BackendUnavailable)) {
398 status = _platformStatus;
399 }
400 QString reason;
401 switch (_sourceMode) {
402 case SourceMode::LegacyPriority:
403 reason = tr("Receiver, then NMEA, then internal positioning; no health-based switching");
404 break;
405 case SourceMode::ReceiverOnly:
406 reason = tr("Pinned to the configured receiver");
407 break;
408 case SourceMode::NmeaOnly:
409 reason = tr("Pinned to the NMEA source");
410 break;
411 case SourceMode::InternalOnly:
412 reason = tr("Pinned to internal positioning");
413 break;
414 case SourceMode::Automatic:
415 reason = _selector.recovering() ? tr("Keeping the current source while the preferred source recovers")
416 : status == SourceStatus::Active ? tr("Using the highest-priority healthy source")
417 : tr("Waiting for a healthy position source");
418 break;
419 }
420 const bool changed = _selectedSource != selected || _sourceStatus != status || _selectionReason != reason;
421 _selectedSource = selected;
422 _sourceStatus = status;
423 _selectionReason = reason;
424 const auto name = selectedSourceName();
425 if (changed || _selectionName != name) {
426 _selectionName = name;
427 emit selectionChanged();
428 }
