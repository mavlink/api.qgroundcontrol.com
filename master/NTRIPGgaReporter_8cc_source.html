45bool reportable(const std::optional<GPSObservation>& observation)
46{

142void NTRIPGgaReporter::_sendGGA()
143{
144 if (!_transport) {
145 return;
146 }
147 const auto requested = _requestedSource;
148 const auto selection = _getBestPosition(requested);
149 _logSelection(requested, selection);
150 if (!selection.position) {
151 _clearSource();
152 return;
153 }
154
155 _awaitingFirstGga = false;
156
157 const GPSObservation& position = *selection.position;
158 const QGeoCoordinate coordinate = position.position.coordinate();
159 const NMEA::GGA fix{
160 .latitude = coordinate.latitude(),
161 .longitude = coordinate.longitude(),
162 .altitude = coordinate.altitude(),
163 .hdop = position.horizontalDop && qIsFinite(*position.horizontalDop) && *position.horizontalDop >= 0
164 ? *position.horizontalDop
165 : qQNaN(),
166 .quality = ggaQuality(position.fixQuality),
167 .satellitesUsed = position.satellitesUsed && *position.satellitesUsed >= 0
168 ? std::optional<unsigned>(static_cast<unsigned>(*position.satellitesUsed))
169 : std::nullopt,
170 };
171 const QByteArray gga = NMEAUtils::makeGGA(fix, QDateTime::currentDateTimeUtc().time());
172 _transport->sendNMEA(gga);
173 if (const QString label = _sourceLabel(selection.source); label != _source) {
174 _source = label;
175 emit sourceChanged(_source);
176 }
177}

179NTRIPGgaReporter::SelectedPosition NTRIPGgaReporter::_getBestPosition(PositionSource requested) const
180{
181 if (requested != PositionSource::Auto) {
182 const auto provider = _providers.value(requested);
183 auto position = provider ? provider() : std::nullopt;
184 return {reportable(position) ? std::move(position) : std::nullopt, requested};
185 }
186
187 static constexpr PositionSource kPriority[] = {
188 PositionSource::VehicleGPS,
189 PositionSource::VehicleEKF,
190 PositionSource::RTKReceiver,
191 PositionSource::GCSPosition,
192 };
193 for (PositionSource source : kPriority) {
194 if (const auto provider = _providers.value(source)) {
195 if (auto position = provider(); reportable(position)) {
196 return {std::move(position), source};
197 }
198 }
199 }
200 return {};
201}

203void NTRIPGgaReporter::_logSelection(PositionSource requested, const SelectedPosition& selection)
204{
205 const auto current = std::make_tuple(requested, selection.source, selection.position.has_value());
206 if (std::exchange(_loggedSelection, current) == current) {
207 return;
208 }
209 if (!selection.position) {
210 qCDebug(NTRIPGgaReporterLog).noquote()

212 return;
213 }
