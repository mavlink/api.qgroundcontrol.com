48void GPSNativeAshtech::receiveWait(unsigned timeout_min)
49{
50 uint64_t time_started = nowUs();
51
52 while (nowUs() < time_started + timeout_min * 1000) {
53 receive(timeout_min);
54 if (ioError()) {

60int GPSNativeAshtech::receive(unsigned timeout)
61{
62 if (const auto deadline = _satelliteAssembler.deadlineUs()) {
63 timeout = std::min(timeout, static_cast<unsigned>(remainingMilliseconds(*deadline)));
64 }
65 const int result = receiveDecoded(timeout);
66 serviceControls();
67 return ioError() ? ioError() : result;
68}
