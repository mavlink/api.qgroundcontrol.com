66GPSReadResult UDPGPSTransport::read(uint8_t* buffer, int length, int timeoutMs)
67{
68 if (isCancelled()) {
69 return {GPSReadStatus::Cancelled};
70 }
71 if (!buffer || length < 0) {
72 return {GPSReadStatus::InvalidData};
73 }
74 if (fatalError()) {
75 return {GPSReadStatus::Closed, 0, _socket ? _socket->errorString() : QString()};
76 }
77 if (length == 0) {
78 return {GPSReadStatus::Data};
79 }
80
81 QDeadlineTimer deadline((std::max) (timeoutMs, 0));
82 while (_pending.isEmpty()) {
83 if (!gpsWaitForSocket(
84 *this, _socket.get(), [this]() { return _socket->hasPendingDatagrams(); }, deadline,
85 kCancellationPollMs)) {
86 return {isCancelled() ? GPSReadStatus::Cancelled
87 : fatalError() ? GPSReadStatus::Error
88 : GPSReadStatus::TimedOut,
89 0, fatalError() ? _socket->errorString() : QString()};
90 }
91 const QNetworkDatagram datagram = _socket->receiveDatagram();
92 if (!datagram.isValid()) {
93 _failed = true;
94 return {GPSReadStatus::Error, 0, _socket->errorString()};
95 }
96 _pending = datagram.data();
97 _pendingOffset = 0;
98 if (_pending.isEmpty() && deadline.hasExpired()) {
99 return {GPSReadStatus::TimedOut};
100 }
101 }
102
103 // Keep the remainder when the driver requests fewer bytes than a whole datagram.
104 const int count = static_cast<int>((std::min) (static_cast<qsizetype>(length), _pending.size() - _pendingOffset));
105 std::memcpy(buffer, _pending.constData() + _pendingOffset, count);
106 _pendingOffset += count;
107 if (_pendingOffset == _pending.size()) {
108 _pending.clear();
109 _pendingOffset = 0;
110 }
111 return {GPSReadStatus::Data, count};
112}
