41void UDPGPSTransport::_receivePending()
42{
43 while (_socket && _socket->hasPendingDatagrams()) {
44 const QNetworkDatagram datagram = _socket->receiveDatagram();
45 if (!datagram.isValid()) {
46 break;
47 }
48 // The dual-stack socket may report an IPv4 sender as an IPv4-mapped IPv6 address.
49 const bool selected = _peerPort != 0 && datagram.senderPort() == _peerPort &&
50 datagram.senderAddress().isEqual(_peerAddress, QHostAddress::TolerantConversion);
51 if (!selected) {
52 if (_peerPort != 0 && _peerIdle.durationElapsed() < _peerIdleTimeout) {
53 continue;
54 }
55 qCDebug(UDPGPSTransportLog) << "Receiving from UDP sender" << datagram.senderAddress()
56 << datagram.senderPort();
57 _peerAddress = datagram.senderAddress();
58 _peerPort = static_cast<quint16>(datagram.senderPort());
59 // A replaced sender's partial sentence must not join the new stream.
60 _pending.clear();
61 }
62 _peerIdle.start();
63 _pending.append(datagram.data());
64 if (const qsizetype excess = _pending.size() - READ_BUFFER_BYTES; excess > 0) {
65 _pending.remove(0, excess);
66 qCWarning(UDPGPSTransportLog)
67 << "Receiver input outran the decoder; dropped the oldest" << excess << "bytes";
68 }
69 }
