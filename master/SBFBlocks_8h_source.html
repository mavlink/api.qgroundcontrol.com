97[[nodiscard]] inline std::optional<BlockHeader> checkedHeader(std::span<const uint8_t> block)
98{
99 // The block CRC covers everything from the id field to the end of the block.
100 constexpr size_t CRC_START = 4;
101 if (block.size() < Wire::SIZE<BlockHeader>) {
102 return std::nullopt;
103 }
104 const auto header = Wire::decode<BlockHeader>(block);
105 if (header.length < Wire::SIZE<BlockHeader> || header.length > block.size() ||
106 header.crc != QGC::crc16Xmodem(block.subspan(CRC_START, header.length - CRC_START))) {
107 return std::nullopt;
108 }
109 return header;
110}
