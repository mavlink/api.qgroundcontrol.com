47int GPSAsciiProtocol::decodeByte(uint8_t byte)
48{
49 if (_rtcm.hasPartialFrame() || byte == RTCMFramer::PREAMBLE) {
50 _lineSize = 0;
51 _discardLine = false;
52 _lineEnded = false;
53 _pendingRTCM = _rtcm.addByte(byte);
54 _drainRTCM();
55 return 0;
56 }
57 if (byte == '\r') {
58 _lineEnded = true;
59 return 0;
60 }
61 if (byte == '\n') {
62 const std::string_view line{_line.data(), _lineSize};
63 int updates = 0;
64 if (!_discardLine && !line.empty()) {
65 updates = _handleNmea(line);
66 updates |= handleReceiverLine(line);
67 }
68 _lineSize = 0;
69 _discardLine = false;
70 _lineEnded = false;
71 _drainSatellites();
72 return updates;
73 }
74 if (byte == '$' || byte == '#') {
75 _lineSize = 0;
76 _discardLine = false;
77 _lineEnded = false;
78 }
79 if (_lineEnded || byte < ' ' || byte > '~' || _lineSize == _line.size()) {
80 _lineSize = 0;
81 _discardLine = true;
82 return 0;
83 }
84 if (!_discardLine) {
85 _line[_lineSize++] = static_cast<char>(byte);
86 }
87 return 0;
88}

90int GPSAsciiProtocol::_handleNmea(std::string_view line)
91{
92 const auto sentence = NMEA::sentence(line);
93 if (!sentence) {
94 return 0;
95 }
96 const uint64_t now = nowUs();
97 int updates = GPSDecodedBatch::PROTOCOL_ACTIVITY;
98 auto satelliteUpdate = _satelliteAssembler.ingest(*sentence, now);
99 _publishSatellites(satelliteUpdate.completed);
100 if (const auto fix = NMEA::gga(*sentence)) {
101 const auto positionTime = NMEA::utcMilliseconds(sentence->fields[NMEA::Field::UTC_TIME]);
102 if (!positionTime || positionTime != _positionTime) {
103 _vdopReceivedAtUs.reset();
104 }
105 _expireVdop(now);
106 applyNMEAGGA(*_position, *fix, now);
107 _positionTime = positionTime;
108 const bool matchingAccuracy = _positionTime && _positionTime == _accuracyTime && now >= _accuracyReceivedAtUs &&
109 now - _accuracyReceivedAtUs <= METADATA_MAX_AGE_US;
110 _position->eph = matchingAccuracy ? _accuracy.horizontalAccuracy : NAN;
111 _position->epv = matchingAccuracy ? _accuracy.verticalAccuracy : NAN;
112 _position->accuracy_timestamp = matchingAccuracy ? _accuracyReceivedAtUs : 0;
113 publishSatelliteUsage(fix->satellitesUsed ? std::optional<int>(*fix->satellitesUsed) : std::nullopt);
114 updates |= 1;
115 } else if (const auto accuracy = NMEA::gst(*sentence)) {
116 _accuracyTime = NMEA::utcMilliseconds(sentence->fields[NMEA::Field::UTC_TIME]);
117 _accuracyReceivedAtUs = now;
118 _accuracy = *accuracy;
119 if (_positionTime && _positionTime == _accuracyTime && now >= _position->timestamp &&
120 now - _position->timestamp <= METADATA_MAX_AGE_US) {
121 _expireVdop(now);
122 _position->eph = _accuracy.horizontalAccuracy;
123 _position->epv = _accuracy.verticalAccuracy;
124 _position->accuracy_timestamp = now;
125 updates |= 1;
126 }
127 } else if (sentence->type() == "GSA" && satelliteUpdate.accepted && _positionTime && now >= _position->timestamp &&
128 now - _position->timestamp <= METADATA_MAX_AGE_US) {
129 // GSA has no UTC field, so it can only supplement the preceding fresh GGA.
130 const auto hdop = NMEA::number<float>(sentence->fields[NMEA::Field::GSA_HDOP]);
131 const auto vdop = NMEA::number<float>(sentence->fields[NMEA::Field::GSA_VDOP]);
132 _position->hdop = hdop && *hdop >= 0 ? *hdop : NAN;
133 _position->vdop = vdop && *vdop >= 0 ? *vdop : NAN;
134 _vdopReceivedAtUs = now;
135 _position->dop_timestamp = now;
136 }
137 return updates;
138}

140void GPSAsciiProtocol::_expireVdop(uint64_t now)
141{

161void GPSAsciiProtocol::_drainSatellites()
162{
163 size_t count = 0;
164 // Leave space for the position and vendor event belonging to the current line.
165 while (count < _pendingSatellites.size() && _decoded.events.size() + 2 < GPSDecodedBatch::MAX_EVENTS) {
166 const auto& system = _pendingSatellites[count++];
167 *_satellites = gpsNMEASatelliteReport(system);
168 publishSatellites(*_satellites);
169 }
170 _pendingSatellites.erase(_pendingSatellites.begin(), _pendingSatellites.begin() + count);
171}

181void GPSAsciiProtocol::_drainRTCM()
182{
183 while (_pendingRTCM && _decoded.events.size() + 2 < GPSDecodedBatch::MAX_EVENTS) {
184 if (_rtcm.valid() && _rtcmEnabled) {
185 const auto frame = _rtcm.frame();
186 gotRTCMMessage(frame.data(), static_cast<int>(frame.size()));
187 _decoded.updates |= GPSDecodedBatch::PROTOCOL_ACTIVITY;
188 }
189 _pendingRTCM = _rtcm.nextFrame();
190 }
