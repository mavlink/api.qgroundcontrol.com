75QString PiperTtsEngine::_searchExecutable()
76{
77 // Desktop launchers often don't have ~/.local/bin in PATH, which is where pip/uv/pipx install Piper.
78 const QStringList fallbackPaths = {QDir::homePath() + QStringLiteral("/.local/bin")};
79
80 // Some distributions name it piper-tts because "piper" is also a mouse configuration tool.
81 QStringList candidates;
82 for (const QString& name : {QStringLiteral("piper-tts"), QStringLiteral("piper")}) {
83 candidates << QStandardPaths::findExecutable(name) << QStandardPaths::findExecutable(name, fallbackPaths);
84 }
85 (void) candidates.removeAll(QString());
86 (void) candidates.removeDuplicates();
87
88 for (const QString& path : candidates) {
89 if (_isPiperTts(path)) {
90 return path;
91 }
92 qCDebug(PiperTtsEngineLog) << "Skipping" << path << "which is not Piper text-to-speech";
93 }
94 return QString();
95}

97bool PiperTtsEngine::_isPiperTts(const QString& path)
98{
99 QProcess process;
100 process.setProcessChannelMode(QProcess::MergedChannels);
101 process.start(path, {QStringLiteral("--help")});
102 if (!process.waitForFinished(5000)) {
103 process.kill();
104 (void) process.waitForFinished(1000);
105 return false;

110QList<QVoice> PiperTtsEngine::_findVoices()
111{
112 QList<QVoice> voices;
113 QStringList names;
114
115 const QStringList voiceDirs = QStandardPaths::locateAll(
116 QStandardPaths::GenericDataLocation, QStringLiteral("piper-voices"), QStandardPaths::LocateDirectory);
117 for (const QString& voiceDir : voiceDirs) {
118 QDirIterator it(voiceDir, {QStringLiteral("*.onnx")}, QDir::Files, QDirIterator::Subdirectories);
119 while (it.hasNext()) {
120 const QFileInfo model(it.next());
121 const QString name = model.completeBaseName();
122 if (names.contains(name)) {
123 continue;
124 }
125
126 QFile configFile(model.filePath() + QStringLiteral(".json"));
127 if (!configFile.open(QIODevice::ReadOnly)) {
128 qCDebug(PiperTtsEngineLog) << "Skipping voice without config:" << model.filePath();
129 continue;
130 }
131 const QJsonObject config = QJsonDocument::fromJson(configFile.readAll()).object();
132 const QString languageCode =
133 config.value(QStringLiteral("language")).toObject().value(QStringLiteral("code")).toString();
134 if (languageCode.isEmpty()) {
135 qCDebug(PiperTtsEngineLog) << "Skipping voice without language code:" << model.filePath();
136 continue;
137 }
138
139 names.append(name);
140 voices.append(createVoice(name, QLocale(languageCode), QVoice::Unknown, QVoice::Other, model.filePath()));
141 }
142 }
143
144 std::ranges::sort(voices, {}, &QVoice::name);
145 return voices;
146}

186void PiperTtsEngine::_readProcessOutput()
187{
188 while (_process->canReadLine()) {
189 const QString line = QString::fromUtf8(_process->readLine()).trimmed();
190 const qsizetype pathStart = line.indexOf(_outputDir.path());
191 if ((pathStart < 0) || !line.endsWith(QStringLiteral(".wav"))) {
192 qCDebug(PiperTtsEngineLog) << "Piper:" << line;
193 continue;
194 }
195
196 const QString path = line.mid(pathStart);
197 // Piper answers in order, so canceled utterances come before the current ones.
198 if (_discardCount > 0) {
199 _discardCount--;
200 (void) QFile::remove(path);
201 continue;
202 }
203 if (_pendingCount > 0) {
204 _pendingCount--;
205 }
206 _playWavFile(path);
207 }
