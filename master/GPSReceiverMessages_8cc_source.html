116 return GPSReceiver::tr("Receiving NMEA data without position messages. Enable GGA output.");

118 case GPSInputProblem::CorrectionsOnly:
119 // Forwarding them is all a corrections-only receiver needs to do.
120 return forwardingCorrections ? QString()
121 : GPSReceiver::tr(
122 "Receiving RTCM corrections without position messages. Turn on "
123 "Forward receiver RTCM to send them to vehicles.");
124 }
125 return {};
