49StreamWatchdog::Result StreamWatchdog::check(qint64 nowMs, bool displaying)
50{
51 // The count before the time: a counted frame's time is then the one read, or a later one. Read the other way
52 // round, a first frame noted in between would get the short timeout measured from before it, a false stall.
53 const bool firstFrameSeen = _sourceFrames.load(std::memory_order_acquire) > 0;
54 qint64 lastSourceFrameMs = 0;
55 // The first check starts the clock, unless a frame's time got there first.
56 if (_lastSourceFrameMs.compare_exchange_strong(lastSourceFrameMs, nowMs, std::memory_order_relaxed)) {
57 lastSourceFrameMs = nowMs;
58 }
59
60 const qint64 sourceTimeoutMs = firstFrameSeen ? _timeoutMs : (_timeoutMs + _firstFrameGraceMs);
61 const qint64 sourceSilentMs = nowMs - lastSourceFrameMs;
62 if (sourceSilentMs > sourceTimeoutMs) {
63 return {.stall = Stall::Source, .silentMs = sourceSilentMs};
64 }
65
66 if (displaying) {
67 const qint64 sinkSilentMs = nowMs - _lastSinkFrameMs.load(std::memory_order_relaxed);
68 if (sinkSilentMs > (_timeoutMs * kDecoderTimeoutFactor)) {
69 return {.stall = Stall::Decoder, .silentMs = sinkSilentMs};
70 }
71 }
72
73 return {};
74}
