22QString openFailure(const QSerialPort& serial, const QString& device)
23{
24 switch (serial.error()) {
25 case QSerialPort::DeviceNotFoundError:
26 //: %1 is a serial device, such as /dev/ttyACM0 or COM3
27 return QCoreApplication::translate("SerialGPSTransport", "%1 was not found").arg(device);
28 case QSerialPort::PermissionError:
29#ifdef Q_OS_UNIX
30 // A device held by another program reports a permission error too.
31 if (const QFileInfo file(device); file.exists() && !(file.isReadable() && file.isWritable())) {
32 //: %1 is a serial device, such as /dev/ttyACM0
33 return QCoreApplication::translate("SerialGPSTransport", "permission denied for %1").arg(device);
34 }
35#endif
36 //: %1 is a serial device, such as /dev/ttyACM0 or COM3
37 return QCoreApplication::translate("SerialGPSTransport", "%1 is in use by another program").arg(device);
38 default:
39 return serial.errorString();
40 }
41}

54GPSOpenResult SerialGPSTransport::open()
55{
56 if (isCancelled()) {
57 return {GPSOpenStatus::Cancelled};
58 }
59 _serial = std::make_unique<QSerialPort>();
60 _inputOverflow = false;
61 _serial->setReadBufferSize(READ_BUFFER_BYTES);
62 QObject::connect(
63 _serial.get(), &QSerialPort::readyRead, _serial.get(),
64 [this]() {
65 // With no serial flow control, exhausting our budget makes stream continuity unverifiable.
66 // Retire the attempt rather than decode bytes across a possible overrun.
67 _inputOverflow = _inputOverflow || _serial->bytesAvailable() >= READ_BUFFER_BYTES;
68 },
69 Qt::DirectConnection);
70 _acceptedTotal = 0;
71 _writtenTotal = 0;
72 QObject::connect(
73 _serial.get(), &QSerialPort::bytesWritten, _serial.get(), [this](qint64 count) { _writtenTotal += count; },
74 Qt::DirectConnection);
75 _serial->setPortName(_device);
76 const QDeadlineTimer openDeadline(_openTimeout);
77 while (!_serial->open(QIODevice::ReadWrite)) {
78 if (isCancelled()) {
79 return {GPSOpenStatus::Cancelled};
80 }
81 // Device can take 10-20s to become accessible after startup.
82 if (_serial->error() != QSerialPort::PermissionError || openDeadline.hasExpired()) {
83 qCWarning(SerialGPSTransportLog)
84 << "GPS: Failed to open Serial Device" << _device << _serial->errorString();
85 return {openDeadline.hasExpired() ? GPSOpenStatus::TimedOut : GPSOpenStatus::Error,
86 openFailure(*_serial, _device)};
87 }
88 qCDebug(SerialGPSTransportLog) << "Cannot open device... retrying";
89 const QDeadlineTimer retryDeadline(
90 (std::min) (std::chrono::milliseconds(openDeadline.remainingTime()), OPEN_RETRY));
91 (void) waitUntil([]() { return false; }, retryDeadline);
92 if (isCancelled()) {
93 return {GPSOpenStatus::Cancelled};
94 }
95 }
96 _serial->clearError();
97
98 const bool configured = _serial->setBaudRate(QSerialPort::Baud9600) && _serial->setDataBits(QSerialPort::Data8) &&
99 _serial->setParity(QSerialPort::NoParity) && _serial->setStopBits(QSerialPort::OneStop) &&
100 _serial->setFlowControl(QSerialPort::NoFlowControl);
101 if (!configured || isCancelled()) {
102 const GPSOpenResult result{isCancelled() ? GPSOpenStatus::Cancelled : GPSOpenStatus::Error,
103 _serial->errorString()};
104 _serial->close();
105 return result;
106 }
107 return {GPSOpenStatus::Opened};
108}

158qint64 SerialGPSTransport::confirmedWritten(int accepted, qint64 drained)
159{
160 Q_UNUSED(drained)
161 const qint64 previousAccepted = _acceptedTotal;
162 _acceptedTotal += accepted;
163 // Late bytesWritten signals belong to earlier connection-wide submissions.
164 const qint64 confirmed =
165 !fatalError() ? (std::max) (_writtenTotal, _acceptedTotal - _serial->bytesToWrite()) : _writtenTotal;
166 return confirmed - previousAccepted;

169bool SerialGPSTransport::setBaudrate(unsigned baudrate)
170{
171 if (isCancelled() || fatalError()) {
172 return false;
173 }
174 if (_serial->setBaudRate(static_cast<qint32>(baudrate))) {
175 return true;
176 }

179 qCDebug(SerialGPSTransportLog) << "Link refused" << baudrate << "baud:" << _serial->errorString();
180 _serial->clearError();
