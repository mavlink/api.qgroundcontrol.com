34GPSOpenResult SerialGPSTransport::open()
35{
36 if (isCancelled()) {
37 return {GPSOpenStatus::Cancelled};
38 }
39 _serial = std::make_unique<QSerialPort>();
40 _inputOverflow = false;
41 _serial->setReadBufferSize(kReadBufferBytes);
42 QObject::connect(
43 _serial.get(), &QSerialPort::readyRead, _serial.get(),
44 [this]() {
45 // With no serial flow control, exhausting our budget makes stream continuity unverifiable.
46 // Retire the attempt rather than decode bytes across a possible overrun.
47 _inputOverflow = _inputOverflow || _serial->bytesAvailable() >= kReadBufferBytes;
48 },
49 Qt::DirectConnection);
50#ifndef Q_OS_ANDROID
51 _acceptedTotal = 0;
52 _writtenTotal = 0;
53 QObject::connect(
54 _serial.get(), &QSerialPort::bytesWritten, _serial.get(), [this](qint64 count) { _writtenTotal += count; },
55 Qt::DirectConnection);
56#endif
57 _serial->setPortName(_device);
58 const QDeadlineTimer openDeadline(kOpenTimeoutMs);
59 while (!_serial->open(QIODevice::ReadWrite)) {
60 if (isCancelled()) {
61 return {GPSOpenStatus::Cancelled};
62 }
63 // Device can take 10-20s to become accessible after startup.
64 if (_serial->error() != QSerialPort::PermissionError || openDeadline.hasExpired()) {
65 qCWarning(SerialGPSTransportLog)
66 << "GPS: Failed to open Serial Device" << _device << _serial->errorString();
67 return {openDeadline.hasExpired() ? GPSOpenStatus::TimedOut : GPSOpenStatus::Error, _serial->errorString()};
68 }
69 qCDebug(SerialGPSTransportLog) << "Cannot open device... retrying";
70 const QDeadlineTimer retryDeadline((std::min) (openDeadline.remainingTime(), qint64(kOpenRetryMs)));
71 while (!retryDeadline.hasExpired() && !isCancelled()) {
72 QThread::msleep(
73 static_cast<unsigned long>((std::min) (retryDeadline.remainingTime(), qint64(kCancellationPollMs))));
74 }
75 if (isCancelled()) {
76 return {GPSOpenStatus::Cancelled};
77 }
78 }
79 _serial->clearError();
80
81 const bool configured = _serial->setBaudRate(QSerialPort::Baud9600) && _serial->setDataBits(QSerialPort::Data8) &&
82 _serial->setParity(QSerialPort::NoParity) && _serial->setStopBits(QSerialPort::OneStop) &&
83 _serial->setFlowControl(QSerialPort::NoFlowControl);
84 if (!configured || isCancelled()) {
85 const GPSOpenResult result{isCancelled() ? GPSOpenStatus::Cancelled : GPSOpenStatus::Error,
86 _serial->errorString()};
87 _serial->close();
88 return result;
89 }
90 return {GPSOpenStatus::Opened};

99GPSReadResult SerialGPSTransport::read(uint8_t* buffer, int length, int timeoutMs)
100{
101 if (isCancelled()) {
102 return {GPSReadStatus::Cancelled};
103 }
104 if (!buffer || length < 0) {
105 return {GPSReadStatus::InvalidData};
106 }
107 if (fatalError()) {
108 return {_inputBudgetExhausted() ? GPSReadStatus::Overflow : GPSReadStatus::Error, 0, _errorDetail()};
109 }
110 if (length == 0) {
111 return {GPSReadStatus::Data};
112 }
113 const QDeadlineTimer deadline((std::max) (timeoutMs, 0));
114 while (_serial->bytesAvailable() == 0) {
115 _serial->waitForReadyRead(static_cast<int>((std::min) (deadline.remainingTime(), qint64(kCancellationPollMs))));
116 if (isCancelled()) {
117 return {GPSReadStatus::Cancelled};
118 }
119 if (fatalError()) {
120 return {_inputBudgetExhausted() ? GPSReadStatus::Overflow : GPSReadStatus::Error, 0, _errorDetail()};
121 }
122 if (_serial->bytesAvailable() == 0 && deadline.hasExpired()) {
123 return {GPSReadStatus::TimedOut};
124 }
125 }
126 const qint64 count = _serial->read(reinterpret_cast<char*>(buffer), length);
127 return {count < 0 ? GPSReadStatus::Error : GPSReadStatus::Data, static_cast<int>((std::max) (count, qint64(0))),
128 count < 0 ? _errorDetail() : QString()};

150GPSWriteResult SerialGPSTransport::write(const uint8_t* buffer, int length)
151{
152 if (isCancelled()) {
153 return {GPSWriteStatus::Cancelled};
154 }
155 if (!buffer || length < 0) {
156 return {GPSWriteStatus::InvalidData};
157 }
158 if (fatalError()) {
159 return {GPSWriteStatus::Error, 0, 0, _errorDetail()};
160 }
161 if (length == 0) {
162 return {GPSWriteStatus::Completed};
163 }
164 // The legacy Android backend writes synchronously with its own timeout and cannot be interrupted.
165 const qint64 count = _serial->write(reinterpret_cast<const char*>(buffer), length);
166 const int written = static_cast<int>(std::clamp(count, qint64(0), qint64(length)));
167 GPSWriteStatus status = GPSWriteStatus::Error;
168 if (isCancelled()) {
169 status = GPSWriteStatus::Cancelled;
170 } else if (count == length && !fatalError()) {
171 status = GPSWriteStatus::Completed;
172 }
173 // A failed legacy write can have delivered bytes without reporting their count.
174 const GPSWriteResult result{status, length, written, status == GPSWriteStatus::Error ? _errorDetail() : QString()};
175 if (status != GPSWriteStatus::Completed) {
176 _serial->close();
177 }
178 return result;
179}
180#endif

182GPSWriteResult SerialGPSTransport::writeBounded(const uint8_t* buffer, int length, QDeadlineTimer deadline)
183{
184#ifdef Q_OS_ANDROID
185 if (isCancelled()) {
186 return {GPSWriteStatus::Cancelled};
187 }
188 if (!buffer || length < 0) {
189 return {GPSWriteStatus::InvalidData};
190 }
191 if (fatalError() || _serial->bytesToWrite() != 0) {
192 return {GPSWriteStatus::Error, 0, 0, _errorDetail()};
193 }
194 if (length == 0) {
195 return {GPSWriteStatus::Completed};
196 }
197 Q_UNUSED(deadline);
198 return {GPSWriteStatus::Unsupported, 0, 0, QStringLiteral("Android serial does not support bounded writes")};
199#else
200 const qint64 previousAccepted = _acceptedTotal;
201 return GPSStreamWrite::writeBounded(
202 *this, _serial.get(), buffer, length, deadline, kWriteBufferBytes,
203 [this](QDeadlineTimer remaining) {
204 _serial->waitForBytesWritten(
205 remaining.isForever()
206 ? kCancellationPollMs
207 : static_cast<int>((std::min) (remaining.remainingTime(), qint64(kCancellationPollMs))));
208 },
209 [this, previousAccepted](int accepted) {
210 _acceptedTotal += accepted;
211 // Late bytesWritten signals belong to earlier connection-wide submissions.
212 const qint64 confirmed =
213 !fatalError() ? (std::max) (_writtenTotal, _acceptedTotal - _serial->bytesToWrite()) : _writtenTotal;
214 return confirmed - previousAccepted;
215 },
216 [this]() { return _errorDetail(); }, [this]() { _serial->close(); });
217#endif
218}
