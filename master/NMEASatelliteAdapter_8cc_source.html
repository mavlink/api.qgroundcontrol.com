66void NMEASatelliteAdapter::_queue(NMEA::SatelliteEpoch epoch)
67{
68 GPSSatelliteObservation observation;
69 observation.updateMode = GPSSatelliteObservation::UpdateMode::ConstellationDelta;
70 for (const auto& system : epoch) {
71 GPSSatelliteProvenance provenance;
72 provenance.constellation = system.constellation;
73 provenance.inViewTimestampUs = system.inViewTimestampUs;
74 provenance.inUseTimestampUs = system.inUseTimestampUs;
75 if (system.usedIds) {
76 provenance.usedSatelliteIds = QList<int>(system.usedIds->begin(), system.usedIds->end());
77 provenance.satellitesUsed = static_cast<int>(system.usedIds->size());
78 }
79 for (const auto& value : system.satellites) {
80 GPSSatellite satellite;
81 satellite.id = value.id;
82 satellite.prn = value.prn;
83 satellite.constellation = value.constellation;
84 satellite.elevationDegrees = value.elevation;
85 satellite.normalizedAzimuthDegrees = value.azimuth;
86 satellite.signalStrength = value.signal;
87 observation.satellites.append(satellite);
88 }
89 observation.provenance.append(provenance);
90 observation.monotonicTimestampUs =
91 std::max<quint64>({observation.monotonicTimestampUs, system.inViewTimestampUs, system.inUseTimestampUs});
92 }
93 if (!observation.provenance.isEmpty()) {
94 // Bound queued epochs if a receiver outpaces the application event loop.
95 if (_pending.size() >= MAX_PENDING_EPOCHS) {
96 _pending.removeFirst();
97 }
98 _pending.append(observation);
99 if (!_deliveryTask.active()) {
100 _deliveryTask.schedule(std::chrono::microseconds::zero(), [this]() { _deliver(); });
101 }
102 }
103}

105void NMEASatelliteAdapter::_deliver()
106{
107 if (!_open || _pending.isEmpty()) {
108 return;
109 }

111 // Schedule before notification; callbacks may close or destroy this assembler.

114 }
115 emit observationReceived(observation);
