36GPSPositionReport::FixType fixType(uint8_t modeType)
37{
38 switch (modeType) {

42 case 6:

45 case 8:

57std::optional<QByteArray> promptDescriptor(QByteArrayView text)
58{
59 for (qsizetype end = text.indexOf('>'); end >= 0; end = text.indexOf('>', end + 1)) {
60 if (end < DESCRIPTOR_SIZE) {
61 continue;
62 }
63 const QByteArrayView descriptor = text.sliced(end - DESCRIPTOR_SIZE, DESCRIPTOR_SIZE);
64 const char before = end > DESCRIPTOR_SIZE ? text[end - DESCRIPTOR_SIZE - 1] : ' ';
65 const bool separated = !(NMEA::isAsciiUpper(before) || NMEA::isAsciiDigit(before));
66 const bool tail = (NMEA::isAsciiUpper(descriptor[2]) || NMEA::isAsciiDigit(descriptor[2])) &&
67 NMEA::isAsciiDigit(descriptor[3]);
68 if (separated && NMEA::isAsciiUpper(descriptor[0]) && NMEA::isAsciiUpper(descriptor[1]) && tail) {
69 return descriptor.toByteArray();
70 }
71 }
72 return std::nullopt;
73}

77bool detectPort(GPSCommandChannel& channel, QByteArray& port)
78{
79 const GPSDeadline deadline = GPSDeadline::after(channel.nowUs(), SBF::Plan::PROMPT_TIMEOUT);
80 if (!channel.write(SBF::Plan::PROMPT)) {
81 return false;
82 }
83 QByteArray received;
84 std::array<uint8_t, GPSCommandChannel::READ_CHUNK_SIZE> chunk{};
85 while (channel.nowUs() < deadline.untilUs) {
86 const int count = channel.read(chunk, deadline.remaining(channel.nowUs()));
87 if (count < 0) {
88 return false;
89 }
90 received.append(reinterpret_cast<const char*>(chunk.data()), count);
91 if (const auto descriptor = promptDescriptor(received)) {
92 port = *descriptor;
93 qCDebug(SBFProtocolLog).noquote() << "Septentrio GNSS receiver COM port:" << port;
94 return true;
95 }
96 // Keep enough for a prompt split across reads.
97 if (received.size() > 2 * static_cast<qsizetype>(chunk.size())) {
98 received = received.last(DESCRIPTOR_SIZE + 1);
99 }
100 }
101 return false;
102}

156bool Protocol::configure(GPSCommandChannel& channel, GPSConfig config, unsigned& baud)
157{
158 // A new configuration stops reporting the base, forgets survey progress and accepts any receiver time.
159 _ready = false;
160 _surveyClock.reset();
161 _surveyActive = false;
162 _lastEpoch.reset();
163 channel.stream().setEnabled(GPSFrameKind::RTCM3, false);
164 channel.stream().reset(GPSFrameKind::SBF);
165 // The link keeps the rate selected or detected; otherwise it runs at Plan::BAUD_RATE.
166 baud = config.linkBaud(baud, SBF::Plan::BAUD_RATE);
167 if (!_configureBase(channel, config.base, baud)) {
168 return false;
169 }
170 _startBase(config.base, channel.nowUs());
171 return true;
172}

174bool Protocol::_configureBase(GPSCommandChannel& channel, const GPSBaseStationConfig& base, unsigned baud)
175{
176 (void) channel.setBaudrate(baud);
177 (void) channel.write(SBF::Plan::FORCE_COMMAND_INPUT);
178
179 if (!channel.runSequence(SBF::Plan::silenceCorrectionOutput()).succeeded()) {
180 return false;
181 }
182 QByteArray port;
183 if (!detectPort(channel, port)) {
184 channel.failControl(QStringLiteral("No command prompt from the Septentrio receiver"));
185 return false;
186 }
187 if (!channel.runRequired(SBF::Plan::port(port, baud))) {
188 return false;
189 }
190 channel.stream().setEnabled(GPSFrameKind::RTCM3, true);
191 return channel.runRequired(SBF::Plan::base(port, base)) && !channel.failed();
192}

212GPSReceiveUpdates Protocol::_decodeBlock(const GPSFrame& frame, GPSDecodeContext& context)
213{
214 const auto header = SBF::checkedHeader(frame.bytes);
215 if (!header || header->number() != SBF::BlockId::PVT_GEODETIC || header->length < Wire::SIZE<SBF::PVTGeodetic>) {
216 return {};
217 }
218 const auto pvt = Wire::decode<SBF::PVTGeodetic>(frame.bytes);
219 if (pvt.tow >= GPSTime::WEEK_MS || pvt.wnc == UINT16_MAX) {
220 return {};
221 }
222 const uint64_t epoch = GPSTime::epochMs(pvt.wnc, pvt.tow);
223 if (_lastEpoch && epoch <= *_lastEpoch) {
224 return GPSReceiveUpdate::Activity;
225 }
226 _lastEpoch = epoch;
227
228 GPSDecodedPosition position;
229 // A base reports in the WGS84/ITRS datum (0) it was configured for. Other receivers report in theirs, such as a
230 // rover in its correction provider's datum (19), and are decoded as reported.
231 if (_ready && pvt.datum != 0) {
232 position.navigation.fixType = GPSPositionReport::FixType::NoFix;
233 position.navigation.timestampUs = context.nowUs();
234 qCWarning(SBFProtocolLog) << "Unsupported Septentrio position datum:" << unsigned(pvt.datum);
235 context.failControl(QStringLiteral("Septentrio position datum is not WGS84/ITRS"));
236 _ready = false;
237 context.stream().setEnabled(GPSFrameKind::RTCM3, false);
238 context.publishSurvey(false, false, {});
239 } else {
240 const bool coordinatesValid = _applyPVTGeodetic(pvt, position, context);
241 if (_ready) {
242 _publishSurveyStatus(pvt, position, coordinatesValid, context);
243 }
244 }
245 context.publishPosition(position);
246 return GPSReceiveUpdate::Activity;
247}

249bool Protocol::_applyPVTGeodetic(const SBF::PVTGeodetic& pvt, GPSDecodedPosition& position,
250 GPSDecodeContext& context) const
251{
252 position.navigation.fixType = fixType(pvt.modeType());
253 if (pvt.error != 0) {
254 position.navigation.fixType = GPSPositionReport::FixType::NoFix;
255 } else if (pvt.mode2D() && position.navigation.fixType >= GPSPositionReport::FixType::Fix3D) {
256 position.navigation.fixType = GPSPositionReport::FixType::Fix2D;
257 }
258
259 // Any value beyond the specified maximum is invalid, not only the do-not-use value (-2*10^10).
260 position.velocityValid = position.navigation.fixType > GPSPositionReport::FixType::NoFix && pvt.error == 0 &&
261 !(fabsf(pvt.vn) > 600.0f || fabsf(pvt.ve) > 600.0f || fabsf(pvt.vu) > 600.0f);
262
263 const bool coordinatesValid = std::isfinite(pvt.latitude) && std::abs(pvt.latitude) <= std::numbers::pi / 2 &&
264 std::isfinite(pvt.longitude) && std::abs(pvt.longitude) <= std::numbers::pi &&
265 std::isfinite(pvt.height) && std::abs(pvt.height) <= DNU;
266 if (!coordinatesValid || !std::isfinite(pvt.undulation) || std::abs(pvt.undulation) > DNU) {
267 position.navigation.fixType = GPSPositionReport::FixType::NoFix;
268 }
269
270 const auto satellitesUsed = gpsSatellitesUsed(pvt.nrSV); // 255 = do not use value
271 position.navigation.satellitesUsed = satellitesUsed;
272 context.publishSatelliteUsage(satellitesUsed);
273
274 position.navigation.latitudeDegrees = pvt.latitude * GPSProtocolMath::RAD_TO_DEG;
275 position.navigation.longitudeDegrees = pvt.longitude * GPSProtocolMath::RAD_TO_DEG;
276 position.navigation.altitudeEllipsoidMeters = pvt.height;
277 position.navigation.altitudeMslMeters = pvt.height - static_cast<double>(pvt.undulation);
278
279 // Accuracy is reported as 2DRMS in cm; halve it for the RMS convention used by the other drivers.
280 position.navigation.horizontalAccuracyMeters =
281 pvt.hAccuracy != UINT16_MAX ? static_cast<float>(pvt.hAccuracy) / 200.0f : NAN;
282 position.navigation.verticalAccuracyMeters =
283 pvt.vAccuracy != UINT16_MAX ? static_cast<float>(pvt.vAccuracy) / 200.0f : NAN;
284
285 position.navigation.speedMetersPerSecond = sqrtf(pvt.vn * pvt.vn + pvt.ve * pvt.ve);
286 position.navigation.courseRadians = std::isfinite(pvt.cog) && pvt.cog >= 0.0f && pvt.cog <= 360.0f
287 ? static_cast<float>(pvt.cog * GPSProtocolMath::DEG_TO_RAD)
288 : NAN;
289
290 // WNc/TOW is GNSS system time, not UTC. Without receiver UTC/leap information,
291 // retain the epoch key internally and let the facade use reception UTC.
292 position.navigation.utcTimeUs = 0;
293 position.navigation.timestampUs = context.nowUs();
294 return coordinatesValid;
295}

297void Protocol::_publishSurveyStatus(const SBF::PVTGeodetic& pvt, const GPSDecodedPosition& position,
298 bool coordinatesValid, GPSDecodeContext& context)
299{
300 // Mode bit 6 means automatic base determination is still in progress, not completed.
301 // Septentrio PolaRx5TR 5.5.0 Reference Guide, SBF Mode definition (p. 382).
302 const bool active = !std::holds_alternative<GPSBaseStationConfig::Fixed>(_base.mode) && pvt.modeAutoSet();
303 const bool valid = !pvt.modeAutoSet() && pvt.modeType() == 3 && !pvt.mode2D() && !pvt.error && coordinatesValid;
304 // The final update on the active-to-inactive transition freezes the survey duration.
305 if (active || _surveyActive) {
306 (void) _surveyClock.update(context.nowUs());
307 }
308 _surveyActive = active;
309 // PVT accuracy describes the navigation solution, not the averaged base survey.
310 GPSEllipsoidPosition surveyPosition;
311 if (coordinatesValid && !pvt.error) {
312 surveyPosition = {.latitudeDegrees = position.navigation.latitudeDegrees,
313 .longitudeDegrees = position.navigation.longitudeDegrees,
314 .altitudeMeters = position.navigation.altitudeEllipsoidMeters};
315 }
316 context.publishSurvey(active, valid, _surveyClock.duration(), surveyPosition);
317}

319QLatin1StringView signature(const GPSFrame& frame)
320{
321 const std::string_view text = frame.text();
322 if (frame.kind == GPSFrameKind::ASCIILine) {
323 return text.starts_with("$R:") || text.starts_with("$R?") ? QLatin1StringView("Septentrio command replies")
324 : QLatin1StringView();
325 }
326 // A candidate the framer truncated at its capture limit cannot be checked, so only short blocks count.
327 if (frame.kind != GPSFrameKind::SBF) {
328 return {};
329 }
330 const auto header = SBF::checkedHeader(frame.bytes);
331 return header && header->length % 4 == 0 ? QLatin1StringView("SBF blocks") : QLatin1StringView();
332}

335bool Protocol::probe(GPSCommandChannel& channel)
336{
337 const auto scope = channel.deadlineScope(SBF::Plan::PROMPT_TIMEOUT);
338 const QByteArrayView prompt = SBF::Plan::PROMPT;
339 channel.beginCommand({prompt.toByteArray(), SBF::Plan::PROMPT_TIMEOUT});
340 QByteArray port;
341 const bool answered = detectPort(channel, port);
342 channel.finishCommand(answered ? GPSCommandOutcome::Acknowledged
343 : channel.failed() ? channel.failureOutcome()
344 : GPSCommandOutcome::TimedOut);
345 return answered;
346}
