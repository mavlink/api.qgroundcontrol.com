63GPSCommandSequence messageRates(std::span<const MessageRate> rates)
64{
65 GPSCommandSequence sequence;
66 for (const auto& rate : rates) {
67 const QByteArray message = QByteArrayView(rate.message).toByteArray();
68 QByteArray suffix;
69 if (!rate.version.empty()) {
70 suffix = ',' + QByteArrayView(rate.version).toByteArray();
71 }
72 const QByteArray write = "PQTMCFGMSGRATE,W," + message + ',' + QByteArray::number(rate.rate) + suffix;
73 const QByteArray read = "PQTMCFGMSGRATE,R," + message + suffix;
74 sequence.steps.emplace_back(command(write, acknowledgement(write)));
75 sequence.steps.emplace_back(command(read, rateReadback(rate)));
