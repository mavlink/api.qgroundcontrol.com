95void GPSCorrectionLedger::pruneDestinationHistory()
96{
97 qsizetype historyCount = 0;
98 for (auto it = _destinations.cbegin(); it != _destinations.cend(); ++it) {
99 if (!_outputs.contains(it.key()) && it->pendingFrames == 0) {
100 ++historyCount;
101 }
102 }
103 while (historyCount > MAX_DESTINATION_HISTORY) {
104 auto oldest = _destinations.end();
105 for (auto it = _destinations.begin(); it != _destinations.end(); ++it) {
106 if (!_outputs.contains(it.key()) && it->pendingFrames == 0 &&
107 (oldest == _destinations.end() || it->lastActivityMs < oldest->lastActivityMs)) {
108 oldest = it;
109 }
110 }
111 if (oldest == _destinations.end()) {
112 return;
113 }
114 _destinations.erase(oldest);
115 --historyCount;
116 }

151bool GPSCorrectionLedger::recordDelivery(const GPSCorrectionDelivery& delivery)
152{
153 const QString key = QString::number(delivery.deliveryId) + QLatin1Char('/') + delivery.destinationId;
154 const auto it = _pendingDeliveries.find(key);
155 if (it == _pendingDeliveries.end()) {
156 return false;
157 }
158 const PendingDelivery pending = it.value();
159 if (pending.destinationSession != delivery.destinationSession || pending.frame.source != delivery.source ||
160 pending.frame.sourceInstance != delivery.sourceInstance || pending.frame.session != delivery.sourceSession ||
161 pending.queuedBytes != delivery.requestedBytes || delivery.writtenBytes > pending.queuedBytes ||
162 delivery.uncertainBytes > pending.queuedBytes - delivery.writtenBytes ||
163 delivery.acceptedBytes > pending.queuedBytes || delivery.writtenBytes > delivery.acceptedBytes ||
164 delivery.uncertainBytes > delivery.acceptedBytes - delivery.writtenBytes) {
165 recordEvent(pending.frame, GPSCorrectionStage::Dropped, GPSCorrectionReason::InvalidDelivery, 0,
166 pending.destination, pending.destinationSession);
167 return false;
168 }
169 _pendingDeliveries.erase(it);
170 auto& destination = _destinations[pending.destination];
171 --destination.pendingFrames;
172 destination.pendingBytes -= pending.queuedBytes;
173 destination.transportAcceptedBytes += delivery.acceptedBytes;
174 if (auto* stats = _currentStatistics(pending.frame)) {
175 stats->transportAcceptedBytes += delivery.acceptedBytes;
176 }
177 const bool complete = pending.complete && delivery.outcome == GPSCorrectionOutcome::Written &&
178 delivery.writtenBytes == pending.queuedBytes;
179 if (delivery.writtenBytes > 0) {
180 destination.writtenBytes += delivery.writtenBytes;
181 destination.writtenFrames += complete ? 1 : 0;
182 if (auto* stats = _currentStatistics(pending.frame)) {
183 stats->writtenBytes += delivery.writtenBytes;
184 stats->writtenFrames += complete ? 1 : 0;
185 }
186 recordEvent(pending.frame, GPSCorrectionStage::Written, GPSCorrectionReason::None, delivery.writtenBytes,
187 pending.destination, pending.destinationSession);
188 }
189 if (delivery.uncertainBytes > 0) {
190 ++destination.unconfirmedFrames;
191 destination.unconfirmedBytes += delivery.uncertainBytes;
192 if (auto* stats = _currentStatistics(pending.frame)) {
193 ++stats->unconfirmedFrames;
194 stats->unconfirmedBytes += delivery.uncertainBytes;
195 }
196 recordEvent(pending.frame, GPSCorrectionStage::Unconfirmed, GPSCorrectionReason::DeliveryUnconfirmed,
197 delivery.uncertainBytes, pending.destination, pending.destinationSession);
198 }
199 const quint64 undelivered = pending.queuedBytes - delivery.writtenBytes - delivery.uncertainBytes;
200 if (undelivered > 0) {
201 const auto reason = delivery.outcome == GPSCorrectionOutcome::Written ? GPSCorrectionReason::PartialWrite
202 : gpsCorrectionReason(delivery.outcome);
203 recordDrop(pending.frame, reason, undelivered, pending.destination, pending.destinationSession);
204 }
205 pruneDestinationHistory();
206 return true;
207}

211{

260 quint64 session, quint64 bytes, bool complete, bool reportsWrites)
261{
262 const QString key = QString::number(frame.deliveryId) + QLatin1Char('/') + id;
263 if (reportsWrites && bytes && _pendingDeliveries.contains(key)) {
264 recordEvent(frame, GPSCorrectionStage::Dropped, GPSCorrectionReason::InvalidDelivery, 0, id, session);
265 return false;
266 }
267 auto& destination = _destinations[id];
268 destination.id = id;

270 destination.lastActivityMs = _clock();
271 destination.reportsWrites = reportsWrites;
272 if (!bytes) {

274 }
275 if (auto* stats = _currentStatistics(frame)) {
276 stats->submittedBytes += bytes;
277 }
278 destination.queuedFrames += complete ? 1 : 0;
279 destination.queuedBytes += bytes;
280 recordEvent(frame, GPSCorrectionStage::Queued, GPSCorrectionReason::None, bytes, id, session);
281 if (!reportsWrites) {
282 return true;
283 }

286 destination.unconfirmedBytes += bytes;

288 ++stats->unconfirmedFrames;
289 stats->unconfirmedBytes += bytes;
290 }

292 session);
293 } else {
294 ++destination.pendingFrames;
295 destination.pendingBytes += bytes;
296 _pendingDeliveries.insert(key, {frame, id, session, bytes, outputId, complete});
