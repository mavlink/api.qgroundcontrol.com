105void GPSRtk::connectGPS(const QString& device, QStringView gps_type)
106{
107 auto reservation = SerialPortManager::instance()->reservePort(device);
108 if (!reservation) {
109 qCDebug(GPSRtkLog) << "Serial port is already reserved:" << device;
110 return;
111 }
112 GPSType type = GPSType::ublox;
113 for (const GPSReceiverTypeEntry& entry : kGPSReceiverTypeTable) {
114 if (gps_type.contains(entry.key, Qt::CaseInsensitive)) {
115 type = entry.type;
116 break;
117 }
118 }
119 connectReceiver(
120 type,
121 [device, reservation](const std::atomic_bool& requestStop) {
122 return std::make_unique<SerialGPSTransport>(device, requestStop);
123 },
124 QStringLiteral("serial:%1").arg(device));

141void GPSRtk::connectReceiver(GPSType type, GPSProvider::TransportFactory transportFactory,
142 const QString& sourceInstance)
143{
144 RTKSettings* const rtkSettings = SettingsManager::instance()->rtkSettings();
145 for (const GPSReceiverTypeEntry& entry : kGPSReceiverTypeTable) {
146 if (entry.type == type) {
147 rtkSettings->baseReceiverManufacturers()->setRawValue(entry.manufacturerId);
148 break;
149 }
150 }
151
152 disconnectGPS();
153
154 _gpsRtkFactGroup->lastError()->setRawValue(static_cast<int>(GPSConnectionError::None));
155 const bool useFixedBase =
156 static_cast<BaseModeDefinition::Mode>(rtkSettings->useFixedBasePosition()->rawValue().toInt()) ==
157 BaseModeDefinition::Mode::BaseFixed;
158 GPSBaseStationConfig rtkConfig;
159 if (useFixedBase) {
160 rtkConfig = GPSBaseStationConfig{
161 .useFixedBase = true,
162 .fixedBaseLatitude = rtkSettings->fixedBasePositionLatitude()->rawValue().toDouble(),
163 .fixedBaseLongitude = rtkSettings->fixedBasePositionLongitude()->rawValue().toDouble(),
164 .fixedBaseAltitudeMeters = rtkSettings->fixedBasePositionAltitude()->rawValue().toFloat(),
165 .fixedBaseAccuracyMeters = rtkSettings->fixedBasePositionAccuracy()->rawValue().toFloat(),
166 };
167 } else {
168 rtkConfig = GPSBaseStationConfig{
169 .surveyInAccMeters = rtkSettings->surveyInAccuracyLimit()->rawValue().toDouble(),
170 .surveyInDurationSecs = rtkSettings->surveyInMinObservationDuration()->rawValue().toLongLong(),
171 };
172 }
173 _gpsProvider = new GPSProvider(std::move(transportFactory), type, GPSReceiverConfig{.base = rtkConfig}, this);
174 const QPointer<GPSProvider> provider = _gpsProvider;
175 const QPointer<GPSCorrectionManager> correctionManager = _correctionManager;
176 if (correctionManager) {
177 auto registration = correctionManager->registerSource(GPSCorrectionSource::LocalReceiver, sourceInstance);
178 if (!provider || _gpsProvider != provider) {
179 return;
180 }
181 _correctionRegistration = std::move(registration);
182 }
183 const auto token = _correctionRegistration.token();
184 const auto current = [this, provider, token, registered = !correctionManager.isNull()]() {
185 return provider && _gpsProvider == provider && (!registered || token.valid());
186 };
187 const auto connectCurrent = [this, provider, current]<typename... Args>(void (GPSProvider::*signal)(Args...),
188 auto handler) {
189 return connect(
190 provider, signal, this,
191 [current, handler](Args... args) {
192 if (current()) {
193 handler(args...);
194 }
195 },
196 Qt::QueuedConnection);
197 };
198 // Queued callbacks retain the producing session's token.
199 (void) connect(
200 provider, &GPSProvider::RTCMDataUpdate, this,
201 [correctionManager, token](const QByteArray& data, qint64 receivedAtMs) {
202 if (!correctionManager) {
203 qCWarning(GPSRtkLog) << "Correction manager not ready; dropping" << data.size() << "bytes";
204 return;
205 }
206 const bool valid = RTCM::isValidFrame(data);
207 const int messageId =
208 data.size() >= 5 ? (static_cast<quint8>(data[3]) << 4) | (static_cast<quint8>(data[4]) >> 4) : 0;
209 correctionManager->acceptIngress(
210 token.event(data, receivedAtMs, messageId, valid, false,
211 valid ? GPSCorrectionReason::None : GPSCorrectionReason::InvalidFrame));
212 },
213 Qt::QueuedConnection);
214 (void) connectCurrent(&GPSProvider::satelliteInfoUpdate, std::bind_front(&GPSRtk::_satelliteInfoUpdate, this));
215 (void) connectCurrent(&GPSProvider::sensorGpsUpdate, std::bind_front(&GPSRtk::_sensorGpsUpdate, this));
216 (void) connectCurrent(&GPSProvider::surveyInStatus, std::bind_front(&GPSRtk::_onGPSSurveyInStatus, this));
217 (void) connectCurrent(&GPSProvider::connectionError, [this](GPSConnectionError error) {
218 _onGPSDisconnect();
219 _onGPSConnectionError(error);
220 });
221 (void) connectCurrent(&GPSProvider::receiverReady, std::bind_front(&GPSRtk::_onGPSConnect, this));
222 (void) connect(
223 provider, &QThread::finished, this,
224 [this, provider]() {
225 if (provider && _gpsProvider == provider) {
226 _gpsProvider = nullptr;
227 _onGPSDisconnect();
228 }
229 },
230 Qt::QueuedConnection);
231 (void) connect(provider, &QThread::finished, provider, &QObject::deleteLater);
232 provider->start();

235void GPSRtk::disconnectGPS()
236{
237 // Invalidate the session before waiting: queued output may still be in the GUI event queue.
238 auto* provider = std::exchange(_gpsProvider, nullptr);
239 _onGPSDisconnect();
240 if (provider) {
241 provider->stop();
242 if (!provider->wait(_disconnectTimeoutMs)) {
243 qCWarning(GPSRtkLog) << "GPS thread did not exit in time; deferring cleanup to finished()";
244 // The worker owns its stop flag and must survive destruction of this manager.
245 provider->setParent(nullptr);
246 }
247 }

260GPSRtk::SatelliteCounts GPSRtk::countSatellites(const GPSSatelliteReport& msg)
261{
262 SatelliteCounts counts;
263 counts.inView = (std::min) (msg.count, GPSSatelliteReport::MAX_SATELLITES);
264 for (uint16_t i = 0; i < counts.inView; ++i) {
265 if (msg.satellites[i].used.value_or(false)) {
266 ++counts.used;
267 }
268 }
269 return counts;
270}
