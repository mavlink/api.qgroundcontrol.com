98void GPSRtk::_onGPSConnectionError(GPSConnectionError error)
99{
100 switch (error) {
101 case GPSConnectionError::OpenFailed:
102 qCWarning(GPSRtkLog) << "Failed to open GPS receiver transport";
103 _setError(error, tr("Failed to open the receiver. Check the device, permissions, and other connections."));
104 break;
105 case GPSConnectionError::ConfigFailed:
106 qCWarning(GPSRtkLog) << "GPS receiver did not accept configuration";
107 if (_gpsRtkFactGroup->lastError()->rawValue().toInt() != static_cast<int>(error) ||
108 _errorMessage.isEmpty()) {
109 _setError(error,
110 tr("Receiver configuration failed. Check the receiver type, baud rate, and base mode."));
111 }
112 break;
113 case GPSConnectionError::DeviceError:
114 qCWarning(GPSRtkLog) << "GPS device error, connection lost";
115 _setError(error, tr("Receiver connection lost. Check the device and reconnect."));
116 break;
117 case GPSConnectionError::None:
118 _setError(error);
119 break;
120 }

487void GPSRtk::_retireSession(quint64 generation)
488{
489 auto retired = std::move(_session);
490 _session = {};
491 _session.generation = generation;
492 const auto provider = retired.provider;
493 const auto timeoutMs = _disconnectTimeoutMs;
494 if (provider) {
495 // Retirement callbacks may delete this owner; the worker must already be independent.
496 provider->setParent(nullptr);
497 provider->stop();
498 }
499 retired.corrections.reset();
500 if (provider) {
501 if (!retired.started) {
502 delete provider.data();
503 } else if (!provider->wait(timeoutMs)) {
504 qCWarning(GPSRtkLog) << "GPS thread did not exit in time; deferring cleanup to finished()";
505 } else {
506 delete provider.data();
507 }
508 }

534GPSRtk::SatelliteCounts GPSRtk::countSatellites(const GPSSatelliteReport& msg)
535{
536 SatelliteCounts counts;
537 if (msg.timestampUs == 0) {
538 return counts;
539 }
540 counts.inView = (std::min) (msg.count, GPSSatelliteReport::MAX_SATELLITES);
541 if (msg.count > GPSSatelliteReport::MAX_SATELLITES) {
542 return counts;
543 }
544 counts.used = 0;
545 for (int i = 0; i < counts.inView; ++i) {
546 if (!msg.satellites[i].used) {
547 counts.used.reset();
548 break;
549 }
550 if (*msg.satellites[i].used) {
551 ++*counts.used;
552 }
553 }
554 return counts;
555}
