130// Best-effort GPU-side wait on the producer's fences: export a sync_file from each dma-buf, import it as an
131// EGL_SYNC_NATIVE_FENCE_ANDROID sync and eglWaitSyncKHR in Qt's (current) context. True only when every plane's
132// dma-buf got a confirmed server-side wait, which makes the CPU barrier redundant.
133bool tryExplicitFenceWait(EGLDisplay eglDpy, GstBuffer* buffer)
134{
135#if defined(QGC_HAS_DMABUF_EXPORT_SYNC)
136 if (!GstEglHelpers::displaySupportsExtension(eglDpy, kNativeFenceExt)) {
137 return false;
138 }
139 static const auto createSyncKHR = reinterpret_cast<PFNEGLCREATESYNCKHRPROC>(eglGetProcAddress("eglCreateSyncKHR"));
140 static const auto waitSyncKHR = reinterpret_cast<PFNEGLWAITSYNCKHRPROC>(eglGetProcAddress("eglWaitSyncKHR"));
141 static const auto destroySyncKHR =
142 reinterpret_cast<PFNEGLDESTROYSYNCKHRPROC>(eglGetProcAddress("eglDestroySyncKHR"));
143 if (!createSyncKHR || !waitSyncKHR || !destroySyncKHR) {
144 return false;
145 }
146
147 int lastFd = -1;
148 for (guint i = 0; i < gst_buffer_n_memory(buffer); ++i) {
149 const int dmaFd = gst_dmabuf_memory_get_fd(gst_buffer_peek_memory(buffer, i));
150 if (dmaFd < 0 || dmaFd == lastFd) {
151 continue;
152 }
153 lastFd = dmaFd;
154 dma_buf_export_sync_file req = {};
155 req.flags = DMA_BUF_SYNC_READ; // wait only on writers (the decoder), not on prior readers
156 req.fd = -1;
157 if (ioctl(dmaFd, DMA_BUF_IOCTL_EXPORT_SYNC_FILE, &req) != 0 || req.fd < 0) {
158 return false;
159 }
160 // EGL owns req.fd once the sync is created (closed by eglDestroySync); close it ourselves only on failure.
161 const EGLint attribs[] = {EGL_SYNC_NATIVE_FENCE_FD_ANDROID, req.fd, EGL_NONE};
162 EGLSyncKHR sync = createSyncKHR(eglDpy, EGL_SYNC_NATIVE_FENCE_ANDROID, attribs);
163 if (sync == EGL_NO_SYNC_KHR) {
164 ::close(req.fd);
165 return false;
166 }
167 const EGLint waited = waitSyncKHR(eglDpy, sync, 0);
168 destroySyncKHR(eglDpy, sync);
169 if (waited != EGL_TRUE) {
170 return false;
171 }
172 }
173 if (!s_loggedExplicitFence.exchange(true, std::memory_order_relaxed)) {
174 qCDebug(GstDmaBufLog) << "waiting on the producer's dma-buf fences GPU-side (CPU barrier skipped)";
175 }
176 GstHwPathTelemetry::record(HwVideoBufferPath::DmaBuf, GstHwPathTelemetry::Counter::GpuWaits);
177 return true;
178#else
179 Q_UNUSED(eglDpy);
180 Q_UNUSED(buffer);
181 return false;
182#endif
183}

185// Bounded so a GPU hang cannot wedge Qt's render thread.
186constexpr int kProducerWaitTimeoutMs = 100;
187
188// CPU-side barrier on the producer's implicit write fences: poll(POLLIN) on a dma-buf fd returns once its write fences
189// signal (the wait linux/dma-buf.h documents). Unlike mmap/DMA_BUF_IOCTL_SYNC it triggers no exporter cache maintenance
190// or buffer migration. False when a wait timed out or failed.
191bool waitForProducerWrites(GstBuffer* buffer)
192{
193 int lastFd = -1;
194 for (guint i = 0; i < gst_buffer_n_memory(buffer); ++i) {
195 GstMemory* mem = gst_buffer_peek_memory(buffer, i);
196 const int fd = (mem && gst_is_dmabuf_memory(mem)) ? gst_dmabuf_memory_get_fd(mem) : -1;
197 if (fd < 0 || fd == lastFd) {
198 continue;
199 }
200 lastFd = fd;
201 pollfd pfd = {fd, POLLIN, 0};
202 int ret = 0;
203 do {
204 ret = ::poll(&pfd, 1, kProducerWaitTimeoutMs);
205 } while (ret < 0 && (errno == EINTR || errno == EAGAIN));
206 if (ret <= 0) {
207 const int pollError = (ret < 0) ? errno : 0;
208 GstHwPathTelemetry::record(HwVideoBufferPath::DmaBuf, GstHwPathTelemetry::Counter::ProducerFenceTimeouts);
209 QGC_HW_WARN_ONCE(GstDmaBufLog, s_loggedProducerWaitTimeout,
210 "dma-buf producer fence wait" << (ret == 0 ? "timed out" : "failed") << "errno="
211 << pollError << "— no producer barrier for this frame");
212 return false;
213 }
214 }
215 GstHwPathTelemetry::record(HwVideoBufferPath::DmaBuf, GstHwPathTelemetry::Counter::CpuWaits);
216 return true;
217}

231bool locatePlanes(GstBuffer* buffer, const GstVideoInfo& info, const GstQtDmaBufImport::PlaneFourccs& fourccs,
232 quint64 modifier, std::span<GstQtDmaBufImport::Plane> planes)
233{
234 const GstVideoMeta* vmeta = gst_buffer_get_video_meta(buffer);
235 for (int i = 0; i < fourccs.count; ++i) {
236 const auto [offset, stride] = GstHw::planeLayout(info, vmeta, i);
237 guint memIdx = 0;
238 guint length = 0;
239 gsize skip = 0;
240 if (!gst_buffer_find_memory(buffer, offset, 1, &memIdx, &length, &skip) || (length != 1)) {
241 return false;
242 }
243 GstMemory* mem = gst_buffer_peek_memory(buffer, memIdx);
244 const quint64 fdOffset = static_cast<quint64>(mem->offset) + skip;
245 constexpr quint64 kMaxAttrib = (std::numeric_limits<EGLint>::max)();
246 if (!gst_is_dmabuf_memory(mem) || (stride <= 0) || (fdOffset > kMaxAttrib)) {
247 return false;
248 }
249 planes[std::size_t(i)] = {.fd = gst_dmabuf_memory_get_fd(mem),
250 .offset = static_cast<quint32>(fdOffset),
251 .pitch = static_cast<quint32>(stride),
252 .drmFourcc = fourccs.fourcc[std::size_t(i)],
253 .modifier = modifier};
254 if (planes[std::size_t(i)].fd < 0) {
255 return false;
256 }
257 }
258 return true;
259}

420QVideoFrameTexturesUPtr GstDmaBufVideoBuffer::importGl(QRhi& rhi, GstBuffer* buffer)
421{
422 const EGLDisplay eglDpy = GstQtDmaBufImport::importDisplay(rhi);
423 if (eglDpy == EGL_NO_DISPLAY) {
424 QGC_HW_WARN_ONCE(GstDmaBufLog, s_loggedNoEglContext,
425 "Qt's GL context is not EGL or can't bind EGLImages — DMABuf frames take the CPU path");
426 GstHwPathTelemetry::recordFallbackReason(HwVideoBufferPath::DmaBuf,
427 GstHwPathTelemetry::HwFallbackReason::NoExt);
428 return GstHwPathTelemetry::fail(HwVideoBufferPath::DmaBuf);
429 }
430
431 const QVideoFrameFormat::PixelFormat pixelFormat = _format.pixelFormat();
432 GstQtDmaBufImport::PlaneFourccs fourccs = GstQtDmaBufImport::planeFourccs(pixelFormat);
433 if (_packedYuvFourcc != 0) {
434 fourccs = {.fourcc = {_packedYuvFourcc}, .count = 1};
435 }
436 const quint64 modifier =
437 ((_drmModifier != DRM_FORMAT_MOD_INVALID) && GstEglHelpers::displaySupportsExtension(eglDpy, kModifiersExt))
438 ? _drmModifier
439 : GstQtDmaBufImport::kImplicitModifier;
440 std::array<GstQtDmaBufImport::Plane, kMaxPlanes> planes{};
441 if ((fourccs.count == 0) || (fourccs.count != int(GST_VIDEO_INFO_N_PLANES(&_videoInfo))) ||
442 !locatePlanes(buffer, _videoInfo, fourccs, modifier, planes)) {
443 QGC_HW_WARN_ONCE(GstDmaBufLog, s_loggedLayoutRejected,
444 "DMABuf layout not importable (format"
445 << gst_video_format_to_string(GST_VIDEO_INFO_FORMAT(&_videoInfo)) << "memories"
446 << gst_buffer_n_memory(buffer) << ") — CPU fallback");
447 GstHwPathTelemetry::recordFallbackReason(HwVideoBufferPath::DmaBuf,
448 GstHwPathTelemetry::HwFallbackReason::ImportUnsupported);
449 return GstHwPathTelemetry::fail(HwVideoBufferPath::DmaBuf);
450 }
451
452 if (!GstHwImportPreflight::preflightOrRecord(&rhi, HwVideoBufferPath::DmaBuf, pixelFormat, _format.frameSize())) {
453 return GstHwPathTelemetry::fail(HwVideoBufferPath::DmaBuf);
454 }
455
456 waitForProducer(eglDpy, buffer);
457
458 // The textures sample the dma-bufs directly: keep the decoder from recycling them until Qt drops the textures.
459 std::shared_ptr<void> keepAlive(gst_buffer_ref(buffer),
460 [](void* ref) { gst_buffer_unref(static_cast<GstBuffer*>(ref)); });
461 GstQtDmaBufImport::ImportResult imported =
462 GstQtDmaBufImport::importTextures(rhi, std::span(planes.data(), std::size_t(fourccs.count)), pixelFormat,
463 _format.frameSize(), std::move(keepAlive));
464 if (!imported.textures) {
465 recordImportFailure(imported.unrecoverable);
466 return GstHwPathTelemetry::fail(HwVideoBufferPath::DmaBuf);
467 }
468 s_consecutiveImportFailures.store(0, std::memory_order_relaxed);
469 if (!s_diag.loggedFirstSuccess.exchange(true, std::memory_order_relaxed)) {
470 qCDebug(GstDmaBufLog) << "First DMABuf import: size=" << _format.frameSize() << "format=" << pixelFormat
471 << "modifier=" << Qt::hex << modifier;
472 }
473 return std::move(imported.textures);
474}
