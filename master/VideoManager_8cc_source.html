178void VideoManager::_updateSourceSummary()
179{
180 int tileCount = 0;
181 VideoStreamController* measured = nullptr;
182 VideoStreamController* measuredInTile = nullptr;
183 bool degraded = false;
184 for (VideoStreamController* source : std::as_const(_sources)) {
185 const VideoStreamStats stats = source->stats();
186 if (source->tile()) {
187 ++tileCount;
188 }
189 if (stats.valid) {
190 VideoStreamController*& first = source->tile() ? measuredInTile : measured;
191 if (!first) {
192 first = source;
193 }
194 }
195 // A stream that isn't decoded, as the thermal stream while its view is off, shows no frames to compare with
196 // the frame rate its camera advertises.
197 const QGCVideoStreamInfo* const streamInfo = source->decoding() ? source->streamInfo() : nullptr;
198 const double cameraFps = (source == _cameraSource) ? _uvcCamera->monitor()->cameraFrameRate()
199 : (streamInfo ? streamInfo->framerate() : 0.0);
200 degraded = degraded || stats.degraded(cameraFps);
201 }
202 if (!measured) {
203 measured = measuredInTile ? measuredInTile : _primaryStream;
204 }
205
206 if (tileCount != _tileCount) {
207 _tileCount = tileCount;
208 emit tileCountChanged();
209 }
210 if (measured != _measuredSource) {
211 _measuredSource = measured;
212 emit measuredSourceChanged();
213 }
214 if (degraded != _degraded) {
215 _degraded = degraded;
216 emit degradedChanged();
217 }

354
355bool VideoManager::waitForVideoBackendReady()
356{
357 if (_initState == InitState::NotStarted) {
358 startVideoBackendInit();
359 }
360
361 if (_initState == InitState::Pending) {
362 _backendInitFuture.waitForFinished();
363 const bool success =
364 !_backendInitFuture.isCanceled() && (_backendInitFuture.resultCount() > 0) && _backendInitFuture.result();
365 _onBackendInitComplete(success);
366 }
367
368 return _initState == InitState::Ready;
369}

371void VideoManager::_onBackendInitComplete(bool success)
372{
373 if (_initState != InitState::Pending) {
374 return;
375 }
376
377 if (!success) {
378 _initState = InitState::Failed;
379 qCCritical(VideoManagerLog) << "video initialization failed";

503void VideoManager::grabImage(const QString& imageFile)
504{
505 // One reading of the clock names the file and dates its EXIF data.
506 const QDateTime now = QDateTime::currentDateTime();
507 QString file = imageFile;
508 if (file.isEmpty()) {
509 const QString directory = SettingsManager::instance()->appSettings()->photoSavePath();
510 if (directory.isEmpty()) {
511 QGC::showAppMessage(tr("Unable to save the photo. The save path must be specified in Settings."));
512 return;
513 }
514 file = QStringLiteral("%1/%2.jpg").arg(directory, now.toString(QStringLiteral("yyyy-MM-dd_hh.mm.ss.zzz")));
515 }
516
517 VideoScreenshot::Exif exif;
518 exif.taken = now;
519 if (_activeVehicle) {
520 exif.position = _activeVehicle->coordinate();
521 }
522
523 bool taken = false;
524 for (VideoStreamController* source : std::as_const(_sources)) {
525 // The main picture and the thermal stream with it: not the tiles.
526 if (!source->tile() && source->canTakeScreenshot()) {
527 taken = source->takeScreenshot(source->taggedFile(file), exif) || taken;
528 }
529 }
530 // What goes wrong from here on, the source reports.
531 if (!taken) {
532 qCWarning(VideoManagerLog) << "No video on display to take a screenshot from";
533 QGC::showAppMessage(tr("Unable to take a photo: no video is on display."));
534 }

595void VideoManager::selectSource(VideoStreamController* source)
596{
597 if (!source || !source->tile()) {
598 return;
599 }
600 qCDebug(VideoManagerLog) << "select" << source->name();
601
602 // A stream takes the main picture back from the camera. The sources follow the setting as it changes.
603 _videoSettings->uvcTileCameraAsMainPicture()->setRawValue(source == _cameraSource);
604 if (source == _cameraSource) {
605 return;
606 }
607
608 const auto stream =
609 std::find_if(_otherStreams.cbegin(), _otherStreams.cend(), [source](const AdvertisedStream& other) {
610 return other.streamInfo && (other.streamInfo == source->streamInfo());
611 });
612 QGCCameraManager* const cameraManager = _activeVehicle ? _activeVehicle->cameraManager() : nullptr;
613 if ((source != _primaryStream) && (stream != _otherStreams.cend()) && stream->camera && cameraManager) {
614 // Copied: the camera manager tells of each change at once, which replaces the streams.
615 MavlinkCameraControlInterface* const camera = stream->camera;
616 const int streamIndex = camera->streamLabels().indexOf(stream->streamInfo->name());
617 // The stream first: the camera then becomes the current one with the stream to show.
618 if (streamIndex >= 0) {
619 camera->setCurrentStream(streamIndex);
620 }
621 cameraManager->setCurrentCamera(cameraManager->cameras()->indexOf(camera));
622 }
623 _updateState();
624}

639void VideoManager::_updateStateNow()
640{
641 if (_cleanedUp) {
642 return;
643 }
644 if (!_videoSettings->disableWhenDisarmed()->rawValue().toBool()) {
645 _disabledWhileDisarmed = false;
646 }
647
648 const QString videoSource = _videoSettings->videoSource()->rawValue().toString();
649 const bool streamWanted = !_disabledWhileDisarmed;
650 // The reason that takes the most to lift comes first: arming doesn't bring back video that was stopped.
651 const NoVideoReason noVideoReason = _videoStopped ? NoVideoReason::Stopped
652 : _disabledWhileDisarmed ? NoVideoReason::DisabledWhileDisarmed
653 : NoVideoReason::Waiting;
654 const bool autoStreamConfigured = VideoStreamResolver::isAutoStream(_primaryStream->streamInfo());
655 const bool isStreamSource = VideoStreamResolver::isNetworkSource(videoSource) || autoStreamConfigured;
656 const int tilesAllowed =
657 static_cast<int>(std::min(_videoSettings->maxVideoTiles()->rawValue().toUInt(), kMaxVideoTiles));
658 const int maxTiles = (_tileRoom < 0) ? tilesAllowed : std::min(tilesAllowed, _tileRoom);
659 // A stream to play takes the place of the camera the video source names: the camera source then plays the camera
660 // to show in a tile.
661 const QString cameraName =
662 !isStreamSource ? videoSource
663 : ((maxTiles > 0) ? _videoSettings->uvcTileCamera()->rawValue().toString() : QString());
664 const bool cameraWanted = streamWanted && !cameraName.isEmpty();
665 if (cameraWanted) {
666 // A camera plugged in or unplugged changes whether the camera source has one to play.

672 // camera that is unplugged, and a restart.

674 isStreamSource && !camera.isEmpty() && _videoSettings->uvcTileCameraAsMainPicture()->rawValue().toBool();

693 for (VideoStreamController* source : std::as_const(_sources)) {

695 }

699 emit autoStreamConfiguredChanged();

703 emit isStreamSourceChanged();
704 }
705 if (uvcVideoSourceID != _uvcVideoSourceID) {
706 qCDebug(VideoManagerLog) << "UVC changed from [" << _uvcVideoSourceID << "] to [" << uvcVideoSourceID << "]";
707 _uvcVideoSourceID = uvcVideoSourceID;
708 emit uvcVideoSourceIDChanged();
709 emit isUvcChanged();
710 }
711 if (hasVideo != _hasVideo) {
712 _hasVideo = hasVideo;
713 emit hasVideoChanged();

715 // The Fly View swaps the video out of its PiP when hasVideo drops; nothing in QML could leave full screen.
716 setFullScreen(false);
717 }

741void VideoManager::_updateTileStreams(int count)
742{
743 const VideoStreamResolver::Settings settings = VideoStreamResolver::snapshot(_videoSettings);
744 QStringList shownAddresses;
745 for (const VideoStreamController* source : {_primaryStream, _thermalStream}) {
746 if (const auto config = VideoStreamResolver::resolve(settings, source->streamInfo(), source->role())) {
747 shownAddresses.append(streamAddress(config->uri));
748 }
749 }
750
751 QList<VideoStreamController*> tiles;
752 for (const AdvertisedStream& stream : std::as_const(_otherStreams)) {
753 if (tiles.size() >= count) {
754 break;
755 }
756 const auto config =
757 VideoStreamResolver::resolve(settings, stream.streamInfo, VideoStreamController::Role::Tile);
758 if (!config || shownAddresses.contains(streamAddress(config->uri))) {
759 continue;
760 }
761 shownAddresses.append(streamAddress(config->uri));
762
763 VideoStreamController* tile = _tileStream(stream.streamInfo);
764 if (!tile) {
765 tile = new VideoStreamController(VideoStreamController::Role::Tile, _videoSettings, this);
766 tile->setName(QStringLiteral("videoTile%1").arg(++_tileStreamSerial));
767 tile->setStreamInfo(stream.streamInfo);
768 // Its source records once it is the primary stream.
769 tile->setRecordingEncoding(recordingEncodingSetting(_videoSettings));
770 tile->setDisplaySuspended(_displaySuspended);
771 _connectSource(tile);
772 _recordingService->addSource(tile);
773 }
774 tile->setDisplayName(stream.name);
775 tile->setTile(true);
776 tiles.append(tile);
777 }
778
779 QList<VideoStreamController*> removed;
780 for (VideoStreamController* const tile : std::as_const(_tileStreams)) {
781 if (!tiles.contains(tile)) {
782 removed.append(tile);
783 }
784 }
785 _tileStreams = tiles;
786
787 const QList<VideoStreamController*> sources =
788 QList<VideoStreamController*>{_primaryStream, _thermalStream, _cameraSource} + _tileStreams;
789 if (sources != _sources) {
790 _sources = sources;
791 // Before the sources that left are deleted: none of them is the measured one from here on.
792 _updateSourceSummary();
793 emit sourcesChanged();
794 }
795 // Once QML has let go of them.
796 for (VideoStreamController* const tile : std::as_const(removed)) {
797 qCDebug(VideoManagerLog) << "tile removed" << tile->name();
798 tile->shutDown();
799 tile->deleteLater();
800 }
801 // The streams have their receivers: so do the new ones.
802 if (_thermalStream->receiver()) {
803 _createVideoReceivers();
804 }
805 _nameExtraStreams();
806}

871void VideoManager::_updateStreamInfo()
872{
873 QGCCameraManager* const cameraManager = _activeVehicle ? _activeVehicle->cameraManager() : nullptr;
874 if (!cameraManager) {
875 _setAdvertisedStreams({}, nullptr, {});
876 return;
877 }
878
879 // The current camera's streams come first, then the other cameras', in the cameras' order.
880 QList<MavlinkCameraControlInterface*> cameras = vehicleCameras(cameraManager);
881 MavlinkCameraControlInterface* const currentCamera = cameraManager->currentCameraInstance();
882 if (cameras.removeOne(currentCamera)) {
883 cameras.prepend(currentCamera);
884 }
885 QGCVideoStreamInfo* const currentStreamInfo = cameraManager->currentStreamInstance();
886 AdvertisedStream current;
887 QList<AdvertisedStream> others;
888 for (MavlinkCameraControlInterface* const camera : std::as_const(cameras)) {
889 QmlObjectListModel* const streams = camera->streams();
890 for (int i = 0; streams && (i < streams->count()); ++i) {
891 auto* const streamInfo = qobject_cast<QGCVideoStreamInfo*>(streams->get(i));
892 if (!streamInfo || streamInfo->isThermal()) {
893 continue;
894 }
895 const QString name = (cameras.size() > 1)
896 ? QStringLiteral("%1: %2").arg(camera->modelName(), streamInfo->name())
897 : streamInfo->name();
898 if (streamInfo == currentStreamInfo) {
899 current = {streamInfo, camera, name};
900 } else {
901 others.append({streamInfo, camera, name});
902 }
903 }
904 }
905 _setAdvertisedStreams(current, cameraManager->thermalStreamInstance(), others);
906}

908void VideoManager::_setAdvertisedStreams(const AdvertisedStream& current, QGCVideoStreamInfo* thermal,
909 const QList<AdvertisedStream>& others)
910{
911 if (_cleanedUp) {
912 return;
913 }
914 _otherStreams = others;
915
916 if (VideoStreamController* const playing = _tileStream(current.streamInfo)) {
917 // The stream to display plays in a tile: its source becomes the primary stream as it is, and the former
918 // primary stream, which plays on, gets a tile if its stream is still one to show.
919 VideoStreamController* const former = _primaryStream;
920 _tileStreams.replace(_tileStreams.indexOf(playing), former);
921 _primaryStream = playing;
922 playing->setTile(former->tile());
923 former->setRole(VideoStreamController::Role::Tile);
924 playing->setRole(VideoStreamController::Role::Primary);
925 emit primaryStreamChanged();
926 emit decodingChanged();
927 emit aspectRatioChanged();
928 } else {
929 _primaryStream->setStreamInfo(current.streamInfo);
930 }
931 if (current.streamInfo) {
932 _primaryStream->setDisplayName(current.name);
933 }
934 _thermalStream->setStreamInfo(thermal);
935 _updateState();
936}
