105void VideoManager::_connectSettings()
106{
107 for (const VideoSettingReactionEntry& entry : kVideoSettingReactions) {
108 Fact* const fact = (_videoSettings->*entry.fact)();
109 switch (entry.reaction) {
110 case VideoSettingReaction::Reconfigure:
111 (void) connect(fact, &Fact::rawValueChanged, this, [this] {
112 _updateState();
113 _primaryStream->scheduleApply();
114 _thermalStream->scheduleApply();
115 });
116 break;
117 case VideoSettingReaction::RebuildSink:
118 (void) connect(fact, &Fact::rawValueChanged, this, [this] {
119 _primaryStream->restart();
120 _thermalStream->restart();
121 });
122 break;
123 }
124 }

151bool VideoManager::waitForVideoBackendReady()
152{
153 if (_initState == InitState::NotStarted) {
154 startVideoBackendInit();
155 }
156
157 if (_initState == InitState::Pending) {
158 _backendInitFuture.waitForFinished();
159 const bool success =
160 !_backendInitFuture.isCanceled() && (_backendInitFuture.resultCount() > 0) && _backendInitFuture.result();
161 _onBackendInitComplete(success);
162 }
163
164 return _initState == InitState::Ready;

327void VideoManager::_updateState()
328{
329 if (!_videoSettings->disableWhenDisarmed()->rawValue().toBool()) {
330 _disabledWhileDisarmed = false;
331 }
332
333 const QString videoSource = _videoSettings->videoSource()->rawValue().toString();
334 const bool streamWanted = _videoSettings->streamEnabled()->rawValue().toBool() && !_disabledWhileDisarmed;
335 const bool autoStreamConfigured = VideoStreamResolver::isAutoStream(_primaryStream->streamInfo());
336 const bool isStreamSource = VideoStreamResolver::isNetworkSource(videoSource) || autoStreamConfigured;
337 const QString uvcVideoSourceID =
338 (streamWanted && !isStreamSource && UVCDevices::deviceExists(videoSource)) ? videoSource : QString();
339 const bool hasVideo = streamWanted && (_primaryStream->configured() || !uvcVideoSourceID.isEmpty());
340 const QGCVideoStreamInfo* const thermalStreamInfo = _thermalStream->streamInfo();
341 const bool hasThermal = thermalStreamInfo && thermalStreamInfo->isThermal();
342
343 _primaryStream->setEnabled(streamWanted && !_videoStopped);
344 _thermalStream->setEnabled(streamWanted && !_videoStopped);
345
346 if (autoStreamConfigured != _autoStreamConfigured) {
347 _autoStreamConfigured = autoStreamConfigured;
348 emit autoStreamConfiguredChanged();
349 }
350 if (isStreamSource != _isStreamSource) {
351 _isStreamSource = isStreamSource;
352 emit isStreamSourceChanged();
353 }
354 if (uvcVideoSourceID != _uvcVideoSourceID) {
355 qCDebug(VideoManagerLog) << "UVC changed from [" << _uvcVideoSourceID << "] to [" << uvcVideoSourceID << "]";
356 _uvcVideoSourceID = uvcVideoSourceID;
357 if (!_uvcVideoSourceID.isEmpty()) {
358 UVCDevices::requestPermission();
359 }
360 emit uvcVideoSourceIDChanged();
361 emit isUvcChanged();
362 }
363 if (hasVideo != _hasVideo) {
364 _hasVideo = hasVideo;
365 emit hasVideoChanged();
366 if (!_hasVideo) {
367 // The Fly View swaps the video out of its PiP when hasVideo drops; nothing in QML could leave full screen.
368 setfullScreen(false);
369 }
370 }
371 if (hasThermal != _hasThermal) {
372 _hasThermal = hasThermal;
373 emit hasThermalChanged();
374 }
