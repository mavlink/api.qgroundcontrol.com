28 /// Size of the video fitted to the view: larger than the view where the fit mode crops, smaller where it doesn't.
29 readonly property real videoWidth: {
30 if (root._ar > 0) {
31 if (root._isModeFitHeight
32 || (root._isModeFill && (root.width / root.height < root._ar))
33 || (root._isModeNoCrop && (root.width / root.height > root._ar))) {
34 // This return value has different implications depending on the mode
35 // For FIT_HEIGHT and FILL
36 // makes so the video width will be larger than (or equal to) the screen width
37 // For NO_CROP Mode
38 // makes so the video width will be smaller than (or equal to) the screen width
39 return root.height * root._ar
40 }
41 }
42 return root.width
43 }
44 readonly property real videoHeight: {
45 if (root._ar > 0) {
46 if (root._isModeFitWidth
47 || (root._isModeFill && (root.width / root.height > root._ar))
48 || (root._isModeNoCrop && (root.width / root.height < root._ar))) {
49 // This return value has different implications depending on the mode
50 // For FIT_WIDTH and FILL
51 // makes so the video height will be larger than (or equal to) the screen height
52 // For NO_CROP Mode
53 // makes so the video height will be smaller than (or equal to) the screen height
54 return root.width * (1 / root._ar)
55 }
56 }
