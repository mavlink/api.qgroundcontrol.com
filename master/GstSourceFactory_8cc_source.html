36#if defined(QGC_GST_ENABLE_LEGACY_PARSEBIN_CAPS_FILTER)
37gboolean filterParserCaps([[maybe_unused]] GstElement* bin, [[maybe_unused]] GstPad* pad,
38 [[maybe_unused]] GstElement* element, GstQuery* query, [[maybe_unused]] gpointer data)
39{
40 if (GST_QUERY_TYPE(query) != GST_QUERY_CAPS) {
41 return FALSE;

46 if (!srcCaps || gst_caps_is_any(srcCaps)) {

49
50 GstCapsPtr sinkCaps;

52 if (gst_structure_has_name(structure, "video/x-h265")) {

54 if (gst_caps_can_intersect(srcCaps, filter.get())) {

61 }
62 }

102gboolean padProbe([[maybe_unused]] GstElement* element, GstPad* pad, gpointer user_data)
103{
104 int* probeRes = static_cast<int*>(user_data);
105 *probeRes |= 1;
106
107 const GstCapsPtr filter = adoptCaps(gst_caps_from_string("application/x-rtp"));
108 const GstCapsPtr caps = adoptCaps(gst_pad_query_caps(pad, nullptr));
109 if (filter && caps && !gst_caps_is_any(caps.get()) && gst_caps_can_intersect(caps.get(), filter.get())) {

155constexpr TransportSpec specFor(Transport transport)
156{
157 switch (transport) {
158 case Transport::Rtsp:
159 return {SourceElement::Rtsp, nullptr, false, false};
160 case Transport::RtpUdpH264:
161 return {SourceElement::Udp, "H264", false, false};
162 case Transport::RtpUdpH265:
163 return {SourceElement::Udp, "H265", false, true};
164 case Transport::MpegTsUdp:
165 return {SourceElement::Udp, nullptr, true, false};
166 case Transport::MpegTsTcp:
167 return {SourceElement::TcpClient, nullptr, true, false};
168 }
169 return {SourceElement::Udp, nullptr, false, false};
170}

172// Shared by the static and dynamic (pad-added) link paths so both apply the identical jitter-buffer
173// policy. Retransmission stays off: these sources have no RTCP back-channel, so the jitterbuffer's
174// retransmission requests would only reach the source element, which drops them.
175GstElement* addJitterBuffer(GstElement* bin, guint latencyMs)
176{
177 GstElementPtr buffer = makeElement("rtpjitterbuffer");
178 if (!buffer) {
179 qCWarning(GstSourceFactoryLog) << "gst_element_factory_make('rtpjitterbuffer') failed";
180 return nullptr;
181 }
182
183 g_object_set(buffer.get(), "latency", latencyMs, "do-lost", TRUE, "drop-on-latency", TRUE, nullptr);
184 return addToBin(bin, std::move(buffer), "rtpjitterbuffer");
185}

187// Heap-owned context for the dynamic (pad-added) link path; freed when @binParser is finalized.
188struct DynamicLinkContext
189{
190 GstElement* binParser;
191 guint latencyMs;
192 bool insertJitterBuffer; // false for RTSP (rtspsrc has its own internal jitterbuffer) and JitterBuffer::None
193};
194
195// Inserts a running rtpjitterbuffer linked to the parser; returns it (bin-owned) or nullptr to link directly.
196GstElement* insertDynamicJitterBuffer(const DynamicLinkContext* ctx)
197{
198 const GstObjectPtr bin = adoptObject(gst_object_get_parent(GST_OBJECT(ctx->binParser)));
199 if (!bin) {
200 return nullptr;
201 }
202
203 GstElement* buffer = addJitterBuffer(GST_ELEMENT(bin.get()), ctx->latencyMs);
204 if (!buffer) {
205 qCWarning(GstSourceFactoryLog) << "dynamic rtpjitterbuffer create/add failed; skipping jitter buffer";
206 return nullptr;
207 }
208
209 if (!gst_element_sync_state_with_parent(buffer) || !gst_element_link(buffer, ctx->binParser)) {
210 qCWarning(GstSourceFactoryLog) << "dynamic rtpjitterbuffer link/sync failed; linking pad straight to parser";
211 disposeChild(buffer);
212 return nullptr;
213 }
214
215 return buffer;
216}

218void linkPad(GstElement* element, GstPad* pad, gpointer data)
219{
220 const auto* ctx = static_cast<const DynamicLinkContext*>(data);
221
222 // tsdemux/rtspsrc fire pad-added for audio/data pads too; only link video src pads so a
223 // non-video pad can't claim the parser ahead of the real video pad.
224 if (GST_PAD_DIRECTION(pad) != GST_PAD_SRC) {
225 return;
226 }
227
228 bool isVideo = false;
229 bool isRtp = false;
230 GstCapsPtr caps = adoptCaps(gst_pad_get_current_caps(pad));
231 if (!caps) {
232 caps = adoptCaps(gst_pad_query_caps(pad, nullptr));
233 }
234 if (caps) {
235 const guint n = gst_caps_get_size(caps.get());
236 for (guint i = 0; i < n; ++i) {
237 const GstStructure* st = gst_caps_get_structure(caps.get(), i);
238 if (isVideoRtp(st)) {
239 // The depayloader/parsebin downstream classifies the video payload.
240 isRtp = true;
241 isVideo = true;
242 } else if (g_str_has_prefix(gst_structure_get_name(st), "video/")) {
243 isVideo = true;
244 }
245 }
246 }
247 if (!isVideo) {
248 return;
249 }
250
251 // Link only the first matching video pad. A second pad-added (duplicate RTP pad, or a later
252 // tsdemux elementary stream) must not stack a second jitterbuffer or re-link an occupied sink.
253 const GstPadPtr parserSink = adoptPad(gst_element_get_static_pad(ctx->binParser, "sink"));
254 if (parserSink && gst_pad_is_linked(parserSink.get())) {
255 return;
256 }
257
258 // Insert a jitterbuffer ahead of the parser for RTP pads on non-RTSP dynamic sources, matching the
259 // static path's policy. RTSP is excluded: rtspsrc owns an internal jitterbuffer (no double-buffering).
260 GstElement* dynamicBuffer = nullptr;
261 if (isRtp && ctx->insertJitterBuffer) {
262 dynamicBuffer = insertDynamicJitterBuffer(ctx);
263 }
264 GstElement* const downstream = dynamicBuffer ? dynamicBuffer : ctx->binParser;
265
266 const GCharPtr name = adoptGChar(gst_pad_get_name(pad));
267 if (!name || !gst_element_link_pads(element, name.get(), downstream, "sink")) {
268 qCWarning(GstSourceFactoryLog) << "gst_element_link_pads() failed for" << (name ? name.get() : "(unnamed pad)");
269 if (dynamicBuffer) {
270 disposeChild(dynamicBuffer);
271 }
272 }
273}

303GstElementPtr buildUdpSource(const QUrl& sourceUrl, const char* rtpEncoding)
304{
305 // Port 0 binds a free port, which udpsrc then reports in its "port" property.
306 const int port = sourceUrl.port();
307 if ((port != 0) && !isValidPort(port)) {
308 qCWarning(GstSourceFactoryLog) << "Invalid UDP port" << port << "in"
309 << QGCNetworkHelper::redactedUrlForLogging(sourceUrl);
310 return nullptr;
311 }
312
313 GstElementPtr source = makeElement("udpsrc", "source");
314 if (!source) {
315 qCWarning(GstSourceFactoryLog) << "gst_element_factory_make('udpsrc') failed";
316 return nullptr;
317 }
318
319 // Normalize udp265:///mpegts:// to udp:// for udpsrc, preserving query params.
320 QUrl udpUrl(sourceUrl);
321 udpUrl.setScheme(QStringLiteral("udp"));
322 udpUrl.setUserInfo(QString());
323 const QByteArray udpUri = udpUrl.toEncoded();
324
325 // SO_RCVBUF above net.core.rmem_max needs CAP_NET_ADMIN; without it gstudpsrc fails with EPERM
326 // (scary warning) then clamps. Request the kernel-permitted max instead (Linux/Android only).
327 int udpBufferSize = kUdpBufferSizeBytes;
328#ifdef Q_OS_LINUX
329 // rmem_max is a boot-time sysctl; parse once and cache (0 = unreadable, leave request unclamped).
330 static const int s_rmemMax = [] {
331 QFile rmemMaxFile(QStringLiteral("/proc/sys/net/core/rmem_max"));
332 if (rmemMaxFile.open(QIODevice::ReadOnly | QIODevice::Text)) {
333 bool ok = false;
334 const int sysMax = rmemMaxFile.readAll().trimmed().toInt(&ok);
335 if (ok && (sysMax > 0)) {
336 return sysMax;
337 }
338 }
339 return 0;
340 }();
341 if ((s_rmemMax > 0) && (s_rmemMax < udpBufferSize)) {
342 qCDebug(GstSourceFactoryLog) << "Clamping UDP buffer-size from" << udpBufferSize << "to net.core.rmem_max"
343 << s_rmemMax;
344 udpBufferSize = s_rmemMax;
345 }
346#endif
347 g_object_set(source.get(), "uri", udpUri.constData(), "buffer-size", udpBufferSize, nullptr);
348
349 if (rtpEncoding) {
350 const GstCapsPtr caps =
351 adoptCaps(gst_caps_new_simple("application/x-rtp", "media", G_TYPE_STRING, "video", "clock-rate",
352 G_TYPE_INT, 90000, "encoding-name", G_TYPE_STRING, rtpEncoding, nullptr));
353 g_object_set(source.get(), "caps", caps.get(), nullptr);
354 }
355 return source;
356}

358struct ParserStage
359{
360 GstElementPtr parser;
361 GstElementPtr depay;
362};

364// The parser (and, for explicit H.265 depayloading, the depayloader) for @p spec, configured but not yet in a bin.
365ParserStage buildParserStage(const TransportSpec& spec, bool isRtsp)
366{
367 const char* const parserFactory = spec.explicitH265Depay ? "h265parse" : "parsebin";
368 ParserStage stage{makeElement(parserFactory, "parser"), {}};
369 if (!stage.parser) {
370 qCWarning(GstSourceFactoryLog) << "gst_element_factory_make(" << parserFactory << ") failed";
371 return {};
372 }
373
374 if (spec.explicitH265Depay) {
375 // UDP H.265 has no SDP/RTSP control plane. Make depayloading and parser config explicit
376 // so hardware decoders, especially Android MediaCodec, see repeated VPS/SPS/PPS at IDR
377 // boundaries instead of depending on parsebin's defaults after startup or reconnect.
378 stage.depay = makeElement("rtph265depay");
379 if (!stage.depay) {
380 qCWarning(GstSourceFactoryLog) << "gst_element_factory_make('rtph265depay') failed";
381 return {};
382 }
383 configureH26xParser(stage.parser.get());
384 configureH26xRtpDepayloader(stage.depay.get(), false);
385 return stage;
386 }
387
388 // parsebin creates the depayloader and codec parser only after it sees the stream caps. Configure them
389 // as soon as they are autoplugged, before data flows.
390 (void) g_signal_connect(stage.parser.get(), "deep-element-added", G_CALLBACK(configureAutopluggedElement),
391 isRtsp ? GINT_TO_POINTER(TRUE) : nullptr);
392
393 // Older Linux/system GStreamer misnegotiates parser->decoder caps; force avc/hvc1 there only.
394 // Bundled 1.28+ SDKs do not need it, and the forced caps break byte-stream HW decoders.
395#if defined(QGC_GST_ENABLE_LEGACY_PARSEBIN_CAPS_FILTER)
396 (void) g_signal_connect(stage.parser.get(), "autoplug-query", G_CALLBACK(filterParserCaps), nullptr);
397#endif
398 return stage;
399}

423// Wire upstream → (optional rtpjitterbuffer) → binParser, topology chosen by RTP probe (MPEG-TS
424// links via pad-added). Created elements join @p bin; returns false (logged) on failure.
425bool linkSourceToParser(GstElement* bin, GstElement* upstream, GstElement* binParser, const Config& config,
426 guint latencyMs, bool isMpegTs, bool isRtsp)
427{
428 // MPEG-TS exposes elementary streams via tsdemux dynamic src pads (none at NULL state) and
429 // isn't raw RTP, so skip the RTP probe and link via pad-added when the video pad appears.
430 int probeRes = 0;
431 if (!isMpegTs) {
432 (void) gst_element_foreach_src_pad(upstream, padProbe, &probeRes);
433 }
434
435 if (probeRes & 1) {
436 if ((probeRes & 2) && config.jitterBuffer != JitterBuffer::None) {
437 GstElement* buffer = addJitterBuffer(bin, latencyMs);
438 if (!buffer) {
439 return false;
440 }
441 // bin owns buffer now; a link failure below is cleaned up via the caller's bin teardown.
442 if (!gst_element_link_many(upstream, buffer, binParser, nullptr)) {
443 qCWarning(GstSourceFactoryLog) << "gst_element_link_many(source, rtpjitterbuffer, parser) failed";
444 return false;
445 }
446 } else {
447 if (!gst_element_link(upstream, binParser)) {
448 qCWarning(GstSourceFactoryLog) << "gst_element_link(source, parser) failed";
449 return false;
450 }
451 }
452 } else {
453 // linkPad applies the jitter-buffer policy itself when an application/x-rtp pad appears on a
454 // non-RTSP source. Context lifetime is tied to binParser (freed on its finalize).
455 auto* ctx =
456 new DynamicLinkContext{binParser, latencyMs, !isRtsp && (config.jitterBuffer != JitterBuffer::None)};
457 g_object_set_data_full(G_OBJECT(binParser), "qgc-dynamic-link-ctx", ctx,
458 [](gpointer p) { delete static_cast<DynamicLinkContext*>(p); });
459 (void) g_signal_connect(upstream, "pad-added", G_CALLBACK(linkPad), ctx);
460 }
461
462 // pad-added fires on the streaming thread after create() returns the NULL-state bin, so it
463 // can't race construction; wrapWithGhostPad ref-checks the parent against teardown.
464 (void) g_signal_connect(binParser, "pad-added", G_CALLBACK(wrapWithGhostPad), nullptr);
465 return true;
466}

468bool linkUdpRtpToDepayAndParser(GstElement* bin, GstElement* source, GstElement* depay, GstElement* parser,
469 const Config& config, guint latencyMs)
470{
471 GstElement* upstream = source;
472 if (config.jitterBuffer != JitterBuffer::None) {
473 GstElement* const buffer = addJitterBuffer(bin, latencyMs);
474 if (!buffer) {
475 return false;
476 }
477 if (!gst_element_link(source, buffer)) {
478 qCWarning(GstSourceFactoryLog) << "gst_element_link(source, rtpjitterbuffer) failed";
479 return false;
480 }
481 upstream = buffer;
482 }
483
484 if (!gst_element_link(upstream, depay)) {
485 qCWarning(GstSourceFactoryLog) << "gst_element_link(upstream, depay) failed";
486 return false;
487 }
488
489 // Left to negotiate, rtph265depay picks hvc1, whose caps it only sets once in-band VPS/SPS/PPS arrive, so its
490 // segment can go out first (GStreamer 1.28 warns) and the parameter-set gate, which handles byte-stream, stands
491 // down. Byte-stream caps are set as soon as the RTP caps arrive.
492 const GstCapsPtr byteStream =
493 adoptCaps(gst_caps_from_string("video/x-h265, stream-format=(string)byte-stream, alignment=(string)au"));
494 if (!byteStream || !gst_element_link_filtered(depay, parser, byteStream.get())) {
495 qCWarning(GstSourceFactoryLog) << "gst_element_link_filtered(depay, parser, byte-stream) failed";
496 return false;
497 }
498
499 return true;
500}

504GstElement* create(const QString& uri, const Config& config)
505{
506 if (uri.isEmpty()) {
507 qCWarning(GstSourceFactoryLog) << "Failed because URI is not specified";
508 return nullptr;
509 }
510
511 const guint latencyMs = (config.latencyMs < 0) ? 0u : static_cast<guint>(config.latencyMs);
512
513 const QUrl sourceUrl(uri);
514 const std::optional<Transport> transport = transportFor(sourceUrl);
515 if (!transport) {
516 qCWarning(GstSourceFactoryLog) << "Unsupported URI scheme:" << sourceUrl.scheme() << "in"
517 << QGCNetworkHelper::redactedUrlForLogging(sourceUrl);
518 return nullptr;
519 }
520
521 const TransportSpec spec = specFor(*transport);
522 const bool isRtsp = (spec.element == SourceElement::Rtsp);
523
524 GstElementPtr source;
525 switch (spec.element) {
526 case SourceElement::Rtsp:
527 source = buildRtspSource(uri, sourceUrl, config, latencyMs);
528 break;
529 case SourceElement::TcpClient:
530 source = buildTcpSource(sourceUrl);
531 break;
532 case SourceElement::Udp:
533 source = buildUdpSource(sourceUrl, spec.rtpEncoding);
534 break;
535 }
536 if (!source) {
537 return nullptr;
538 }
539
540 GstElementPtr bin = adoptElement(gst_bin_new("sourcebin"));
541 if (!bin) {
542 qCWarning(GstSourceFactoryLog) << "gst_bin_new('sourcebin') failed";
543 return nullptr;
544 }
545
546 ParserStage parserStage = buildParserStage(spec, isRtsp);
547 if (!parserStage.parser) {
548 return nullptr;
549 }
550
551 // From here on the bin owns every element; the borrowed pointers stay valid while bin is alive.
552 GstElement* upstream = addToBin(bin.get(), std::move(source), "source");
553 if (!upstream) {
554 return nullptr;
555 }
556
557 GstElement* depay = nullptr;
558 if (parserStage.depay) {
559 depay = addToBin(bin.get(), std::move(parserStage.depay), "rtph265depay");
560 if (!depay) {
561 return nullptr;
562 }
563 }
564
565 GstElement* const binParser = addToBin(bin.get(), std::move(parserStage.parser), "parser");
566 if (!binParser) {
567 return nullptr;
568 }
569
570 if (spec.demuxMpegTs) {
571 upstream = addTsDemux(bin.get(), upstream);
572 if (!upstream) {
573 return nullptr;
574 }
575 }
576
577 if (spec.explicitH265Depay) {
578 if (!linkUdpRtpToDepayAndParser(bin.get(), upstream, depay, binParser, config, latencyMs)) {
579 return nullptr;
580 }
581 if (!addStaticGhostPad(binParser)) {
582 return nullptr;
583 }
584 } else if (!linkSourceToParser(bin.get(), upstream, binParser, config, latencyMs, spec.demuxMpegTs, isRtsp)) {
585 return nullptr;
586 }
587
588 return bin.release();
589}
