81bool _verifyPlugins()
82{
83 GstRegistry* registry = gst_registry_get();
84 if (!registry) {
85 qCCritical(GStreamerLog) << "Failed to get GStreamer registry";
86 return false;
87 }
88
89 qCDebug(GStreamerLog) << "Installed GStreamer plugins:";
90 GStreamer::forEachPlugin(registry, [](GstPlugin* plugin) {
91 qCDebug(GStreamerLog) << " " << gst_plugin_get_name(plugin) << gst_plugin_get_version(plugin);
92 });
93
94 bool result = true;
95 const auto hasPlugin = [registry](const char* name) {
96 const GstObjectPtr plugin(GST_OBJECT(gst_registry_find_plugin(registry, name)));
97 return plugin != nullptr;
98 };
99 // qgc plus GSTREAMER_RUNTIME_REQUIRED_PLUGINS (PluginPolicy.cmake), so a stripped registry fails loudly instead
100 // of at the first stream attempt. Each entry lists, '|'-separated, the plugin sets that satisfy it, a set's
101 // plugins joined by '+' (videoconvert+videoscale stand in for the fused videoconvertscale of 1.22+).
102 static constexpr const char* kRequiredPlugins[] = {"qgc", QGC_GST_RUNTIME_REQUIRED_PLUGINS};
103 for (const char* requirement : kRequiredPlugins) {
104 const QList<QByteArray> alternatives = QByteArray(requirement).split('|');
105 const bool satisfied = std::ranges::any_of(alternatives, [&hasPlugin](const QByteArray& alternative) {
106 const QList<QByteArray> plugins = alternative.split('+');
107 return std::ranges::all_of(
108 plugins, [&hasPlugin](const QByteArray& plugin) { return hasPlugin(plugin.constData()); });
109 });
110 if (!satisfied) {
111 qCCritical(GStreamerLog) << "Required GStreamer plugin not found:" << requirement;
112 result = false;
113 }
114 }
115
116 if (!result) {
117 // Surface blacklisted plugins so a failure from a corrupt/incompatible plugin file
118 // shows up here instead of looking like a missing-plugin problem.
119 GStreamer::forEachPlugin(registry, [](GstPlugin* p) {
120 const gchar* desc = gst_plugin_get_description(p);
121 if (!desc || !g_str_has_prefix(desc, "BLACKLIST"))
122 return;
123 const gchar* filename = gst_plugin_get_filename(p);
124 qCWarning(GStreamerLog) << "Blacklisted plugin:" << gst_plugin_get_name(p)
125 << "file:" << (filename ? filename : "(null)");
126 });
127
128 // Path / scanner env vars belong to the environment layer that set them.
129 Environment::logDiagnostics();
130 }
131
132 return result;
133}

154// Before gst-plugins-good 1.24.10, udpsrc's unlock_stop() replaces its GCancellable on the streaming thread (basesrc
155// defers it to the first create() after PLAYING) while unlock() cancels it from the thread leaving PLAYING. A pipeline
156// stopped right after it started then cancels a freed GCancellable and corrupts the heap. Serialize the two, as the
157// upstream fix does (gstreamer!7897).
158void _serializeUdpSrcUnlock()
159{
160 static std::once_flag s_serialized;
161 std::call_once(s_serialized, [] {
162 const GstFactoryPtr factory = adoptFactory(gst_element_factory_find("udpsrc"));
163 if (!factory || gst_plugin_feature_check_version(GST_PLUGIN_FEATURE(factory.get()), 1, 24, 10)) {
164 return;
165 }
166
167 const GstFeaturePtr loaded = adoptFeature(gst_plugin_feature_load(GST_PLUGIN_FEATURE(factory.get())));
168 const GType type =
169 loaded ? gst_element_factory_get_element_type(GST_ELEMENT_FACTORY(loaded.get())) : G_TYPE_INVALID;
170 if (!g_type_is_a(type, GST_TYPE_BASE_SRC)) {
171 qCWarning(GStreamerLog) << "udpsrc failed to load; stopping a UDP stream right after it starts may crash";
172 return;
173 }
174
175 // Never released, so the patched class outlives every udpsrc.
176 auto* const klass = static_cast<GstBaseSrcClass*>(g_type_class_ref(type));
177 s_udpSrcUnlock = klass->unlock;
178 s_udpSrcUnlockStop = klass->unlock_stop;
179 if (s_udpSrcUnlock && s_udpSrcUnlockStop) {
180 klass->unlock = _udpSrcUnlockSerialized;
181 klass->unlock_stop = _udpSrcUnlockStopSerialized;
182 qCDebug(GStreamerLog) << "Serialized udpsrc unlock/unlock_stop (gst-plugins-good before 1.24.10)";
183 }
184 });
185}

196bool _initGstRuntime(const QStringList& args, const Environment::ValidationResult& env)
197{
198 if (!env.ok) {
199 qCCritical(GStreamerLog) << "Invalid GStreamer environment configuration:" << env.error;
200 return false;
201 }
202
203 // args is snapshotted on the GUI thread: QCoreApplication::arguments() is not thread-safe
204 // and initialize() runs on a QtConcurrent pool thread.
205 QByteArrayList argStorage;
206 argStorage.reserve(args.size());
207 for (const QString& arg : args) {
208 argStorage.append(arg.toUtf8());
209 }
210
211 QVarLengthArray<char*, 16> argv;
212 for (QByteArray& arg : argStorage) {
213 argv.append(arg.data());
214 }
215
216 int argc = argv.size();
217 char** argvPtr = argv.data();
218 GError* error = nullptr;
219
220 if (!gst_init_check(&argc, &argvPtr, &error)) {
221 qCCritical(GStreamerLog) << "Failed to initialize GStreamer:" << (error ? error->message : "unknown error");
222 g_clear_error(&error);
223 return false;
224 }
225
226 return true;
227}

303// Video sink refcount protocol: createVideoSink returns one owned, non-floating ref, which the caller
304// drops with releaseVideoSink. While decoding, the receiver holds its own ref and the pipeline bin
305// takes another on gst_bin_add (dropped by gst_bin_remove). Keep the ref/unref sites balanced.
306void* createVideoSink(const VideoSinkConfig& config)
307{
308 const gboolean disablePar = config.disablePixelAspectRatio ? TRUE : FALSE;
309 const char* const conversion = config.conversionElement.isEmpty() ? nullptr : config.conversionElement.constData();
310
311 const GstFactoryPtr factory = adoptFactory(gst_element_factory_find("qgcvideosinkbin"));
312 if (!factory) {
313 // completeInit verified this factory at startup; absence here means the registry changed underfoot.
314 qCCritical(GStreamerLog) << "qgcvideosinkbin factory not found";
315 return nullptr;
316 }
317
318#if defined(QGC_HAS_ANY_GPU_PATH)
319 GstElement* videoSinkBin =
320 gst_element_factory_create_full(factory.get(), "gpu-zerocopy", config.gpuZeroCopy ? TRUE : FALSE,
321 "conversion-element", conversion, "disable-par", disablePar, NULL);
322#else
323 GstElement* videoSinkBin = gst_element_factory_create_full(factory.get(), "conversion-element", conversion,
324 "disable-par", disablePar, NULL);
325#endif
326 if (!videoSinkBin) {
327 qCCritical(GStreamerLog) << "qgcvideosinkbin element creation failed";
328 return nullptr;
329 }
330 // Sink the floating ref so adding the sink to a bin never consumes the caller's reference.
331 return gst_object_ref_sink(videoSinkBin);
332}

347bool setupQVideoSinkElement(void* sinkBin, QVideoSink* videoSink, QObject* controllerParent)
348{
349 if (!sinkBin || !videoSink || !controllerParent) {
350 // controllerParent owns the controller's QObject lifetime — else it leaks (caller has no handle).
351 qCWarning(GStreamerLog) << "setupQVideoSinkElement: null sinkBin, videoSink, or controllerParent";
352 return false;
353 }
354
355 // Idempotent re-setup: tear down prior controllers so repeated startDecoding cycles
356 // don't accumulate dangling ones.
357 for (auto* c : QGCQVideoSinkController::controllersOf(controllerParent)) {
358 c->setActive(false);
359 // Stop the poll timer synchronously: deleteLater is deferred and the timer would otherwise
360 // keep binding the same element while the new controller is being installed.
361 c->prepareForRelease();
362 c->deleteLater();
363 }
364
365 // Clear the GL/Vulkan bridges' exhausted-retry latches so a restart after Qt's RHI appears can
366 // prime on the next NEED_CONTEXT.
367 HwBuffers::onPipelineRestart();
368
369 // Accessor returns a transfer-full ref; the guard releases it once the controller has taken
370 // its own ref for deferred QObject teardown.
371 const GstObjectPtr element(GST_OBJECT(gst_qgc_video_sink_bin_get_qvideosink(GST_QGC_VIDEO_SINK_BIN(sinkBin))));
372 if (!element) {
373 qCCritical(GStreamerLog) << "setupQVideoSinkElement: bin has no qgcqvideosink child";
374 return false;
375 }
376 GstElement* const elementRaw = GST_ELEMENT(element.get());
377
378 auto* controller = new QGCQVideoSinkController(elementRaw, controllerParent);
379
380 // Route the initial install through the controller so its destroyed-sink QPointer guard
381 // covers setup, not just later swaps.
382 controller->setVideoSink(QPointer<QVideoSink>(videoSink));
383 controller->setActive(true);
384
385 return true;
386}

437static const char* zeroCopyFamilyForApi(QSGRendererInterface::GraphicsApi api)
438{
439 switch (api) {

441#if defined(QGC_HAS_GST_GLMEMORY_GPU_PATH) || defined(QGC_HAS_GST_DMABUF_GPU_PATH)

444 return "CPU";
445#endif

464 return "CPU (Vulkan import dormant)";

470void onMainWindowReady(QQuickWindow* window)
471{
472 HwBuffers::connectMainWindow(window);
473 // Prefer the resolved backend (cachedRhi) over the configured API once the scene graph is up; QGCRhiCapture exists
474 // only when a GPU path is compiled, hence the guard.
475 QSGRendererInterface::GraphicsApi api = QQuickWindow::graphicsApi();
476#if defined(QGC_HAS_ANY_GPU_PATH)
477 if (QRhi* rhi = QGCRhiCapture::cachedRhi()) {
478 switch (rhi->backend()) {
479 case QRhi::OpenGLES2: api = QSGRendererInterface::OpenGL; break;
480 case QRhi::D3D11: api = QSGRendererInterface::Direct3D11; break;
481 case QRhi::D3D12: api = QSGRendererInterface::Direct3D12; break;
482 case QRhi::Metal: api = QSGRendererInterface::Metal; break;
483 case QRhi::Vulkan: api = QSGRendererInterface::Vulkan; break;
484 default: break;
485 }
486 }
487#endif
488 qCInfo(GStreamerLog) << "Resolved RHI backend:" << graphicsApiName(api) << "→ zero-copy path:"
489 << zeroCopyFamilyForApi(api);
