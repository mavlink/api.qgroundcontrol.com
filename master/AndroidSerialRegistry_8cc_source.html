14// Java gets a handle that is never used twice instead of a C++ pointer, so a report that arrives after its port
15// has closed finds nothing. A report looks the listener up and calls it under the read lock; unregisterPort()
16// takes the write lock, and so returns only when no report is still using the listener. The pattern follows Qt
17// Bluetooth's LowEnergyNotificationHub.
18static QReadWriteLock s_listenersLock;
19static QHash<PortHandle, PortListener*> s_listeners;
20static PortHandle s_lastHandle = kNoPort;
21
22PortHandle registerPort(PortListener* listener)
23{
24 if (!listener) {
25 qCWarning(AndroidSerialLog) << "registerPort called with null listener";
26 return kNoPort;
27 }
28
29 QWriteLocker locker(&s_listenersLock);
30
31 const PortHandle handle = ++s_lastHandle;
32 s_listeners.insert(handle, listener);
33 return handle;
34}
35
36void unregisterPort(PortHandle handle)
37{
38 QWriteLocker locker(&s_listenersLock);
39 s_listeners.remove(handle);
40}

42bool deliverNewData(PortHandle handle, const char* bytes, int length)
43{
44 QReadLocker locker(&s_listenersLock);
45 PortListener* const listener = s_listeners.value(handle, nullptr);
46 if (!listener) {
47 qCDebug(AndroidSerialLog) << "nativeDeviceNewData: stale token, port already closed";
48 return true;
49 }
50
51 return listener->newDataArrived(bytes, length);
52}
