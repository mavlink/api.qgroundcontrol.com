134QStringList VideoRecordingService::pruneStorage(const QString& directory, quint64 maxBytes)
135{
136 QStringList nameFilters{QStringLiteral("*.ass")};

139 nameFilters.append(QStringLiteral("*.") + VideoReceiver::fileExtension(format));
140 }
141
142 QDir dir(directory);
143 const QFileInfoList files = dir.entryInfoList(

145
146 quint64 total = 0;
147 for (const QFileInfo& file : files) {
148 total += static_cast<quint64>(file.size());
149 }
150
