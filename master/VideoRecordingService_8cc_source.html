154QStringList VideoRecordingService::pruneStorage(const QString& directory, quint64 maxBytes)
155{
156 QStringList nameFilters{QStringLiteral("*.ass")};
157 for (const VideoReceiver::FileFormat format :
158 {VideoReceiver::FileFormat::Mkv, VideoReceiver::FileFormat::Mov, VideoReceiver::FileFormat::Mp4}) {
159 nameFilters.append(QStringLiteral("*.") + VideoReceiver::fileExtension(format));
160 }
161
162 QDir dir(directory);
163 const QFileInfoList files = dir.entryInfoList(
164 nameFilters, QDir::Files | QDir::Readable | QDir::Writable | QDir::NoSymLinks, QDir::Time | QDir::Reversed);
165
166 quint64 total = 0;
167 for (const QFileInfo& file : files) {
168 total += static_cast<quint64>(file.size());
169 }
170
171 QStringList removed;
172 for (const QFileInfo& file : files) {
173 if (total < maxBytes) {
174 break;
175 }
176 total -= static_cast<quint64>(file.size());
177 const QString path = file.filePath();
178 if (QFile::remove(path)) {
179 qCDebug(VideoRecordingServiceLog) << "Removed old recording" << path;
180 removed.append(path);
181 } else {
182 qCWarning(VideoRecordingServiceLog) << "Failed to remove old recording" << path;
183 }
184 }
185 return removed;
186}
