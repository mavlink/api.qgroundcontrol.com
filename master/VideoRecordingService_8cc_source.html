191QStringList VideoRecordingService::pruneStorage(const QString& directory, quint64 maxBytes,
192 const QStringList& openFiles)
193{
194 QSet<QString> open;
195 for (const QString& file : openFiles) {
196 open.insert(QFileInfo(file).absoluteFilePath());
197 }
198 const QDateTime beingWrittenSince = QDateTime::currentDateTime().addDuration(-kBeingWritten);
199
200 QStringList nameFilters{QStringLiteral("*.ass")};
201 for (const VideoReceiver::FileFormat format :
202 {VideoReceiver::FileFormat::Mkv, VideoReceiver::FileFormat::Mov, VideoReceiver::FileFormat::Mp4}) {
203 nameFilters.append(QStringLiteral("*.") + VideoReceiver::fileExtension(format));
204 }
205
206 QDir dir(directory);
207 const QFileInfoList files = dir.entryInfoList(
208 nameFilters, QDir::Files | QDir::Readable | QDir::Writable | QDir::NoSymLinks, QDir::Time | QDir::Reversed);
209
210 quint64 total = 0;
211 for (const QFileInfo& file : files) {
212 total += static_cast<quint64>(file.size());
213 }
214
215 QStringList removed;
216 for (const QFileInfo& file : files) {
217 if (total < maxBytes) {
218 break;
219 }
220 if (open.contains(file.absoluteFilePath()) || (file.lastModified() > beingWrittenSince)) {
221 continue;
222 }
223 total -= static_cast<quint64>(file.size());
224 const QString path = file.filePath();
225 if (QFile::remove(path)) {
226 qCDebug(VideoRecordingServiceLog) << "Removed old recording" << path;
227 removed.append(path);
228 } else {
229 qCWarning(VideoRecordingServiceLog) << "Failed to remove old recording" << path;
230 }
231 }
232 return removed;
