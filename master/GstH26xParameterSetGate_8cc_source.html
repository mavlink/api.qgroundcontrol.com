23// Joining a stream mid-GOP, slices before the next parameter sets can't be decoded. GStreamer 1.28's h264parse warns
24// for every such slice and, when converting byte-stream to avc for the decoder, fails the pipeline with "No caps set"
25// if it ever pushes an access unit before it has both SPS and PPS, e.g. an SPS whose PPS a jitter buffer dropped.
26// The depayloader's DELTA_UNIT flag isn't enough either: a sliced encoder's IDR can be joined after its SPS/PPS.
27struct NalSummary
28{
29 bool vps = false;
30 bool sps = false;
31 bool pps = false;
32 bool slice = false;
33};
34
35NalSummary summarizeNals(GstBuffer* buffer, bool h265)
36{
37 NalSummary summary;
38 GstMapInfo map;
39 if (!gst_buffer_map(buffer, &map, GST_MAP_READ)) {
40 return summary;
41 }
42 const guint8* const data = map.data;
43 for (gsize i = 0; (i + 3) < map.size; ++i) {
44 if ((data[i] != 0) || (data[i + 1] != 0) || (data[i + 2] != 1)) {
45 continue;
46 }
47 const guint8 header = data[i + 3];
48 if (h265) {
49 const guint nalType = (header >> 1) & 0x3FU;
50 summary.vps |= (nalType == 32U);
51 summary.sps |= (nalType == 33U);
52 summary.pps |= (nalType == 34U);
53 summary.slice |= (nalType <= 31U);
54 } else {
55 const guint nalType = header & 0x1FU;
56 summary.sps |= (nalType == 7U);
57 summary.pps |= (nalType == 8U);
58 summary.slice |= ((nalType >= 1U) && (nalType <= 5U));
59 }
60 i += 2;
61 }
62 gst_buffer_unmap(buffer, &map);
63 return summary;
64}

98GstBuffer* prependMemories(GstBuffer* buffer, GstBuffer* prefix)
99{
100 buffer = gst_buffer_make_writable(buffer);
101 for (guint i = gst_buffer_n_memory(prefix); i > 0; --i) {
102 gst_buffer_prepend_memory(buffer, gst_buffer_get_memory(prefix, i - 1));
103 }
104 return buffer;

107GstPadProbeReturn holdUntilParameterSets(GstPad* pad, GstPadProbeInfo* info, gpointer data)
108{
109 GstBuffer* buffer = GST_PAD_PROBE_INFO_BUFFER(info);
110 if (!buffer) {
111 return GST_PAD_PROBE_OK;
112 }
113 // Length-prefixed avc/hvc1 output carries its parameter sets in the caps' codec_data.
114 if (!outputsByteStream(pad)) {
115 return GST_PAD_PROBE_REMOVE;
116 }
117
118 auto* gate = static_cast<ParameterSetGate*>(data);
119 const NalSummary nals = summarizeNals(buffer, gate->h265);
120 if (!nals.slice) {
121 if (nals.vps) {
122 gst_buffer_replace(&gate->vps, buffer);
123 }
124 if (nals.sps) {
125 gst_buffer_replace(&gate->sps, buffer);
126 }
127 if (nals.pps) {
128 gst_buffer_replace(&gate->pps, buffer);
129 }
130 return GST_PAD_PROBE_DROP;
131 }
132
133 const bool complete = (nals.sps || gate->sps) && (nals.pps || gate->pps) && (!gate->h265 || nals.vps || gate->vps);
134 if (!complete) {
135 return GST_PAD_PROBE_DROP;
136 }
137
138 // Prepend, in VPS/SPS/PPS order, the stored sets this buffer lacks; one buffer may hold several of them.
139 GstBuffer* const stored[] = {(gate->h265 && !nals.vps) ? gate->vps : nullptr, nals.sps ? nullptr : gate->sps,
140 nals.pps ? nullptr : gate->pps};
141 for (int i = 2; i >= 0; --i) {
142 const bool duplicate = (i > 0 && stored[i] == stored[i - 1]) || (i > 1 && stored[i] == stored[i - 2]);
143 if (stored[i] && !duplicate) {
144 buffer = prependMemories(buffer, stored[i]);
145 }
146 }
147 GST_PAD_PROBE_INFO_DATA(info) = buffer;
148 return GST_PAD_PROBE_REMOVE;
149}

153void configureH26xParser(GstElement* element)
154{
155 const char* factoryName = elementFactoryName(element);
156 if ((g_strcmp0(factoryName, "h264parse") == 0) || (g_strcmp0(factoryName, "h265parse") == 0)) {
157 // Recording can begin long after RTSP setup. Repeat codec parameter sets at every IDR so

164void configureH26xRtpDepayloader(GstElement* element, bool requestKeyframes)
165{
166 const char* factoryName = elementFactoryName(element);
167 if (!isH26xRtpDepayloader(factoryName)) {
168 return;

172 g_object_set(element, "request-keyframe", TRUE, nullptr);

174 if (const GstPadPtr srcPad = adoptPad(gst_element_get_static_pad(element, "src"))) {
175 gateOnParameterSets(srcPad.get(), g_strcmp0(factoryName, "rtph265depay") == 0);
176 }
