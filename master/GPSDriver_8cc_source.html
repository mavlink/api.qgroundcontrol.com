47bool GPSDriver::configure()
48{
49 _state->driver.reset();
50 GPSPx4Data::initialize(_state->position);
51 _state->satellites = {};
52 if (const QString error = gpsReceiverConfigError(_type, _config); !error.isEmpty()) {
53 qCWarning(GPSDriverLog) << error;
54 return false;
55 }
56
57 unsigned baudrate = _transport.fixedBaudrate();
58 const float headingOffset = _config.headingOffsetRadians.value_or(0.0f);
59 switch (_type) {
60 case GPSType::trimble:
61 _state->driver = std::make_unique<GPSDriverAshtech>(&callbackTrampoline, this, &_state->position,
62 &_state->satellites, headingOffset);
63 baudrate = 115200;
64 break;
65 case GPSType::septentrio:
66 _state->driver = std::make_unique<GPSDriverSBF>(&callbackTrampoline, this, &_state->position,
67 &_state->satellites, headingOffset);
68 break;
69 case GPSType::ublox: {
70 const GPSDriverUBX::Settings settings{
71 .dynamic_model = static_cast<uint8_t>(_config.dynamicModel.value_or(7)),
72 .dgnss_timeout = 0,
73 .min_cno = 0,
74 .min_elev = 0,
75 .output_rate = 0,
76 .heading_offset = headingOffset,
77 .uart2_baudrate = 57600,
78 .ppk_output = false,
79 .jam_det_sensitivity_hi = false,
80 .mode = GPSDriverUBX::UBXMode::Normal,
81 };
82 _state->driver = std::make_unique<GPSDriverUBX>(GPSDriverUBX::Interface::UART, &callbackTrampoline, this,
83 &_state->position, &_state->satellites, settings);
84 break;
85 }
86 case GPSType::femto:
87 _state->driver = std::make_unique<GPSDriverFemto>(&callbackTrampoline, this, &_state->position,
88 &_state->satellites, headingOffset);
89 break;
90 }
91
92 if (!_state->driver) {
93 qCWarning(GPSDriverLog) << "Unsupported GPS type:" << static_cast<int>(_type);
94 return false;
95 }
96
97 if (_config.role == GPSReceiverConfig::Role::RTKBase) {
98 const auto& base = _config.base;
99 if (base.useFixedBase) {
100 _state->driver->setBasePosition(base.fixedBaseLatitude, base.fixedBaseLongitude,
101 base.fixedBaseAltitudeMeters, base.fixedBaseAccuracyMeters * 1000.0f);
102 } else {
103 _state->driver->setSurveyInSpecs(static_cast<uint32_t>(base.surveyInAccMeters * 10000.0),
104 static_cast<uint32_t>(base.surveyInDurationSecs));
105 }
106 }
107
108 GPSHelper::GPSConfig gpsConfig{};
109 gpsConfig.output_mode =
110 _config.role == GPSReceiverConfig::Role::RTKBase ? GPSHelper::OutputMode::RTCM : GPSHelper::OutputMode::GPS;
111 gpsConfig.gnss_systems = static_cast<GPSHelper::GNSSSystemsMask>(_config.constellationMask);
112
113 if (_state->driver->configure(baudrate, gpsConfig) != 0) {
114 qCWarning(GPSDriverLog) << "Driver configuration failed for type" << static_cast<int>(_type);
115 _state->driver.reset();
116 return false;
117 }

119 GPSPx4Data::initialize(_state->position);
120 return true;

123int GPSDriver::receive(unsigned timeoutMs)
124{
125 if (!_state->driver) {
126 return -1;
127 }
128
129 const int ret = _state->driver->receive(timeoutMs);
130 if (ret < 0) {
131 return ret;
132 }
133
134 if ((ret & 0x01) && _sinks.onPosition) {
135 _sinks.onPosition(GPSPx4Data::position(_state->position));
136 }
137 if ((ret & 0x02) && _sinks.onSatelliteInfo) {
138 _sinks.onSatelliteInfo(GPSPx4Data::satellites(_state->satellites, _type));
139 }
140 return ret;

143int GPSDriver::handleCallback(int type, void* data1, int data2)
144{
145 switch (static_cast<GPSCallbackType>(type)) {
146 case GPSCallbackType::readDeviceData: {
147 int timeoutMs = 0;
148 memcpy(&timeoutMs, data1, sizeof(timeoutMs)); // px4 packs the timeout into data1's first bytes (unaligned)
149 const auto result = _transport.read(static_cast<uint8_t*>(data1), data2, timeoutMs);
150 if (result.status == GPSReadStatus::Data && result.bytesRead >= 0 && result.bytesRead <= data2) {
151 return result.bytesRead;
152 }
153 return result.status == GPSReadStatus::TimedOut ? 0 : -1;
154 }
155 case GPSCallbackType::writeDeviceData: {
156 const auto result = _transport.write(static_cast<const uint8_t*>(data1), data2);
157 return result.status == GPSWriteStatus::Completed && result.acceptedBytes == data2 &&
158 result.writtenBytes == data2 && result.uncertainBytes() == 0
159 ? data2
160 : -1;
161 }
162 case GPSCallbackType::setBaudrate:
163 return _transport.setBaudrate(static_cast<unsigned>(data2)) ? 0 : -1;
164 case GPSCallbackType::gotRTCMMessage:
165 if (!data1 || data2 <= 0) {
166 qCWarning(GPSDriverLog) << "Invalid RTCM callback payload";
167 return -1;
168 }
169 if (_sinks.onRTCM) {
170 _sinks.onRTCM({static_cast<const uint8_t*>(data1), static_cast<std::size_t>(data2)});
171 }
172 break;
173 case GPSCallbackType::surveyInStatus:
174 if (data1 && _sinks.onSurveyIn) {
175 const SurveyInStatus* const status = static_cast<const SurveyInStatus*>(data1);
176 GPSSurveyReport out;
177 out.latitudeDegrees = status->latitude;
178 out.longitudeDegrees = status->longitude;
179 out.altitudeEllipsoidMeters = status->altitude;
180 // Ashtech and Femto use zero for unknown accuracy; UBX can round a valid value to zero.
181 if (status->mean_accuracy != 0 || (_type != GPSType::trimble && _type != GPSType::femto)) {
182 out.meanAccuracyMeters = static_cast<double>(status->mean_accuracy) / 1000.0;
183 }
184 out.duration = std::chrono::seconds(status->duration);
185 out.valid = status->flags & 0x01;
186 out.active = (status->flags >> 1) & 0x01;
187 _sinks.onSurveyIn(out);
188 }
189 break;
190 case GPSCallbackType::setClock:
191 default:
192 break;
193 }
194
195 return 0;
196}
