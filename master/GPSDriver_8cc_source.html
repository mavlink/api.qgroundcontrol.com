27QString mismatchHint(const GPSReceiverSignatures& signatures, GPSType configured)
28{
29 const auto& match = signatures.match();
30 if (!match || match->family->type == configured) {
31 return {};
32 }
33 QString evidence = match->evidence;
34 evidence[0] = evidence[0].toUpper();
35 return QStringLiteral("%1 were received; this looks like a %2 receiver")
36 .arg(evidence, gpsReceiverName(match->family->type));
37}

96bool GPSDriver::configure()
97{
98 _state = std::make_unique<State>();
99 if (const QString error = gpsReceiverConfigError(_type, _config); !error.isEmpty()) {
100 _state->configurationError = error;
101 qCWarning(GPSDriverLog) << error;
102 return false;
103 }
104 if (_type != GPSType::passive) {
105 const auto& mode = _config.base.mode;
106 if (std::holds_alternative<GPSBaseStationConfig::Fixed>(mode)) {
107 qCDebug(GPSDriverLog) << "Fixed base configured";
108 } else if (const auto* averaging = std::get_if<GPSBaseStationConfig::ReceiverAveraging>(&mode)) {
109 qCDebug(GPSDriverLog) << "Receiver-managed averaging maximum duration (s):"
110 << averaging->maximumDuration.count();
111 } else if (const auto* survey = std::get_if<GPSBaseStationConfig::SurveyIn>(&mode)) {
112 qCDebug(GPSDriverLog) << "Survey-in accuracy (m):" << survey->accuracyMeters
113 << "minimum duration (s):" << survey->duration.count();
114 }
115 }
116
117 GPSRuntimeIO io;
118 io.clock = _clock;
119 io.read = [this](std::span<uint8_t> bytes, GPSDeadline deadline) {
120 const auto result = _transport.read(bytes, deadline.remaining(_clock.nowUs()));
121 const bool data = result.status == GPSReadStatus::Data && result.bytesRead > 0 &&
122 static_cast<size_t>(result.bytesRead) <= bytes.size();
123 _state->cycle.activity |= data;
124 if (data && _state->signatures && _state->configuring) {
125 _state->signatures->push(bytes.first(static_cast<size_t>(result.bytesRead)));
126 }
127 return result;
128 };
129 io.write = [this](std::span<const uint8_t> bytes, GPSDeadline deadline) {
130 // Do not submit another part of a multipart command after its absolute deadline.
131 const auto remaining = deadline.remaining(_clock.nowUs());
132 if (remaining == std::chrono::milliseconds::zero()) {
133 return GPSWriteResult{GPSWriteStatus::TimedOut};
134 }
135 // The transport waits in real time; only the remaining time carries over from the driver's clock.
136 const QDeadlineTimer transportDeadline = deadline.untilUs == UINT64_MAX
137 ? QDeadlineTimer(QDeadlineTimer::Forever, Qt::PreciseTimer)
138 : QDeadlineTimer(remaining, Qt::PreciseTimer);
139 return _transport.write(bytes, transportDeadline);
140 };
141 io.setBaudrate = [this](unsigned baud) {
142 if (_transport.setBaudrate(baud)) {
143 return GPSBaudStatus::Configured;
144 }
145 return _transport.fatalError() ? GPSBaudStatus::Error : GPSBaudStatus::Unsupported;
146 };
147 io.cancelToken = _transport.cancelToken();
148 GPSRuntimeObserver observer;
149 observer.commandFinished = [this](const GPSConfigurationEvidence& result) {
150 if (_state->configuring) {
151 _state->evidence.push_back(result);
152 }
153 };
154 observer.decoded = [this](const GPSEventBatch& batch) { _publish(batch); };
155
156 unsigned baudrate = _config.baudRate ? _config.baudRate : _transport.fixedBaudrate();
157 if (_config.baudRate && _transport.fixedBaudrate() && _config.baudRate != _transport.fixedBaudrate()) {
158 _state->configurationError = QStringLiteral("Selected baud rate differs from the transport's fixed baud rate");
159 qCWarning(GPSDriverLog) << _state->configurationError;
160 return false;
161 }
162 if (_type == GPSType::automatic) {
163 return _configureDetected(std::move(io), std::move(observer), baudrate);
164 }
165 const GPSReceiverFamily* family = gpsReceiverFamily(_type);
166 if (!family) {
167 _state->configurationError =
168 QStringLiteral("Unsupported GPS type: %1")
169 .arg(QLatin1StringView(QMetaEnum::fromType<GPSType>().valueToKey(static_cast<int>(_type))));
170 qCWarning(GPSDriverLog) << "Unsupported GPS type:" << _type;
171 return false;
172 }
173 // Without a selected rate or a fixed link rate, the family may prefer one to detection.
174 if (family->autoBaudRate && !baudrate) {
175 baudrate = family->autoBaudRate;
176 }
177 if (_type != GPSType::passive) {
178 _state->signatures = std::make_unique<GPSReceiverSignatures>(gpsReceiverFamilies());
179 }
180 return _configureFamily(*family, std::move(io), std::move(observer),
181 {.base = _config.base, .allowPersistentChanges = _config.allowPersistentChanges}, baudrate,
182 {});
183}

185bool GPSDriver::_configureDetected(GPSRuntimeIO io, GPSRuntimeObserver observer, unsigned baudrate)
186{
187 _state->configuring = true;
188 const GPSReceiverDetection detection =
189 GPSReceiverDetector(gpsReceiverFamilies(), io, observer.commandFinished).detect(baudrate);
190 _state->configuring = false;
191 if (!detection.found()) {
192 _state->configurationError = !detection.errorDetail.isEmpty() ? detection.errorDetail
193 : detection.error == GPSProtocolError::Cancelled
194 ? QStringLiteral("Receiver detection cancelled")
195 : QStringLiteral("No supported receiver was identified");
196 qCWarning(GPSDriverLog) << "Receiver detection failed:" << _state->configurationError;
197 return false;
198 }
199 const GPSType detected = detection.family->type;
200 const QString name = gpsReceiverName(detected);
201 if (_sinks.onReceiverDetected) {
202 _sinks.onReceiverDetected(detected);
203 }
204 bool compactFallback = false;
205 const GPSReceiverConfig config = gpsReceiverConfigForDetected(detected, _config, &compactFallback);
206 if (compactFallback) {
207 qCWarning(GPSDriverLog).noquote()
208 << QStringLiteral("Detected %1 receiver has no compact (MSM4) RTCM option; it sends its standard set")
209 .arg(name);
210 }
211 if (const QString error = gpsDetectedReceiverConfigError(detected, config); !error.isEmpty()) {
212 _state->configurationError = error;
213 qCWarning(GPSDriverLog).noquote() << error;
214 return false;
215 }
216 // Without a fixed or selected rate, the family's baud search starts at the detected rate and moves the link to
217 // the rate the family prefers.
218 return _configureFamily(
219 *detection.family, std::move(io), std::move(observer),
220 {.base = config.base, .allowPersistentChanges = config.allowPersistentChanges, .detectedBaud = detection.baud},
221 baudrate, QStringLiteral("Detected %1 receiver at %2 baud").arg(name).arg(detection.baud));
222}

224bool GPSDriver::_configureFamily(const GPSReceiverFamily& family, GPSRuntimeIO io, GPSRuntimeObserver observer,
225 const GPSConfig& config, unsigned baudrate, const QString& detected)
226{
227 _state->runtime = std::make_unique<GPSProtocolRuntime>(family, std::move(io), std::move(observer));
228 _state->configuring = true;
229 const bool configured = _state->runtime->configure(config, baudrate);
230 _state->configuring = false;
231 const auto signatures = std::exchange(_state->signatures, nullptr);
232 if (!configured) {
233 QString error = _state->runtime->errorDetail();
234 if (error.isEmpty()) {
235 error = QStringLiteral("Receiver configuration failed");
236 }
237 if (!detected.isEmpty()) {
238 error = QStringLiteral("%1: %2").arg(detected, error);
239 } else if (const QString hint = signatures ? mismatchHint(*signatures, family.type) : QString();
240 !hint.isEmpty()) {
241 error += (error.endsWith(u'.') ? QStringLiteral(" ") : QStringLiteral(". ")) + hint;
242 }
243 _state->configurationError = error;
244 _state->consentRequired = _state->runtime->error() == GPSProtocolError::ConsentRequired;
245 qCWarning(GPSDriverLog) << "Driver configuration failed for type" << family.type << error;
246 _state->runtime.reset();
247 return false;
248 }
249 _state->configurationError.clear();
250 return true;
251}

253GPSReceiveResult GPSDriver::receiveOutcome(std::chrono::milliseconds timeout)
254{
255 if (!_state->runtime) {
256 return {GPSProtocolError::InvalidArgument};
257 }
258 _state->cycle = {};
259 _publishExpiredSatellites();
260 const GPSReceiveUpdates decoded = _state->runtime->receive(timeout);
261 _publishExpiredSatellites();
262 if (_state->runtime->error() != GPSProtocolError::None) {
263 return {.error = _state->runtime->error(),
264 .updates = _state->cycle.updates,
265 .detail = _state->runtime->errorDetail()};
266 }
267 if (_transport.isCancelled()) {
268 return {.error = GPSProtocolError::Cancelled, .updates = _state->cycle.updates};
269 }
270 if (_transport.fatalError()) {
271 // An empty read reports why the link failed without consuming input.
272 const GPSReadResult failure = _transport.read({}, std::chrono::milliseconds::zero());
273 return {.error = GPSProtocolError::Transport, .updates = _state->cycle.updates, .detail = failure.detail};
274 }
275 return {.liveness = _state->cycle.usefulData ? GPSReceiveLiveness::Data
276 : _state->cycle.activity || decoded != GPSReceiveUpdates{} ? GPSReceiveLiveness::Activity
277 : GPSReceiveLiveness::Idle,
278 .updates = _state->cycle.updates};
279}
