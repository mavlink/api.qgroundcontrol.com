131bool GPSDriver::configure()
132{
133 if (_operationInProgress) {
134 _state->configurationError = QStringLiteral("Receiver operation already in progress; configuration rejected");
135 qCWarning(GPSDriverLog) << _state->configurationError;
136 return false;
137 }
138 const QScopedValueRollback operation(_operationInProgress, true);
139 _state = std::make_unique<State>();
140 if (const QString error = gpsReceiverConfigError(_type, _config); !error.isEmpty()) {
141 _state->configurationError = error;
142 qCWarning(GPSDriverLog) << error;
143 return false;
144 }
145
146 GPSProtocolIO io;
147 io.nowUs = MonotonicClock::nowUs;
148 io.read = [this](std::span<uint8_t> bytes, GPSDeadline deadline) {
149 const int timeout = deadline.remainingMilliseconds(MonotonicClock::nowUs());
150 const auto result = _transport.read(bytes.data(), static_cast<int>(bytes.size()), timeout);
151 _state->activity |= result.status == GPSReadStatus::Data && result.bytesRead > 0;
152 return result;
153 };
154 io.write = [this](std::span<const uint8_t> bytes, GPSDeadline deadline) {
155 const int remaining = deadline.remainingMilliseconds(MonotonicClock::nowUs());
156 // Do not submit another part of a multipart command after its absolute deadline.
157 if (_transport.isCancelled() || remaining == 0) {
158 return GPSWriteResult{_transport.isCancelled() ? GPSWriteStatus::Cancelled : GPSWriteStatus::TimedOut};
159 }
160 return _transport.writeConfiguration(bytes.data(), static_cast<int>(bytes.size()), deadline.toQDeadlineTimer());
161 };
162 io.setBaudrate = [this](unsigned baud) {
163 if (_transport.isCancelled()) {
164 return GPSBaudStatus::Cancelled;
165 }
166 return _transport.setBaudrate(baud) ? GPSBaudStatus::Configured : GPSBaudStatus::Error;
167 };
168 io.wait = [this](std::chrono::microseconds duration) {
169 const auto until = std::chrono::steady_clock::now() + duration;
170 while (!_transport.isCancelled() && std::chrono::steady_clock::now() < until) {
171 std::this_thread::sleep_for(std::min(until - std::chrono::steady_clock::now(),
172 std::chrono::steady_clock::duration(std::chrono::milliseconds(20))));
173 }
174 return !_transport.isCancelled();
175 };
176 io.log = [](GPSProtocolLogLevel level, QStringView message) {
177 if (level == GPSProtocolLogLevel::Debug) {
178 qCDebug(GPSNativeDriversLog) << message;
179 } else {
180 qCWarning(GPSNativeDriversLog) << message;
181 }
182 };
183 io.commandFinished = [this](const GPSCommandResult& result) {
184 if (_state->configuring) {
185 _state->evidence.push_back(result.evidence);
186 }
187 };
188 io.decoded = [this](const GPSDecodedBatch& batch) {
189 _state->activity |= !batch.events.empty();
190 for (const auto& event : batch.events) {
191 std::visit(
192 [this](const auto& report) {
193 using Report = std::decay_t<decltype(report)>;
194 if constexpr (std::is_same_v<Report, GPSNativeIntegrityReport>) {
195 _state->integrity = report;
196 } else if constexpr (std::is_same_v<Report, GPSNativePositionReport>) {
197 if (!_state->configuring) {
198 _state->usefulData = true;
199 _state->updates |= 1;
200 if (_sinks.onPosition) {
201 _sinks.onPosition(
202 GPSNativeData::position(report, _state->integrity, MonotonicClock::nowUs()));
203 }
204 }
205 } else if constexpr (std::is_same_v<Report, GPSNativeSatelliteReport>) {
206 if (!_state->configuring) {
207 _state->usefulData = true;
208 _state->updates |= 2;
209 const auto snapshot = _state->satelliteSnapshot.update(report, MonotonicClock::nowUs());
210 if (_sinks.onSatelliteInfo) {
211 _sinks.onSatelliteInfo(snapshot);
212 }
213 }
214 } else if constexpr (std::is_same_v<Report, GPSSatelliteUsageReport>) {
215 if (!_state->configuring) {
216 _state->usefulData = true;
217 _state->updates |= 2;
218 if (_sinks.onSatelliteUsage) {
219 _sinks.onSatelliteUsage(report);
220 }
221 }
222 } else if constexpr (std::is_same_v<Report, GPSNativeSurveyReport>) {
223 _state->usefulData = true;
224 if (_sinks.onSurveyIn) {
225 _sinks.onSurveyIn(GPSNativeData::survey(report));
226 }
227 } else if constexpr (std::is_same_v<Report, GPSRTCMReport>) {
228 _state->usefulData = true;
229 if (!_state->configuring && _sinks.onRTCM) {
230 _sinks.onRTCM(std::span(report.bytes).first(report.size));
231 }
232 }
233 },
234 event);
235 }
236 };
237
238 unsigned baudrate = _config.baudRate ? _config.baudRate : _transport.fixedBaudrate();
239 if (_config.baudRate && _transport.fixedBaudrate() && _config.baudRate != _transport.fixedBaudrate()) {
240 _state->configurationError = QStringLiteral("Selected baud rate differs from the transport's fixed baud rate");
241 qCWarning(GPSDriverLog) << _state->configurationError;
242 return false;
243 }
244 const auto factory = findProtocolFactory(_type);
245 if (factory == PROTOCOL_FACTORIES.end()) {
246 _state->configurationError = QStringLiteral("Unsupported GPS type: %1").arg(static_cast<int>(_type));
247 qCWarning(GPSDriverLog) << "Unsupported GPS type:" << static_cast<int>(_type);
248 return false;
249 }
250 _state->driver = factory->create(std::move(io), &_state->position, &_state->satellites);
251 if (_type == GPSType::trimble && !_config.baudRate) {
252 baudrate = 115200;
253 }
254 GPSProtocol::GPSConfig config{};
255 config.base = _config.base;
256 const bool asciiReceiver = _type == GPSType::unicore || _type == GPSType::quectel || _type == GPSType::passive;
257 config.dynamicModel = static_cast<uint8_t>(_config.dynamicModel.value_or(asciiReceiver ? 0 : 7));
258 config.output_mode =
259 _config.role == GPSReceiverConfig::Role::RTKBase ? GPSProtocol::OutputMode::RTCM : GPSProtocol::OutputMode::GPS;
260 config.gnss_systems = static_cast<GPSProtocol::GNSSSystemsMask>(_config.constellationMask);
261 config.allowPersistentChanges = _config.allowPersistentChanges;
262 _state->configuring = true;
263 const int result = _state->driver->configure(baudrate, config);
264 _state->driver->finishConfigurationEvidence();
265 if (result >= 0) {
266 // Configuration can finish by publishing a fixed-base or survey-start event without another read.
267 _state->driver->consume({});
268 }
269 _state->configuring = false;
270 if (result < 0) {
271 _state->configurationError = _state->driver->ioErrorDetail();
272 if (_state->configurationError.isEmpty()) {
273 _state->configurationError = QStringLiteral("Receiver configuration failed");
274 }
275 qCWarning(GPSDriverLog) << "Driver configuration failed for type" << static_cast<int>(_type)
276 << _state->configurationError;
277 _state->driver.reset();
278 return false;
279 }
280 _state->configurationError.clear();
281 return true;
282}

284GPSReceiveResult GPSDriver::receiveOutcome(unsigned timeoutMs)
285{
286 if (_operationInProgress) {
287 const QString detail = QStringLiteral("Receiver operation already in progress; receive rejected");
288 qCWarning(GPSDriverLog) << detail;
289 return {GPSReceiveStatus::Busy, 0, -EBUSY, detail};
290 }
291 const QScopedValueRollback operation(_operationInProgress, true);
292 if (!_state->driver) {
293 return {GPSReceiveStatus::NotConfigured, 0, -1};
294 }
295 _state->updates = 0;
296 _state->usefulData = false;
297 _state->activity = false;
298 _publishExpiredSatellites();
299 const int result = _state->driver->receive(timeoutMs);
300 _publishExpiredSatellites();
301 const int error = _state->driver->ioError() ? _state->driver->ioError() : (result < -1 ? result : 0);
302 if (error) {
303 return {error == -ECANCELED ? GPSReceiveStatus::Cancelled
304 : error == -EPROTO ? GPSReceiveStatus::ProtocolError
305 : GPSReceiveStatus::TransportError,
306 _state->updates, error, _state->driver->ioErrorDetail()};
307 }
308 if (_transport.isCancelled()) {
309 return {GPSReceiveStatus::Cancelled, _state->updates, -ECANCELED};
310 }
311 if (_transport.fatalError()) {
312 return {GPSReceiveStatus::TransportError, _state->updates, -EIO};
313 }
314 return {_state->usefulData ? GPSReceiveStatus::Data
315 : _state->activity || result > 0 ? GPSReceiveStatus::Activity
316 : GPSReceiveStatus::Idle,
317 _state->updates, 0};
318}
