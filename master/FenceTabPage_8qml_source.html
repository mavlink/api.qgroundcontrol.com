101 // A GeoMap viewport corner can point into the sky in a 3D tilt: moves it toward the center until it meets the

103 function _groundCoordinateInViewport(corner, center) {
104 for (let step = 0; step < 10; step++) {
105 const fraction = step / 10
106 const coordinate = editorMap.toCoordinate(

118 while (normalized > 180) {

127 // Whether point is inside the convex quadrilateral quad, edges included; a point sits on an edge within a tolerance
128 // of about a centimeter (the points are {x: longitude offset, y: latitude} in degrees)
129 function _insideQuadrilateral(point, quad) {
130 const tolerance = 1e-7
131 let side = 0
132 for (let i = 0; i < quad.length; i++) {
133 const from = quad[i]
134 const to = quad[(i + 1) % quad.length]
135 const edgeX = to.x - from.x
136 const edgeY = to.y - from.y
137 const edgeLength = Math.hypot(edgeX, edgeY)
138 if (edgeLength === 0) {
139 continue
140 }
141 const distance = (edgeX * (point.y - from.y) - edgeY * (point.x - from.x)) / edgeLength
142 if (distance > tolerance) {
143 if (side < 0) {
144 return false
145 }
146 side = 1
147 } else if (distance < -tolerance) {
148 if (side > 0) {
149 return false
150 }
151 side = -1
152 }
153 }
154 return true
155 }

157 // Geographic bounds {north, south, west, east} of a rectangle, centered in the viewport, that lies within the
158 // ground the viewport shows. The controller adds fences inside the bounds it is given, which the bounds of the
159 // viewport corners are not once the map is rotated or tilted: they enclose the ground shown instead of lying
160 // within it. The rectangle is shrunk from them until its corners are inside the quadrilateral of the corners'
161 // ground. This only compares ground picks with each other, so it does not depend on the map projecting back to
162 // the screen. Longitudes are taken relative to the viewport center so a viewport over the dateline works. Null if
163 // the viewport shows no ground.
164 function _viewportGroundBounds() {
165 const viewport = editorMap.centerViewport
166 const centerPoint = Qt.point(viewport.x + viewport.width / 2, viewport.y + viewport.height / 2)
167 const corners = [
168 Qt.point(viewport.x, viewport.y),
169 Qt.point(viewport.x + viewport.width, viewport.y),
170 Qt.point(viewport.x + viewport.width, viewport.y + viewport.height),
171 Qt.point(viewport.x, viewport.y + viewport.height)
172 ]
173 const cornerCoordinates = []
174 for (const corner of corners) {
175 const coordinate = _groundCoordinateInViewport(corner, centerPoint)
176 if (!coordinate) {
177 return null
178 }
179 cornerCoordinates.push(coordinate)
180 }
181 // Relative to the center, a viewport of any width up to the whole world keeps its corners within a half
182 // turn of it on either side
183 const centerCoordinate = editorMap.toCoordinate(centerPoint, false /* clipToViewPort */)
184 const referenceLongitude = centerCoordinate.isValid ? centerCoordinate.longitude : cornerCoordinates[0].longitude
185 const quad = cornerCoordinates.map(
186 coordinate => Qt.point(_normalizedLongitude(coordinate.longitude - referenceLongitude), coordinate.latitude))
187 const minX = Math.min(...quad.map(point => point.x))
188 const maxX = Math.max(...quad.map(point => point.x))
189 const minY = Math.min(...quad.map(point => point.y))
190 const maxY = Math.max(...quad.map(point => point.y))
191 if (maxX - minX < 1e-9 || maxY - minY < 1e-9) {
192 return null
193 }
194
195 const center = centerCoordinate.isValid ? Qt.point(0, centerCoordinate.latitude)
196 : Qt.point((minX + maxX) / 2, (minY + maxY) / 2)
197 const halfX = Math.max(maxX - center.x, center.x - minX)
198 const halfY = Math.max(maxY - center.y, center.y - minY)
199
200 for (let scale = 1; scale > 0.01; scale *= 0.9) {
201 const west = center.x - halfX * scale
202 const east = center.x + halfX * scale
203 const south = center.y - halfY * scale
204 const north = center.y + halfY * scale
205 if ([Qt.point(west, south), Qt.point(east, south), Qt.point(east, north), Qt.point(west, north)]
206 .every(corner => _insideQuadrilateral(corner, quad))) {
207 return {
208 north: Math.min(north, 90),
209 south: Math.max(south, -90),
210 west: _normalizedLongitude(referenceLongitude + west),
211 east: _normalizedLongitude(referenceLongitude + east)
212 }
213 }
214 }
215 return null
216 }
