36void glib_log_handler(const gchar* log_domain, GLogLevelFlags log_level, const gchar* message, gpointer user_data)
37{
38 Q_UNUSED(user_data);
39 const QString domain = log_domain ? QString::fromUtf8(log_domain) : QStringLiteral("GLib");
40 const QString msg = QString::fromUtf8(message);
41
42 if (msg.contains(QStringLiteral("External plugin loader failed"))) {
43 s_externalPluginLoaderFailed.store(true);
44 }
45
46 if (msg.contains(QStringLiteral("pygobject initialization failed"), Qt::CaseInsensitive)) {
47 qCDebug(GStreamerLoggingLog) << domain << msg;
48 return;
49 }
50
51 switch (log_level & G_LOG_LEVEL_MASK) {
52 case G_LOG_LEVEL_ERROR:
53 case G_LOG_LEVEL_CRITICAL:
54 qCCritical(GStreamerLoggingLog) << domain << msg;
55 break;
56 case G_LOG_LEVEL_WARNING:
57 qCWarning(GStreamerLoggingLog) << domain << msg;
58 break;
59 case G_LOG_LEVEL_MESSAGE:
60 case G_LOG_LEVEL_INFO:
61 qCInfo(GStreamerLoggingLog) << domain << msg;
62 break;
63 case G_LOG_LEVEL_DEBUG:
64 default:
65 qCDebug(GStreamerLoggingLog) << domain << msg;
66 break;
67 }
68}

97void qtGstLog(GstDebugCategory* category, GstDebugLevel level, const gchar* file, const gchar* function, gint line,
98 GObject* object, GstDebugMessage* message, gpointer data)
99{
100 Q_UNUSED(category);
101 Q_UNUSED(data);
102
103 // gst_debug_log() has already applied the category threshold.
104 QtMsgType type = QtDebugMsg;
105 switch (level) {
106 case GST_LEVEL_ERROR:
107 type = QtCriticalMsg;
108 break;
109 case GST_LEVEL_WARNING:
110 type = QtWarningMsg;
111 break;
112 case GST_LEVEL_FIXME:
113 case GST_LEVEL_INFO:
114 type = QtInfoMsg;
115 break;
116 default:
117 break;
118 }
119 if (!GStreamerAPILog().isEnabled(type)) {
120 return;
121 }
122
123 const GStreamer::GCharPtr objectInfo =
124 object ? GStreamer::adoptGChar(gst_info_strdup_printf("%" GST_PTR_FORMAT " ", object)) : nullptr;
125 const char* const prefix = objectInfo ? objectInfo.get() : "";
126 qt_message_output(type, QMessageLogContext(file, line, function, GStreamerAPILog().categoryName()),
127 QString::fromUtf8(prefix) + QString::fromUtf8(gst_debug_message_get(message)));
128}

130void configureDebugLogging(std::optional<int> level)
131{
132 if (nativeDebugLogRequested()) {
133 // Keep GStreamer's own output; a bridge left by an earlier call would duplicate every line.
134 gst_debug_remove_log_function(GStreamer::qtGstLog);
135 } else {
136 gst_debug_remove_log_function(gst_debug_log_default);
137 gst_debug_remove_log_function(GStreamer::qtGstLog);
138 gst_debug_add_log_function(GStreamer::qtGstLog, nullptr, nullptr);
139 }
140
141 if (!qEnvironmentVariableIsEmpty("GST_DEBUG")) {
142 return;
143 }
144
145 // A GStreamer development build (nano > 0, as the official 1.28.8 Windows SDK is) defaults to GST_LEVEL_ERROR
146 // rather than GST_LEVEL_NONE, so the setting's default is applied explicitly.
147 setDebugLevel(level.value_or(GST_LEVEL_NONE));
148}
