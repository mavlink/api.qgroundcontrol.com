272void GPSReceiver::_endSession(GPSConnectionError error, const QString& detail, bool portRemoved)
273{
274 // The family an Automatic session detected, else the selected one.
275 const auto* family = _session ? gpsReceiverDescriptorForManufacturer(_session->manufacturer()) : nullptr;
276 _retireSession();
277 _resetStatus();
278 _setPersistentChangesAllowed(false);
279 const GPSReceiverSessionOutcome outcome = _connection->sessionEnded(portRemoved);
280 if (portRemoved) {
281 qCDebug(GPSReceiverLog) << "Receiver serial device removed";
282 }
283 // Automatic connections never pass the flash-save consent this receiver asked for.
284 if (outcome == GPSReceiverSessionOutcome::AutoRetrying && error == GPSConnectionError::ConsentRequired &&
285 !detail.isEmpty() && family) {
286 qCWarning(GPSReceiverLog) << "GPS receiver did not accept configuration";
287 _setError(GPSReceiverMessages::consentRefused(detail, family->type));
288 return;
289 }
290 if (outcome == GPSReceiverSessionOutcome::AutoRetrying || outcome == GPSReceiverSessionOutcome::Retrying) {
291 qCWarning(GPSReceiverLog) << "GPS receiver session ended:" << error << detail;
292 } else if (outcome == GPSReceiverSessionOutcome::None && !portRemoved) {
293 logConnectionError(error, detail);
294 }
295 _setError(GPSReceiverMessages::sessionEnded(outcome, error, detail, portRemoved));
296}

357void GPSReceiver::_updateSerialPortEntries()
358{
359 // Just after a scan the manager returns that inventory without scanning again.
360 const QList<SerialPortManager::Port> available = _serialPorts->availablePorts();
361 QList<GPSSerialPortEntry> entries;
362 entries.reserve(available.size());
363 for (const auto& port : available) {
364 // The USB product string, else the name of a known board.
365 const QString name = port.description.trimmed().isEmpty() ? port.boardName : port.description.trimmed();
366 //: %1 is a serial device's name, such as u-blox GNSS receiver; %2 its path, such as /dev/ttyACM0 or COM3
367 QString label = name.isEmpty() ? port.systemLocation : tr("%1 – %2").arg(name, port.systemLocation);
368 if (_serialPorts->isPortReserved(port.systemLocation) && _serialClaims.value(port.systemLocation).expired()) {
369 //: %1 is a serial device that another QGroundControl connection, such as a vehicle link, uses
370 label = tr("%1 (in use)").arg(label);
371 }
372 entries.append({.value = port.systemLocation, .label = label});
373 }
374 if (entries != _serialPortEntries) {
375 _serialPortEntries = std::move(entries);
376 emit serialPortEntriesChanged();
377 }

538bool GPSReceiver::_connectReceiver(GPSType type, GPSReceiverWorker::TransportFactory transportFactory,
539 const QString& sourceInstance, uint32_t baudRate, bool allowPersistentChanges,
540 const QString& serialDevice, const QString& endpoint)
541{
542 QString configError;
543 GPSReceiverConfig config =
544 gpsReceiverConfigFor(_configuration.base, type, baudRate, allowPersistentChanges, &configError);
545 if (!configError.isEmpty()) {
546 _setError(configError);
547 return false;
548 }
549 _retireSession();
550 _resetStatus();
551 _setError();
552 const RTKSettings::ReceiverRole role =
553 type == GPSType::passive ? RTKSettings::Passive : RTKSettings::ConfiguredBase;
554 auto* const session = new GPSReceiverSession(
555 ++_sessionCount,
556 {.type = type,
557 .role = role,
558 .forwardsCorrections = role == RTKSettings::ConfiguredBase || _configuration.forwardReceiverRtcm,
559 .config = std::move(config),
560 .serialDevice = serialDevice,
561 .endpoint = endpoint},
562 this);
563 session->registerCorrections(_correctionManager, sourceInstance);
564 _session = session;
565 connect(session, &GPSReceiverSession::satelliteInfoUpdated, this, &GPSReceiver::_onSatelliteInfoUpdated);
566 connect(session, &GPSReceiverSession::positionUpdated, this, &GPSReceiver::_onPositionUpdated);
567 connect(session, &GPSReceiverSession::surveyInStatusUpdated, this, &GPSReceiver::_onGPSSurveyReport);
568 connect(session, &GPSReceiverSession::receiverReady, this, &GPSReceiver::_onGPSConnect);
569 connect(session, &GPSReceiverSession::receiverDetected, this, &GPSReceiver::_onReceiverDetected);
570 connect(session, &GPSReceiverSession::inputProblem, this, &GPSReceiver::_onInputProblem);
571 connect(session, &GPSReceiverSession::ended, this, [this](GPSConnectionError error, const QString& detail) {
572 _endSession(error, detail, false);
573 _emitChanges();
574 });
575 if (!session->start(_workerFactory, std::move(transportFactory))) {
576 _retireSession();
577 _setError(tr("Failed to create the receiver session."));
578 return false;
579 }
580 _setPersistentChangesAllowed(false);
581 return true;
582}

609void GPSReceiver::_joinRetiredWorkers()
610{
611 // At application exit the event loop that deletes finished workers has stopped, so they are joined here.
612 const QDeadlineTimer deadline(SHUTDOWN_TIMEOUT);
613 for (const QPointer<GPSReceiverWorker>& worker : std::exchange(_retiredWorkers, {})) {
614 if (!worker) {
615 continue;
616 }
617 if (worker->wait(deadline)) {
618 delete worker.data();
619 } else {
620 qCWarning(GPSReceiverLog) << "GPS receiver worker did not stop within" << SHUTDOWN_TIMEOUT.count() << "s";
621 }
622 }
