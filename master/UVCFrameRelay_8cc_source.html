28QVideoFrame UVCFrameRelay::transformed(const QVideoFrame& frame, QtVideo::Rotation rotation, bool mirrored)
29{
30 if (!frame.isValid()) {
31 return frame;
32 }
33 QVideoFrame result = SharedVideoFrame::withOwnMetadata(frame);
34 // Qt mirrors a frame after rotating it, so a rotation applied on top of a mirrored frame turns the other way
35 // (Rot(a) * Mirror == Mirror * Rot(-a)), as Qt's own VideoTransformation::rotate() composes it. QtVideo::Rotation's
36 // values are degrees.
37 const int added = frame.mirrored() ? (360 - static_cast<int>(rotation)) : static_cast<int>(rotation);
38 result.setRotation(static_cast<QtVideo::Rotation>((static_cast<int>(frame.rotation()) + added) % 360));
39 result.setMirrored(frame.mirrored() != mirrored);
40 return result;
41}
