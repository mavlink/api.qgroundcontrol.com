118template <class Api>
119int resolvePlanes(GstBuffer* buffer, Api& api, MapDiagnostics& diag, PlaneResourceGuard<typename Api::Handle>& guard)
120{
121 using Memory = typename Api::Memory;
122 using Device = typename Api::Device;
123
124 const int memCount = (std::min) (int(gst_buffer_n_memory(buffer)), kMaxPlanes);
125 for (int i = 0; i < memCount; ++i) {
126 GstMemory* mem = gst_buffer_peek_memory(buffer, guint(i));
127 if (!mem || !Api::isMemory(mem)) {
128 QGC_HW_WARN_ONCE(Api::log, diag.loggedNonD3DMemory,
129 "resolve: plane" << i << "memory is not" << Api::kMemoryName << "(allocator="
130 << (mem && mem->allocator ? mem->allocator->mem_type : "null") << ")");
131 return 0;
132 }
133 auto* d3dmem = reinterpret_cast<Memory*>(mem);
134 // gst-d3d may land on an isolated device when NEED_CONTEXT was preempted; sampling a foreign-device resource
135 // from QRhi corrupts silently.
136 if (i == 0) {
137 // currentDevice() is transfer-full — unref on every branch to avoid UAF after reset().
138 Device* bridgeDev = Api::bridgeDevice();
139 if (bridgeDev && !Api::sameDevice(d3dmem->device, bridgeDev)) {
140 const gint64 bridgeLuid = GstD3DContextBridge::readAdapterLuid(bridgeDev);
141 const gint64 bufLuid = GstD3DContextBridge::readAdapterLuid(d3dmem->device);
142 gst_object_unref(bridgeDev);
143 QGC_HW_WARN_ONCE(Api::log, diag.loggedDeviceMismatch,
144 "resolve:" << Api::kMemoryName << "on foreign device (bridge LUID=" << bridgeLuid
145 << "buffer LUID=" << bufLuid
146 << "); bridge missed NEED_CONTEXT race — rejecting frame");
147 return 0;
148 }
149 if (bridgeDev) {
150 gst_object_unref(bridgeDev);
151 }
152 }
153 if (!api.prepare(d3dmem, i)) {
154 return 0;
155 }
156 auto* resource = Api::resource(d3dmem);
157 if (!resource) {
158 QGC_HW_WARN_ONCE(Api::log, diag.loggedNullResource,
159 "resolve:" << Api::kMemoryName << "has no resource handle for plane" << i);
160 return 0;
161 }
162 const guint subIdx = Api::subresource(d3dmem, i);
163 if (Api::isArraySlice(resource, subIdx)) {
164 auto* staging = api.copySlice(resource, subIdx, i, d3dmem);
165 if (!staging) {
166 return 0;
167 }
168 guard.handles[i] = staging;
169 } else {
170 resource->AddRef();
171 guard.handles[i] = resource;
172 }
173 guard.owned = i + 1;
174 }
175 return api.finish() ? memCount : 0;
176}

211template <class HandleT>
212QVideoFrameTexturesUPtr mapResolvedTextures(GstHwVideoBuffer& self, QRhi& rhi, QVideoFrameTexturesUPtr& old,
213 const GstHwMapTail& tail,
214 const std::array<HandleT*, kMaxPlanes>& resolvedHandles, int resolvedCount,
215 const std::shared_ptr<void>& keepAlive = {})
216{
217 using BundleT = FrameTextures<HandleT>;
218
219 const auto* desc = QVideoTextureHelper::textureDescription(tail.pixelFormat);
220 if (!desc || desc->nplanes <= 0) {
221 return GstHwPathTelemetry::fail(tail.path);
222 }
223
224 const int textureCount = (std::min)(desc->nplanes, kMaxPlanes);
225 if (resolvedCount <= 0 || resolvedCount > kMaxPlanes) {
226 return GstHwPathTelemetry::fail(tail.path);
227 }
228
229 std::array<HandleT*, kMaxPlanes> handles{};
230 for (int i = 0; i < (std::min)(resolvedCount, textureCount); ++i) {
231 handles[i] = resolvedHandles[i];
232 }
233
234 if (resolvedCount < textureCount) {
235 if (resolvedCount != 1 || !handles[0] || !canAliasSingleResourceAcrossPlanes(tail.pixelFormat)) {
236 return GstHwPathTelemetry::fail(tail.path);
237 }
238 // D3D NV12/P010 output can expose one texture with multiple shader-resource planes.
239 for (int i = resolvedCount; i < textureCount; ++i) {
240 handles[i] = handles[0];
241 }
242 }
243
244 // Reuse-eligible when the pooled staging handles are identical: the pool rotates a stable set of resources, so the
245 // existing QRhiTexture views transparently sample the newly-copied frame.
246 return GstHwFrameTexturesBase::finishMap<BundleT>(
247 self, rhi, old, tail,
248 [&](BundleT& prev) {
249 if (!prev.matchesImport(&rhi, tail.frameSize, tail.pixelFormat,
250 std::span<HandleT* const>(handles.data(), std::size_t(textureCount)))) {
251 return false;
252 }
253 prev.setKeepAlive(keepAlive);
254 return true;
255 },
256 [&] {
257 // The bundle owns these AddRef'd refs; the buffer dtor releases the originals independently.
258 for (int i = 0; i < textureCount; ++i) {
259 if (handles[i])
260 handles[i]->AddRef();
261 }
262 auto textures = std::make_unique<BundleT>(&rhi, tail.frameSize, tail.pixelFormat, handles, textureCount);
263 textures->setKeepAlive(keepAlive);
264 return textures;
265 });
266}

343template <class Api>
344bool proposePool(GstQuery* query, GstCaps* caps, const GstVideoInfo* vinfo, gsize size, guint minBuffers)
345{
346 GstCapsFeatures* features = gst_caps_get_features(caps, 0);
347 if (!features || !gst_caps_features_contains(features, Api::kCapsFeature)) {
348 return false;
349 }
350 typename Api::Device* device = Api::bridgeDevice();
351 if (!device) {
352 return false;
353 }
354 bool proposed = false;
355 if (GstBufferPool* pool = Api::newPool(device)) {
356 GstStructure* config = gst_buffer_pool_get_config(pool);
357 gst_buffer_pool_config_set_params(config, caps, size, minBuffers, 0);
358 gst_buffer_pool_config_add_option(config, GST_BUFFER_POOL_OPTION_VIDEO_META);
359 Api::setAllocationParams(config, device, vinfo);
360 if (gst_buffer_pool_set_config(pool, config)) {
361 // The pool may grow the buffer size to the allocated texture's pitch; re-read it.
362 GstStructure* updated = gst_buffer_pool_get_config(pool);
363 guint poolSize = static_cast<guint>(size);
364 gst_buffer_pool_config_get_params(updated, nullptr, &poolSize, nullptr, nullptr);
365 gst_structure_free(updated);
366 gst_query_add_allocation_pool(query, pool, poolSize, minBuffers, 0);
367 proposed = true;
368 }
369 gst_object_unref(pool);
370 }
371 gst_object_unref(device); // bridgeDevice() returns a transfer-full ref.
372 return proposed;
373}
