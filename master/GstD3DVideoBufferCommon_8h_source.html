118template <class Api>
119int resolvePlanes(GstBuffer* buffer, Api& api, MapDiagnostics& diag, PlaneResourceGuard<typename Api::Handle>& guard)
120{
121 using Memory = typename Api::Memory;
122 using Device = typename Api::Device;
123
124 const int memCount = (std::min) (int(gst_buffer_n_memory(buffer)), kMaxPlanes);
125 for (int i = 0; i < memCount; ++i) {
126 GstMemory* mem = gst_buffer_peek_memory(buffer, guint(i));
127 if (!mem || !Api::isMemory(mem)) {
128 QGC_HW_WARN_ONCE(Api::log, diag.loggedNonD3DMemory,
129 "resolve: plane" << i << "memory is not" << Api::kMemoryName << "(allocator="
130 << (mem && mem->allocator ? mem->allocator->mem_type : "null") << ")");
131 return 0;
132 }
133 auto* d3dmem = reinterpret_cast<Memory*>(mem);
134 // gst-d3d may land on an isolated device when NEED_CONTEXT was preempted; sampling a foreign-device resource
135 // from QRhi corrupts silently.
136 if (i == 0) {
137 // currentDevice() is transfer-full — unref on every branch to avoid UAF after reset().
138 Device* bridgeDev = Api::bridgeDevice();
139 if (bridgeDev && !Api::sameDevice(d3dmem->device, bridgeDev)) {
140 const gint64 bridgeLuid = GstD3DContextBridge::readAdapterLuid(bridgeDev);
141 const gint64 bufLuid = GstD3DContextBridge::readAdapterLuid(d3dmem->device);
142 gst_object_unref(bridgeDev);
143 QGC_HW_WARN_ONCE(Api::log, diag.loggedDeviceMismatch,
144 "resolve:" << Api::kMemoryName << "on foreign device (bridge LUID=" << bridgeLuid
145 << "buffer LUID=" << bufLuid
146 << "); bridge missed NEED_CONTEXT race — rejecting frame");
147 return 0;
148 }
149 if (bridgeDev) {
150 gst_object_unref(bridgeDev);
151 }
152 }
153 if (!api.prepare(d3dmem, i)) {
154 return 0;
155 }
156 auto* resource = Api::resource(d3dmem);
157 if (!resource) {
158 QGC_HW_WARN_ONCE(Api::log, diag.loggedNullResource,
159 "resolve:" << Api::kMemoryName << "has no resource handle for plane" << i);
160 return 0;
161 }
162 const guint subIdx = Api::subresource(d3dmem, i);
163 if (Api::isArraySlice(resource, subIdx)) {
164 auto* staging = api.copySlice(resource, subIdx, i, d3dmem);
165 if (!staging) {
166 return 0;
167 }
168 guard.handles[i] = staging;
169 } else {
170 resource->AddRef();
171 guard.handles[i] = resource;
172 }
173 guard.owned = i + 1;
174 }
175 return api.finish() ? memCount : 0;
176}

214template <class HandleT>
215QVideoFrameTexturesUPtr mapResolvedTextures(GstHwVideoBuffer& self, QRhi& rhi, QVideoFrameTexturesUPtr& old,
216 const GstHwMapTail& tail,
217 const std::array<HandleT*, kMaxPlanes>& resolvedHandles, int resolvedCount,
218 const std::shared_ptr<void>& keepAlive = {})
219{
220 using BundleT = FrameTextures<HandleT>;
221
222 const auto* desc = QVideoTextureHelper::textureDescription(tail.pixelFormat);
223 if (!desc || desc->nplanes <= 0) {
224 return GstHwPathTelemetry::fail(tail.path);
225 }
226
227 const int textureCount = (std::min)(desc->nplanes, kMaxPlanes);
228 if (resolvedCount <= 0 || resolvedCount > kMaxPlanes) {
229 return GstHwPathTelemetry::fail(tail.path);
230 }
231
232 std::array<HandleT*, kMaxPlanes> handles{};
233 for (int i = 0; i < (std::min)(resolvedCount, textureCount); ++i) {
234 handles[i] = resolvedHandles[i];
235 }
236
237 if (resolvedCount < textureCount) {
238 if (resolvedCount != 1 || !handles[0] || !canAliasSingleResourceAcrossPlanes(tail.pixelFormat)) {
239 return GstHwPathTelemetry::fail(tail.path);
240 }
241 // D3D NV12/P010 output can expose one texture with multiple shader-resource planes.
242 for (int i = resolvedCount; i < textureCount; ++i) {
243 handles[i] = handles[0];
244 }
245 }
246
247 // Reuse-eligible when the pooled staging handles are identical: the pool rotates a stable set of resources, so the
248 // existing QRhiTexture views transparently sample the newly-copied frame.
249 return GstHwFrameTexturesBase::finishMap<BundleT>(
250 self, rhi, old, tail,
251 [&](BundleT& prev) {
252 if (!prev.matchesImport(&rhi, tail.frameSize, tail.pixelFormat,
253 std::span<HandleT* const>(handles.data(), std::size_t(textureCount)))) {
254 return false;
255 }
256 prev.setKeepAlive(keepAlive);
257 return true;
258 },
259 [&] {
260 // The bundle owns these AddRef'd refs; the buffer dtor releases the originals independently.
261 for (int i = 0; i < textureCount; ++i) {
262 if (handles[i])
263 handles[i]->AddRef();
264 }
265 auto textures = std::make_unique<BundleT>(&rhi, tail.frameSize, tail.pixelFormat, handles, textureCount);
266 textures->setKeepAlive(keepAlive);
267 return textures;
268 });
269}

273template <class Api>
274bool proposePool(GstQuery* query, GstCaps* caps, const GstVideoInfo* vinfo, gsize size, guint minBuffers)
275{
276 GstCapsFeatures* features = gst_caps_get_features(caps, 0);
277 if (!features || !gst_caps_features_contains(features, Api::kCapsFeature)) {
278 return false;
279 }
280 typename Api::Device* device = Api::bridgeDevice();
281 if (!device) {
282 return false;
283 }
284 bool proposed = false;
285 if (GstBufferPool* pool = Api::newPool(device)) {
286 GstStructure* config = gst_buffer_pool_get_config(pool);
287 gst_buffer_pool_config_set_params(config, caps, size, minBuffers, 0);
288 gst_buffer_pool_config_add_option(config, GST_BUFFER_POOL_OPTION_VIDEO_META);
289 Api::setAllocationParams(config, device, vinfo);
290 if (gst_buffer_pool_set_config(pool, config)) {
291 // The pool may grow the buffer size to the allocated texture's pitch; re-read it.
292 GstStructure* updated = gst_buffer_pool_get_config(pool);
293 guint poolSize = static_cast<guint>(size);
294 gst_buffer_pool_config_get_params(updated, nullptr, &poolSize, nullptr, nullptr);
295 gst_structure_free(updated);
296 gst_query_add_allocation_pool(query, pool, poolSize, minBuffers, 0);
297 proposed = true;
298 }
299 gst_object_unref(pool);
300 }
301 gst_object_unref(device); // bridgeDevice() returns a transfer-full ref.
302 return proposed;
303}
