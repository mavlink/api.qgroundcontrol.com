36int GPSNativeFemto::writeAckedCommandFemto(const char* command, const char* reply, const unsigned int timeout)
37{
38 const GPSConfigurationStep step{command, std::chrono::milliseconds(timeout)};
39 const size_t command_length = strlen(command);
40 const size_t reply_length = strlen(reply);
41 uint8_t buf[GPS_READ_BUFFER_SIZE];
42
43 // Keep one full ACK in the bounded receive window.
44 if (reply_length == 0 || reply_length > sizeof(buf) ||
45 !writeCommand(step, {reinterpret_cast<const uint8_t*>(command), command_length})) {
46 return -1;
47 }
48
49 size_t buffered = 0;
50
51 bool acknowledged = false;
52 const auto result = awaitCommand(
53 step,
54 [&] {
55 const int count = read(buf + buffered, sizeof(buf) - buffered, timeout);
56 if (count <= 0) {
57 return;
58 }
59 buffered += static_cast<size_t>(count);
60 for (size_t index = 0; index + reply_length <= buffered; ++index) {
61 if (memcmp(buf + index, reply, reply_length) == 0) {
62 acknowledged = true;
63 }
64 }
65 if (buffered >= reply_length) {
66 const size_t retained = reply_length - 1;
67 memmove(buf, buf + buffered - retained, retained);
68 buffered = retained;
69 }
70 },
71 [&] { return acknowledged ? GPSCommandOutcome::Acknowledged : GPSCommandOutcome::Pending; });
72 return result.evidence.outcome == GPSCommandOutcome::Acknowledged ? 0 : -1;
73}

75int GPSNativeFemto::configure(unsigned& baudrate, const GPSConfig& config)
76{
77 _configure_done = false;
78 resetIOError();
79 _survey_duration = 0;
80 _survey_in_start = 0;
81 _correction_output_activated = false;
82 _rtcmActivationPending = false;
83 _rtcm_parsing.reset();
84 decodeInit();
85 if (!validateConfiguration(config)) {
86 return -1;
87 }
88 _baseConfig = config.base;
89 _output_mode = config.output_mode;
90 constexpr unsigned supportedBaudrate = 115200;
91 bool success = false;
92
93 if (baudrate == 0 || baudrate == supportedBaudrate) {
94 setBaudrate(supportedBaudrate);
95 for (int run = 0; run < 2; ++run) {
96 if (writeAckedCommandFemto("UNLOGALL THISPORT\r\n", "<UNLOGALL OK", FEMTO_RESPONSE_TIMEOUT) == 0 &&
97 writeAckedCommandFemto("VERSION\r\n", "<VERSION OK", FEMTO_RESPONSE_TIMEOUT) == 0) {
98 success = true;
99 break;
100 }
101 }
102 }
103
104 if (!success) {
105 return -1;
106 }
107
108 baudrate = supportedBaudrate;
109 decodeInit();
110
112 if (_output_mode == OutputMode::RTCM) {
113 if (!_rtcm_parsing) {
114 _rtcm_parsing.emplace();
115 }
116
117 _rtcm_parsing->reset();
118 }
119
120 if (_output_mode == OutputMode::GPS) {
121 // Stop base averaging and release its fixed position before enabling navigation output.
122 if (writeAckedCommandFemto("POSAVE OFF\r\n", "<POSAVE OK", FEMTO_RESPONSE_TIMEOUT) != 0 ||
123 writeAckedCommandFemto("FIX NONE\r\n", "<FIX OK", FEMTO_RESPONSE_TIMEOUT) != 0) {
124 return -1;
125 }
126
127 if (writeAckedCommandFemto("LOG UAVGPSB 0.1\r\n", "<LOG OK", FEMTO_RESPONSE_TIMEOUT) == 0) {
129 if (writeAckedCommandFemto("LOG UAVGPSB 0.05\r\n", "<LOG OK", FEMTO_RESPONSE_TIMEOUT) == 0) {
130 } else if (!ioError()) {
131 log(GPSProtocolLogLevel::Warning, "Femto: command LOG UAVGPSB 0.05 failed,maybe no authorization");
132 }
133
134 } else {
135 if (!ioError()) {
136 log(GPSProtocolLogLevel::Warning, "Femto: command LOG UAVGPSB 0.1 failed");
137 }
138 return -1;
139 }
140
141 if (_satellite_info) {
142 if (writeAckedCommandFemto("LOG UAVSTATUSB 1\r\n", "<LOG OK", FEMTO_RESPONSE_TIMEOUT) == 0) {
143 } else if (!ioError()) {
144 log(GPSProtocolLogLevel::Warning, "Femto: command LOG UAVSTATUSB 1 failed");
145 }
146 }
147
148 } else {
149 activateCorrectionOutput();
150 }
151
152 _configure_done = true;
153
154 return ioError();
155}

157void GPSNativeFemto::activateCorrectionOutput()
158{
159 if (_output_mode != OutputMode::RTCM || _correction_output_activated) {
160 return;
161 }
162 if (!_baseConfig.useFixedBase) {
163 if (writeAckedCommandFemto("POSAVE ON \r\n", "<POSAVE OK", FEMTO_RESPONSE_TIMEOUT) != 0 ||
164 writeAckedCommandFemto("LOG GPGGA 1 \r\n", "<LOG OK", FEMTO_RESPONSE_TIMEOUT) != 0) {
165 controlFailed();
166 return;
167 }
168 _survey_duration = 0;
169 _survey_in_start = nowUs();
170 sendSurveyInStatusUpdate(true, false);
171 return;
172 }
173 const auto& settings = _baseConfig;
174 char buffer[100];
175 const int length =
176 snprintf(buffer, sizeof(buffer), "FIX POSITION %.8lf %.8lf %.5f\r\n", settings.fixedPosition.latitudeDegrees,
177 settings.fixedPosition.longitudeDegrees, double(settings.fixedPosition.altitudeMeters));
178 if (length < 0 || length >= int(sizeof(buffer)) ||
179 writeAckedCommandFemto(buffer, "FIX OK", FEMTO_RESPONSE_TIMEOUT) != 0 ||
180 writeAckedCommandFemto("LOG GPGGA 1 \r\n", "<LOG OK", FEMTO_RESPONSE_TIMEOUT) != 0) {
181 controlFailed();
182 return;
183 }
184 activateRTCMOutput();
185 if (_correction_output_activated) {
186 sendSurveyInStatusUpdate(false, true, settings.fixedPosition.latitudeDegrees,
187 settings.fixedPosition.longitudeDegrees, settings.fixedPosition.altitudeMeters);
188 }
189}

191void GPSNativeFemto::activateRTCMOutput()
192{
193 if (writeAckedCommandFemto("LOG RTCM 1\r\n", "<LOG OK", FEMTO_RESPONSE_TIMEOUT) != 0) {
194 controlFailed();
195 return;
196 }
197 _correction_output_activated = true;
198}
