106 for (const auto& [id, name] : MESSAGE_NAMES) {

118inline constexpr uint8_t SYNC1 = 0xb5;
119inline constexpr uint8_t SYNC2 = 0x62;
120inline constexpr size_t HEADER_SIZE = 6;
122inline constexpr size_t FRAME_OVERHEAD = HEADER_SIZE + 2;
124inline constexpr size_t MAX_PAYLOAD_SIZE = 4096;
127inline constexpr size_t MAX_SKIPPED_PAYLOAD_SIZE = 8192;
128
130[[nodiscard]] constexpr std::span<const uint8_t> encodeFrame(MessageId message, std::span<const uint8_t> payload,
131 std::span<uint8_t> out)
132{
133 const size_t size = payload.size() + FRAME_OVERHEAD;
134 if (payload.size() > MAX_PAYLOAD_SIZE || size > out.size()) {
135 return {};
136 }
137 out[0] = SYNC1;
138 out[1] = SYNC2;
139 out[2] = message.cls;
140 out[3] = message.id;
141 out[4] = static_cast<uint8_t>(payload.size());
142 out[5] = static_cast<uint8_t>(payload.size() >> 8);
143 std::copy(payload.begin(), payload.end(), out.begin() + HEADER_SIZE);
144 const auto checksum = QGC::fletcher8(out.subspan(2, size - 4));
145 out[size - 2] = checksum.a;
146 out[size - 1] = checksum.b;
147 return out.first(size);
148}

151struct Frame
152{
153 uint16_t message = 0;
155 std::span<const uint8_t> bytes;
156 std::span<const uint8_t> payload;
157};

159class FrameDecoder
160{
161public:
162 [[nodiscard]] bool idle() const { return _state == State::Sync1; }
163
164 void reset()
165 {
166 _state = State::Sync1;
167 _checksum = {};
168 }
169
170 [[nodiscard]] std::optional<Frame> consume(uint8_t byte)
171 {
172 switch (_state) {
173 case State::Sync1:
174 if (byte == SYNC1) {
175 _state = State::Sync2;
176 }
177 break;
178 case State::Sync2:
179 if (byte == SYNC2) {
180 _size = 2;
181 _state = State::Header;
182 } else if (byte != SYNC1) {
183 reset();
184 }
185 break;
186 case State::Header:
187 _add(byte);
188 if (_size == HEADER_SIZE) {
189 _length = static_cast<uint16_t>(_bytes[4] | (_bytes[5] << 8));
190 _payloadRead = 0;
191 if (_length > MAX_SKIPPED_PAYLOAD_SIZE) {
192 reset();
193 } else {
194 _state = _length ? State::Payload : State::Checksum1;
195 }
196 }
197 break;
198 case State::Payload:
199 if (_length <= MAX_PAYLOAD_SIZE) {
200 _add(byte);
201 } else {
202 _checksum = QGC::fletcher8({&byte, 1}, _checksum);
203 }
204 if (++_payloadRead == _length) {
205 _state = State::Checksum1;
206 }
207 break;
208 case State::Checksum1:
209 if (byte == _checksum.a) {
210 _bytes[_size++] = byte;
211 _state = State::Checksum2;
212 } else {
213 reset();
214 }
215 break;
216 case State::Checksum2: {
217 const bool valid = byte == _checksum.b && _length <= MAX_PAYLOAD_SIZE;
218 reset();
219 if (valid) {
220 _bytes[_size++] = byte;
221 const std::span<const uint8_t> bytes(_bytes.data(), _size);
222 return Frame{static_cast<uint16_t>(_bytes[2] | (_bytes[3] << 8)), bytes,
223 bytes.subspan(HEADER_SIZE, _length)};
224 }
225 break;
226 }
227 }
228 return std::nullopt;
229 }
230
231private:
232 enum class State
233 {
234 Sync1,
235 Sync2,
236 Header,
237 Payload,
238 Checksum1,
239 Checksum2
240 };
241
242 void _add(uint8_t byte)
243 {
244 _checksum = QGC::fletcher8({&byte, 1}, _checksum);
245 _bytes[_size++] = byte;
246 }
247
248 std::array<uint8_t, MAX_PAYLOAD_SIZE + FRAME_OVERHEAD> _bytes{SYNC1, SYNC2};
249 size_t _size = 0;
250 uint16_t _length = 0;
251 uint16_t _payloadRead = 0;
252 State _state = State::Sync1;
253 QGC::Fletcher8 _checksum;
254};
