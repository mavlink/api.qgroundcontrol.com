51std::string gpuMemoryCaps(const SinkCapsPolicy& policy, const std::string& formats)
52{
53 std::string caps;
54 [[maybe_unused]] const auto offer = [&caps, &formats](const char* mediaType) {
55 caps += mediaType;
56 caps += ", format=";
57 caps += formats;
58 caps += "; ";
59 };
60 switch (policy.target.backend) {
61 case RenderBackend::OpenGL:
62 // DMABuf and AHardwareBuffer reach Qt as EGLImages on its EGLDisplay; GLMemory through the GL share group,
63 // which gst-gl joins on EGL and GLX.
64#if defined(QGC_HAS_GST_DMABUF_GPU_PATH)
65 if (policy.target.egl && policy.dmaBufDisplay) {
66 caps += sinkDmaDrmCaps(policy.dmaBufDisplay);
67 // Legacy memory:DMABuf covers v4l2h264dec/Mali/V3D LINEAR. Unrestricted DMA_DRM is never offered (only
68 // the importable modifiers above): gst-va iHD would negotiate tiled+CCS, which no per-plane import can
69 // read.
70 offer("video/x-raw(memory:DMABuf)");
71 }
72#endif
73#if defined(QGC_HAS_GST_GLMEMORY_GPU_PATH)
74 if (policy.target.sharesGlContext() && (policy.gpuBin != GpuBinMode::DirectDmaBuf)) {
75 offer(kGlMemory2D);
76 }
77#endif
78#if defined(QGC_HAS_GST_AHARDWAREBUFFER_GPU_PATH)
79 if (policy.target.egl) {
80 offer("video/x-raw(memory:AHardwareBuffer)");
81 }
82#endif
83 break;
84 case RenderBackend::Direct3D11:
85#if defined(QGC_HAS_GST_D3D11_GPU_PATH)
86#if defined(QGC_HAS_GST_D3D12_GPU_PATH)
87 // First: its decoder works on a device of its own, a D3D11Memory one on the device Qt renders with.
88 if (policy.d3d11OpensD3D12) {
89 caps += std::string(kD3D12Memory) + ", format=" + kD3D12DecoderFormats + "; ";
90 }
91#endif
92 offer("video/x-raw(memory:D3D11Memory)");
93#endif
94 break;
95 case RenderBackend::Vulkan:
96 // DMA_DRM only, in the layouts Qt's Vulkan device imports: Vulkan needs the modifier, which legacy
97 // memory:DMABuf caps don't carry.
98#if defined(QGC_HAS_GST_DMABUF_VULKAN_IMPORT) && GST_CHECK_VERSION(1, 24, 0)
99 if (policy.dmaBufDisplay) {
100 caps += GstHw::buildVulkanDmaDrmCaps(advertisedFormats(kOfferDmaDrm));
101 }
102#endif
103 break;
104 case RenderBackend::Direct3D12:
105 case RenderBackend::Metal:
106 case RenderBackend::Null:
107 case RenderBackend::Unknown:
108 // Direct3D12: D3D12Memory only through the D3D12Convert bin (buildSinkCaps). Metal: vtdec's
109 // IOSurface-backed CoreVideo buffers arrive as system memory.
110 break;
111 }
112 return caps;
113}

116std::string sinkMemoryCaps(const SinkCapsPolicy& policy)
117{
118 if (policy.gpuZeroCopy && (policy.gpuBin == GpuBinMode::GlUpload)) {
119 return std::string(kGlMemory2D) + ", format=" + advertisedFormatList(kOfferGlUpload);
120 }
121 if (policy.gpuZeroCopy && (policy.gpuBin == GpuBinMode::D3D12Convert)) {
122 return std::string(kD3D12Memory) + ", format=" + kD3D12ConvertFormats;
123 }
124 const std::string formats = advertisedFormatList();
125 return (policy.gpuZeroCopy ? gpuMemoryCaps(policy, formats) : std::string()) + "video/x-raw, format=" + formats;
126}

142std::string buildSinkCaps(const SinkCapsPolicy& policy)
143{
144 // Square pixels in GPU memory as well as in system memory: the frame's size is all the VideoOutput has to go by.
145 // Only system memory is scaled to them, by the bin's converter (and D3D12Memory by d3d12convert), so a decoder
146 // with other pixels negotiates system memory.
147 const std::string caps = sinkMemoryCaps(policy);
148 return policy.squarePixels ? sinkCapsWithSquarePixels(caps) : caps;
149}
