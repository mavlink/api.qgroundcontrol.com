46std::string gpuMemoryCaps(const SinkCapsPolicy& policy, const std::string& formats)
47{
48 std::string caps;
49 [[maybe_unused]] const auto offer = [&caps, &formats](const char* mediaType) {
50 caps += mediaType;
51 caps += ", format=";
52 caps += formats;
53 caps += "; ";
54 };
55 switch (policy.target.backend) {
56 case RenderBackend::OpenGL:
57 // DMABuf and AHardwareBuffer reach Qt as EGLImages on its EGLDisplay; GLMemory through the GL share group,
58 // which gst-gl joins on EGL and GLX.
59#if defined(QGC_HAS_GST_DMABUF_GPU_PATH)
60 if (policy.target.egl && policy.dmaBufDisplay) {
61 caps += sinkDmaDrmCaps(policy.dmaBufDisplay);
62 // Legacy memory:DMABuf covers v4l2h264dec/Mali/V3D LINEAR. Unrestricted DMA_DRM is never offered (only
63 // the importable modifiers above): gst-va iHD would negotiate tiled+CCS, which no per-plane import can
64 // read.
65 offer("video/x-raw(memory:DMABuf)");
66 }
67#endif
68#if defined(QGC_HAS_GST_GLMEMORY_GPU_PATH)
69 if (policy.target.sharesGlContext() && (policy.gpuBin != GpuBinMode::DirectDmaBuf)) {
70 offer(kGlMemory2D);
71 }
72#endif
73#if defined(QGC_HAS_GST_AHARDWAREBUFFER_GPU_PATH)
74 if (policy.target.egl) {
75 offer("video/x-raw(memory:AHardwareBuffer)");
76 }
77#endif
78 break;
79 case RenderBackend::Direct3D11:
80#if defined(QGC_HAS_GST_D3D11_GPU_PATH)
81 offer("video/x-raw(memory:D3D11Memory)");
82#endif
83 break;
84 default:
85 // Direct3D12: D3D12Memory only through the D3D12Convert bin (buildSinkCaps). Metal: vtdec's
86 // IOSurface-backed CoreVideo buffers arrive as system memory. Vulkan: no import path.
87 break;
88 }
89 return caps;
90}

94std::string buildSinkCaps(const SinkCapsPolicy& policy)
95{
96 if (policy.gpuZeroCopy && (policy.gpuBin == GpuBinMode::GlUpload)) {
97 return std::string(kGlMemory2D) + ", format=" + advertisedFormatList(kOfferGlUpload);
98 }
99 if (policy.gpuZeroCopy && (policy.gpuBin == GpuBinMode::D3D12Convert)) {
100 // d3d12convert scales as well, so it gives square pixels on the GPU.
101 std::string caps = std::string(kD3D12MemoryRgb) + ", format=" + kD3D12ConvertFormats;
102 if (policy.squarePixels) {
103 caps += ", pixel-aspect-ratio=(fraction)1/1";
104 }
105 return caps;
106 }
107
108 const std::string formats = advertisedFormatList();
109 std::string caps = policy.gpuZeroCopy ? gpuMemoryCaps(policy, formats) : std::string();
110 caps += "video/x-raw, format=" + formats;
111 if (policy.squarePixels) {
112 caps += ", pixel-aspect-ratio=(fraction)1/1";
113 }
114 return caps;
115}
