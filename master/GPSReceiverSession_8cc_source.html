14void GPSBaseStationState::applySurvey(const GPSSurveyReport& report)
15{
16 if (!mode || *mode == BaseModeDefinition::Mode::BaseFixed) {
17 return;
18 }
19 const bool located =
20 std::isfinite(report.position.latitudeDegrees) && std::isfinite(report.position.longitudeDegrees);
21 const double accuracy = report.meanAccuracyMeters.value_or(
22 surveyAccuracyLimitMeters.value_or(std::numeric_limits<double>::quiet_NaN()));
23 if (report.valid && located && std::isfinite(accuracy)) {
24 position = std::pair(report.position, accuracy);
25 } else {
26 position.reset();
27 }
28}

46GPSBaseStationState gpsBaseStationStateFor(GPSType type, const GPSReceiverConfig& config)
47{
48 GPSBaseStationState state;
49 if (type == GPSType::passive) {
50 return state;
51 }
52 if (const auto* fixed = std::get_if<GPSBaseStationConfig::Fixed>(&config.base.mode)) {
53 state.mode = BaseModeDefinition::Mode::BaseFixed;
54 state.position = std::pair(fixed->position, static_cast<double>(fixed->accuracyMeters));
55 } else if (const auto* survey = std::get_if<GPSBaseStationConfig::SurveyIn>(&config.base.mode)) {
56 state.mode = BaseModeDefinition::Mode::BaseSurveyIn;
57 state.surveyAccuracyLimitMeters = survey->accuracyMeters;
58 } else {
59 state.mode = BaseModeDefinition::Mode::BaseReceiverAveraging;
60 }
61 return state;
62}

145void GPSReceiverSession::retire()
146{
147 if (std::exchange(_retired, true)) {
148 return;
149 }
150 const QPointer<GPSReceiverWorker> worker = _worker;
151 if (worker) {
152 (void) disconnect(worker, nullptr, this, nullptr);
153 // Registration observers may delete the owner; the worker must already be independent.
154 worker->setParent(nullptr);
