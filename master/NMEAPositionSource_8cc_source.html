245 static GPSObservation::FixQuality _fixQuality(const EpochMetadata& epoch)
246 {
247 using Quality = GPSObservation::FixQuality;
248 switch (epoch.ggaQuality.value_or(NMEA::GgaQuality::GPS)) {
249 case NMEA::GgaQuality::INVALID:
250 return Quality::NoFix;
251 case NMEA::GgaQuality::GPS:
252 if (epoch.dimension == NMEA::FixDimension::TWO_D)
253 return Quality::Fix2D;
254 if (epoch.dimension == NMEA::FixDimension::THREE_D)
255 return Quality::Fix3D;
256 return Quality::Unknown;
257 case NMEA::GgaQuality::DIFFERENTIAL:
258 return Quality::Differential;
259 case NMEA::GgaQuality::RTK_FIXED:
260 return Quality::RTKFixed;
261 case NMEA::GgaQuality::RTK_FLOAT:
262 return Quality::RTKFloat;
263 case NMEA::GgaQuality::ESTIMATED:
264 return Quality::Extrapolated;
265 default:
266 return Quality::Unknown;
267 }

294 bool _acceptNavigationStatus(const QTime& time, const QDate& date, quint64 receivedAtUs)
295 {
296 if (!receivedAtUs || receivedAtUs < _statusReceiptUs) {
297 return false;
298 }
299 if (time.isValid() && _statusTime.isValid()) {
300 if (date.isValid() && _statusDate.isValid()) {
301 if (QDateTime(date, time, QTimeZone::UTC) < QDateTime(_statusDate, _statusTime, QTimeZone::UTC)) {
302 return false;
303 }
304 } else {
305 int difference = _statusTime.msecsTo(time);
306 // Undated GGA timestamps wrap at midnight; ordinary negative deltas are late sentences.
307 if (difference < -HALF_DAY_MS) {
308 difference += DAY_MS;
309 } else if (difference > HALF_DAY_MS) {
310 difference -= DAY_MS;
311 }
312 if (difference < 0) {
313 return false;
314 }
315 }
316 }
317 _statusReceiptUs = receivedAtUs;
318 if (time.isValid()) {
319 _statusTime = time;
320 _statusDate = date;
321 }
322 return true;
323 }

358void NMEAPositionSource::_resetDecoder()
359{
360 ++_generation;
361 _requestTask.cancel();
362 _publicationTask.cancel();
363 _lossTask.cancel();
364 _errorTask.cancel();
365 _pendingLoss.reset();
366 _pendingObservation.reset();
367 _pendingRequested = false;
368 _lastObservation = {};
369 _error = NoError;
370 _decoder = std::make_unique<NMEATimestampedPositionDecoder>(
371 _device, _scheduler, [this](GPSObservation loss) { _fixLost(std::move(loss)); });
372 _decoder->setUserEquivalentRangeError(USER_EQUIVALENT_RANGE_ERROR_METERS);
373 // Qt owns epoch merging; the outer source owns requested publication cadence.
374 _decoder->setUpdateInterval(0);
375 const quint64 generation = _generation;
376 connect(_decoder.get(), &QGeoPositionInfoSource::positionUpdated, this,
377 [this, generation](const QGeoPositionInfo& update) {
378 if (generation != _generation || (!_started && !_requestTask.active())) {
379 return;
380 }
381 _pendingRequested |= _requestTask.active();
382 if (!static_cast<NMEATimestampedPositionDecoder*>(_decoder.get())->hasFix(update)) {
383 if (!_started && _requestTask.active()) {
384 _decoder->requestUpdate(DEFAULT_REQUEST_TIMEOUT_MS);
385 }
386 return;
387 }
388 _pendingObservation = update;
389 if (_pendingRequested) {
390 _publicationTask.cancel();
391 }
392 _schedulePublication();
393 });
394 connect(_decoder.get(), &QGeoPositionInfoSource::errorOccurred, this, [this, generation](Error error) {
395 // The outer request task owns deadlines, including after Qt rejects a retained fix.
396 if (error == UpdateTimeoutError) {
397 return;
398 }
399 if (_scheduler) {
400 _errorTask.schedule(std::chrono::microseconds::zero(), [this, generation, error]() {
401 if (generation == _generation) {
402 _error = error;
403 emit errorOccurred(error);
404 }
405 });
406 }
407 });
