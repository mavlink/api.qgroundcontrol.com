20bool isRetryable(NTRIPError error)

241bool NTRIPManager::_dispatch(Event ev, const QString& detail, std::chrono::milliseconds retryAfter)
242{
243 using CS = ConnectionStatus;
244
245 struct Transition
246 {
247 CS from;
248 Event event;
249 CS to;
250 };
251
252 // Every legal transition; events not listed for the current state are ignored. Entry actions are in
253 // _onEnterState().
254 static constexpr Transition kTransitions[] = {
255 // Disconnected: user or settings can kick us into Connecting.
256 {CS::Disconnected, Event::StartRequested, CS::Connecting},
257
258 // Connecting: transport handshake outcome.
259 {CS::Connecting, Event::TransportConnected, CS::Connected},
260 {CS::Connecting, Event::TransportError, CS::Reconnecting},
261 {CS::Connecting, Event::TransportFatalError, CS::Error},
262 {CS::Connecting, Event::ConfigInvalid, CS::Error},
263 {CS::Connecting, Event::StopRequested, CS::Disconnected},
264
265 // Connected: streaming; may lose link or be stopped.
266 {CS::Connected, Event::TransportError, CS::Reconnecting},
267 {CS::Connected, Event::TransportFatalError, CS::Error},
268 {CS::Connected, Event::StopRequested, CS::Disconnected},
269 {CS::Connected, Event::HotReconfigure, CS::Connecting},
270
271 // Self-transition: a transport-affecting setting changed mid-handshake. Re-runs the Connecting entry action
272 // (teardown + restart) without a state change so the in-flight attempt picks up the new config.
273 {CS::Connecting, Event::HotReconfigure, CS::Connecting},
274
275 // Reconnecting: backoff timer owns when we try again, but user/settings can short-circuit it.
276 {CS::Reconnecting, Event::ReconnectDue, CS::Connecting},
277 {CS::Reconnecting, Event::ReconnectGaveUp, CS::Error},
278 {CS::Reconnecting, Event::StopRequested, CS::Disconnected},
279 {CS::Reconnecting, Event::StartRequested, CS::Connecting},
280
281 // Error: user-visible fault. User retry or stop both resolve it.
282 {CS::Error, Event::StartRequested, CS::Connecting},
283 {CS::Error, Event::StopRequested, CS::Disconnected},
284 };
285
286 for (const auto& row : kTransitions) {
287 if (row.from == _connectionStatus && row.event == ev) {
288 _enterState(row.to, detail, retryAfter);
289 return true;
290 }
291 }
292 qCDebug(NTRIPManagerLog) << "NTRIP event" << ev << "ignored in state" << _connectionStatus;
293 return false;
294}

296void NTRIPManager::_enterState(ConnectionStatus to, const QString& detail, std::chrono::milliseconds retryAfter)
297{
298 const ConnectionStatus from = _connectionStatus;
299 const QString msg = detail.isEmpty() ? _defaultMessageFor(to) : detail;
300 if (from != to) {
301 _waitingForNetwork = false;
302 qCDebug(NTRIPManagerLog) << "NTRIP state" << from << "→" << to << msg;
303 }
304
305 // Commit state + message before running entry actions so that a recursive
306 // _dispatch() from inside an entry action (e.g. _openSession → ConfigInvalid)
307 // observes the already-committed state, not the stale caller value.
308 _connectionStatus = to;
309 _statusMessage = msg;
310
311 // Entry action runs on every dispatched transition, including self-transitions
312 // (e.g. Connecting→Connecting on HotReconfigure).
313 _onEnterState(to, retryAfter);
314}

333void NTRIPManager::_onEnterState(ConnectionStatus to, std::chrono::milliseconds retryAfter)
334{
335 switch (to) {
336 case ConnectionStatus::Disconnected:
337 case ConnectionStatus::Error:
338 _cancelReconnect();
339 _stopStreaming();
340 _running = {};
341 if (to == ConnectionStatus::Disconnected) {
342 _correctionSource.reset();
343 }
344 break;
345
346 case ConnectionStatus::Connecting:
347 _cancelReconnect();
348 _openSession();
349 break;
350
351 case ConnectionStatus::Connected:
352 _ggaReporter.start(_transport);
353 _stats.start();
354 break;
355
356 case ConnectionStatus::Reconnecting:
357 _stopStreaming();
358 _scheduleReconnect(retryAfter);
359 break;
360 }
361}

379void NTRIPManager::_scheduleReconnect(std::chrono::milliseconds retryAfter)
380{
381 if (_shouldWaitForNetwork()) {
382 _waitForNetwork();
383 return;
384 }
385
386 if (_reconnectExhausted()) {
387 _dispatch(Event::ReconnectGaveUp,
388 tr("Gave up after %1 reconnect attempts: %2").arg(MAX_RECONNECT_ATTEMPTS).arg(_lastFailureDetail));
389 return;
390 }
391 const ExponentialBackoff beforeAttempt = _reconnectBackoff;
392 const auto backoff = _reconnectBackoff.next(retryAfter);
393 _reconnectTask.schedule(backoff, [this, beforeAttempt]() {
394 if (_shouldWaitForNetwork()) {
395 // The attempt never ran, so the next one keeps its delay.
396 _reconnectBackoff = beforeAttempt;
397 _waitForNetwork();
398 } else {
399 _dispatch(Event::ReconnectDue);
400 }
401 _publish();
402 });
403}

433void NTRIPManager::_openSession()
434{
435 _stopStreaming();
436
437 const Configuration config = _configuration;
438 const auto& connection = config.connection;
439
440 _correctionSource.reset();
441 if (_correctionManager) {
442 _correctionSource =
443 _correctionManager->openSource(GPSCorrectionSettings::Ntrip, correctionSourceInstance(connection));
444 }
445
446 if (const QString err = connection.streamValidationError(); !err.isEmpty()) {
447 qCWarning(NTRIPManagerLog) << "NTRIP config invalid:" << err << "host=" << connection.host
448 << " port=" << connection.port;
449 _dispatch(Event::ConfigInvalid, err);
450 return;
451 }
452
453 qCDebug(NTRIPManagerLog) << "Starting NTRIP transport: host=" << connection.host << " port=" << connection.port
454 << " mount=" << connection.mountpoint;
455
456 // Replace the generic "Connecting..." with a host-specific message.
457 _statusMessage = tr("Connecting to %1:%2...").arg(connection.host).arg(connection.port);
458
459 _running = config;
460 _stats.reset();
461
462 NTRIPTransport* transport = _transportFactory ? _transportFactory(_configuration, this) : nullptr;
463 if (!transport) {
464 transport = new NTRIPHttpTransport(connection, config.filter.messageIds(), this, _scheduler);
465 }
466 _transport = transport;
467
468 // Qt still delivers events queued before a transport was retired; they belong to the ended stream.
469 const auto isCurrent = [this, guard = QPointer<NTRIPTransport>(transport)]() {
470 return guard && guard == _transport;
471 };
472
473 // Queued: error handling stops and deletes the transport, which must not happen inside the transport's (or its
474 // socket's) own signal emission.
475 connect(
476 transport, &NTRIPTransport::error, this,
477 [this, isCurrent](const NTRIPFailure& failure) {
478 if (isCurrent()) {
479 _onTransportError(failure);
480 }
481 },
482 Qt::QueuedConnection);
483
484 // Direct, so the handshake state precedes any error or frame queued after it. _stopStreaming() disconnects a
485 // retired transport before stopping it, so direct signals only ever come from the current one.
486 connect(transport, &NTRIPTransport::connected, this, &NTRIPManager::_onTransportConnected);
487
488 // Queued, like errors, so routing and statistics run outside the transport's parse loop and keep their order
489 // relative to a queued failure.
490 connect(
491 transport, &NTRIPTransport::correctionFrameReceived, this,
492 [this, isCurrent](const RTCMDecodedFrame& frame) {
493 if (isCurrent()) {
494 _onCorrectionFrame(frame);
495 }
496 },
497 Qt::QueuedConnection);
498
499 connect(transport, &NTRIPTransport::certificatePinned, this, &NTRIPManager::_onCertificatePinned);
500
501 transport->start();
502 qCDebug(NTRIPManagerLog) << "NTRIP transport started";
503}

515void NTRIPManager::_onTransportError(const NTRIPFailure& failure)
516{
517 const auto code = failure.code;
518 const auto& detail = failure.detail;
519 if (_connectionStatus != ConnectionStatus::Connecting && _connectionStatus != ConnectionStatus::Connected) {
520 return;
521 }
522 qCWarning(NTRIPManagerLog) << "NTRIP error:" << code << detail;
523 _lastFailureDetail = detail;
524
525 if (_isEnabled() && isRetryable(code)) {
526 const auto backoffMs = _reconnectBackoff.peek(failure.retryAfter).count();
527 qCDebug(NTRIPManagerLog) << "NTRIP reconnecting in" << backoffMs << "ms (attempt"
528 << (_reconnectBackoff.attempts() + 1) << ")";
529 _dispatch(Event::TransportError, tr("Reconnecting in %1s: %2").arg(backoffMs / 1000).arg(detail),
530 failure.retryAfter);
531 } else {
532 _dispatch(Event::TransportFatalError, detail);
533 }
534 _publish();
535}

565void NTRIPManager::_applyConfiguration()
566{
567 if (_shutdown) {
568 return;
569 }
570 const Configuration& newConfig = _configuration;
571 const bool isActive =
572 (_connectionStatus == ConnectionStatus::Connecting || _connectionStatus == ConnectionStatus::Connected);
573
574 if (!_isEnabled()) {
575 _reconnectBackoff.reset();
