41// Single name-based decoder classification for the hardware/software split, the forceVideoDecoder
42// rank overrides and availableDecoderFamilies(). Prefix (not substring) matching, so e.g. avdec_wnv1
43// isn't taken for an NVIDIA decoder. The first match wins: specific prefixes precede general ones.
44constexpr DecoderFamilyPrefix kDecoderFamilyPrefixes[] = {
45 // Android MediaCodec: Google/Codec2 software wrappers; every other amcviddec is a hardware codec.
46 {"amcviddec-omxgoogle", VideoDecoderOption::Default, false},
47 {"amcviddec-c2android", VideoDecoderOption::Default, false},
48 {"amcviddec-", VideoDecoderOption::Default, true},
49 // Legacy gstreamer-vaapi is demoted on Linux and never bumped by VideoDecoderOption::VAAPI (modern va* only).
50 {"vaapi", VideoDecoderOption::Default, true},
51 {"va", VideoDecoderOption::VAAPI, true},
52 {"nv", VideoDecoderOption::NVIDIA, true},
53 {"d3d11", VideoDecoderOption::DirectX3D, true},
54 {"d3d12", VideoDecoderOption::DirectX3D, true},
55 {"dxva", VideoDecoderOption::DirectX3D, true},
56 // vtdec and vtdec_hw, and from GStreamer 1.30 one decoder per codec (vth264dec, vth264dec_hw, ...).
57 {"vt", VideoDecoderOption::VideoToolbox, true},
58 {"qsv", VideoDecoderOption::Intel, true},
59 {"msdk", VideoDecoderOption::Intel, true},

86bool isHardwareDecoderFactory(GstElementFactory* factory)
87{
88 if (!factory) {
89 return false;
90 }
91
92 const gchar* factoryName = gst_plugin_feature_get_name(GST_PLUGIN_FEATURE(factory));
93 if (!factoryName) {
94 return false;
95 }
96
97 if (const DecoderFamilyPrefix* entry = findDecoderFamilyPrefix(factoryName)) {
98 return entry->isHardware;
99 }
100
101 const auto containsHardware = [](const gchar* value) {
102 if (!value) {
103 return false;
104 }
105 const GCharPtr lower = adoptGChar(g_ascii_strdown(value, -1));
106 return g_strrstr(lower.get(), "hardware") != nullptr;
107 };
108
109 return containsHardware(gst_element_factory_get_metadata(factory, GST_ELEMENT_METADATA_KLASS)) ||
110 containsHardware(gst_element_factory_get_metadata(factory, GST_ELEMENT_METADATA_DESCRIPTION));
111}

127// gst_init() applies GST_PLUGIN_FEATURE_RANK ("name:rank,..."), and nothing re-applies it after QGC's overrides.
128bool rankedByUser(const gchar* featureName)
129{
130 const QList<QByteArray> userRanks = qgetenv("GST_PLUGIN_FEATURE_RANK").split(',');
131 return std::ranges::any_of(userRanks, [featureName](const QByteArray& entry) {
132 return entry.left(entry.indexOf(':')).trimmed() == featureName;

179RankTraits currentRankTraits()
180{
181 RankTraits traits;
182#if defined(Q_OS_LINUX)
183 traits.preferZeroCopyDecoders = true;
184#elif defined(Q_OS_WIN)
185 // Ranks usually apply before the scene graph resolves its backend, while the render target is still Unknown: fall
186 // back to QSG_RHI_BACKEND (Qt's D3D11 default) then.
187 bool d3d12 = false;
188 switch (GstQgc::currentRenderTarget().backend) {
189 case GstQgc::RenderBackend::Direct3D12:
190 d3d12 = true;
191 break;
192 case GstQgc::RenderBackend::Unknown:
193 d3d12 = qEnvironmentVariable("QSG_RHI_BACKEND").compare(QLatin1String("d3d12"), Qt::CaseInsensitive) == 0;
194 break;
195 case GstQgc::RenderBackend::Direct3D11:
196 // No D3D API is sampled zero-copy under these: the D3D11 decoders stay the ones preferred.
197 case GstQgc::RenderBackend::Null:
198 case GstQgc::RenderBackend::OpenGL:
199 case GstQgc::RenderBackend::Metal:
200 case GstQgc::RenderBackend::Vulkan:
201 break;
202 }
203 traits.rhiD3DApi = d3d12 ? D3DApi::D3D12 : D3DApi::D3D11;
204 // Both known once the scene graph has published Qt's device, which has the ranks applied again (reapply()).
205 traits.d3d11OpensD3D12 = !d3d12 && HwBuffers::d3d11OpensD3D12();
206 traits.rhiAdapterLuid = HwBuffers::graphicsAdapterLuid();
207#endif
208 traits.softwareRendering = HwBuffers::softwareRenderingRequested();
209 return traits;
210}

220void DecoderRankApplier::apply(VideoDecoderOption option)
221{
222 GstRegistry* const registry = _registry();
223 if (!registry) {
224 return;
225 }
226 const QMutexLocker locker(&_mutex);
227 _restore(registry);
228
229 const RankTraits traits = currentRankTraits();
230 const RankPlan plan = planRanks(option, installedDecoders(), traits);
231 qCDebug(GStreamerDecoderRanksLog) << "Applying decoder option" << static_cast<int>(option) << "with"
232 << plan.changes.size() << "rank changes";
233 for (const QString& warning : plan.warnings) {
234 qCWarning(GStreamerDecoderRanksLog).noquote() << warning;
235 }
236 for (const RankChange& change : plan.changes) {
237 (void) _setRank(registry, change.feature.constData(), change.rank);
238 }
239 _applied = option;
240 _appliedTraits = traits;
241}

264bool DecoderRankApplier::demoteFailedDecoder(const char* featureName, const GstCaps* caps)
265{
266 GstRegistry* const registry = _registry();
267 if (!registry || !featureName || !caps || rankedByUser(featureName)) {
268 return false;
269 }
270 // decodebin3's candidates for the stream: the autoplugged decoders whose sink template covers its caps.
271 GList* const decoders = gst_element_factory_list_get_elements(GST_ELEMENT_FACTORY_TYPE_DECODER, GST_RANK_MARGINAL);
272 GList* const candidates = gst_element_factory_list_filter(decoders, caps, GST_PAD_SINK, TRUE);
273 gst_plugin_feature_list_free(decoders);
274 bool otherDecoder = false;
275 for (GList* node = candidates; node != nullptr; node = node->next) {
276 otherDecoder =
277 otherDecoder || (g_strcmp0(gst_plugin_feature_get_name(GST_PLUGIN_FEATURE(node->data)), featureName) != 0);
278 }
279 gst_plugin_feature_list_free(candidates);
280 if (!otherDecoder) {
281 return false;
282 }
283 const QMutexLocker locker(&_mutex);
284 return _setRank(registry, featureName, GST_RANK_NONE);
285}

287bool DecoderRankApplier::_setRank(GstRegistry* registry, const char* featureName, uint16_t rank)
288{
289 if (!registry || !featureName) {
290 return false;
291 }
292 const GstFeaturePtr feature = adoptFeature(gst_registry_lookup_feature(registry, featureName));
293 if (!feature) {
294 return false;
295 }
296
297 if (rankedByUser(featureName)) {
298 qCDebug(GStreamerDecoderRanksLog) << " Keeping the GST_PLUGIN_FEATURE_RANK rank of" << featureName;
299 return true;
300 }
301 const QByteArray key(featureName);
302 if (!_originalRanks.contains(key)) {
303 _originalRanks.insert(key, gst_plugin_feature_get_rank(feature.get()));
304 }
305 qCDebug(GStreamerDecoderRanksLog) << " Changing feature (" << featureName << ") to use rank:" << rank;
306 gst_plugin_feature_set_rank(feature.get(), rank);
307 return true;
308}

363QList<VideoDecoderOption> availableDecoderFamilies()
364{
365 QList<VideoDecoderOption> families;
366 forEachVideoDecoderFactory(GST_RANK_NONE, [&families](GstElementFactory* factory) {
367 const VideoDecoderOption family =
368 decoderFamilyForFactory(gst_plugin_feature_get_name(GST_PLUGIN_FEATURE(factory)));
369#ifndef QGC_HAS_GST_VULKAN_GPU_PATH
370 // Only a build with the opt-in Vulkan GPU path has a use for them: no sink takes their VulkanImage memory.

373 }
374#endif
375 if ((family != VideoDecoderOption::Default) && !families.contains(family)) {
376 families.append(family);
377 }
378 });
379 return families;

382void logDecoderRanks()
383{
384 if (!GStreamerDecoderRanksLog().isDebugEnabled()) {
385 return;
386 }
387
388 struct Decoder
389 {
390 guint rank;
391 QByteArray name;
392 QString description;
393 };
394
395 QList<Decoder> decoders;
396 forEachVideoDecoderFactory(GST_RANK_NONE, [&decoders](GstElementFactory* factory) {
397 GstPluginFeature* const feature = GST_PLUGIN_FEATURE(factory);
398 decoders.append({gst_plugin_feature_get_rank(feature), QByteArray(gst_plugin_feature_get_name(feature)),
399 describeDecoderFactory(factory)});
400 });
401
402 if (decoders.isEmpty()) {
403 qCDebug(GStreamerDecoderRanksLog) << "No video decoder factories found";
404 return;
405 }
406
407 std::ranges::sort(decoders, [](const Decoder& lhs, const Decoder& rhs) {
408 return (lhs.rank != rhs.rank) ? (lhs.rank > rhs.rank) : (lhs.name < rhs.name);
409 });
410
411 qCDebug(GStreamerDecoderRanksLog) << "Video decoder ranks:";
412 for (const Decoder& decoder : std::as_const(decoders)) {
413 qCDebug(GStreamerDecoderRanksLog).noquote() << " " + decoder.description;
414 }
415}
