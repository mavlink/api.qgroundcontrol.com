26JitterBufferTotals jitterBufferTotals(GstElement* bin)
27{
28 JitterBufferTotals totals;
29 if (!bin || !GST_IS_BIN(bin)) {
30 return totals;
31 }
32 const auto addStats = [](const GValue* item, gpointer data) {
33 auto* const element = GST_ELEMENT(g_value_get_object(item));
34 GstElementFactory* const factory = gst_element_get_factory(element);
35 if (!factory || (g_strcmp0(GST_OBJECT_NAME(factory), "rtpjitterbuffer") != 0)) {
36 return;
37 }
38 GstStructure* rawStats = nullptr;
39 g_object_get(element, "stats", &rawStats, nullptr);
40 const GStreamer::GstStructurePtr stats = GStreamer::adoptStructure(rawStats);
41 const auto field = [&stats](const char* name) {
42 guint64 value = 0;
43 if (stats) {
44 (void) gst_structure_get_uint64(stats.get(), name, &value);
45 }
46 return value;
47 };
48 auto* const sum = static_cast<JitterBufferTotals*>(data);
49 sum->found = true;
50 sum->packetsPushed += field("num-pushed");
51 sum->packetsLost += field("num-lost");
52 sum->packetsLate += field("num-late");
53 sum->packetsDuplicate += field("num-duplicates");
54 sum->rtxRequested += field("rtx-count");
55 sum->rtxRecovered += field("rtx-success-count");
56 sum->maxJitterNs = std::max(sum->maxJitterNs, field("avg-jitter"));
57 };
58 GstIterator* const elements = gst_bin_iterate_recurse(GST_BIN(bin));
59 // rtspsrc adds its rtpbin, and rtpbin its jitter buffers, while the stream runs.
60 while (gst_iterator_foreach(elements, addStats, &totals) == GST_ITERATOR_RESYNC) {
61 gst_iterator_resync(elements);
62 totals = {};
63 }
64 gst_iterator_free(elements);
65 return totals;
66}

106void PipelineStatsProbe::_noteKeyframe(GstClockTime time)
107{
108 if (_keyframeTimeCount > 0) {
109 const GstClockTime previous = _keyframeTimes[_keyframeTimeCount - 1];
110 // Parameter sets that arrive in a buffer of their own share the keyframe's timestamp.
111 if (time == previous) {
112 return;
113 }
114 // Timestamps that went back start over, as after a new segment.
115 if (time < previous) {
116 _keyframeTimeCount = 0;
117 }
118 }
119 if (_keyframeTimeCount == kKeyframeWindow) {
120 std::shift_left(_keyframeTimes.begin(), _keyframeTimes.end(), 1);
121 --_keyframeTimeCount;
122 }
123 _keyframeTimes[_keyframeTimeCount++] = time;
124 if (_keyframeTimeCount >= 2) {
125 const GstClockTime span = _keyframeTimes[_keyframeTimeCount - 1] - _keyframeTimes[0];
126 _keyframeInterval.store(span / (_keyframeTimeCount - 1), std::memory_order_relaxed);
127 }
