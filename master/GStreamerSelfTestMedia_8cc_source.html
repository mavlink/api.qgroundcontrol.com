49QString playToEos(GstElement* pipeline, GstElement* source, const QByteArray& data, std::chrono::milliseconds timeout)
50{
51 const GstBusPtr bus = adoptBus(gst_element_get_bus(pipeline));
52 const bool started = gst_element_set_state(pipeline, GST_STATE_PLAYING) != GST_STATE_CHANGE_FAILURE;
53 if (started && source) {
54 (void) gst_app_src_push_buffer(GST_APP_SRC(source),
55 gst_buffer_new_memdup(data.constData(), static_cast<gsize>(data.size())));
56 (void) gst_app_src_end_of_stream(GST_APP_SRC(source));
57 }
58 const GstClockTime wait =
59 started ? static_cast<GstClockTime>(std::chrono::nanoseconds(std::max(timeout, 0ms)).count()) : 0;
60 const GstMessagePtr message = adoptMessage(
61 gst_bus_timed_pop_filtered(bus.get(), wait, static_cast<GstMessageType>(GST_MESSAGE_EOS | GST_MESSAGE_ERROR)));
62 (void) gst_element_set_state(pipeline, GST_STATE_NULL);
63
64 if (message && (GST_MESSAGE_TYPE(message.get()) == GST_MESSAGE_EOS)) {
65 return {};
66 }
67 if (message) {
68 return errorText(message.get());
69 }
70 return started ? QStringLiteral("no end of stream within %1 ms").arg(timeout.count())
71 : QStringLiteral("the pipeline failed to start");
72}

121DecodeResult decodeSample(const Sample& sample, const QByteArray& data, std::chrono::milliseconds timeout)
122{
123 DecoderWatch watch;
124 std::atomic<int> frames = 0;
125 const GstElementPtr pipeline = newPipeline("selftest-decode");
126 GstElementPtr sourceOwner = makeElement("appsrc");
127 GstElementPtr parserOwner = makeElement(sample.parser);
128 GstElementPtr decodebinOwner = makeElement("decodebin3");
129 GstElementPtr sinkOwner = makeElement("appsink");
130 DecodeResult result;
131 if (!pipeline || !sourceOwner || !parserOwner || !decodebinOwner || !sinkOwner) {
132 result.failure = QStringLiteral("cannot create appsrc, %1, decodebin3 or appsink").arg(sample.parser);
133 return result;
134 }
135
136 const GstCapsPtr inputCaps = byteStreamCaps(sample);
137 const GstCapsPtr rawCaps = adoptCaps(gst_caps_new_empty_simple("video/x-raw"));
138 g_object_set(sourceOwner.get(), "caps", inputCaps.get(), nullptr);
139 // Nothing pulls the samples: keep only the latest, and post the EOS without waiting for them to be taken.
140 g_object_set(sinkOwner.get(), "caps", rawCaps.get(), "sync", FALSE, "max-buffers", 1u, "drop", TRUE, "wait-on-eos",
141 FALSE, nullptr);
142
143 GstElement* const source = addToBin(pipeline.get(), std::move(sourceOwner), "appsrc");
144 GstElement* const parser = addToBin(pipeline.get(), std::move(parserOwner), sample.parser);
145 GstElement* const decodebin = addToBin(pipeline.get(), std::move(decodebinOwner), "decodebin3");
146 GstElement* const sink = addToBin(pipeline.get(), std::move(sinkOwner), "appsink");
147 if (!source || !parser || !decodebin || !sink || !gst_element_link_many(source, parser, decodebin, nullptr)) {
148 result.failure = QStringLiteral("cannot link appsrc ! %1 ! decodebin3").arg(sample.parser);
149 return result;
150 }
151
152 const GstPadPtr sinkPad = adoptPad(gst_element_get_static_pad(sink, "sink"));
153 (void) gst_pad_add_probe(sinkPad.get(), GST_PAD_PROBE_TYPE_BUFFER, countBuffer, &frames, nullptr);
154 (void) g_signal_connect(decodebin, "pad-added", G_CALLBACK(linkNewSrcPad), sinkPad.get());
155 (void) g_signal_connect(pipeline.get(), "deep-element-added", G_CALLBACK(watchDecoder), &watch);
156
157 result.failure = playToEos(pipeline.get(), source, data, timeout);
158 if (result.failure.isEmpty() && (frames.load() == 0)) {
159 result.failure = QStringLiteral("no decoded frame reached the sink");
160 }
161
162 const std::lock_guard lock(watch.mutex);
163 result.decoders = watch.decoders;
164 result.decoders.removeDuplicates();
165 result.hardware = watch.hardware;
166 return result;
167}

192{

205 GstElement* const demux = addToBin(pipeline.get(), std::move(demuxerOwner), demuxer);

207 if (!source || !demux || !sink || !gst_element_link(source, demux)) {
208 return QStringLiteral("cannot link filesrc ! %1").arg(demuxer);
209 }
210

212 (void) gst_pad_add_probe(sinkPad.get(), GST_PAD_PROBE_TYPE_BUFFER, countBuffer, &buffers, nullptr);
213 (void) g_signal_connect(demux, "pad-added", G_CALLBACK(linkNewSrcPad), sinkPad.get());
214
215 const QString failure = playToEos(pipeline.get(), nullptr, {}, timeout);
216 if (!failure.isEmpty()) {
217 return failure;
218 }
219 return (buffers.load() > 0) ? QString() : QStringLiteral("%1 found no video in it").arg(demuxer);
