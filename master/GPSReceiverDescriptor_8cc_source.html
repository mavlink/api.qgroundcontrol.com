78consteval bool registryComplete()
79{
80 for (int type = 0; type < static_cast<int>(GPSType::automatic); ++type) {
81 if (std::ranges::count(DESCRIPTORS, static_cast<GPSType>(type), &GPSReceiverDescriptor::type) != 1) {
82 return false;
83 }
84 }
85 for (const auto& descriptor : DESCRIPTORS) {
86 if (descriptor.type == GPSType::automatic || descriptor.manufacturerId == GPS_AUTOMATIC_MANUFACTURER ||
87 std::ranges::count(DESCRIPTORS, descriptor.manufacturerId, &GPSReceiverDescriptor::manufacturerId) != 1) {
88 return false;
89 }
90 }
91 return DESCRIPTORS.size() == static_cast<size_t>(GPSType::automatic);
92}

94static_assert(registryComplete(), "Each GPSType needs one descriptor with a unique, nonzero manufacturer ID");

147GPSReceiverCapabilities gpsReceiverCapabilities(GPSType type)
148{
149 if (type == GPSType::automatic) {
150 // Detection finds a base family; until then every option some base family supports is accepted.
151 GPSReceiverCapabilities combined;
152 for (const auto& descriptor : DESCRIPTORS) {
153 const auto& capabilities = descriptor.capabilities;
154 if (!capabilities.rtkBase) {
155 continue;
156 }
157 combined.recognized = true;
158 combined.rtkBase = true;
159 combined.surveyIn |= capabilities.surveyIn;
160 combined.receiverAveraging |= capabilities.receiverAveraging;
161 combined.persistentConfiguration |= capabilities.persistentConfiguration;
162 combined.compactObservations |= capabilities.compactObservations;
163 }
164 return combined;
165 }
166
167 const auto* descriptor = gpsReceiverDescriptor(type);
168 if (!descriptor) {
169 return {};
170 }
171 return descriptor->capabilities;
172}

175GPSReceiverPresentation buildPresentation(int manufacturer)
176{
177 const auto* selected = gpsReceiverDescriptorForManufacturer(manufacturer);
178 GPSReceiverPresentation presentation;
179 for (const auto& descriptor : DESCRIPTORS) {
180 if (manufacturer != GPS_AUTOMATIC_MANUFACTURER && &descriptor != selected) {
181 continue;
182 }
183 presentation.rtkBase |= descriptor.capabilities.rtkBase;
184 presentation.surveyIn |= descriptor.capabilities.surveyIn;
185 presentation.receiverAveraging |= descriptor.capabilities.receiverAveraging;
186 presentation.compactObservations |= descriptor.capabilities.compactObservations;
187 presentation.surveyAccuracy |= descriptor.surveyAccuracy != Accuracy::Unavailable;
188 presentation.surveyDuration |= descriptor.configurableSurveyDuration;
189 presentation.fixedBaseAccuracy |= descriptor.fixedBaseAccuracy;
190 // Automatic offers these for whichever family it identifies: the consent a connection may pass and the notes
191 // on that family's side effects.
192 presentation.persistentConfiguration |= descriptor.capabilities.persistentConfiguration;
193 presentation.restartOnConnect |= descriptor.restartOnConnect;
194 presentation.surveyMaySavePosition |= descriptor.surveyMaySavePosition;
195 }
196 // Labels and live status describe one family, which Automatic only knows once connected.
197 presentation.specificReceiver = selected != nullptr;
198 presentation.automatic = manufacturer == GPS_AUTOMATIC_MANUFACTURER;
199 presentation.passive = selected && selected->capabilities.passive;
200 presentation.observationAccuracyFilter = selected && selected->surveyAccuracy == Accuracy::ObservationFilter;
201 presentation.acceptedObservationTime = selected && selected->surveyDuration == Duration::AcceptedObservations;
202 presentation.reportsSurveyDuration = selected && selected->surveyDuration != Duration::Unavailable;
203 return presentation;
204}

207const GPSReceiverPresentation& gpsReceiverPresentation(int manufacturer)
208{
209 // Immutable after first use, so QML bindings see one value per receiver family.
210 static const QHash<int, GPSReceiverPresentation> presentations = [] {
211 QHash<int, GPSReceiverPresentation> result{
212 {GPS_AUTOMATIC_MANUFACTURER, buildPresentation(GPS_AUTOMATIC_MANUFACTURER)}};
213 for (const auto& descriptor : DESCRIPTORS) {
214 result.insert(descriptor.manufacturerId, buildPresentation(descriptor.manufacturerId));
215 }
216 return result;
217 }();
218 static const GPSReceiverPresentation unknown = buildPresentation(-1);
219 const auto presentation = presentations.constFind(manufacturer);
220 return presentation != presentations.cend() ? *presentation : unknown;
