94QVariantMap gpsReceiverPresentation(int manufacturer)
95{
96 const auto* selected = gpsReceiverDescriptorForManufacturer(manufacturer);
97 GPSReceiverCapabilities capabilities;
98 bool surveyAccuracy = false;
99 bool surveyDuration = false;
100 bool fixedBaseAccuracy = false;
101 for (const auto& descriptor : DESCRIPTORS) {
102 if (manufacturer != 0 && &descriptor != selected) {
103 continue;
104 }
105 capabilities.recognized |= descriptor.capabilities.recognized;
106 capabilities.rtkBase |= descriptor.capabilities.rtkBase;
107 capabilities.surveyIn |= descriptor.capabilities.surveyIn;
108 capabilities.receiverAveraging |= descriptor.capabilities.receiverAveraging;
109 surveyAccuracy |= descriptor.surveyAccuracy != Accuracy::Unavailable;
110 surveyDuration |= descriptor.configurableSurveyDuration;
111 fixedBaseAccuracy |= descriptor.fixedBaseAccuracy;
112 }
113 // "All" combines editable fields, not receiver-specific side effects or live status.
114 return {
115 {QStringLiteral("recognized"), capabilities.recognized},
116 {QStringLiteral("specificReceiver"), selected != nullptr},
117 {QStringLiteral("rtkBase"), capabilities.rtkBase},
118 {QStringLiteral("surveyIn"), capabilities.surveyIn},
119 {QStringLiteral("receiverAveraging"), capabilities.receiverAveraging},
120 {QStringLiteral("passive"), selected && selected->capabilities.passive},
121 {QStringLiteral("surveyAccuracy"), surveyAccuracy},
122 {QStringLiteral("surveyDuration"), surveyDuration},
123 {QStringLiteral("fixedBaseAccuracy"), fixedBaseAccuracy},
124 {QStringLiteral("observationAccuracyFilter"),
125 selected && selected->surveyAccuracy == Accuracy::ObservationFilter},
126 {QStringLiteral("acceptedObservationTime"),
127 selected && selected->surveyDuration == Duration::AcceptedObservations},
128 {QStringLiteral("reportsSurveyDuration"), selected && selected->surveyDuration != Duration::Unavailable},
129 {QStringLiteral("persistentConfiguration"), selected && selected->capabilities.persistentConfiguration},
