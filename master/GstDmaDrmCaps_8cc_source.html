31// Qt's importer binds one EGLImage per format plane, so compressed layouts and those with auxiliary planes (CCS, DCC,
32// AFBC, UBWC) can't be imported. Plain vendor tilings can: the modifier goes with each plane.
33bool modifierImportable(guint64 modifier) noexcept
34{
35 const guint64 code = modifier & 0x00ffffffffffffffULL;
36 switch (modifier >> 56) {
37 case DRM_FORMAT_MOD_VENDOR_INTEL:
38 // X, Y, Yf and 4 tiling; every other Intel modifier is a CCS variant.
39 return (code == 1) || (code == 2) || (code == 3) || (code == 9);
40 case DRM_FORMAT_MOD_VENDOR_AMD:
41 return ((code >> 13) & 0x1) == 0; // AMD_FMT_MOD DCC bit
42 case DRM_FORMAT_MOD_VENDOR_ARM:
43 return ((modifier >> 52) & 0xf) == 0x01; // DRM_FORMAT_MOD_ARM_TYPE_MISC: not AFBC/AFRC compression
44 case DRM_FORMAT_MOD_VENDOR_QCOM:
45 return code != 1; // DRM_FORMAT_MOD_QCOM_COMPRESSED (UBWC)
46 default:
47 return modifier != DRM_FORMAT_MOD_INVALID;
48 }
49}

51// What GstQtDmaBufImport::planeFourccs() picks for the Qt counterpart of @p format: one R8/GR88 (8-bit) or
52// R16/GR1616 (16-bit) image per plane of a multi-plane format, the format's own fourcc for a single-plane one.
53guint32 planeImportFourcc(GstVideoFormat format, guint plane) noexcept
54{
55 const GstVideoFormatInfo* finfo = gst_video_format_get_info(format);
56 if (!finfo || (plane >= GST_VIDEO_FORMAT_INFO_N_PLANES(finfo))) {
57 return 0;
58 }
59 if (GST_VIDEO_FORMAT_INFO_N_PLANES(finfo) == 1) {
60 return gst_video_dma_drm_fourcc_from_format(format);
61 }
62 gint components[GST_VIDEO_MAX_COMPONENTS];
63 gst_video_format_info_component(finfo, plane, components);
64 const int count = static_cast<int>(std::ranges::count_if(components, [](gint c) { return c >= 0; }));
65 const int pixelStride = (components[0] >= 0) ? GST_VIDEO_FORMAT_INFO_PSTRIDE(finfo, components[0]) : 0;
66 if (count == 1) {
67 return (pixelStride == 1) ? DRM_FORMAT_R8 : (pixelStride == 2) ? DRM_FORMAT_R16 : 0;
68 }
69 if (count == 2) {
70 return (pixelStride == 2) ? DRM_FORMAT_GR88 : (pixelStride == 4) ? DRM_FORMAT_GR1616 : 0;
71 }
72 return 0;
73}

77bool appendDrmFormat(std::string& entries, guint32 fourcc, guint64 modifier)

95public:

99 {}

102
104 std::vector<EGLuint64KHR> all(EGLint fourcc) const { return query(fourcc, false); }
105
107 const std::vector<EGLuint64KHR>& sampleable(guint32 fourcc)
108 {
109 const auto [it, inserted] = _sampleable.try_emplace(fourcc);
110 if (inserted) {
111 it->second = query(static_cast<EGLint>(fourcc), true);
112 }
113 return it->second;
114 }
115
116private:
117 std::vector<EGLuint64KHR> query(EGLint fourcc, bool texture2DOnly) const
118 {
119 EGLint count = 0;
120 if ((_query(_display, fourcc, 0, nullptr, nullptr, &count) != EGL_TRUE) || (count <= 0)) {
121 return {};
122 }
123 std::vector<EGLuint64KHR> mods(static_cast<std::size_t>(count));
124 std::vector<EGLBoolean> externalOnly(static_cast<std::size_t>(count));
125 if ((_query(_display, fourcc, count, mods.data(), externalOnly.data(), &count) != EGL_TRUE) || (count <= 0)) {
126 return {};
127 }
128 mods.resize(static_cast<std::size_t>(count));
129 if (!texture2DOnly) {
130 return mods;
131 }
132 std::vector<EGLuint64KHR> texture2D;
133 for (std::size_t i = 0; i < mods.size(); ++i) {
134 if (!externalOnly[i]) {
135 texture2D.push_back(mods[i]);
136 }
137 }
138 return texture2D;
139 }

142 PFNEGLQUERYDMABUFMODIFIERSEXTPROC _query;
143 std::map<guint32, std::vector<EGLuint64KHR>> _sampleable;

148std::string buildSupportedDmaDrmCaps(std::span<const GstVideoFormat> formats, void* display)
149{
150 if (formats.empty()) {
151 return std::string();
152 }
153
154 const auto dpy = static_cast<EGLDisplay>(display);
155 if ((dpy == EGL_NO_DISPLAY) || !GstEglHelpers::displaySupportsExtension(dpy, kModifiersExt)) {
156 qCDebug(GstDmaDrmCapsLog) << "DMA_DRM modifier query unavailable (no DMABuf import display, or no "
157 "EGL_EXT_image_dma_buf_import_modifiers); offering no DMA_DRM caps";
158 return std::string();
159 }
160
161 const auto queryFormats =
162 reinterpret_cast<PFNEGLQUERYDMABUFFORMATSEXTPROC>(eglGetProcAddress("eglQueryDmaBufFormatsEXT"));
163 ModifierQuery modifiers(dpy);
164 if (!queryFormats || !modifiers.valid()) {
165 return std::string();
166 }
167
168 EGLint numFormats = 0;
169 if ((queryFormats(dpy, 0, nullptr, &numFormats) != EGL_TRUE) || (numFormats <= 0)) {
170 return std::string();
171 }
172 std::vector<EGLint> fourccs(static_cast<std::size_t>(numFormats));
173 if ((queryFormats(dpy, numFormats, fourccs.data(), &numFormats) != EGL_TRUE) || (numFormats <= 0)) {
174 return std::string();
175 }
176 fourccs.resize(static_cast<std::size_t>(numFormats));
177
178 std::string entries;
179 int offeredCount = 0;
180 int excludedCount = 0;
181 for (const EGLint fourcc : fourccs) {
182 // GStreamer's own fourcc mapping, the one it reads drm-format caps with.
183 const GstVideoFormat format = gst_video_dma_drm_fourcc_to_format(static_cast<guint32>(fourcc));
184 if (std::ranges::find(formats, format) == formats.end()) {
185 continue;
186 }
187 const guint planes = GST_VIDEO_FORMAT_INFO_N_PLANES(gst_video_format_get_info(format));
188 for (const EGLuint64KHR mod : modifiers.all(fourcc)) {
189 bool importable = modifierImportable(mod);
190 for (guint plane = 0; importable && (plane < planes); ++plane) {
191 const guint32 planeFourcc = planeImportFourcc(format, plane);
192 importable = (planeFourcc != 0) && std::ranges::any_of(modifiers.sampleable(planeFourcc),
193 [mod](EGLuint64KHR m) { return m == mod; });
194 }
195 if (importable && appendDrmFormat(entries, static_cast<guint32>(fourcc), mod)) {
196 ++offeredCount;
197 } else {
198 ++excludedCount;
199 }
200 }
201 }
202
203 if (entries.empty()) {
204 return std::string();
205 }
206 qCDebug(GstDmaDrmCapsLog) << "EGL-derived DMA_DRM caps:" << offeredCount << "format/modifier pairs offered,"
207 << excludedCount << "not importable per plane";
208 return "video/x-raw(memory:DMABuf), format=DMA_DRM, drm-format={ " + entries + " }; ";
209}
