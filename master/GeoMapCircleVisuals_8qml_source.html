109 MouseArea {
110 anchors.fill: parent

114 // Cursor-to-anchor offset at press, so the handle doesn't jump to the cursor
115 property point _grabOffset
116
117 function _parentPosition(mouse) {
118 return handle.mapToItem(handle.parent, mouse.x, mouse.y)
119 }
120
121 onPressed: (mouse) => {
122 const pos = _parentPosition(mouse)
123 _grabOffset = Qt.point(handle.x + handle.anchorPoint.x - pos.x, handle.y + handle.anchorPoint.y - pos.y)
124 }
125
126 onPositionChanged: (mouse) => {
127 const pos = _parentPosition(mouse)
128 const target = Qt.point(pos.x + _grabOffset.x, pos.y + _grabOffset.y)
129 const scene = handle.scene
130 // Absolute handles sit on the horizontal plane at their altitude,
131 // ground-clamped ones on the rendered terrain
132 const coordinate = (handle.altitudeMode === GeoMapItem.Absolute)
133 ? scene.camera.coordinateAtScreenPoint(target, handle.scenePosition.z)
134 : handle.surfaceModel.surfaceCoordinateAtScreenPoint(scene.camera, target, scene.verticalScale * scene.terrainScale)
135 if (coordinate.isValid) {
136 handle.dragged(coordinate)
137 }
138 }
139 }
