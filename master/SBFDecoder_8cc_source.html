40sbf_buf_t decodeBlock(std::span<const uint8_t> bytes)
41{
42 sbf_buf_t value{};
43 value.sync = LittleEndian::read<uint16_t>(bytes, 0).value_or(0);
44 value.crc16 = LittleEndian::read<uint16_t>(bytes, 2).value_or(0);
45 const auto id = LittleEndian::read<uint16_t>(bytes, 4).value_or(0);
46 value.msg_id = id & 0x1fff;
47 value.msg_revision = id >> 13;
48 value.length = LittleEndian::read<uint16_t>(bytes, 6).value_or(0);
49 value.TOW = LittleEndian::read<uint32_t>(bytes, 8).value_or(0);
50 value.WNc = LittleEndian::read<uint16_t>(bytes, 12).value_or(0);
51 switch (value.msg_id) {
52 case SBF_ID_PVTGeodetic:
53 value.payload_pvt_geodetic.mode_type = (LittleEndian::read<uint8_t>(bytes, 14).value_or(0) >> 0) & 15;
54 value.payload_pvt_geodetic.mode_reserved = (LittleEndian::read<uint8_t>(bytes, 14).value_or(0) >> 4) & 3;
55 value.payload_pvt_geodetic.mode_base_fixed = (LittleEndian::read<uint8_t>(bytes, 14).value_or(0) >> 6) & 1;
56 value.payload_pvt_geodetic.mode_2d = (LittleEndian::read<uint8_t>(bytes, 14).value_or(0) >> 7) & 1;
57 value.payload_pvt_geodetic.error = LittleEndian::read<uint8_t>(bytes, 15).value_or(0);
58 value.payload_pvt_geodetic.latitude = LittleEndian::read<double>(bytes, 16).value_or(0);
59 value.payload_pvt_geodetic.longitude = LittleEndian::read<double>(bytes, 24).value_or(0);
60 value.payload_pvt_geodetic.height = LittleEndian::read<double>(bytes, 32).value_or(0);
61 value.payload_pvt_geodetic.undulation = LittleEndian::read<float>(bytes, 40).value_or(0);
62 value.payload_pvt_geodetic.vn = LittleEndian::read<float>(bytes, 44).value_or(0);
63 value.payload_pvt_geodetic.ve = LittleEndian::read<float>(bytes, 48).value_or(0);
64 value.payload_pvt_geodetic.vu = LittleEndian::read<float>(bytes, 52).value_or(0);
65 value.payload_pvt_geodetic.cog = LittleEndian::read<float>(bytes, 56).value_or(0);
66 value.payload_pvt_geodetic.rx_clk_bias = LittleEndian::read<double>(bytes, 60).value_or(0);
67 value.payload_pvt_geodetic.RxClkDrift = LittleEndian::read<float>(bytes, 68).value_or(0);
68 value.payload_pvt_geodetic.time_system = LittleEndian::read<uint8_t>(bytes, 72).value_or(0);
69 value.payload_pvt_geodetic.datum = LittleEndian::read<uint8_t>(bytes, 73).value_or(0);
70 value.payload_pvt_geodetic.nr_sv = LittleEndian::read<uint8_t>(bytes, 74).value_or(0);
71 value.payload_pvt_geodetic.wa_corr_info = LittleEndian::read<uint8_t>(bytes, 75).value_or(0);
72 value.payload_pvt_geodetic.reference_id = LittleEndian::read<uint16_t>(bytes, 76).value_or(0);
73 value.payload_pvt_geodetic.mean_corr_age = LittleEndian::read<uint16_t>(bytes, 78).value_or(0);
74 value.payload_pvt_geodetic.signal_info = LittleEndian::read<uint32_t>(bytes, 80).value_or(0);
75 value.payload_pvt_geodetic.alert_flag = LittleEndian::read<uint8_t>(bytes, 84).value_or(0);
76 value.payload_pvt_geodetic.nr_bases = LittleEndian::read<uint8_t>(bytes, 85).value_or(0);
77 value.payload_pvt_geodetic.ppp_info = LittleEndian::read<uint16_t>(bytes, 86).value_or(0);
78 value.payload_pvt_geodetic.latency = LittleEndian::read<uint16_t>(bytes, 88).value_or(0);
79 value.payload_pvt_geodetic.h_accuracy = LittleEndian::read<uint16_t>(bytes, 90).value_or(0);
80 value.payload_pvt_geodetic.v_accuracy = LittleEndian::read<uint16_t>(bytes, 92).value_or(0);
81 break;
82 case SBF_ID_VelCovGeodetic:
83 value.payload_vel_col_geodetic.mode_type = (LittleEndian::read<uint8_t>(bytes, 14).value_or(0) >> 0) & 15;
84 value.payload_vel_col_geodetic.mode_reserved =
85 (LittleEndian::read<uint8_t>(bytes, 14).value_or(0) >> 4) & 3;
86 value.payload_vel_col_geodetic.mode_base_fixed =
87 (LittleEndian::read<uint8_t>(bytes, 14).value_or(0) >> 6) & 1;
88 value.payload_vel_col_geodetic.mode_2d = (LittleEndian::read<uint8_t>(bytes, 14).value_or(0) >> 7) & 1;
89 value.payload_vel_col_geodetic.error = LittleEndian::read<uint8_t>(bytes, 15).value_or(0);
90 value.payload_vel_col_geodetic.cov_vn_vn = LittleEndian::read<float>(bytes, 16).value_or(0);
91 value.payload_vel_col_geodetic.cov_ve_ve = LittleEndian::read<float>(bytes, 20).value_or(0);
92 value.payload_vel_col_geodetic.cov_vu_vu = LittleEndian::read<float>(bytes, 24).value_or(0);
93 value.payload_vel_col_geodetic.cov_dt_dt = LittleEndian::read<float>(bytes, 28).value_or(0);
94 value.payload_vel_col_geodetic.cov_vn_ve = LittleEndian::read<float>(bytes, 32).value_or(0);
95 value.payload_vel_col_geodetic.cov_vn_vu = LittleEndian::read<float>(bytes, 36).value_or(0);
96 value.payload_vel_col_geodetic.cov_vn_dt = LittleEndian::read<float>(bytes, 40).value_or(0);
97 value.payload_vel_col_geodetic.cov_ve_vu = LittleEndian::read<float>(bytes, 44).value_or(0);
98 value.payload_vel_col_geodetic.cov_ve_dt = LittleEndian::read<float>(bytes, 48).value_or(0);
99 value.payload_vel_col_geodetic.cov_vu_dt = LittleEndian::read<float>(bytes, 52).value_or(0);
100 break;
101 case SBF_ID_DOP:
102 value.payload_dop.nr_sv = LittleEndian::read<uint8_t>(bytes, 14).value_or(0);
103 value.payload_dop.reserved = LittleEndian::read<uint8_t>(bytes, 15).value_or(0);
104 value.payload_dop.pDOP = LittleEndian::read<uint16_t>(bytes, 16).value_or(0);
105 value.payload_dop.tDOP = LittleEndian::read<uint16_t>(bytes, 18).value_or(0);
106 value.payload_dop.hDOP = LittleEndian::read<uint16_t>(bytes, 20).value_or(0);
107 value.payload_dop.vDOP = LittleEndian::read<uint16_t>(bytes, 22).value_or(0);
108 value.payload_dop.hpl = LittleEndian::read<float>(bytes, 24).value_or(0);
109 value.payload_dop.vpl = LittleEndian::read<float>(bytes, 28).value_or(0);
110 break;
111 case SBF_ID_AttEuler:
112 value.payload_att_euler.nr_sv = LittleEndian::read<uint8_t>(bytes, 14).value_or(0);
113 value.payload_att_euler.error_aux1 = (LittleEndian::read<uint8_t>(bytes, 15).value_or(0) >> 0) & 3;
114 value.payload_att_euler.error_aux2 = (LittleEndian::read<uint8_t>(bytes, 15).value_or(0) >> 2) & 3;
115 value.payload_att_euler.error_reserved = (LittleEndian::read<uint8_t>(bytes, 15).value_or(0) >> 4) & 7;
116 value.payload_att_euler.error_not_requested = (LittleEndian::read<uint8_t>(bytes, 15).value_or(0) >> 7) & 1;
117 value.payload_att_euler.mode = LittleEndian::read<uint16_t>(bytes, 16).value_or(0);
118 value.payload_att_euler.reserved = LittleEndian::read<uint16_t>(bytes, 18).value_or(0);
119 value.payload_att_euler.heading = LittleEndian::read<float>(bytes, 20).value_or(0);
120 value.payload_att_euler.pitch = LittleEndian::read<float>(bytes, 24).value_or(0);
121 value.payload_att_euler.roll = LittleEndian::read<float>(bytes, 28).value_or(0);
122 value.payload_att_euler.pitch_dot = LittleEndian::read<float>(bytes, 32).value_or(0);
123 value.payload_att_euler.roll_dot = LittleEndian::read<float>(bytes, 36).value_or(0);
124 value.payload_att_euler.heading_dot = LittleEndian::read<float>(bytes, 40).value_or(0);
125 break;
126 case SBF_ID_AttCovEuler:
127 value.payload_att_cov_euler.reserved = LittleEndian::read<uint8_t>(bytes, 14).value_or(0);
128 value.payload_att_cov_euler.error_aux1 = (LittleEndian::read<uint8_t>(bytes, 15).value_or(0) >> 0) & 3;
129 value.payload_att_cov_euler.error_aux2 = (LittleEndian::read<uint8_t>(bytes, 15).value_or(0) >> 2) & 3;
130 value.payload_att_cov_euler.error_reserved = (LittleEndian::read<uint8_t>(bytes, 15).value_or(0) >> 4) & 7;
131 value.payload_att_cov_euler.error_not_requested =
132 (LittleEndian::read<uint8_t>(bytes, 15).value_or(0) >> 7) & 1;
133 value.payload_att_cov_euler.cov_headhead = LittleEndian::read<float>(bytes, 16).value_or(0);
134 value.payload_att_cov_euler.cov_pitchpitch = LittleEndian::read<float>(bytes, 20).value_or(0);
135 value.payload_att_cov_euler.cov_rollroll = LittleEndian::read<float>(bytes, 24).value_or(0);
136 value.payload_att_cov_euler.cov_headpitch = LittleEndian::read<float>(bytes, 28).value_or(0);
137 value.payload_att_cov_euler.cov_headroll = LittleEndian::read<float>(bytes, 32).value_or(0);
138 value.payload_att_cov_euler.cov_pitchroll = LittleEndian::read<float>(bytes, 36).value_or(0);
139 break;
140 default:
141 break;
142 }
143 return value;
144}

147int GPSNativeSBF::parseChar(const uint8_t b)
148{
149 int ret = 0;
150
151 // A native frame owns its payload, including any embedded RTCM preambles.
152 if (_rtcm_parsing && (_decode_state == SBF_DECODE_SYNC1 || _rtcm_parsing->hasPartialFrame())) {
153 const bool complete = _rtcm_parsing->addByte(b);
154 if (complete) {
155 if (_rtcm_parsing->valid()) {
156 gotRTCMMessage(_rtcm_parsing->frame().data(), _rtcm_parsing->frame().size());
157 }
158 _rtcm_parsing->reset();
159 return GPSDecodedBatch::PROTOCOL_ACTIVITY;
160 }
161 if (_rtcm_parsing->hasPartialFrame()) {
162 return 0;
163 }
164 }
165
166 switch (_decode_state) {
167 // Expecting Sync1
168 case SBF_DECODE_SYNC1:
169 if (b == SBF_SYNC1) { // Sync1 found --> expecting Sync2
170 payloadRxAdd(b); // add a payload byte
171 _decode_state = SBF_DECODE_SYNC2;
172 }
173
174 break;
175
176 // Expecting Sync2
177 case SBF_DECODE_SYNC2:
178 if (b == SBF_SYNC2) { // Sync2 found --> expecting CRC
179 payloadRxAdd(b); // add a payload byte
180 _decode_state = SBF_DECODE_PAYLOAD;
181
182 } else { // Sync1 not followed by Sync2: reset parser
183 decodeInit();
184 }
185
186 break;
187
188 // Expecting payload
189 case SBF_DECODE_PAYLOAD:
190 ret = payloadRxAdd(b); // add a payload byte
191
192 if (ret < 0) {
193 // payload not handled, discard message
194 ret = 0;
195 decodeInit();
196
197 } else if (ret > 0) {
198 ret = payloadRxDone(); // finish payload processing
199
200 if (_rtcm_parsing) {
201 _rtcm_parsing->reset();
202 }
203
204 decodeInit();
205
206 } else {
207 // expecting more payload, stay in state SBF_DECODE_PAYLOAD
208 ret = 0;
209 }
210
211 break;
212
213 default:
214 break;
215 }
216
217 return ret;
218}

220int GPSNativeSBF::payloadRxAdd(const uint8_t b)
221{
222 int ret = 0;
223 _wire[_rx_payload_index++] = b;
224 const auto length = LittleEndian::read<uint16_t>(_wire, 6).value_or(0);
225
226 if ((_rx_payload_index > 7 && _rx_payload_index >= length) || _rx_payload_index >= _wire.size()) {
227 ret = 1; // payload received completely
228 }
229
230 return ret;
231}

233uint16_t crc16(const uint8_t* data_p, uint32_t length)
234{
235 uint8_t x;
236 uint16_t crc = 0;
237
238 while (length--) {
239 x = crc >> 8 ^ *data_p++;
240 x ^= x >> 4;
241 crc = static_cast<uint16_t>((crc << 8) ^ (x << 12) ^ (x << 5) ^ x);
242 }
243
244 return crc;
245}

247int GPSNativeSBF::payloadRxDone()
248{
249 int ret = 0;
250
251 _buf = decodeBlock(std::span<const uint8_t>(_wire).first(_rx_payload_index));
252 if (_buf.length < 14 || _buf.length > _rx_payload_index || _buf.crc16 != crc16(_wire.data() + 4, _buf.length - 4)) {
253 return 0;
254 }
255
256 size_t requiredLength = 0;
257 switch (_buf.msg_id) {
258 case SBF_ID_PVTGeodetic:
259 requiredLength = 94;
260 break;
261 case SBF_ID_VelCovGeodetic:
262 requiredLength = 56;
263 break;
264 case SBF_ID_DOP:
265 requiredLength = 32;
266 break;
267 case SBF_ID_AttEuler:
268 requiredLength = 44;
269 break;
270 case SBF_ID_AttCovEuler:
271 requiredLength = 40;
272 break;
273 default:
274 return 0;
275 }
276 if (_buf.length < requiredLength || _buf.TOW >= WEEK_MS || _buf.WNc == UINT16_MAX) {
277 return 0;
278 }
279
280 auto* epoch = navigationEpoch(uint64_t(_buf.WNc) * WEEK_MS + _buf.TOW);
281 if (!epoch) {
282 return GPSDecodedBatch::PROTOCOL_ACTIVITY;
283 }
284 auto* output = _gps_position;
285 _gps_position = &epoch->position;
286 switch (_buf.msg_id) {
287 case SBF_ID_PVTGeodetic: {
288 epoch->hasPosition = true;
289
290 if (_buf.payload_pvt_geodetic.mode_type < 1) {
291 _gps_position->fix_type = GPSPositionReport::FixType::NoFix;
292
293 } else {
294 switch (_buf.payload_pvt_geodetic.mode_type) {
295 case 2:
296 case 6:
297 _gps_position->fix_type = GPSPositionReport::FixType::Differential;
298 break;
299
300 case 5:
301 case 8:
302 _gps_position->fix_type = GPSPositionReport::FixType::RTKFloat;
303 break;
304
305 case 4:
306 case 7:
307 _gps_position->fix_type = GPSPositionReport::FixType::RTKFixed;
308 break;
309
310 default:
311 _gps_position->fix_type = GPSPositionReport::FixType::Fix3D;
312 break;
313 }
314 }
315
316 if (_buf.payload_pvt_geodetic.error != 0) {
317 _gps_position->fix_type = GPSPositionReport::FixType::NoFix;
318 } else if (_buf.payload_pvt_geodetic.mode_2d &&
319 _gps_position->fix_type >= GPSPositionReport::FixType::Fix3D) {
320 _gps_position->fix_type = GPSPositionReport::FixType::Fix2D;
321 }
322
323 // Check fix and error code
324 _gps_position->vel_ned_valid =
325 _gps_position->fix_type > GPSPositionReport::FixType::NoFix && _buf.payload_pvt_geodetic.error == 0;
326
327 // Check boundaries and invalidate GPS velocities
328 // We're not just checking for the do-not-use value (-2*10^10) but for any value beyond the specified max
329 // values
330 if (fabsf(_buf.payload_pvt_geodetic.vn) > 600.0f || fabsf(_buf.payload_pvt_geodetic.ve) > 600.0f ||
331 fabsf(_buf.payload_pvt_geodetic.vu) > 600.0f) {
332 _gps_position->vel_ned_valid = false;
333 }
334
335 // Check boundaries and invalidate position
336 // We're not just checking for the do-not-use value (-2*10^10) but for any value beyond the specified max
337 // values
338 const auto& pvt = _buf.payload_pvt_geodetic;
339 const bool coordinatesValid = std::isfinite(pvt.latitude) &&
340 std::abs(pvt.latitude) <= std::numbers::pi / 2 &&
341 std::isfinite(pvt.longitude) && std::abs(pvt.longitude) <= std::numbers::pi &&
342 std::isfinite(pvt.height) && std::abs(pvt.height) <= DNU;
343 if (!coordinatesValid || !std::isfinite(pvt.undulation) || std::abs(pvt.undulation) > DNU) {
344 _gps_position->fix_type = GPSPositionReport::FixType::NoFix;
345 }
346
347 if (_buf.payload_pvt_geodetic.nr_sv < 255) { // 255 = do not use value
348 _gps_position->satellites_used = _buf.payload_pvt_geodetic.nr_sv;
349
350 if (_satellite_info) {
351 publishSatelliteUsage(_gps_position->satellites_used);
352 }
353
354 } else {
355 _gps_position->satellites_used = UINT8_MAX;
356 if (_satellite_info) {
357 publishSatelliteUsage(std::nullopt);
358 }
359 }
360
361 _gps_position->latitude_deg = _buf.payload_pvt_geodetic.latitude * GPS_RAD_TO_DEG;
362 _gps_position->longitude_deg = _buf.payload_pvt_geodetic.longitude * GPS_RAD_TO_DEG;
363 _gps_position->altitude_ellipsoid_m = _buf.payload_pvt_geodetic.height;
364 _gps_position->altitude_msl_m =
365 _buf.payload_pvt_geodetic.height - static_cast<double>(_buf.payload_pvt_geodetic.undulation);
366
367 /* H and V accuracy are reported in 2DRMS, but based off the uBlox reporting we expect RMS.
368 * Devide by 100 from cm to m and in addition divide by 2 to get RMS. */
369 _gps_position->eph = _buf.payload_pvt_geodetic.h_accuracy != UINT16_MAX
370 ? static_cast<float>(_buf.payload_pvt_geodetic.h_accuracy) / 200.0f
371 : NAN;
372 _gps_position->accuracy_timestamp = nowUs();
373 _gps_position->epv = _buf.payload_pvt_geodetic.v_accuracy != UINT16_MAX
374 ? static_cast<float>(_buf.payload_pvt_geodetic.v_accuracy) / 200.0f
375 : NAN;
376
377 _gps_position->vel_n_m_s = static_cast<float>(_buf.payload_pvt_geodetic.vn);
378 _gps_position->vel_e_m_s = static_cast<float>(_buf.payload_pvt_geodetic.ve);
379 _gps_position->vel_d_m_s = -1.0f * static_cast<float>(_buf.payload_pvt_geodetic.vu);
380 _gps_position->vel_m_s = sqrtf(_gps_position->vel_n_m_s * _gps_position->vel_n_m_s +
381 _gps_position->vel_e_m_s * _gps_position->vel_e_m_s);
382
383 const float course = _buf.payload_pvt_geodetic.cog;
384 _gps_position->cog_rad =
385 std::isfinite(course) && course >= 0.0f && course <= 360.0f ? course * GPS_DEG_TO_RAD : NAN;
386 _gps_position->courseAccuracyRadians = 1.0f * GPS_DEG_TO_RAD;
387
388 // WNc/TOW is GNSS system time, not UTC. Without receiver UTC/leap information,
389 // retain the epoch key internally and let the facade use reception UTC.
390 _gps_position->time_utc_usec = 0;
391 _gps_position->timestamp = nowUs();
392
393 // In RTCM mode, PVTGeodetic is used to get base station survey-in
394 if (_configured && _output_mode == OutputMode::RTCM) {
395 // Mode bit 6 means automatic base determination is still in progress, not completed.
396 // Septentrio PolaRx5TR 5.5.0 Reference Guide, SBF Mode definition (p. 382).
397 const bool active = !_baseConfig.useFixedBase && pvt.mode_base_fixed;
398 const bool valid =
399 !pvt.mode_base_fixed && pvt.mode_type == 3 && !pvt.mode_2d && !pvt.error && coordinatesValid;
400 if (!_baseConfig.useFixedBase && (active || _survey_active)) {
401 _survey_duration = (nowUs() - _survey_activation_date) / 1000000;
402 }
403 _survey_active = active;
404 GPSNativeSurveyReport status{};
405 // PVT accuracy describes the navigation solution, not the averaged base survey.
406 status.altitudeDatum = GPSNativeSurveyReport::AltitudeDatum::Ellipsoid;
407 status.latitude = coordinatesValid && !pvt.error ? _gps_position->latitude_deg : NAN;
408 status.longitude = coordinatesValid && !pvt.error ? _gps_position->longitude_deg : NAN;
409 status.altitude = coordinatesValid && !pvt.error ? _gps_position->altitude_ellipsoid_m : NAN;
410 status.duration = _baseConfig.useFixedBase ? 0 : _survey_duration;
411 status.flags = static_cast<uint8_t>(valid) | (static_cast<uint8_t>(active) << 1);
412 surveyInStatus(status);
413 ret |= 4; // RTCM infos have been updated
414 }
415
416 //
417 break;
418 }
419
420 case SBF_ID_VelCovGeodetic: {
421 const auto& covariance = _buf.payload_vel_col_geodetic;
422 const std::array variances{covariance.cov_vn_vn, covariance.cov_ve_ve, covariance.cov_vu_vu};
423 const bool valid = !covariance.error && std::all_of(variances.begin(), variances.end(), [](float variance) {
424 return std::isfinite(variance) && variance >= 0;
425 });
426 _gps_position->speedAccuracyMetersPerSecond =
427 valid ? std::sqrt(*std::max_element(variances.begin(), variances.end())) : NAN;
428 break;
429 }
430 case SBF_ID_DOP:
431 _gps_position->hdop = _buf.payload_dop.hDOP != UINT16_MAX ? _buf.payload_dop.hDOP * 0.01f : NAN;
432 _gps_position->vdop = _buf.payload_dop.vDOP != UINT16_MAX ? _buf.payload_dop.vDOP * 0.01f : NAN;
433 _gps_position->dop_timestamp = nowUs();
434 break;
435
436 case SBF_ID_AttEuler: {
437 const auto& attitude = _buf.payload_att_euler;
438 _gps_position->heading = NAN;
439 _gps_position->heading_timestamp = nowUs();
440 if (!attitude.error_not_requested && !attitude.error_aux1 && !attitude.error_aux2 && attitude.mode >= 1 &&
441 attitude.mode <= 4 && std::isfinite(attitude.heading) && std::abs(attitude.heading) <= 360) {
442 _gps_position->heading = std::remainder(attitude.heading, 360.0f) * GPS_DEG_TO_RAD;
443 }
444 break;
445 }
446 case SBF_ID_AttCovEuler: {
447 const auto& covariance = _buf.payload_att_cov_euler;
448 const float variance = covariance.cov_headhead;
449 const bool valid = !covariance.error_not_requested && !covariance.error_aux1 && !covariance.error_aux2 &&
450 std::isfinite(variance) && variance >= 0;
451 _gps_position->heading_accuracy = valid ? std::sqrt(variance) * GPS_DEG_TO_RAD : NAN;
452 break;
453 }
454
455 default:
456
457 break;
458 }
459
460 _gps_position = output;

464GPSNativeSBF::NavigationEpoch* GPSNativeSBF::navigationEpoch(uint64_t receiverTimeMs)
465{
466 flushDecoded();
467 if (_lastPublishedEpoch && receiverTimeMs <= *_lastPublishedEpoch) {
468 return nullptr;
469 }
470 for (auto& epoch : _epochs) {
471 if (epoch && epoch->receiverTimeMs == receiverTimeMs) {
472 return &*epoch;
473 }
474 }
475 auto slot = _epochs.begin();
476 while (slot != _epochs.end() && *slot) {
477 ++slot;
478 }
479 if (slot == _epochs.end()) {
480 slot = _epochs[0]->receiverTimeMs < _epochs[1]->receiverTimeMs ? _epochs.begin() : _epochs.begin() + 1;
481 if (receiverTimeMs <= (*slot)->receiverTimeMs) {
482 return nullptr;
483 }
484 finishEpoch(*slot);
485 }
486 *slot = NavigationEpoch{.receiverTimeMs = receiverTimeMs, .receiptUs = nowUs(), .position = {}};
487 return &**slot;
488}

490void GPSNativeSBF::finishEpoch(std::optional<NavigationEpoch>& epoch)
491{
492 if (epoch->hasPosition && (!_lastPublishedEpoch || epoch->receiverTimeMs > *_lastPublishedEpoch)) {
493 *_gps_position = epoch->position;
494 _gps_position->timestamp = epoch->receiptUs;
495 _decoded.updates |= 1;
496 _decoded.events.emplace_back(*_gps_position);
497 }
498 if (!_lastPublishedEpoch || epoch->receiverTimeMs > *_lastPublishedEpoch) {
499 _lastPublishedEpoch = epoch->receiverTimeMs;
500 }
501 epoch.reset();
502}

504void GPSNativeSBF::flushDecoded()
505{
506 if (_epochs[0] && _epochs[1] && _epochs[0]->receiverTimeMs > _epochs[1]->receiverTimeMs) {
507 std::swap(_epochs[0], _epochs[1]);
508 }
509 const auto now = nowUs();
510 for (auto& epoch : _epochs) {
511 if (epoch && now >= epoch->receiptUs && now - epoch->receiptUs >= EPOCH_MAX_AGE_US) {
512 finishEpoch(epoch);
513 }
514 }
