97void PlanMasterController::_activeVehicleChanged(Vehicle* activeVehicle)
98{
99 if (_managerVehicle == activeVehicle) {
100 // We are already setup for this vehicle
101 return;
102 }
103
104 qCDebug(PlanMasterControllerLog) << "_activeVehicleChanged" << activeVehicle;
105
106 if (_managerVehicle) {
107 // Disconnect old vehicle. Be careful of wildcarding disconnect too much since _managerVehicle may equal _controllerVehicle
108 disconnect(_managerVehicle, &Vehicle::initialPlanRequestCompleteChanged, this, nullptr);
109 disconnect(_managerVehicle->missionManager(), nullptr, this, nullptr);
110 disconnect(_managerVehicle->geoFenceManager(), nullptr, this, nullptr);
111 disconnect(_managerVehicle->rallyPointManager(), nullptr, this, nullptr);
112
113 // Any in-flight transfer chain can never complete against the new vehicle's managers
114 _loadSequence = SyncSequence::Idle;
115 _sendSequence = SyncSequence::Idle;
116 }
117
118 bool newOffline = false;
119 if (activeVehicle == nullptr) {
120 // Since there is no longer an active vehicle we use the offline controller vehicle as the manager vehicle
121 _managerVehicle = _controllerVehicle;
122 newOffline = true;
123 } else {
124 newOffline = false;
125 _managerVehicle = activeVehicle;
126
127 // Update controllerVehicle to the currently connected vehicle
128 AppSettings* appSettings = SettingsManager::instance()->appSettings();
129 appSettings->offlineEditingFirmwareClass()->setRawValue(QGCMAVLink::firmwareClass(_managerVehicle->firmwareType()));
130 appSettings->offlineEditingVehicleClass()->setRawValue(QGCMAVLink::vehicleClass(_managerVehicle->vehicleType()));
131
132 // We use these signals to sequence upload and download to the multiple controller/managers
133 connect(_managerVehicle, &Vehicle::initialPlanRequestCompleteChanged, this, &PlanMasterController::_initialPlanRequestCompleteChanged);
134 connect(_managerVehicle->missionManager(), &MissionManager::newMissionItemsAvailable, this, &PlanMasterController::_loadMissionComplete);
135 connect(_managerVehicle->geoFenceManager(), &GeoFenceManager::loadComplete, this, &PlanMasterController::_loadGeoFenceComplete);
136 connect(_managerVehicle->rallyPointManager(), &RallyPointManager::loadComplete, this, &PlanMasterController::_loadRallyPointsComplete);
137 connect(_managerVehicle->missionManager(), &MissionManager::sendComplete, this, &PlanMasterController::_sendMissionComplete);
138 connect(_managerVehicle->geoFenceManager(), &GeoFenceManager::sendComplete, this, &PlanMasterController::_sendGeoFenceComplete);
139 connect(_managerVehicle->rallyPointManager(), &RallyPointManager::sendComplete, this, &PlanMasterController::_sendRallyPointsComplete);
140 }
141
142 _offline = newOffline;
143 emit offlineChanged(offline());
144 emit managerVehicleChanged(_managerVehicle);
145
146 if (_flyView) {
147 // We are in the Fly View
148 if (newOffline) {
149 // No active vehicle, clear mission
150 qCDebug(PlanMasterControllerLog) << "_activeVehicleChanged: Fly View - No active vehicle, clearing stale plan";
151 removeAll();
152 } else {
153 // Fly view has changed to a new active vehicle, update to show correct mission
154 qCDebug(PlanMasterControllerLog) << "_activeVehicleChanged: Fly View - New active vehicle, loading new plan from manager vehicle";
155 _showPlanFromManagerVehicle();
156 }
157 } else {
158 // We are in the Plan view.
159 if (containsItems()) {
160 // We have a plan which is from a different vehicle than the new active vehicle. By definition this plan requires and upload.
161 _setDirtyForUpload(true);
162
163 // The plan view has a stale plan in it
164 if (dirtyForSave()) {
165 // Plan is dirty, the user must decide what to do in all cases
166 qCDebug(PlanMasterControllerLog) << "_activeVehicleChanged: Plan View - Previous dirty plan exists, no new active vehicle, sending promptForPlanUsageOnVehicleChange signal";
167 emit promptForPlanUsageOnVehicleChange();
168 } else {
169 // Plan is not dirty
170 if (newOffline) {
171 // The active vehicle went away with no new active vehicle
172 qCDebug(PlanMasterControllerLog) << "_activeVehicleChanged: Plan View - Previous clean plan exists, no new active vehicle, clear stale plan";
173 removeAll();
174 } else {
175 // We are transitioning from one active vehicle to another. Show the plan from the new vehicle.
176 qCDebug(PlanMasterControllerLog) << "_activeVehicleChanged: Plan View - Previous clean plan exists, new active vehicle, loading from new manager vehicle";
177 _showPlanFromManagerVehicle();
178 }
179 }
180 } else {
181 // There is no previous Plan in the view
182 _setDirtyStates(false, false);
183 if (newOffline) {
184 // Nothing special to do in this case
185 qCDebug(PlanMasterControllerLog) << "_activeVehicleChanged: Plan View - No previous plan, no longer connected to vehicle, nothing to do";
186 } else {
187 // Just show the plan from the new vehicle
188 qCDebug(PlanMasterControllerLog) << "_activeVehicleChanged: Plan View - No previous plan, new active vehicle, loading from new manager vehicle";
189 _showPlanFromManagerVehicle();
190 }
191 }
192 }
193
194 // Vehicle changed so we need to signal everything
195 emit containsItemsChanged();
196 emit syncInProgressChanged();
197 emit dirtyForSaveChanged(dirtyForSave());
198 emit dirtyForUploadChanged(dirtyForUpload());
199
200 _updatePlanCreatorsList();
201}

343void PlanMasterController::loadFromFile(const QString& filename)
344{
345 QString errorString;
346 QString errorMessage = tr("Error loading Plan file (%1). %2").arg(filename).arg("%1");
347
348 if (filename.isEmpty()) {
349 return;
350 }
351
352 QFileInfo fileInfo(filename);
353 QFile file(filename);
354
355 if (!file.open(QIODevice::ReadOnly | QIODevice::Text)) {
356 errorString = file.errorString() + QStringLiteral(" ") + filename;
357 QGC::showAppMessage(errorMessage.arg(errorString));
358 return;
359 }
360
361 bool success = false;
362 if (fileInfo.suffix() == AppSettings::waypointsFileExtension || fileInfo.suffix() == QStringLiteral("txt")) {
363 success = _missionController.loadTextFile(file, errorString);
364 } else {
365 success = _loadPlanJson(file.readAll(), errorString);
366 }
367 if (!success) {
368 QGC::showAppMessage(errorMessage.arg(errorString));
369 }
370
371 if (success){
372 _currentPlanFile = QString::asprintf("%s/%s.%s", fileInfo.path().toLocal8Bit().data(), fileInfo.completeBaseName().toLocal8Bit().data(), AppSettings::planFileExtension);
373 emit currentPlanFileChanged();
374 _setDirtyStates(false /* dirtyForSave */, true /* dirtyForUpload */);
375 } else {
376 _clearCurrentPlanFile();
377 }
378}

380bool PlanMasterController::_loadPlanJson(const QByteArray& bytes, QString& errorString)
381{
382 QJsonDocument jsonDoc;
383 if (!JsonParsing::isJsonFile(bytes, jsonDoc, errorString)) {
384 return false;
385 }
386
387 QJsonObject json = jsonDoc.object();
388 //-- Allow plugins to pre process the load
389 QGCCorePlugin::instance()->preLoadFromJson(this, json);
390
391 int version;
392 if (!JsonParsing::validateExternalQGCJsonFile(json, kPlanFileType, kPlanFileVersion, kPlanFileVersion, version, errorString)) {
393 return false;
394 }
395
396 const QList<JsonParsing::KeyValidateInfo> rgKeyInfo = {
397 { kJsonMissionObjectKey, QJsonValue::Object, true },
398 { kJsonGeoFenceObjectKey, QJsonValue::Object, true },
399 { kJsonRallyPointsObjectKey, QJsonValue::Object, true },
400 };
401 if (!JsonParsing::validateKeys(json, rgKeyInfo, errorString)) {
402 return false;
403 }
404
405 if (!_missionController.load(json[kJsonMissionObjectKey].toObject(), errorString) ||
406 !_geoFenceController.load(json[kJsonGeoFenceObjectKey].toObject(), errorString) ||
407 !_rallyPointController.load(json[kJsonRallyPointsObjectKey].toObject(), errorString)) {
408 return false;
409 }
410
411 //-- Allow plugins to post process the load
412 QGCCorePlugin::instance()->postLoadFromJson(this, json);
413 return true;
414}

416QJsonDocument PlanMasterController::saveToJson()
417{
418 QJsonObject planJson;
419 QGCCorePlugin::instance()->preSaveToJson(this, planJson);
420 QJsonObject missionJson;
421 QJsonObject fenceJson;
422 QJsonObject rallyJson;
423 JsonParsing::saveQGCJsonFileHeader(planJson, kPlanFileType, kPlanFileVersion);
424 //-- Allow plugin to preemptly add its own keys to mission
425 QGCCorePlugin::instance()->preSaveToMissionJson(this, missionJson);
426 _missionController.save(missionJson);
427 //-- Allow plugin to add its own keys to mission
428 QGCCorePlugin::instance()->postSaveToMissionJson(this, missionJson);
429 _geoFenceController.save(fenceJson);
430 _rallyPointController.save(rallyJson);
431 planJson[kJsonMissionObjectKey] = missionJson;
432 planJson[kJsonGeoFenceObjectKey] = fenceJson;
433 planJson[kJsonRallyPointsObjectKey] = rallyJson;
434 QGCCorePlugin::instance()->postSaveToJson(this, planJson);
435 return QJsonDocument(planJson);
436}

449bool PlanMasterController::saveToFile(const QString& filename)
450{
451 if (filename.isEmpty()) {
452 return false;
453 }
454
455 QString planFilename = filename;
456 if (!QFileInfo(filename).fileName().contains(".")) {
457 planFilename += QString(".%1").arg(fileExtension());
458 }
459
460 QFile file(planFilename);
461
462 if (!file.open(QIODevice::WriteOnly | QIODevice::Text)) {
463 QGC::showAppMessage(tr("Plan save error %1 : %2").arg(filename).arg(file.errorString()));
464 return false;
465 } else {
466 const QByteArray saveBytes = saveToJson().toJson();
467 const qint64 bytesWritten = file.write(saveBytes);
468 if (bytesWritten != saveBytes.size()) {
469 QGC::showAppMessage(tr("Plan save error %1 : %2").arg(filename).arg(file.errorString()));
470 return false;
471 }
472 if(_currentPlanFile != planFilename) {
473 _currentPlanFile = planFilename;
474 emit currentPlanFileChanged();
475 }
476 _setDirtyForSave(false);
477 }
478
479 return true;
480}

613{

674void PlanMasterController::_setDirtyForSave(bool dirtyForSave)
675{
676 if (_dirtyForSave != dirtyForSave) {
677 _dirtyForSave = dirtyForSave;
678 emit dirtyForSaveChanged(_dirtyForSave);

710void PlanMasterController::_updatePlanCreatorsList(void)
711{
712 if (_flyView) {
713 return;
714 }
715
716 const auto vehicleClass = _managerVehicle->vehicleClass();
717
718 // Only rebuild if the vehicle class actually changed
719 if (_planCreators && _planCreatorsVehicleClass == vehicleClass) {
720 return;
721 }
722
723 if (!_planCreators) {
724 _planCreators = new QmlObjectListModel(this);
725 } else {
726 _planCreators->clearAndDeleteContents();
727 }
728
729 _planCreatorsVehicleClass = vehicleClass;
730
731 // Allow custom builds to provide their own list of plan creators
732 const QList<PlanCreator*> creators = QGCCorePlugin::instance()->planCreators(this);
733
734 // Filter by vehicle class and add to the model
735 for (PlanCreator* creator : creators) {
736 if (creator->supportsVehicleClass(vehicleClass)) {
737 _planCreators->append(creator);
738 } else {
739 delete creator;
740 }
741 }
742
743 emit planCreatorsChanged(_planCreators);
744}
