65inline bool validPayload(uint16_t message, std::span<const uint8_t> payload)
66{
67 const auto* schema = messageSchema(message);
68 if (!schema) {
69 return true;
70 }
71 if (payload.size() < schema->minimum || payload.size() > schema->maximum ||
72 (payload.size() - schema->minimum) % schema->stride != 0) {
73 return false;
74 }
75 if (message == UBX_MSG_NAV_SAT || message == UBX_MSG_NAV_SVINFO) {
76 return payload.size() == 8 + 12 * size_t(payload[message == UBX_MSG_NAV_SAT ? 5 : 4]) &&
77 (message != UBX_MSG_NAV_SAT || payload[4] == 1);
78 }
79 if (message == UBX_MSG_MON_HW) {
80 return payload.size() == 56 || payload.size() == 60 || payload.size() == 68;
81 }
82 if (message == UBX_MSG_MON_RF) {
83 return payload[0] == 0 && payload[1] > 0 && payload.size() == 4 + 24 * size_t(payload[1]);
84 }
85 if (message == UBX_MSG_MON_COMMS) {
86 return payload[0] == 0 && payload.size() == 8 + 40 * size_t(payload[1]);
87 }
88 if (message == UBX_MSG_SEC_SIG) {
89 return payload[0] == 1 ? payload.size() >= 5
90 : (payload[0] == 2 || payload[0] == 3) && payload.size() == 4 + 4 * size_t(payload[3]);
91 }
92 if (message == UBX_MSG_RXM_COR) {
93 return payload[0] == 1;
94 }
95 return true;
96}
