430struct MessageSchema
431{
432 uint16_t message;
433 size_t minimum;
434 size_t maximum;
435 size_t stride;
436 int towOffset;
437};
438
439template <typename T>
440[[nodiscard]] constexpr MessageSchema fixedSchema(int towOffset = -1)
441{
442 return {T::ID.value(), WIRE_SIZE<T>, WIRE_SIZE<T>, 1, towOffset};
443}
444
445template <typename Header, typename Block>
446[[nodiscard]] constexpr MessageSchema repeatedSchema(MessageId message, size_t maximumBlocks)
447{
448 return {message.value(), WIRE_SIZE<Header>, WIRE_SIZE<Header> + WIRE_SIZE<Block> * maximumBlocks, WIRE_SIZE<Block>,
449 -1};
450}

472[[nodiscard]] constexpr const MessageSchema* messageSchema(uint16_t message)
473{
474 for (const auto& schema : MESSAGE_SCHEMAS) {
475 if (schema.message == message) {
476 return &schema;
477 }
478 }
479 return nullptr;
480}

483[[nodiscard]] inline bool validPayload(uint16_t message, std::span<const uint8_t> payload, const MessageSchema* schema)
484{
485 if (!schema) {
486 return true;
487 }
488 if (payload.size() < schema->minimum || payload.size() > schema->maximum ||
489 (payload.size() - schema->minimum) % schema->stride != 0) {
490 return false;
491 }
492 switch (message) {
493 case Msg::NAV_SAT.value():
494 return payload.size() == schema->minimum + schema->stride * size_t(payload[5]) && payload[4] == 1;
495 case Msg::NAV_SVINFO.value():
496 return payload.size() == schema->minimum + schema->stride * size_t(payload[4]);
497 case Msg::MON_RF.value():
498 return payload[0] == 0 && payload[1] > 0 &&
499 payload.size() == schema->minimum + schema->stride * (size_t(payload[1]) - 1);
500 case Msg::SEC_SIG.value():
501 return payload[0] == 1
502 ? payload.size() >= 5
503 : (payload[0] == 2 || payload[0] == 3) && payload.size() == 4 + 4 * size_t(payload[3]);
504 default:
505 return true;
506 }
507}
