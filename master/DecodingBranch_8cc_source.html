22bool DecodingBranch::build(GstElement* pipeline, GstElement* tee)
23{
24 _pipeline = pipeline;
25
26 const GStreamer::TeeBranch branch = GStreamer::addTeeBranch(_pipeline, tee, "display");
27 if (!branch.valve) {
28 return false;
29 }
30 _valve = branch.valve;
31
32 // Leaky downstream + tiny depth: the live-display branch must drop the oldest
33 // buffer on backpressure, not stall the streaming thread. The recording branch
34 // keeps default non-leaky semantics so every frame reaches the muxer.
35 gst_util_set_object_arg(G_OBJECT(branch.queue), "leaky", "downstream");
36 g_object_set(branch.queue, "max-size-buffers", kDisplayQueueMaxBuffers, "max-size-bytes", 0, "max-size-time",
37 G_GUINT64_CONSTANT(0), nullptr);
38 return true;
39}

41void DecodingBranch::reset()
42{
43 // The video sink outlives the pipeline (owned by the caller of startDecoding), so its probe
44 // and pipeline membership must be dropped even when start() failed before decoding began.
45 if (_decoder || _videoSink) {
46 _shutdown();
47 }
48 // Owned by the pipeline.
49 _valve = nullptr;
50 _pipeline = nullptr;
51}

53void DecodingBranch::start(GstElement* videoSink, bool hasWidget, bool streaming)
54{
55 qCDebug(GstVideoReceiverLog) << "Starting decoding" << _receiver._redactedUri();
56
57 if (!hasWidget) {
58 qCDebug(GstVideoReceiverLog) << "Video Widget is NULL" << _receiver._redactedUri();
59 emit _receiver.onStartDecodingComplete(VideoReceiver::STATUS_FAIL);
60 return;
61 }
62
63 if (!_pipeline && _videoSink) {
64 // Replace a sink stored before start(); also drops its frame probe.
65 _shutdown();
66 }
67
68 if (_videoSink) {
69 qCDebug(GstVideoReceiverLog) << "Already decoding!" << _receiver._redactedUri();
70 emit _receiver.onStartDecodingComplete(VideoReceiver::STATUS_INVALID_STATE);
71 return;
72 }
73
74 const GStreamer::GstPadPtr pad = GStreamer::adoptPad(gst_element_get_static_pad(videoSink, "sink"));
75 if (!pad) {
76 qCCritical(GstVideoReceiverLog) << "Unable to find sink pad of video sink" << _receiver._redactedUri();
77 emit _receiver.onStartDecodingComplete(VideoReceiver::STATUS_FAIL);
78 return;
79 }
80
81 _receiver._watchdog.clearSinkFrames();
82
83 (void) _sinkFrameProbe.add(pad.get(), GST_PAD_PROBE_TYPE_BUFFER, _sinkProbe, this);
84
85 _videoSink = GStreamer::adoptElement(GST_ELEMENT(gst_object_ref(videoSink)));
86 _sinkStatsAtStart = GstQgc::frameStats(_videoSink.get());
87
88 if (!streaming) {
89 emit _receiver.onStartDecodingComplete(VideoReceiver::STATUS_OK);
90 return;
91 }
92
93 emit _receiver.onStartDecodingComplete(_insertDecoder() ? VideoReceiver::STATUS_OK : VideoReceiver::STATUS_FAIL);
94}

96void DecodingBranch::stop()
97{
98 qCDebug(GstVideoReceiverLog) << "Stopping decoding" << _receiver._redactedUri();
99
100 // Gate on _videoSink (set by startDecoding) instead of _decoding (which only flips on the first frame),
101 // so a stop before the first frame still removes the branch.
102 if (!_videoSink) {
103 qCDebug(GstVideoReceiverLog) << "Not decoding!" << _receiver._redactedUri();
104 emit _receiver.onStopDecodingComplete(VideoReceiver::STATUS_INVALID_STATE);
105 return;
106 }
107
108 // The display branch has nothing to flush, so it is torn down at once instead of draining an EOS.
109 if (_valve) {
110 g_object_set(_valve, "drop", TRUE, nullptr);
111 }
112 _shutdown();
113
114 emit _receiver.onStopDecodingComplete(VideoReceiver::STATUS_OK);
115}

137void DecodingBranch::onBusCapsInfo(GstMessage* message)
138{
139 const GstQgc::CapsInfo info = GstQgc::parseCapsInfo(message);
140 // src compared by address only on the GUI thread; never dereferenced (may be gone by then).
141 void* src = GST_MESSAGE_SRC(message);
142 QMetaObject::invokeMethod(
143 &_receiver,
144 [this, info, src]() {
145 for (auto* c : QGCQVideoSinkController::controllersOf(&_receiver)) {
146 if (static_cast<const void*>(c->element()) == src) {
147 c->updateNegotiation(info.format, info.size);
148 }
149 }
150 // Unlike the encoded caps upstream of the decoder, the sink's are always fixed.
151 emit _receiver.videoSizeChanged(info.size);
152 },
153 Qt::QueuedConnection);

184bool DecodingBranch::_addDecoder()
185{
186 _decoder = GStreamer::adoptElement(gst_element_factory_make("decodebin3", nullptr));
187 if (!_decoder) {
188 qCWarning(GstVideoReceiverLog) << "gst_element_factory_make('decodebin3') failed";
189 return false;
190 }
191 // Ours alongside the pipeline's, so decoder() stays valid until _shutdown().
192 (void) gst_object_ref_sink(_decoder.get());
193
194 (void) g_signal_connect(_decoder.get(), "deep-element-added", G_CALLBACK(_onDecoderElementAdded), this);
195
196 (void) gst_bin_add(GST_BIN(_pipeline), _decoder.get());
197 (void) gst_element_sync_state_with_parent(_decoder.get());
198
199 _receiver._dotDumper.dump(_pipeline, "pipeline-with-decoder");
200
201 if (!gst_element_link(_valve, _decoder.get())) {
202 qCWarning(GstVideoReceiverLog) << "Unable to link decoder";
203 GStreamer::disposeChild(_decoder.get());
204 _decoder.reset();
205 return false;
206 }
207
208 if (const GStreamer::GstPadPtr srcPad = GStreamer::firstSrcPad(_decoder.get())) {
209 onDecoderPad(srcPad.get());
210 } else {
211 (void) g_signal_connect(_decoder.get(), "pad-added", G_CALLBACK(GstVideoReceiver::_onNewPad), &_receiver);
212 }
213
214 return true;
215}

236bool DecodingBranch::_addVideoSink(GstPad* pad)
237{
238 _ensureVideoSinkInPipeline();
239
240 const GStreamer::GstPadPtr sinkPad = GStreamer::adoptPad(gst_element_get_static_pad(_videoSink.get(), "sink"));
241 const GstPadLinkReturn linkRet = sinkPad ? gst_pad_link(pad, sinkPad.get()) : GST_PAD_LINK_WRONG_HIERARCHY;
242 if (linkRet != GST_PAD_LINK_OK) {
243 qCWarning(GstVideoReceiverLog) << "Unable to link decoder pad to video sink:" << gst_pad_link_get_name(linkRet);

246 GStreamer::disposeChild(_videoSink.get());
247 return false;
248 }
249

251
252 _receiver._dotDumper.dump(_pipeline, "pipeline-with-videosink");
253

267void DecodingBranch::_shutdown()
268{
269 if (_decoder) {
270 GStreamer::disposeChild(_decoder.get());
271 _decoder.reset();
272 }
273
274 _sinkFrameProbe.remove();
275
276 _receiver._watchdog.clearSinkFrames();
277
278 if (_videoSink) {
279 GStreamer::disposeChild(_videoSink.get());
280 _videoSink.reset();
281 _releaseDisplayedFrame();
282 }
283 _sinkStatsAtStart.reset();
284
285 if (_decoding.exchange(false)) {
286 qCDebug(GstVideoReceiverLog) << "Decoding stopped";
287 emit _receiver.decodingChanged(false);
288 }

290 _receiver._dotDumper.dump(_pipeline, "pipeline-decoding-stopped");

305void DecodingBranch::_onDecoderElementAdded(GstBin* /*decodebin*/, GstBin* /*subBin*/, GstElement* element,
306 gpointer data)
307{
308 GstElementFactory* factory = gst_element_get_factory(element);
309 if (!factory || !gst_element_factory_list_is_type(factory, GST_ELEMENT_FACTORY_TYPE_DECODER)) {
310 return;
311 }
312
313 // Live streams: once a frame is late, the videodecoder base class advances earliest_time and drops every frame
314 // after it. Turned off here, before the decoder sees data, for each decoder decodebin3 plugs, later swaps included.
315 if (g_object_class_find_property(G_OBJECT_GET_CLASS(element), "qos")) {
316 g_object_set(element, "qos", FALSE, nullptr);
317 }
318
319 qCDebug(GstVideoReceiverLog).noquote()
320 << "Decodebin3 selected" << GStreamer::describeDecoderFactory(factory) << "- QoS disabled";
321 const QString decoderName = QString::fromUtf8(gst_plugin_feature_get_name(GST_PLUGIN_FEATURE(factory)));
322
323 auto* self = static_cast<DecodingBranch*>(data);
324 self->_receiver._runOnWorkerForPipeline([self, decoderName]() {
325 {
326 QMutexLocker locker(&self->_decoderNameMutex);
327 if (decoderName == self->_decoderName) {
328 return;
329 }
330 self->_decoderName = decoderName;
331 }
332 (void) QMetaObject::invokeMethod(
333 &self->_receiver, [self]() { emit self->_receiver.decoderNameChanged(); }, Qt::QueuedConnection);
334 });
335}
