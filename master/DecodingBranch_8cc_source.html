19void setSinkSync(GstElement* sink, bool sync)
20{
21 if (g_object_class_find_property(G_OBJECT_GET_CLASS(sink), "sync")) {

59bool DecodingBranch::build()
60{
61 const GStreamer::TeeBranch branch = GStreamer::addTeeBranch(_context.pipeline, _context.tee, "display");
62 if (!branch.valve) {
63 return false;
64 }
65 _queue = branch.queue;
66 _valve = branch.valve;
67 GStreamer::markBranch(_queue, GStreamer::PipelineBranch::Decoding);
68 GStreamer::markBranch(_valve, GStreamer::PipelineBranch::Decoding);
69
70 // Tiny and leaky: the live display must drop its oldest frame on backpressure, never stall the tee.
71 gst_util_set_object_arg(G_OBJECT(_queue), "leaky", "downstream");
72 g_object_set(_queue, "max-size-buffers", kDisplayQueueMaxBuffers, "max-size-bytes", 0, "max-size-time",
73 G_GUINT64_CONSTANT(0), nullptr);
74
75 // "overrun" fires before the queue drops its oldest frame.
76 (void) _queueOverrun.connect(_queue, "overrun", G_CALLBACK(_onQueueOverrun), this);
77 const GStreamer::GstPadPtr queueSrc = GStreamer::adoptPad(gst_element_get_static_pad(_queue, "src"));
78 if (!queueSrc || !_keyframeGate.add(queueSrc.get(), GST_PAD_PROBE_TYPE_BUFFER, _gateProbe, this)) {
79 qCCritical(GstVideoReceiverLog) << "Failed to add the display keyframe gate";
80 return false;
81 }
82 return true;
83}

173void DecodingBranch::_insertDecoder()
174{
175 setSinkSync(_sink.get(), !_lowLatency);
176 if (!gst_bin_add(GST_BIN(_context.pipeline), _sink.get())) {
177 qCWarning(GstVideoReceiverLog) << "Failed to add the video sink" << _context.redactedUri;
178 return;
179 }
180 // PAUSED (not READY) negotiates downstream caps before the decoder pushes; PLAYING once linked.
181 (void) gst_element_set_state(_sink.get(), GST_STATE_PAUSED);
182
183 _decoder = GStreamer::makeElement("decodebin3");
184 if (!_decoder) {
185 qCWarning(GstVideoReceiverLog) << "gst_element_factory_make('decodebin3') failed";
186 GStreamer::disposeChild(_sink.get());
187 return;
188 }
189 GStreamer::markBranch(_decoder.get(), GStreamer::PipelineBranch::Decoding);
190 (void) _decoderElementAdded.connect(_decoder.get(), "deep-element-added", G_CALLBACK(_onDecoderElementAdded), this);
191 (void) _decoderPadAdded.connect(_decoder.get(), "pad-added", G_CALLBACK(_onDecoderPadAdded), this);
192
193 if (!gst_bin_add(GST_BIN(_context.pipeline), _decoder.get()) ||
194 !gst_element_sync_state_with_parent(_decoder.get()) || !gst_element_link(_valve, _decoder.get())) {
195 qCWarning(GstVideoReceiverLog) << "Unable to insert the decoder" << _context.redactedUri;
196 _removeDecoder();
197 return;
198 }
199 _context.dotDumper->dump(_context.pipeline, "pipeline-with-decoder");
200
201 (void) _insertions.fetch_add(1, std::memory_order_relaxed);
202 _context.watchdog->startSinkClock(StreamWatchdog::now());
203 _gateClosed.store(false, std::memory_order_relaxed);
204 g_object_set(_valve, "drop", FALSE, nullptr);
205
206 // Without a keyframe the new decoder has nothing to start from until the stream's next one.
207 const GStreamer::GstPadPtr valveSink = GStreamer::adoptPad(gst_element_get_static_pad(_valve, "sink"));
208 GStreamer::requestKeyframe(valveSink.get());
209
210 qCDebug(GstVideoReceiverLog) << "Decoder inserted" << _context.redactedUri;
211}

213void DecodingBranch::_removeDecoder()
214{
215 if (_valve) {
216 g_object_set(_valve, "drop", TRUE, nullptr);
217 }
218 // The sink first: going NULL releases a decoder thread waiting inside it, so the decoder can stop.
219 if (_sink) {
220 GStreamer::disposeChild(_sink.get());
221 }
222 if (_decoder) {
223 _decoderPadAdded.disconnect();
224 _decoderElementAdded.disconnect();
225 GStreamer::disposeChild(_decoder.get());
226 _decoder.reset();
227 _context.dotDumper->dump(_context.pipeline, "pipeline-decoding-stopped");
228 }

273void DecodingBranch::_onQueueOverrun(GstElement* /*queue*/, gpointer data)
274{
275 auto* const self = static_cast<DecodingBranch*>(data);
276 if (self->_gateClosed.exchange(true, std::memory_order_relaxed)) {
277 return;
278 }
279 self->_gateDrops.store(0, std::memory_order_relaxed);
280 self->_context.post([self]() {
281 if (self->_valve) {
282 const GStreamer::GstPadPtr valveSink =
283 GStreamer::adoptPad(gst_element_get_static_pad(self->_valve, "sink"));
284 GStreamer::requestKeyframe(valveSink.get());
285 }

289GstPadProbeReturn DecodingBranch::_gateProbe(GstPad* /*pad*/, GstPadProbeInfo* info, gpointer data)
290{
291 auto* const self = static_cast<DecodingBranch*>(data);
292 if (!self->_gateClosed.load(std::memory_order_relaxed)) {
293 return GST_PAD_PROBE_OK;
294 }
295 const GstBuffer* const buffer = GST_PAD_PROBE_INFO_BUFFER(info);
296 const bool delta = buffer && GST_BUFFER_FLAG_IS_SET(buffer, GST_BUFFER_FLAG_DELTA_UNIT);
297 if (delta && (self->_gateDrops.fetch_add(1, std::memory_order_relaxed) < kMaxGateDrops)) {
298 return GST_PAD_PROBE_DROP;
299 }
300 self->_gateClosed.store(false, std::memory_order_relaxed);
301 return GST_PAD_PROBE_OK;
302}
