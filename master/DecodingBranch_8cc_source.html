150SinkFrameCounts sinkFrameCountsSince(const GstStructure* stats, const GstStructure* since)
151{
152 const auto growth = [stats, since](const char* field) { return GstQgc::frameStatsGrowth(stats, since, field); };

176bool DecodingBranch::build()
177{
178 if (!_head.build(_context.pipeline, _context.tee, GStreamer::PipelineBranch::Decoding, "display")) {
179 return false;
180 }
181 GstElement* const queue = _head.queue();
182
183 // Tiny and leaky: the live display must drop its oldest frame on backpressure, never stall the tee. The
184 // backpressure comes late, though: ahead of its decoder, decodebin3's multiqueue first buffers the stream, by
185 // time and with no limit on buffers. It holds its min-interleave-time: 250 ms by default, the decoder's latency
186 // plus 100 ms once the decoder reports one, and no less than the pipeline's latency (GStreamer 1.24, and the
187 // later 1.22 releases). Only a decoder that falls further behind than that makes this queue leak.
188 gst_util_set_object_arg(G_OBJECT(queue), "leaky", "downstream");
189 g_object_set(queue, "max-size-buffers", kDisplayQueueMaxBuffers, "max-size-bytes", 0, "max-size-time",
190 G_GUINT64_CONSTANT(0), nullptr);
191
192 // "overrun" announces a leak; the gate's probe sees whether one followed.
193 _head.onOverrun(G_CALLBACK(_onQueueOverrun), this);
194 const GStreamer::GstPadPtr queueSrc = GStreamer::adoptPad(gst_element_get_static_pad(queue, "src"));
195 if (!queueSrc || !_keyframeGate.add(queueSrc.get(), GST_PAD_PROBE_TYPE_BUFFER, _gateProbe, this)) {
196 qCCritical(GstVideoReceiverLog) << "Failed to add the display keyframe gate";
197 return false;
198 }
199 return true;
200}

259{
260 if (!_decoder) {
261 return;
262 }
263

366bool DecodingBranch::_tryInsertDecoder()
367{
368 setSinkSync(_sink.get(), !_lowLatency);
369 if (!gst_bin_add(GST_BIN(_context.pipeline), _sink.get())) {
370 qCWarning(GstVideoReceiverLog) << "Failed to add the video sink" << _context.redactedUri;
371 return false;
372 }
373 // PAUSED (not READY) negotiates downstream caps before the decoder pushes; PLAYING once linked.
374 (void) gst_element_set_state(_sink.get(), GST_STATE_PAUSED);
375
376 _decoder = GStreamer::makeElement("decodebin3");
377 if (!_decoder) {
378 qCWarning(GstVideoReceiverLog) << "gst_element_factory_make('decodebin3') failed";
379 GStreamer::disposeChild(_sink.get());
380 return false;
381 }
382 GStreamer::markBranch(_decoder.get(), GStreamer::PipelineBranch::Decoding);
383 TeeBranchHead::absorbFlowErrors(_decoder.get());
384 (void) _decoderElementAdded.connect(_decoder.get(), "deep-element-added", G_CALLBACK(_onDecoderElementAdded), this);
385 (void) _decoderPadAdded.connect(_decoder.get(), "pad-added", G_CALLBACK(_onDecoderPadAdded), this);
386 (void) g_signal_connect(_decoder.get(), "select-stream", G_CALLBACK(decodeVideoOnly), nullptr);
387
388 if (!gst_bin_add(GST_BIN(_context.pipeline), _decoder.get()) ||
389 !gst_element_sync_state_with_parent(_decoder.get()) || !gst_element_link(_head.valve(), _decoder.get())) {
390 qCWarning(GstVideoReceiverLog) << "Unable to insert the decoder" << _context.redactedUri;
391 _removeDecoder();
392 return false;
393 }
394 _context.dotDumper->dump(_context.pipeline, "pipeline-with-decoder");
395
396 (void) _insertions.fetch_add(1, std::memory_order_relaxed);
397 _context.watchdog->startSinkClock(StreamWatchdog::now());
398 _gateClosed.store(false, std::memory_order_relaxed);
399 // Without a keyframe the new decoder has nothing to start from until the stream's next one.
400 _head.open();
401
402 qCDebug(GstVideoReceiverLog) << "Decoder inserted" << _context.redactedUri;
403 return true;
404}

406void DecodingBranch::_removeDecoder()
407{
408 _head.close();
409 // The sink first: going NULL releases a decoder thread waiting inside it, so the decoder can stop.
410 if (_sink) {

414 _sinkLateFramesSeen = 0;

416 if (_decoder) {
417 _decoderPadAdded.disconnect();
418 _decoderElementAdded.disconnect();
419 GStreamer::disposeChild(_decoder.get());
420 _decoder.reset();
421 _pluggedDecoder.reset();

500GstPadProbeReturn DecodingBranch::_gateProbe(GstPad* /*pad*/, GstPadProbeInfo* info, gpointer data)
501{
502 auto* const self = static_cast<DecodingBranch*>(data);
503 const GstBuffer* const buffer = GST_PAD_PROBE_INFO_BUFFER(info);
504 // The queue marks the buffer that follows a leak. Each overrun before it leaked one buffer, as the queue fills by
505 // buffer count only. An overrun before a buffer without the mark leaked nothing, and is forgotten: it would be
506 // taken for the leak behind a mark that upstream sets later, for a loss ahead of the tee.
507 if (buffer && GST_BUFFER_FLAG_IS_SET(buffer, GST_BUFFER_FLAG_DISCONT)) {
508 const quint64 leaked = self->_overruns.exchange(0, std::memory_order_relaxed);
509 (void) self->_framesDropped.fetch_add(leaked, std::memory_order_relaxed);
510 if ((leaked > 0) && !self->_gateClosed.exchange(true, std::memory_order_relaxed)) {
511 self->_gateClosedAtMs.store(StreamWatchdog::now(), std::memory_order_relaxed);
512 self->_context.post([self]() { self->_head.requestKeyframe(); });
513 }
514 } else {
515 self->_overruns.store(0, std::memory_order_relaxed);
516 }
517 if (!self->_gateClosed.load(std::memory_order_relaxed)) {
518 return GST_PAD_PROBE_OK;
519 }
520 if (buffer && GST_BUFFER_FLAG_IS_SET(buffer, GST_BUFFER_FLAG_DELTA_UNIT)) {
521 const qint64 now = StreamWatchdog::now();
522 if ((now - self->_gateClosedAtMs.load(std::memory_order_relaxed)) < kMaxGateMs) {
523 (void) self->_framesDropped.fetch_add(1, std::memory_order_relaxed);
524 // The gate's time is not the decoder's: with the shortest stream timeout the decoder counts as stalled
525 // after as long as the gate may stay closed.
526 self->_context.watchdog->noteFrameWithheld(now);
527 return GST_PAD_PROBE_DROP;
528 }
529 // Told once at a level that shows: the same holds for every later request to this stream's sender.
530 if (!self->_gateTimeoutWarned.exchange(true, std::memory_order_relaxed)) {
531 qCWarning(GstVideoReceiverLog) << "No keyframe within" << kMaxGateMs
532 << "ms of the request that followed a dropped frame: the picture goes on "
533 "without one. The stream's sender may not take keyframe requests."
534 << self->_context.redactedUri;
535 } else {
536 qCDebug(GstVideoReceiverLog) << "No keyframe within" << kMaxGateMs << "ms of the request"
537 << self->_context.redactedUri;
538 }
539 }
540 self->_gateClosed.store(false, std::memory_order_relaxed);
541 return GST_PAD_PROBE_OK;
542}

567void DecodingBranch::_onDecoderElementAdded(GstBin* decodebin, GstBin* /*subBin*/, GstElement* element, gpointer data)
568{
569 // A message can outlive the decoder that held its source, which leaves nothing to blame it on but its own mark.
570 GStreamer::markBranch(element, GStreamer::PipelineBranch::Decoding);
571
572 GstElementFactory* factory = gst_element_get_factory(element);
573 if (!factory || !gst_element_factory_list_is_type(factory, GST_ELEMENT_FACTORY_TYPE_DECODER)) {
574 return;
575 }
576
577 // Live streams: once a frame is late, a QoS event has the videodecoder base class drop the frames due before the
578 // time it names. Turned off here, before the decoder sees data, for each decoder decodebin3 plugs, later swaps
579 // included. That stops the base class only: avdec and amcvideodec act on the event's deadline themselves, and it
580 // is the sink bin, which keeps its sink's QoS off, that sends them none.
581 if (g_object_class_find_property(G_OBJECT_GET_CLASS(element), "qos")) {
582 g_object_set(element, "qos", FALSE, nullptr);
583 }
584 // A decoder takes a source that isn't live (tcp://) for a file, and trades delay for throughput.
585 const bool live = fedByLiveSource(GST_ELEMENT(decodebin));
586 if (!live) {
587 decodeWithoutDelay(element);
588 }
