56SinkFrameCounts sinkFrameCountsSince(const GstStructure* stats, const GstStructure* since)

82bool DecodingBranch::build()
83{
84 if (!_head.build(_context.pipeline, _context.tee, GStreamer::PipelineBranch::Decoding, "display")) {
85 return false;
86 }
87 GstElement* const queue = _head.queue();
88
89 // Tiny and leaky: the live display must drop its oldest frame on backpressure, never stall the tee.
90 gst_util_set_object_arg(G_OBJECT(queue), "leaky", "downstream");
91 g_object_set(queue, "max-size-buffers", kDisplayQueueMaxBuffers, "max-size-bytes", 0, "max-size-time",
92 G_GUINT64_CONSTANT(0), nullptr);
93
94 // "overrun" fires before the queue drops its oldest frame.
95 _head.onOverrun(G_CALLBACK(_onQueueOverrun), this);
96 const GStreamer::GstPadPtr queueSrc = GStreamer::adoptPad(gst_element_get_static_pad(queue, "src"));
97 if (!queueSrc || !_keyframeGate.add(queueSrc.get(), GST_PAD_PROBE_TYPE_BUFFER, _gateProbe, this)) {
98 qCCritical(GstVideoReceiverLog) << "Failed to add the display keyframe gate";
99 return false;
100 }
101 return true;
102}

195DecodingBranch::StatsSample DecodingBranch::sampleStats()
196{
197 const GStreamer::GstStructurePtr stats = GstQgc::frameStats(_sink.get());
198 const SinkFrameCounts sinkCounts = sinkFrameCountsSince(stats.get(), _sinkStatsAtStart.get());
199
200 quint64 busiestPathFrames = 0;
201 for (std::size_t i = 0; i < GstHw::kPathCount; ++i) {
202 const auto path = HwVideoBufferPath(i);
203 const quint64 frames = GstQgc::frameStatsField(stats.get(), GstQgc::sinkFramesField(path));
204 const quint64 sinceSample = (frames > _pathFramesAtSample[i]) ? (frames - _pathFramesAtSample[i]) : 0;
205 if (sinceSample > busiestPathFrames) {
206 busiestPathFrames = sinceSample;
207 _displayPath = QString::fromLatin1(GstHw::toString(path));
208 }
209 _pathFramesAtSample[i] = frames;
210 }
211
212 return {
213 .framesDisplayed = _retiredFramesDisplayed + sinkCounts.displayed,
214 .framesDropped =
215 _framesDropped.load(std::memory_order_relaxed) + _retiredFramesSuperseded + sinkCounts.superseded,
216 .displayPath = _displayPath,
217 .gpuFallbacks = _retiredGpuFallbacks + sinkCounts.gpuFallbacks,
218 .decoderRestarts = _decoderRestarts,
219 };

234void DecodingBranch::_insertDecoder()
235{
236 setSinkSync(_sink.get(), !_lowLatency);
237 if (!gst_bin_add(GST_BIN(_context.pipeline), _sink.get())) {
238 qCWarning(GstVideoReceiverLog) << "Failed to add the video sink" << _context.redactedUri;
239 return;
240 }
241 // PAUSED (not READY) negotiates downstream caps before the decoder pushes; PLAYING once linked.
242 (void) gst_element_set_state(_sink.get(), GST_STATE_PAUSED);
243
244 _decoder = GStreamer::makeElement("decodebin3");
245 if (!_decoder) {
246 qCWarning(GstVideoReceiverLog) << "gst_element_factory_make('decodebin3') failed";
247 GStreamer::disposeChild(_sink.get());
248 return;
249 }
250 GStreamer::markBranch(_decoder.get(), GStreamer::PipelineBranch::Decoding);
251 (void) _decoderElementAdded.connect(_decoder.get(), "deep-element-added", G_CALLBACK(_onDecoderElementAdded), this);
252 (void) _decoderPadAdded.connect(_decoder.get(), "pad-added", G_CALLBACK(_onDecoderPadAdded), this);
253
254 if (!gst_bin_add(GST_BIN(_context.pipeline), _decoder.get()) ||
255 !gst_element_sync_state_with_parent(_decoder.get()) || !gst_element_link(_head.valve(), _decoder.get())) {
256 qCWarning(GstVideoReceiverLog) << "Unable to insert the decoder" << _context.redactedUri;
257 _removeDecoder();
258 return;
259 }
260 _context.dotDumper->dump(_context.pipeline, "pipeline-with-decoder");
261
262 (void) _insertions.fetch_add(1, std::memory_order_relaxed);
263 _context.watchdog->startSinkClock(StreamWatchdog::now());
264 _gateClosed.store(false, std::memory_order_relaxed);
265 // Without a keyframe the new decoder has nothing to start from until the stream's next one.
266 _head.open();
267
268 qCDebug(GstVideoReceiverLog) << "Decoder inserted" << _context.redactedUri;
269}

329void DecodingBranch::_onQueueOverrun(GstElement* /*queue*/, gpointer data)
330{
331 auto* const self = static_cast<DecodingBranch*>(data);
332 // Each overrun leaks the oldest buffer: the queue fills by buffer count only.
333 (void) self->_framesDropped.fetch_add(1, std::memory_order_relaxed);
334 if (self->_gateClosed.exchange(true, std::memory_order_relaxed)) {
335 return;
336 }
337 self->_gateDrops.store(0, std::memory_order_relaxed);
338 self->_context.post([self]() { self->_head.requestKeyframe(); });
339}

341GstPadProbeReturn DecodingBranch::_gateProbe(GstPad* /*pad*/, GstPadProbeInfo* info, gpointer data)
342{
343 auto* const self = static_cast<DecodingBranch*>(data);
344 if (!self->_gateClosed.load(std::memory_order_relaxed)) {
345 return GST_PAD_PROBE_OK;
346 }
347 const GstBuffer* const buffer = GST_PAD_PROBE_INFO_BUFFER(info);
348 const bool delta = buffer && GST_BUFFER_FLAG_IS_SET(buffer, GST_BUFFER_FLAG_DELTA_UNIT);
349 if (delta && (self->_gateDrops.fetch_add(1, std::memory_order_relaxed) < kMaxGateDrops)) {
350 (void) self->_framesDropped.fetch_add(1, std::memory_order_relaxed);
351 return GST_PAD_PROBE_DROP;
352 }
353 self->_gateClosed.store(false, std::memory_order_relaxed);
354 return GST_PAD_PROBE_OK;
355}
