39inline femto_uav_gps_t decodePosition(std::span<const uint8_t> bytes)
40{
41 femto_uav_gps_t value{};
42 value.time_utc_usec = LittleEndian::read<uint64_t>(bytes, 0).value_or(0);
43 value.lat = LittleEndian::read<int32_t>(bytes, 8).value_or(0);
44 value.lon = LittleEndian::read<int32_t>(bytes, 12).value_or(0);
45 value.alt = LittleEndian::read<int32_t>(bytes, 16).value_or(0);
46 value.alt_ellipsoid = LittleEndian::read<int32_t>(bytes, 20).value_or(0);
47 value.s_variance_m_s = LittleEndian::read<float>(bytes, 24).value_or(0);
48 value.c_variance_rad = LittleEndian::read<float>(bytes, 28).value_or(0);
49 value.eph = LittleEndian::read<float>(bytes, 32).value_or(0);
50 value.epv = LittleEndian::read<float>(bytes, 36).value_or(0);
51 value.hdop = LittleEndian::read<float>(bytes, 40).value_or(0);
52 value.vdop = LittleEndian::read<float>(bytes, 44).value_or(0);
53 value.noise_per_ms = LittleEndian::read<int32_t>(bytes, 48).value_or(0);
54 value.jamming_indicator = LittleEndian::read<int32_t>(bytes, 52).value_or(0);
55 value.vel_m_s = LittleEndian::read<float>(bytes, 56).value_or(0);
56 value.vel_n_m_s = LittleEndian::read<float>(bytes, 60).value_or(0);
57 value.vel_e_m_s = LittleEndian::read<float>(bytes, 64).value_or(0);
58 value.vel_d_m_s = LittleEndian::read<float>(bytes, 68).value_or(0);
59 value.cog_rad = LittleEndian::read<float>(bytes, 72).value_or(0);
60 value.timestamp_time_relative = LittleEndian::read<int32_t>(bytes, 76).value_or(0);
61 value.heading = LittleEndian::read<float>(bytes, 80).value_or(0);
62 value.fix_type = LittleEndian::read<uint8_t>(bytes, 84).value_or(0);
63 value.vel_ned_valid = LittleEndian::read<uint8_t>(bytes, 85).value_or(0);
64 value.satellites_used = LittleEndian::read<uint8_t>(bytes, 86).value_or(0);
65 value.heading_type = LittleEndian::read<uint8_t>(bytes, 87).value_or(0);
66 return value;
67}

70int GPSNativeFemto::handleMessage(int len)
71{
72 int ret = 0;
73 if (_output_mode != OutputMode::RTCM &&
74 (len < _femto_msg.header[3] || _femto_msg.payloadLength > len - _femto_msg.header[3])) {
75 return 0;
76 }
77 uint16_t messageid = _femto_msg.messageId;
78
79 if (messageid == FEMTO_MSG_ID_UAVGPS) {
80 if (_femto_msg.payloadLength < Femto::GPS_PAYLOAD_SIZE) {
81 return 0;
82 }
83 const auto position = decodePosition({_femto_msg.data, _femto_msg.payloadLength});
84
85 _gps_position->time_utc_usec = position.time_utc_usec;
86 _gps_position->latitude_deg = position.lat / 1e7;
87 _gps_position->longitude_deg = position.lon / 1e7;
88 _gps_position->altitude_msl_m = position.alt / 1e3;
89 _gps_position->altitude_ellipsoid_m = position.alt_ellipsoid / 1e3;
90 _gps_position->speedAccuracyMetersPerSecond = position.s_variance_m_s;
91 _gps_position->courseAccuracyRadians = position.c_variance_rad;
92 _gps_position->eph = position.eph;
93 _gps_position->accuracy_timestamp = nowUs();
94 _gps_position->epv = position.epv;
95 _gps_position->hdop = position.hdop;
96 _gps_position->dop_timestamp = nowUs();
97 _gps_position->vdop = position.vdop;
98 _integrity.noise_per_ms = position.noise_per_ms;
99 _integrity.rf_timestamp = nowUs();
100 _integrity.jamming_indicator = position.jamming_indicator;
101 _integrity.rf_timestamp = nowUs();
102 publishIntegrity();
103 _gps_position->vel_m_s = position.vel_m_s;
104 _gps_position->vel_n_m_s = position.vel_n_m_s;
105 _gps_position->vel_e_m_s = position.vel_e_m_s;
106 _gps_position->vel_d_m_s = position.vel_d_m_s;
107 _gps_position->cog_rad = position.cog_rad;
108 _gps_position->timestamp_time_relative = position.timestamp_time_relative;
109 _gps_position->fix_type = GPSPositionReport::fixTypeFromValue(position.fix_type);
110 _gps_position->vel_ned_valid = position.vel_ned_valid;
111 _gps_position->satellites_used = position.satellites_used;
112
113 if (position.heading_type == 6) {
114 float heading = position.heading;
115 heading *= GPS_PI / 180.0f; // deg to rad, now in range [0, 2pi]
116 heading -= _heading_offset; // range: [-pi, 3pi]
117
118 if (heading > GPS_PI) {
119 heading -= 2.f * GPS_PI; // final range is [-pi, pi]
120 }
121
122 _gps_position->heading = heading;
123 _gps_position->heading_timestamp = nowUs();
124
125 } else {
126 _gps_position->heading = NAN;
127 _gps_position->heading_timestamp = nowUs();
128 }
129
130 _gps_position->timestamp = nowUs();
131
132 ret = 1;
133
134 } else if (_satellite_info && messageid == FEMTO_MSG_ID_UAVSTATUS) {
135 if (_femto_msg.payloadLength < Femto::STATUS_HEADER_SIZE) {
136 return 0;
137 }
138 const std::span<const uint8_t> status(_femto_msg.data, _femto_msg.payloadLength);
139 const auto count = LittleEndian::read<uint32_t>(status, 36).value_or(0);
140 if (count > 64 || status.size() < 40 + count * 8) {
141 return 0;
142 }
143
144 _satellite_info->timestamp = nowUs();
145 _satellite_info->count = std::min<uint32_t>(count, GPSNativeSatelliteReport::SAT_INFO_MAX_SATELLITES);
146
147 for (size_t i = 0; i < _satellite_info->count; i++) {
148 _satellite_info->entries[i].id = LittleEndian::read<uint8_t>(status, 40 + i * 8 + 0).value_or(0);
149 _satellite_info->entries[i].used.reset();
150 _satellite_info->entries[i].elevation = LittleEndian::read<uint8_t>(status, 40 + i * 8 + 3).value_or(0);
151 _satellite_info->entries[i].azimuth = LittleEndian::read<uint16_t>(status, 40 + i * 8 + 4).value_or(0);
152 _satellite_info->entries[i].signal = LittleEndian::read<uint8_t>(status, 40 + i * 8 + 2).value_or(0);
153 _satellite_info->entries[i].prn = LittleEndian::read<uint8_t>(status, 40 + i * 8 + 0).value_or(0);
154 }
155
156 ret = 2;
157
158 } else if (OutputMode::RTCM == _output_mode && messageid == FEMTO_MSG_ID_GPGGA &&
159 (memcmp(_femto_msg.data + 3, "GGA,", 3) == 0)) {
160 const auto parsed = NMEA::sentence({reinterpret_cast<const char*>(_femto_msg.data), static_cast<size_t>(len)});
161 const auto fix = parsed ? NMEA::gga(*parsed) : std::nullopt;
162 if (!fix) {
163 return 0;
164 }
165 if (!_correction_output_activated && fix->quality == 7) {
166 _survey_in_start = 0;
167 sendSurveyInStatusUpdate(false, true, fix->latitude, fix->longitude, fix->altitude + fix->geoidSeparation);
168 _rtcmActivationPending = true;
169 }
170 if (_satellite_info) {
171 publishSatelliteUsage(fix->satellitesUsed);
172 }
173 }
174
175 // handle survey-in status update
176 if (_survey_in_start != 0) {
177 const uint64_t now = nowUs();
178 uint32_t survey_in_duration = (now - _survey_in_start) / 1000000;
179
180 if (survey_in_duration != _survey_duration) {
181 _survey_duration = survey_in_duration;
182 sendSurveyInStatusUpdate(true, false);
183 }
184 }
185
186 return ret;
187}

189int GPSNativeFemto::parseChar(uint8_t temp)
190{
191 int iRet = 0;
192
193 if (_rtcm_parsing) {
194 if (_rtcm_parsing->addByte(temp) && _rtcm_parsing->valid()) {
195 gotRTCMMessage(_rtcm_parsing->frame().data(), _rtcm_parsing->frame().size());
196 decodeInit();
197 _rtcm_parsing->reset();
198 return iRet;
199 }
200 }
201
202 if (_output_mode == OutputMode::GPS) {
203 switch (_decode_state) {
204 case FemtoDecodeState::pream_ble1:
205 if (temp == FEMTO_PREAMBLE1) {
206 _decode_state = FemtoDecodeState::pream_ble2;
207 _femto_msg.read = 0;
208 }
209
210 break;
211
212 case FemtoDecodeState::pream_ble2:
213 if (temp == FEMTO_PREAMBLE2) {
214 _decode_state = FemtoDecodeState::pream_ble3;
215
216 } else {
217 _decode_state = FemtoDecodeState::pream_ble1;
218 }
219
220 break;
221
222 case FemtoDecodeState::pream_ble3:
223 if (temp == FEMTO_PREAMBLE3) {
224 _decode_state = FemtoDecodeState::head_length;
225
226 } else {
227 _decode_state = FemtoDecodeState::pream_ble1;
228 }
229
230 break;
231
232 case FemtoDecodeState::head_length:
233 if (temp != Femto::HEADER_SIZE) {
234 _decode_state = FemtoDecodeState::pream_ble1;
235 break;
236 }
237 _femto_msg.header[0] = FEMTO_PREAMBLE1;
238 _femto_msg.header[1] = FEMTO_PREAMBLE2;
239 _femto_msg.header[2] = FEMTO_PREAMBLE3;
240 _femto_msg.header[3] = temp;
241 _decode_state = FemtoDecodeState::head_data;
242 _femto_msg.read = 4;
243 break;
244
245 case FemtoDecodeState::head_data:
246 if (_femto_msg.read >= sizeof(_femto_msg.header)) {
247 _decode_state = FemtoDecodeState::pream_ble1;
248 break;
249 }
250
251 _femto_msg.header[_femto_msg.read] = temp;
252 _femto_msg.read++;
253
254 if (_femto_msg.read >= _femto_msg.header[3]) {
255 _femto_msg.messageId = LittleEndian::read<uint16_t>(_femto_msg.header, 4).value_or(0);
256 _femto_msg.payloadLength = LittleEndian::read<uint16_t>(_femto_msg.header, 8).value_or(0);
257 if (_femto_msg.payloadLength > sizeof(_femto_msg.data)) {
258 _decode_state = FemtoDecodeState::pream_ble1;
259 } else {
260 _decode_state = _femto_msg.payloadLength == 0 ? FemtoDecodeState::crc1 : FemtoDecodeState::data;
261 }
262 }
263
264 break;
265
266 case FemtoDecodeState::data:
267 if (static_cast<size_t>(_femto_msg.read - _femto_msg.header[3]) >= sizeof(_femto_msg.data)) {
268 _decode_state = FemtoDecodeState::pream_ble1;
269 break;
270 }
271
272 _femto_msg.data[_femto_msg.read - _femto_msg.header[3]] = temp;
273 _femto_msg.read++;
274
275 if (_femto_msg.read >= (_femto_msg.payloadLength + _femto_msg.header[3])) {
276 _decode_state = FemtoDecodeState::crc1;
277 }
278
279 break;
280
281 case FemtoDecodeState::crc1:
282 _femto_msg.crc = (uint32_t(temp) << 0);
283 _decode_state = FemtoDecodeState::crc2;
284 break;
285
286 case FemtoDecodeState::crc2:
287 _femto_msg.crc += (uint32_t(temp) << 8);
288 _decode_state = FemtoDecodeState::crc3;
289 break;
290
291 case FemtoDecodeState::crc3:
292 _femto_msg.crc += (uint32_t(temp) << 16);
293 _decode_state = FemtoDecodeState::crc4;
294 break;
295
296 case FemtoDecodeState::crc4: {
297 _femto_msg.crc += (uint32_t(temp) << 24);
298 _decode_state = FemtoDecodeState::pream_ble1;
299
300 uint32_t crc = QGC::crc32Update({_femto_msg.header, _femto_msg.header[3]});
301 crc = QGC::crc32Update({_femto_msg.data, _femto_msg.payloadLength}, crc);
302
303 if (_femto_msg.crc == crc) {
304 iRet = _femto_msg.read;
305
306 } else {
307 }
308 } break;
309
310 default:
311 break;
312 }
313
314 } else {
316 iRet = static_cast<int>(_nmeaFramer.addByte(temp));
317 if (iRet > 0) {
318 _femto_msg.messageId = FEMTO_MSG_ID_GPGGA;
319 if (_rtcm_parsing) {
320 _rtcm_parsing->reset();
321 }
322 }
323 }
324
325 return iRet;
326}

334void GPSNativeFemto::sendSurveyInStatusUpdate(bool active, bool valid, double latitude, double longitude,
335 float altitude)
336{
337 GPSNativeSurveyReport status;
338 status.altitudeDatum = GPSNativeSurveyReport::AltitudeDatum::Ellipsoid;
339 status.latitude = latitude;
340 status.longitude = longitude;
341 status.altitude = altitude;
342 status.duration = !_baseConfig.useFixedBase ? _survey_duration : 0;
343 status.mean_accuracy = 0; // unknown
344 status.flags = (int) valid | ((int) active << 1);
345 surveyInStatus(status);
346}
