63GpuBinMode selectGpuBin(GpuBinPreference preference, const RenderTarget& target) noexcept
64{
65#if defined(QGC_HAS_GST_D3D12_GPU_PATH)
66 if (target.backend == RenderBackend::Direct3D12) {
67 return GpuBinMode::D3D12Convert;
68 }
69#endif
70 bool suited = true;
71 switch (preference.mode) {
72 case GpuBinMode::GlUpload:
73 suited = target.sharesGlContext();
74 break;
75 case GpuBinMode::DirectDmaBuf:
76 suited = target.isOpenGlOnEgl();
77 break;
78 case GpuBinMode::D3D12Convert:
79 suited = (target.backend == RenderBackend::Direct3D12);
80 break;
81 case GpuBinMode::Passthrough:
82 break;
83 }
84 return (suited || preference.explicitChoice) ? preference.mode : GpuBinMode::Passthrough;
85}
