6GPSSatellite satelliteObservation(const GPSNativeSatelliteData& source, GPSConstellation constellation)

18 }

87GPSSatelliteReport SatelliteSnapshot::update(const GPSNativeSatelliteReport& source, uint64_t nowUs)
88{
89 GPSSatelliteObservation observation;
90 observation.monotonicTimestampUs = source.timestamp;
91 observation.updateMode = source.constellation ? GPSSatelliteObservation::UpdateMode::ConstellationDelta
92 : GPSSatelliteObservation::UpdateMode::FullSnapshot;
93 const auto viewCount =
94 std::min({source.count, GPSNativeSatelliteReport::SAT_INFO_MAX_SATELLITES, GPSSatelliteReport::MAX_SATELLITES});
95 for (uint16_t i = 0; i < viewCount; ++i) {
96 const auto& satellite = source.entries[i];
97 observation.satellites.append(
98 satelliteObservation(satellite, source.constellation.value_or(satellite.constellation)));
99 }
100 if (source.constellation) {
101 GPSSatelliteProvenance provenance;
102 provenance.constellation = *source.constellation;
103 provenance.inViewTimestampUs = source.timestamp;
104 if (source.usage) {
105 provenance.inUseTimestampUs = source.usage->timestamp;
106 const auto count = std::min(source.usage->count, GPSNativeSatelliteReport::SAT_INFO_MAX_SATELLITES);
107 provenance.satellitesUsed = count;
108 provenance.usedSatelliteIds.emplace(source.usage->ids.begin(), source.usage->ids.begin() + count);
109 } else {
110 int used = 0;
111 bool known = !observation.satellites.isEmpty();
112 for (const auto& satellite : observation.satellites) {
113 known &= satellite.used.has_value();
114 used += satellite.used.value_or(false);
115 }
116 if (known) {
117 provenance.inUseTimestampUs = source.timestamp;
118 provenance.satellitesUsed = used;
119 }
120 }
121 observation.provenance.append(provenance);
122 }
123 _latestReceiptUs =
124 nowUs ? nowUs : std::max({_latestReceiptUs, source.timestamp, source.usage ? source.usage->timestamp : 0});
125 _state.updateObservation(observation, _latestReceiptUs);
126 return _snapshot(_latestReceiptUs);
127}

139GPSSatelliteReport SatelliteSnapshot::_snapshot(uint64_t nowUs)
140{
141 const auto accepted = _state.snapshot(nowUs);
142 GPSSatelliteReport snapshot;
143 for (const auto& provenance : accepted.provenance) {
144 snapshot.timestampUs = std::max<uint64_t>(snapshot.timestampUs, provenance.inViewTimestampUs);
145 }
146 for (const auto& satellite : accepted.satellites) {
147 if (snapshot.count == GPSSatelliteReport::MAX_SATELLITES) {
148 break;
149 }
150 auto& entry = snapshot.satellites[snapshot.count++];
151 entry.id = satellite.id;
152 entry.prn = satellite.prn;
153 entry.constellation = satellite.constellation;
154 entry.used = satellite.used;
155 entry.elevationDegrees = satellite.elevationDegrees;
156 entry.azimuthDegrees = satellite.normalizedAzimuthDegrees;
157 entry.signalStrength = satellite.signalStrength;
158 for (const auto& provenance : accepted.provenance) {
159 if (provenance.constellation == satellite.constellation) {
160 entry.inViewTimestampUs = provenance.inViewTimestampUs;
161 entry.inUseTimestampUs = provenance.inUseTimestampUs;
162 break;
163 }
164 }
165 }
166 return snapshot;
167}
