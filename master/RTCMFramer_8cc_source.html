48uint16_t RTCMFramer::_discardCandidate()
49{
50 uint16_t validatedFrameSize = 0;
51 uint16_t searchOffset = _frameSize;
52 if (!valid()) {
53 _recoverySize = (std::max) (_recoverySize, _frameSize);
54 searchOffset = 1;
55 }
56 uint16_t discardSize = _frameSize;
57 if (searchOffset < _recoverySize) {
58 discardSize = _recoverySize;
59 // Fresh bytes may belong to an incomplete outer frame.
60 const std::span<const uint8_t> bytes{_bytes.data(), _recoverySize};
61 for (uint16_t offset = searchOffset; offset < _recoverySize; ++offset) {
62 const auto suffix = bytes.subspan(offset);
63 if (suffix[0] != PREAMBLE || (suffix.size() >= 2 && (suffix[1] & 0xfc))) {
64 continue;
65 }
66 if (suffix.size() < HEADER_SIZE) {
67 discardSize = (std::min) (discardSize, offset);
68 continue;
69 }
70 const size_t payloadSize = ((suffix[1] & 3) << 8) | suffix[2];
71 if (payloadSize < 2) {
72 continue;
73 }
74 const size_t length = HEADER_SIZE + payloadSize + CRC_SIZE;
75 if (length > suffix.size()) {
76 discardSize = (std::min) (discardSize, offset);
77 continue;
78 }
79 if (isValidFrame(suffix.first(length))) {
80 discardSize = offset;
81 validatedFrameSize = static_cast<uint16_t>(length);
82 break;
83 }
84 }
85 }
86 _discardPrefix(discardSize);
87 _frameSize = 0;
88 _valid = false;
89 return validatedFrameSize;
90}

92bool RTCMFramer::_inspect(uint16_t validatedFrameSize)
93{
94 _valid = false;
95 _frameSize = 0;
96 _payloadSize = 0;
97 const auto preamble = std::find(_bytes.begin(), _bytes.begin() + _size, PREAMBLE);
98 _discardPrefix(static_cast<uint16_t>(preamble - _bytes.begin()));
99 if (_size >= 2 && (_bytes[1] & 0xfc)) {
