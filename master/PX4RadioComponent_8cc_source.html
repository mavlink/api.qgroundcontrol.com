33bool PX4RadioComponent::setupComplete(void) const
34{
35 if (_vehicle->parameterManager()->getParameter(-1, "COM_RC_IN_MODE")->rawValue().toInt() != 1) {
36 // The best we can do to detect the need for a radio calibration is look for attitude
37 // controls to be mapped.
38 for(const char* const&mapParam : {"RC_MAP_ROLL", "RC_MAP_PITCH", "RC_MAP_YAW", "RC_MAP_THROTTLE"}) {
39 if (_vehicle->parameterManager()->getParameter(ParameterManager::defaultComponentId, mapParam)->rawValue().toInt() == 0) {
40 return false;
41 }
42 }
43 }
44
45 return true;
46}
