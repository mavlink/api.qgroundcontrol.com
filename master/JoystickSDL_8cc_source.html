30static bool sdlEventWatcher(void *userdata, SDL_Event *event)
31{
32 Q_UNUSED(userdata);
33
34 JoystickManager *manager = JoystickManager::instance();
35 if (!manager) {
36 return true;
37 }
38
39 switch (event->type) {
40 case SDL_EVENT_JOYSTICK_ADDED:
41 qCInfo(JoystickSDLLog) << "SDL event: Joystick added, instance ID:" << event->jdevice.which;
42 QMetaObject::invokeMethod(manager, "_checkForAddedOrRemovedJoysticks", Qt::QueuedConnection);
43 break;
44 case SDL_EVENT_JOYSTICK_REMOVED:
45 qCInfo(JoystickSDLLog) << "SDL event: Joystick removed, instance ID:" << event->jdevice.which;
46 QMetaObject::invokeMethod(manager, "_checkForAddedOrRemovedJoysticks", Qt::QueuedConnection);
47 break;
48 // Gamepad events ignored - SDL fires both joystick and gamepad events for gamepads
49 case SDL_EVENT_GAMEPAD_ADDED:
50 qCDebug(JoystickSDLLog) << "SDL event: Gamepad added (ignored, handled via joystick event), instance ID:" << event->gdevice.which;
51 break;
52 case SDL_EVENT_GAMEPAD_REMOVED:
53 qCDebug(JoystickSDLLog) << "SDL event: Gamepad removed (ignored, handled via joystick event), instance ID:" << event->gdevice.which;
54 break;
55
56 case SDL_EVENT_JOYSTICK_BATTERY_UPDATED:
57 qCDebug(JoystickSDLLog) << "Battery updated for joystick" << event->jbattery.which
58 << "state:" << event->jbattery.state
59 << "percent:" << event->jbattery.percent;
60 QMetaObject::invokeMethod(manager, "_handleBatteryUpdated",
61 Qt::QueuedConnection,
62 Q_ARG(int, static_cast<int>(event->jbattery.which)));
63 break;
64
65 case SDL_EVENT_GAMEPAD_REMAPPED:
66 qCDebug(JoystickSDLLog) << "Gamepad remapped:" << event->gdevice.which;
67 QMetaObject::invokeMethod(manager, "_handleGamepadRemapped",
68 Qt::QueuedConnection,
69 Q_ARG(int, static_cast<int>(event->gdevice.which)));
70 break;
71
72 case SDL_EVENT_GAMEPAD_TOUCHPAD_DOWN:
73 case SDL_EVENT_GAMEPAD_TOUCHPAD_MOTION:
74 case SDL_EVENT_GAMEPAD_TOUCHPAD_UP:
75 QMetaObject::invokeMethod(manager, "_handleTouchpadEvent",
76 Qt::QueuedConnection,
77 Q_ARG(int, static_cast<int>(event->gtouchpad.which)),
78 Q_ARG(int, event->gtouchpad.touchpad),
79 Q_ARG(int, event->gtouchpad.finger),
80 Q_ARG(bool, event->type != SDL_EVENT_GAMEPAD_TOUCHPAD_UP),
81 Q_ARG(float, event->gtouchpad.x),
82 Q_ARG(float, event->gtouchpad.y),
83 Q_ARG(float, event->gtouchpad.pressure));
84 break;
85
86 case SDL_EVENT_GAMEPAD_SENSOR_UPDATE:
87 QMetaObject::invokeMethod(manager, "_handleSensorUpdate",
88 Qt::QueuedConnection,
89 Q_ARG(int, static_cast<int>(event->gsensor.which)),
90 Q_ARG(int, event->gsensor.sensor),
91 Q_ARG(float, event->gsensor.data[0]),
92 Q_ARG(float, event->gsensor.data[1]),
93 Q_ARG(float, event->gsensor.data[2]));
94 break;
95
96 case SDL_EVENT_JOYSTICK_UPDATE_COMPLETE:
97 case SDL_EVENT_GAMEPAD_UPDATE_COMPLETE:
98 QMetaObject::invokeMethod(manager, "_handleUpdateComplete",
99 Qt::QueuedConnection,
100 Q_ARG(int, static_cast<int>(event->common.type == SDL_EVENT_GAMEPAD_UPDATE_COMPLETE
101 ? event->gdevice.which : event->jdevice.which)));
102 break;
103
104 default:
105 break;
106 }
107
108 return true;
109}

187QMap<QString, Joystick*> JoystickSDL::discover()
188{
189 if (!QThread::isMainThread()) {
190 // An empty map would make the caller delete joysticks that the cache still points to
191 qCCritical(JoystickSDLLog) << "Internal Error: called off the main thread, returning the previous discovery";
192 const QMutexLocker cacheLocker(&s_discoveryCacheMutex);
193 return s_discoveryCache;
194 }
195
196 QMap<QString, Joystick*> current;
197 // Working copy of the cache: joysticks found again move to current, the rest are deleted at the end
198 QMap<QString, Joystick*> previous;
199 {
200 const QMutexLocker cacheLocker(&s_discoveryCacheMutex);
201 previous = s_discoveryCache;
202 }
203
204 qCDebug(JoystickSDLLog) << "Discovering joysticks";
205
206 // Required on Android before SDL_GetJoysticks() returns devices
207 SDLJoystick::pumpEvents();
208
209 SDLJoystick::JoystickLock lock;
210
211 int count = 0;
212 SDL_JoystickID *ids = SDL_GetJoysticks(&count);
213 if (!ids) {
214 qCWarning(JoystickSDLLog) << "SDL_GetJoysticks failed:" << SDL_GetError();
215 return current;
216 }
217
218 qCDebug(JoystickSDLLog) << "SDL_GetJoysticks returned" << count << "joysticks";
219 for (int n = 0; n < count; ++n) {
220 const QString joystickName = SDLJoystick::getNameForInstanceId(ids[n]);
221 qCDebug(JoystickSDLLog) << " [" << n << "] ID:" << ids[n]
222 << "Name:" << joystickName
223 << "IsGamepad:" << SDL_IsGamepad(ids[n]);
224 }
225
226 for (int n = 0; n < count; ++n) {
227 const SDL_JoystickID jid = ids[n];
228 QString baseName = SDLJoystick::getNameForInstanceId(jid);
229 if (baseName.isEmpty()) {
230 baseName = QStringLiteral("Joystick %1").arg(jid);
231 }
232 QString name = baseName;
233
234 // Check cache by instance ID (reconnection of same device)
235 bool foundInCache = false;
236 for (auto it = previous.begin(); it != previous.end(); ++it) {
237 auto *cachedJs = static_cast<JoystickSDL*>(it.value());
238 if (static_cast<SDL_JoystickID>(cachedJs->instanceId()) == jid) {
239 if (cachedJs->name() == baseName) {
240 name = it.key();
241 current[name] = cachedJs;
242 previous.erase(it);
243 foundInCache = true;
244 } else {
245 // SDL instance ids can be recycled. If the id matches but the reported
246 // name changed, treat it as a new device and drop the stale cached object.
247 cachedJs->deleteLater();
248 previous.erase(it);
249 }
250 break;
251 }
252 }
253 if (foundInCache) {
254 continue;
255 }
256
257 // Check cache by name (for joysticks that were disconnected and reconnected)
258 if (previous.contains(name) && !current.contains(name)) {
259 current[name] = previous[name];
260 auto *js = static_cast<JoystickSDL*>(current[name]);
261 js->setInstanceId(jid);
262 (void) previous.remove(name);
263 continue;
264 }
265
266 // Handle duplicate names by appending a number
267 int duplicateIndex = 2;
268 while (current.contains(name) || previous.contains(name)) {
269 name = QStringLiteral("%1 #%2").arg(baseName).arg(duplicateIndex++);
270 }
271
272 QList<int> gamepadAxes;
273 QSet<int> joyAxesMappedToGamepad;
274
275 if (SDL_IsGamepad(jid)) {
276 auto tmpGamepad = SDL_OpenGamepad(jid);
277 if (!tmpGamepad) {
278 qCWarning(JoystickSDLLog) << "Failed to open gamepad" << jid << SDL_GetError();
279 continue;
280 }
281
282 // Determine if this gamepad axis is one we should show to the user
283 for (int i = 0; i < SDL_GAMEPAD_AXIS_COUNT; i++) {
284 if (SDL_GamepadHasAxis(tmpGamepad, static_cast<SDL_GamepadAxis>(i))) {
285 gamepadAxes.append(i);
286 }
287 }
288
289 // If a sdlJoystick axis is mapped to a sdlGamepad axis, then the axis is represented
290 // by both the sdlJoystick interface and the sdlGamepad interface. If this is the case,
291 // We'll only show the sdlGamepad interface version of the axis to the user.
292 int bindingCount = 0;
293 SDL_GamepadBinding **bindings = SDL_GetGamepadBindings(tmpGamepad, &bindingCount);
294 if (bindings) {
295 for (int i = 0; i < bindingCount; ++i) {
296 SDL_GamepadBinding *binding = bindings[i];
297 if (binding && binding->input_type == SDL_GAMEPAD_BINDTYPE_AXIS && binding->output_type == SDL_GAMEPAD_BINDTYPE_AXIS) {
298 joyAxesMappedToGamepad.insert(binding->input.axis.axis);
299 }
300 }
301 SDL_free(bindings);
302 } else {
303 qCWarning(JoystickSDLLog) << "Failed to get bindings for" << name << "error:" << SDL_GetError();
304 }
305
306 SDL_CloseGamepad(tmpGamepad);
307 }
308
309 SDL_Joystick *tmpJoy = SDL_OpenJoystick(jid);
310 if (!tmpJoy) {
311 qCWarning(JoystickSDLLog) << "Failed to open joystick" << jid << SDL_GetError();
312 continue;
313 }
314
315 QList<int> nonGamepadAxes;
316 const int axisCount = SDL_GetNumJoystickAxes(tmpJoy);
317 for (int i = 0; i < axisCount; i++) {
318 if (!joyAxesMappedToGamepad.contains(i)) {
319 nonGamepadAxes.append(i);
320 }
321 }
322
323 const int buttonCount = SDL_GetNumJoystickButtons(tmpJoy);
324 const int hatCount = SDL_GetNumJoystickHats(tmpJoy);
325 SDL_CloseJoystick(tmpJoy);
326
327 qCDebug(JoystickSDLLog) << "Creating JoystickSDL for" << name << "jid:" << jid;
328
329 current[name] = new JoystickSDL(name,
330 gamepadAxes,
331 nonGamepadAxes,
332 qMax(0, buttonCount),
333 qMax(0, hatCount),
334 jid);
335 }
336
337 SDL_free(ids);
338
339 qCDebug(JoystickSDLLog) << "Discovered" << current.size() << "joysticks:";
340 for (auto it = current.begin(); it != current.end(); ++it) {
341 auto *js = static_cast<JoystickSDL*>(it.value());
342 qCDebug(JoystickSDLLog) << " " << it.key() << "instanceId:" << js->instanceId();
343 }
344
345 for (auto* joystick : std::as_const(previous)) {
346 joystick->deleteLater();
347 }
348
349 const QMutexLocker cacheLocker(&s_discoveryCacheMutex);
350 s_discoveryCache = current;
351 return current;
352}

440int JoystickSDL::_getAxisValue(int idx) const
441{
442 if (idx < 0) {
443 return 0;
444 }
445
446#ifdef TEST_WITH_VIRTUAL_AXES
447 // Handle virtual axes (last 2 axes)
448 const int totalPhysicalAxes = _gamepadAxes.length() + _nonGamepadAxes.length();
449 if (idx >= totalPhysicalAxes && idx < totalPhysicalAxes + 2) {
450 // Check if button 0 is pressed
451 bool button0Down = false;
452 if (_sdlGamepad) {
453 button0Down = SDL_GetGamepadButton(_sdlGamepad, static_cast<SDL_GamepadButton>(0));
454 } else if (_sdlJoystick) {
455 button0Down = SDL_GetJoystickButton(_sdlJoystick, 0);
456 }
457
458 if (button0Down) {
459 // When button 0 is down, return values from axis 0 and 1
460 const int virtualAxisIdx = idx - totalPhysicalAxes;
461 if (virtualAxisIdx == 0) {
462 // First virtual axis maps to axis 0
463 if (_sdlGamepad && _gamepadAxes.length() > 0) {
464 return SDL_GetGamepadAxis(_sdlGamepad, static_cast<SDL_GamepadAxis>(_gamepadAxes[0]));
465 } else if (_sdlJoystick) {
466 return SDL_GetJoystickAxis(_sdlJoystick, 0);
467 }
468 } else {
469 // Second virtual axis maps to axis 1
470 if (_sdlGamepad && _gamepadAxes.length() > 1) {
471 return SDL_GetGamepadAxis(_sdlGamepad, static_cast<SDL_GamepadAxis>(_gamepadAxes[1]));
472 } else if (_sdlJoystick) {
473 return SDL_GetJoystickAxis(_sdlJoystick, 1);
474 }
475 }
476 } else {
477 // When button 0 is up, return centered values
478 return 0;
479 }
480 }
481#endif
482
483 if (_sdlGamepad) {
484 if (idx < _gamepadAxes.length()) {
485 return SDL_GetGamepadAxis(_sdlGamepad, static_cast<SDL_GamepadAxis>(_gamepadAxes[idx]));
486 }
487 const int nonGamepadIdx = idx - static_cast<int>(_gamepadAxes.length());
488 if (nonGamepadIdx < _nonGamepadAxes.length()) {
489 return SDL_GetJoystickAxis(_sdlJoystick, _nonGamepadAxes[nonGamepadIdx]);
490 }
491 return 0;
492 }
493
494 return SDL_GetJoystickAxis(_sdlJoystick, idx);
495}
