24void GPSReceiverConnectionPolicy::configurationChanged(const GPSReceiverConfiguration& previous)
25{
26 const bool connectionChanged = !previous.sameConnection(_configuration);
27 const bool turnedOn = !previous.autoConnect && _configuration.autoConnect;
28 const bool turnedOff = previous.autoConnect && !_configuration.autoConnect;
29 if (connectionChanged || turnedOff) {
30 _cancelRetries();
31 }
32 if (turnedOn) {
33 _paused = false;
34 _startPending = true;
35 // A saved receiver the user already connected is kept connected from now on.
36 _keepConnected = _owner == Owner::Saved && !connectionChanged;
37 }

40void GPSReceiverConnectionPolicy::_cancelRetries()
41{
42 if (_keepConnected) {

99GPSReceiverSessionOutcome GPSReceiverConnectionPolicy::sessionEnded(bool portRemoved)
100{
101 if (_owner == Owner::Discovered && _discoveryEnabled()) {
102 if (portRemoved) {
103 // Discovery connects the receiver again when it returns.
104 _owner = Owner::None;
105 _cancelRetries();
106 return GPSReceiverSessionOutcome::Unplugged;
107 }
108 _scheduleRetry();
109 return GPSReceiverSessionOutcome::AutoRetrying;
110 }
111 if (_owner == Owner::Saved && _keepConnected) {
112 if (portRemoved) {
113 _waitingForPort = true;
114 return GPSReceiverSessionOutcome::WaitingForPort;
115 }
116 _scheduleRetry();
117 return GPSReceiverSessionOutcome::Retrying;
118 }
119 _owner = Owner::None;
120 _cancelRetries();
121 return GPSReceiverSessionOutcome::None;
122}

147bool GPSReceiverConnectionPolicy::_hasSavedReceiver() const
148{
149 switch (_configuration.effectiveConnectionType()) {
150 case RTKSettings::Serial:
151 return !_configuration.serialDevice.trimmed().isEmpty();
152 case RTKSettings::Tcp:
153 return !_configuration.tcpHost.trimmed().isEmpty() && _configuration.tcpPort != 0;
154 case RTKSettings::Udp:
155 // UDP only receives, so it cannot carry a configured base.
156 return _configuration.receiverRole != RTKSettings::ConfiguredBase;
157 }
158 return false;

173void GPSReceiverConnectionPolicy::_updateSaved()
174{
175 if (_configuration.effectiveConnectionType() == RTKSettings::Serial) {
176 if (!_serialDeviceAvailable(_configuration.serialDevice.trimmed())) {
177 if (!std::exchange(_waitingForPort, true)) {
178 //: %1 is a serial device, such as /dev/ttyACM0 or COM3
179 _receiver.setConnectionError(
180 tr("Waiting for the receiver on %1.").arg(_configuration.serialDevice.trimmed()));
181 }
182 return;
183 }
184 if (std::exchange(_waitingForPort, false)) {
185 // Retry as soon as the receiver returns instead of waiting out the backoff.
186 _attemptSaved();
187 return;
188 }
189 }
190 if (!_retryDeadlineUs || _scheduler->nowUs() >= *_retryDeadlineUs) {
191 _attemptSaved();
192 }
193}

213bool GPSReceiverConnectionPolicy::_connectConfigured(bool allowPersistentChanges)
214{
215 const bool configuredBase = _configuration.receiverRole == RTKSettings::ConfiguredBase;
216 const auto type = configuredBase ? gpsReceiverTypeForManufacturer(_configuration.baseReceiverManufacturer)
217 : std::optional(GPSType::passive);
218 if (!type) {
219 _receiver.setConnectionError(tr("Select a receiver type before connecting."));
220 return false;
221 }
222 const auto connection = _configuration.effectiveConnectionType();
223 if (connection == RTKSettings::Tcp) {
224 const QString host = _configuration.tcpHost.trimmed();
225 const uint tcpPort = _configuration.tcpPort;
226 if (host.isEmpty() || tcpPort == 0 || tcpPort > 65535) {
227 _receiver.setConnectionError(tr("Enter the receiver's TCP host and port."));
228 return false;
229 }
230 return _receiver.connectTcp(host, static_cast<quint16>(tcpPort), *type, allowPersistentChanges);
231 }
232 if (connection == RTKSettings::Udp) {
233 const uint udpPort = _configuration.udpPort;
234 if (configuredBase) {
235 _receiver.setConnectionError(
236 tr("A configured base needs a serial or TCP connection. UDP only receives data."));
237 return false;
238 }
239 if (udpPort == 0 || udpPort > 65535) {
240 _receiver.setConnectionError(tr("Enter the UDP port that receives the data."));
241 return false;
242 }
243 return _receiver.connectUdp(static_cast<quint16>(udpPort), *type);
244 }
245#ifndef QGC_NO_SERIAL_LINK
246 const QString device = _configuration.serialDevice.trimmed();
247 if (!_serialDeviceAvailable(device)) {
248 _receiver.setConnectionError(tr("Select an available serial device that is not a bootloader."));
249 return false;
250 }
251 // Zero asks configurable receivers to detect the rate.
252 const auto baud = _configuration.serialBaudRate;
253 if (!gpsValidBaudRate(baud, *type == GPSType::passive)) {
254 _receiver.setConnectionError(gpsReceiverConfigErrorText(GPSReceiverConfigError::InvalidBaudRate));
255 return false;
256 }
257 auto claim = _receiver.claimSerialPort(device);
258 if (!claim) {
259 _receiver.setConnectionError(tr("The selected serial device is already in use."));
260 return false;
261 }
262 return _receiver.connectSerial(device, *type, baud, allowPersistentChanges, std::move(claim));
263#else
264 return false;
265#endif
266}

268void GPSReceiverConnectionPolicy::_updateDiscovery()
269{
270#ifndef QGC_NO_SERIAL_LINK
271 if (_owner == Owner::Discovered) {
272 if (_serialDeviceAvailable(_autoPort)) {
273 if (!_retryDeadlineUs || _scheduler->nowUs() >= *_retryDeadlineUs) {
274 _connectDiscovered(_autoPort);
275 }
276 return;
277 }
278 // The receiver being retried was unplugged; discovery finds it again when it returns.
279 _owner = Owner::None;
280 _cancelRetries();
281 }
282 const QStringList receivers = _receiver.rtkReceiverPorts();
283 _waitingPorts.removeIf([&receivers](const QString& device) { return !receivers.contains(device); });
284 const quint64 nowUs = _scheduler->nowUs();
285 for (const QString& device : receivers) {
286 if (_waitingPorts.settled(device, nowUs)) {
287 _connectDiscovered(device);
288 return;
289 }
290 }
291#endif
292}

296{
297 auto claim = _receiver.claimSerialPort(device);
298 if (!claim) {
299 return;
300 }
301 _owner = Owner::Discovered;
302 _autoPort = device;
303 _waitingPorts.clear();
304 _retryDeadlineUs.reset();
305 // The USB identity only chose the port; the saved manufacturer chooses the family, detecting it when Automatic.

307 if (!type) {
308 _receiver.setConnectionError(tr("Select a receiver type before connecting."));
309 }
310 if (!type || !_receiver.connectSerial(device, *type, 0, false, std::move(claim))) {

312 }
