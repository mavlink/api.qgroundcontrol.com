42{

62 if (events::externalLogLevel(check.log_levels) <= events::Log::Error) {
63 severity = "error";
64 } else if (events::externalLogLevel(check.log_levels) <= events::Log::Warning) {
65 severity = "warning";
66 }
67

69 checks.append(Check{
70 .ackKey = (static_cast<quint64>(check.id) << 32) | check.affected_health_component_index,
71 .message = QString::fromStdString(check.message),
72 .description = description.replace("\n", "<br/>"),
73 .severity = severity,
74 // Without a current mode every check is listed, since any of them might apply
75 .inCurrentMode = !currentModeKnown || ((check.affected_mode_groups & (1ull << flightModeGroup)) != 0),
76 });
77 }

79

81 _modeGroupResults.canArm = 0;
82 _modeGroupResults.canRun = 0;

85 if (results.canArm(modeGroup)) {

132void HealthAndArmingCheckReport::_rebuild(const QList<Check>& checks)
133{
134 // Checks are matched by key rather than text, so a warning whose values change stays acknowledged.
135 // Mode-independent so acknowledgements survive a mode switch.
136 QSet<quint64> reportedWarningKeys;
137 for (const Check& check : checks) {
138 if (check.severity == QStringLiteral("warning")) {
139 reportedWarningKeys.insert(check.ackKey);
140 }
141 }
142 _acknowledgedWarningKeys.intersect(reportedWarningKeys);
143
144 _problemsForCurrentMode->clearAndDeleteContents();
145 _currentModeWarningKeys.clear();
146 for (const Check& check : checks) {
147 if (!check.inCurrentMode) {
148 continue;
149 }
150
151 auto* const problem = new HealthAndArmingCheckProblem(check.message, check.description, check.severity, this);
152 if (check.severity == QStringLiteral("warning")) {
153 _currentModeWarningKeys.insert(check.ackKey);
154 problem->setAcknowledged(_acknowledgedWarningKeys.contains(check.ackKey));
155 }
156 _problemsForCurrentMode->append(problem);
157 }
158}
