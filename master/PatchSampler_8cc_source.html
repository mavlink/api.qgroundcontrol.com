30QList<float> PatchSampler::sample()
31{
32 QList<float> heights;
33 heights.reserve(qsizetype(_gridSize + 1) * (_gridSize + 1));
34 for (int row = 0; row <= _gridSize; row++) {
35 const qint64 m = (qint64(_key.y) * _gridSize) + row;
36 for (int col = 0; col <= _gridSize; col++) {
37 const qint64 n = (qint64(_key.x) * _gridSize) + col;
38 if ((row == 0) || (row == _gridSize) || (col == 0) || (col == _gridSize)) {
39 heights.append(_boundaryHeight(n, m));
40 } else {
41 heights.append(_patchView.isValid() ? _viewHeight(_patchView, n, m) : 0.0f);
42 }
43 }
44 }
45 return heights;
46}

51float PatchSampler::_viewHeight(const ElevationTilePyramid::View& view, qint64 n, qint64 m) const
52{
53 const double u = (std::ldexp(double(n), view.key.zoom - _key.zoom) / _gridSize) - view.key.x;
54 const double v = (std::ldexp(double(m), view.key.zoom - _key.zoom) / _gridSize) - view.key.y;
55 return static_cast<float>(heightAtUV(*view.grid, u, v));
56}

60int PatchSampler::_touchingCells(qint64 s, qint64 (&cells)[2]) const
61{
62 const qint64 cellCount = qint64(1) << TileMath::kMaxZoom;
63 int count = 0;
64 const qint64 cell = s / _gridSize;
65 if (cell < cellCount) {
66 cells[count++] = cell;
67 }
68 if (((s % _gridSize) == 0) && (cell > 0)) {
69 cells[count++] = cell - 1;
70 }
71 return count;
72}

74ElevationTilePyramid::View PatchSampler::_resolveCell(qint64 cx, qint64 cy)
75{
76 const TileMath::TileKey tile{int(cx >> _shiftToMax), int(cy >> _shiftToMax), _key.zoom};
77 for (const TileMemo& memo : _memos) {
78 if (memo.tile == tile) {
79 return memo.view;
80 }
81 }
82 const ElevationTilePyramid::View view =
83 _pyramid.bestTileFor(TileMath::TileKey{int(cx), int(cy), TileMath::kMaxZoom});
84 if (!_pyramid.hasDescendant(tile) && (_memos.size() < _memos.capacity())) {
85 _memos.append({tile, view});
86 }
87 return view;
88}

90float PatchSampler::_boundaryHeight(qint64 n, qint64 m)
91{
92 qint64 xCells[2];
93 qint64 yCells[2];
94 const int xCount = _touchingCells(n << _shiftToMax, xCells);
95 const int yCount = _touchingCells(m << _shiftToMax, yCells);
96
97 // Fixed candidate order derived purely from the position: every patch
98 // sharing this vertex walks the same cells and returns the first
99 // resolvable view, so the height is canonical
100 for (int yi = 0; yi < yCount; yi++) {
101 for (int xi = 0; xi < xCount; xi++) {
102 const ElevationTilePyramid::View view = _resolveCell(xCells[xi], yCells[yi]);
103 if (view.isValid()) {
104 return _viewHeight(view, n, m);
105 }
106 }
107 }
108 return 0.0f; // no stored data touches the vertex: every sharer agrees on zero
109}
