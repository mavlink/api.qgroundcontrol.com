55bool TeeBranchHead::build(GstElement* pipeline, GstElement* tee, GStreamer::PipelineBranch branch, const char* what)
56{
57 reset();
58
59 GStreamer::GstElementPtr queue = GStreamer::makeElement("queue");
60 GStreamer::GstElementPtr valve = GStreamer::makeElement("valve");
61 if (!queue || !valve) {
62 qCWarning(GstVideoReceiverLog) << "Failed to create the queue and valve of the" << what << "branch";
63 return false;
64 }
65 g_object_set(valve.get(), "drop", TRUE, nullptr);
66 GStreamer::markBranch(queue.get(), branch);
67 GStreamer::markBranch(valve.get(), branch);
68
69 GstElement* const binQueue = GStreamer::addToBin(pipeline, std::move(queue), "queue");
70 GstElement* const binValve = GStreamer::addToBin(pipeline, std::move(valve), "valve");
71 if (!binQueue || !binValve || !gst_element_link_many(tee, binQueue, binValve, nullptr)) {
72 qCWarning(GstVideoReceiverLog) << "Unable to link the" << what << "branch";
73 return false;
74 }
75 _queue = binQueue;
76 _valve = binValve;
77 return true;
78}
