56{
57 // Try to show a GUI dialog — important for AppImage users where stderr is invisible.
58 // Fork a child and attempt dialog tools in order of preference; no shell is invoked.
59 const pid_t pid = fork();
60 if (pid == 0) {
61 const QByteArray zenityText = QByteArrayLiteral("--text=") + msg;
62 execlp("zenity", "zenity", "--error", "--title=Error", zenityText.constData(), nullptr);
63 execlp("kdialog", "kdialog", "--error", msg.constData(), nullptr);
64 execlp("xmessage", "xmessage", "-center", msg.constData(), nullptr);
65 _exit(1);
66 } else if (pid > 0) {
67 int status = 0;
68 (void) waitpid(pid, &status, 0);
69 }
70 // Always write to stderr as well
71 fprintf(stderr, "Error: %s\n", msg.constData());

178// Runs on the dump thread, whose own buffer keeps it apart from the filter's messages.
179void writeCrashDump()
180{
181 char narrow[6 * MAX_PATH] = {};
182 const HANDLE file =
183 CreateFileW(g_crashDumpPartPath, GENERIC_WRITE, 0, nullptr, CREATE_ALWAYS, FILE_ATTRIBUTE_NORMAL, nullptr);
184 if (file == INVALID_HANDLE_VALUE) {
185 writeCrashMessage(L"QGC: no crash dump, cannot create a file in QGC_CRASH_DUMP_DIR\n", narrow);
186 return;
187 }
188
189 MINIDUMP_EXCEPTION_INFORMATION exceptionInfo = {};
190 exceptionInfo.ThreadId = g_crashThreadId;
191 exceptionInfo.ExceptionPointers = g_crashException;
192 exceptionInfo.ClientPointers = FALSE;
193 const BOOL written =
194 g_miniDumpWriteDump(GetCurrentProcess(), GetCurrentProcessId(), file,
195 static_cast<MINIDUMP_TYPE>(MiniDumpWithUnloadedModules | MiniDumpWithThreadInfo),
196 g_crashException ? &exceptionInfo : nullptr, nullptr, nullptr);
197 DWORD error = written ? ERROR_SUCCESS : GetLastError();
198 (void) CloseHandle(file);
199
200 // Only a complete dump gets the .dmp name that CI collects.
201 bool kept = false;
202 if (written) {
203 kept = MoveFileExW(g_crashDumpPartPath, g_crashDumpPath, MOVEFILE_REPLACE_EXISTING) != FALSE;
204 if (!kept) {
205 error = GetLastError();
206 }
207 }
208
209 wchar_t message[2 * MAX_PATH] = {};
210 if (kept) {
211 (void) formatWide(message, L"QGC: wrote crash dump %ls\n", g_crashDumpPath);
212 } else {
213 (void) DeleteFileW(g_crashDumpPartPath);
214 (void) formatWide(message, L"QGC: crash dump failed with error 0x%08lX\n", static_cast<unsigned long>(error));
215 }
216 writeCrashMessage(message, narrow);
217}

228// Resolved before any crash: the exception filter must not load libraries or start threads.
229void setupCrashDumps()
230{
231 wchar_t directory[MAX_PATH] = {};
232 const DWORD length = GetEnvironmentVariableW(L"QGC_CRASH_DUMP_DIR", directory, MAX_PATH);
233 if ((length == 0) || (length >= MAX_PATH)) {
234 return;
235 }
236 if (!formatWide(g_crashDumpPath, L"%ls\\QGC-crash-%lu.dmp", directory,
237 static_cast<unsigned long>(GetCurrentProcessId())) ||
238 !formatWide(g_crashDumpPartPath, L"%ls.part", g_crashDumpPath)) {
239 g_crashDumpPath[0] = L'\0';
240 return;
241 }
242
243 const HMODULE dbghelp = LoadLibraryExW(L"dbghelp.dll", nullptr, LOAD_LIBRARY_SEARCH_SYSTEM32);
244 if (dbghelp) {
245 g_miniDumpWriteDump = reinterpret_cast<MiniDumpWriteDumpFn>(
246 reinterpret_cast<void (*)()>(GetProcAddress(dbghelp, "MiniDumpWriteDump")));
247 }
248 if (!g_miniDumpWriteDump) {
249 return;
250 }
251
252 g_crashDumpRequested = CreateEventW(nullptr, FALSE, FALSE, nullptr);
253 g_crashDumpWritten = CreateEventW(nullptr, TRUE, FALSE, nullptr);
254 if (!g_crashDumpRequested || !g_crashDumpWritten) {
255 return;
256 }
257 const HANDLE thread = CreateThread(nullptr, 256 * 1024, crashDumpThread, nullptr, 0, nullptr);
258 if (!thread) {
259 return;
260 }
261 (void) CloseHandle(thread);
262 g_crashDumpReady = true;
263}

265LONG WINAPI WindowsUnhandledExceptionFilter(EXCEPTION_POINTERS* ep)
266{
267 // A second crashing thread lets the first finish its report, then ends the process.
268 if (InterlockedExchange(&g_crashHandled, 1) != 0) {
269 if (g_crashDumpReady) {
270 (void) WaitForSingleObject(g_crashDumpWritten, CRASH_DUMP_WAIT_MS);
271 }
272 return EXCEPTION_EXECUTE_HANDLER;
273 }
274
275 const EXCEPTION_RECORD* const record = ep ? ep->ExceptionRecord : nullptr;
276 const DWORD code = record ? record->ExceptionCode : 0;
277 const ULONG_PTR address = record ? reinterpret_cast<ULONG_PTR>(record->ExceptionAddress) : 0;
278
279 const wchar_t* moduleName = L"<unknown module>";
280 ULONG_PTR offset = address;
281 HMODULE module = nullptr;
282 if ((address != 0) &&
283 GetModuleHandleExW(GET_MODULE_HANDLE_EX_FLAG_FROM_ADDRESS | GET_MODULE_HANDLE_EX_FLAG_UNCHANGED_REFCOUNT,
284 reinterpret_cast<LPCWSTR>(address), &module) &&
285 (GetModuleFileNameW(module, g_crashModulePath, MAX_PATH) > 0)) {
286 moduleName = g_crashModulePath;
287 offset = address - reinterpret_cast<ULONG_PTR>(module);
288 }
289
290 (void) formatWide(g_crashMessage, L"QGC: unhandled SEH 0x%08lX at 0x%llX in %ls+0x%llX\n",
291 static_cast<unsigned long>(code), static_cast<unsigned long long>(address), moduleName,
292 static_cast<unsigned long long>(offset));
293 writeCrashMessage(g_crashMessage, g_crashNarrow);
294
295 if (g_crashDumpReady) {
296 g_crashException = ep;
297 g_crashThreadId = GetCurrentThreadId();
298 if (SetEvent(g_crashDumpRequested) &&
299 (WaitForSingleObject(g_crashDumpWritten, CRASH_DUMP_WAIT_MS) == WAIT_TIMEOUT)) {
300 writeCrashMessage(L"QGC: crash dump unfinished after 20 s; exiting without it\n", g_crashNarrow);
301 }
302 } else if (g_crashDumpPath[0] != L'\0') {
303 writeCrashMessage(L"QGC: no crash dump, the dump writer could not start\n", g_crashNarrow);
304 }
305
306 return EXCEPTION_EXECUTE_HANDLER;
307}

416#ifdef Q_OS_ANDROID
417void Platform::skipStaticDestructorsAtExit(int exitCode)
418{
419 // Once the activity is destroyed, Qt calls exit(), whose static destructors include libhwui's CommonPool: it
420 // destroys its mutex while the pool's detached hwuiTask threads still lock it, which aborts the process ("FORTIFY:
421 // pthread_mutex_lock called on a destroyed mutex"). Android apps normally end without running static destructors.
422 // exit() runs the handler registered last first, so this one ends the process before the others run.
423 g_exitCode = exitCode;
424 (void) std::atexit([] { std::_Exit(g_exitCode); });
425}
