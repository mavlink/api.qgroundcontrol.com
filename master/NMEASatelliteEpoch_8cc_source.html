29bool validNavigation(const NMEA::Sentence& input)
30{
31 if (input.type() == "GGA") {
32 return NMEA::gga(input).has_value() ||
33 (input.count >= NMEA::Field::GGA_MIN_FIELDS &&
34 NMEA::number<unsigned>(input.fields[NMEA::Field::GGA_QUALITY]) == NMEA::GgaQuality::INVALID);
35 }
36 if (input.type() != "RMC" || input.count < 10) {
37 return false;
38 }
39 const auto date = input.fields[9];
40 if (date.size() != 6 ||
41 !std::all_of(date.begin(), date.end(), [](char digit) { return digit >= '0' && digit <= '9'; })) {
42 return false;
43 }
44 const auto day = NMEA::number<unsigned>(date.substr(0, 2));
45 const auto month = NMEA::number<unsigned>(date.substr(2, 2));
46 const auto year = NMEA::number<unsigned>(date.substr(4, 2));
47 if (!day || !month || !year ||
48 !std::chrono::year_month_day{std::chrono::year{2000 + static_cast<int>(*year)}, std::chrono::month{*month},
49 std::chrono::day{*day}}
50 .ok()) {
51 return false;
52 }
53 return input.fields[NMEA::Field::RMC_STATUS] == "V" ||
54 (input.fields[NMEA::Field::RMC_STATUS] == "A" && NMEA::coordinate(input.fields[3], input.fields[4], true) &&
55 NMEA::coordinate(input.fields[5], input.fields[6], false));
56}

60std::optional<GSV> gsv(const Sentence& input)
61{
62 if (input.type() != "GSV" || input.count < GSV_FIRST_SATELLITE_FIELD)
63 return {};
64 const auto& f = input.fields;
65 const auto total = number<int>(f[GSV_PAGE_COUNT_FIELD]);
66 const auto message = number<int>(f[GSV_PAGE_NUMBER_FIELD]);
67 const auto count = number<int>(f[GSV_SATELLITE_COUNT_FIELD]);
68 const auto system = satelliteConstellation(input.talker(), {}, {});
69 if (!total || !message || !count || (system == GPSConstellation::Unknown && input.talker() != "GN") || *total < 1 ||
70 *total > MAX_GSV_PAGES || *message < 1 || *message > *total || *count < 0 || *count > MAX_GSV_SATELLITES ||
71 *total != std::max(1, (*count + GSV_SATELLITES_PER_PAGE - 1) / GSV_SATELLITES_PER_PAGE))
72 return {};
73 const int entries = std::min(GSV_SATELLITES_PER_PAGE, *count - (*message - 1) * GSV_SATELLITES_PER_PAGE);
74 const size_t end = GSV_FIRST_SATELLITE_FIELD + entries * GSV_SATELLITE_FIELDS;
75 const size_t fields = input.count - GSV_FIRST_SATELLITE_FIELD;
76 const size_t slots = fields / GSV_SATELLITE_FIELDS;
77 const size_t trailingFields = fields % GSV_SATELLITE_FIELDS;
78 if (slots < static_cast<size_t>(entries) || slots > GSV_SATELLITES_PER_PAGE || trailingFields > 1)
79 return {};
80 const size_t signalIndex = GSV_FIRST_SATELLITE_FIELD + slots * GSV_SATELLITE_FIELDS;
81 // Some receivers retain all four satellite slots on short or empty pages.
82 for (size_t index = end; index < signalIndex; ++index) {
83 if (!f[index].empty())
84 return {};
85 }
86 GSV result{system, *total, *message, *count};
87 if (trailingFields == 1 && !f[signalIndex].empty()) {
88 const auto signal = number<int>(f[signalIndex], HEX_BASE);
89 if (!signal || *signal < 0 || *signal > MAX_GSV_SIGNAL_ID)
90 return {};
91 result.signal = *signal;
92 }
93 for (size_t index = GSV_FIRST_SATELLITE_FIELD; index < end; index += GSV_SATELLITE_FIELDS) {
94 const auto id = number<int>(f[index]);
95 if (!id || *id < 1 || *id > MAX_GSV_SATELLITE_ID)
96 return {};
97 SatelliteData satellite;
98 satellite.constellation = satelliteConstellation(input.talker(), {}, *id);
99 if (satellite.constellation == GPSConstellation::Unknown)
100 return {};
101 satellite.id = satelliteId(satellite.constellation, *id);
102 satellite.prn = *id;
103 if (const auto value = number<double>(f[index + GSV_ELEVATION_OFFSET]);
104 value && *value >= 0 && *value <= MAX_ELEVATION_DEGREES)
105 satellite.elevation = value;
106 if (const auto value = number<double>(f[index + GSV_AZIMUTH_OFFSET]);
107 value && *value >= 0 && *value <= MAX_AZIMUTH_DEGREES)
108 satellite.azimuth = value;
109 if (const auto value = number<int>(f[index + GSV_SIGNAL_OFFSET]);
110 value && *value >= 0 && *value <= MAX_SIGNAL_STRENGTH)
111 satellite.signal = value;
112 result.satellites.push_back(satellite);
113 }
114 return result;
115}

126SatelliteAssembler::Update SatelliteAssembler::ingest(const Sentence& input, uint64_t receivedAtUs, uint64_t nowUs)
127{
128 Update update{false, flushDue(nowUs)};
129 const auto complete = [&]() {
130 auto epoch = flush();
131 update.completed.insert(update.completed.end(), std::make_move_iterator(epoch.begin()),
132 std::make_move_iterator(epoch.end()));
133 };
134 if (validNavigation(input)) {
135 const auto time = utcMilliseconds(input.fields[Field::UTC_TIME]);
136 if (time && time != _time) {
137 complete();
138 _time = time;
139 }
140 return update;
141 }
142 if (input.type() == "GSV") {
143 const auto parsed = gsv(input);
144 if (!parsed)
145 return update;
146 const auto& page = *parsed;
147 auto& signalReports = _views[page.constellation];
148 if (page.message == 1 && signalReports[page.signal].complete())
149 complete();
150 auto& report = _views[page.constellation][page.signal];
151 if (page.message == 1)
152 report = {page.messages, page.satelliteCount, 1, receivedAtUs, {}};
153 if (report.messages != page.messages || report.count != page.satelliteCount || report.next != page.message) {
154 _views[page.constellation].erase(page.signal);
155 return update;
156 }
157 report.satellites.insert(report.satellites.end(), page.satellites.begin(), page.satellites.end());
158 report.timestamp = std::min(report.timestamp, receivedAtUs);
159 ++report.next;
160 update.accepted = true;
161 } else if (input.type() == "GSA" && input.count >= Field::GSA_MIN_FIELDS) {
162 const auto& f = input.fields;
163 const auto fix = number<unsigned>(f[Field::GSA_DIMENSION]);
164 if (!fix || *fix < FixDimension::NO_FIX || *fix > FixDimension::THREE_D)
165 return update;
166 std::optional<int> system;
167 if (input.talker() == "GN" && input.count > Field::GSA_SYSTEM_ID && !f[Field::GSA_SYSTEM_ID].empty()) {
168 system = number<int>(f[Field::GSA_SYSTEM_ID], HEX_BASE);
169 if (!system)
170 return update;
171 }
172 std::map<GPSConstellation, Used> reports;
173 const auto explicitSystem = satelliteConstellation(input.talker(), system, {});
174 if (explicitSystem != GPSConstellation::Unknown)
175 reports[explicitSystem].timestamp = receivedAtUs;
176 if (explicitSystem == GPSConstellation::GPS)
177 reports[GPSConstellation::SBAS].timestamp = receivedAtUs;
178 for (size_t index = Field::GSA_FIRST_SATELLITE; index < Field::GSA_FIRST_SATELLITE + Field::GSA_SATELLITE_SLOTS;
179 ++index) {
180 if (f[index].empty())
181 continue;
182 const auto id = number<int>(f[index]);
183 if (!id || *id <= 0)
184 return update;
185 const auto constellation = satelliteConstellation(input.talker(), system, id);
186 if (constellation == GPSConstellation::Unknown)
187 return update;
188 auto& report = reports[constellation];
189 report.timestamp = receivedAtUs;
190 if (*fix != FixDimension::NO_FIX)
191 report.ids.insert(satelliteId(constellation, *id));
192 }
193 if (reports.empty())
194 return update;
195 if (!_views.empty() && std::any_of(reports.begin(), reports.end(),
196 [this](const auto& report) { return _used.contains(report.first); }))
197 complete();
198 for (auto& [constellation, report] : reports)
199 _used[constellation] = std::move(report);
200 update.accepted = true;
201 }
202 if (update.accepted) {
203 if (!_batchStartedUs) {
204 _batchStartedUs = nowUs;
205 }
206 _lastAcceptedUs = nowUs;
207 }
208 return update;
209}

225SatelliteEpoch SatelliteAssembler::flush()
226{
227 _batchStartedUs.reset();
228 std::map<GPSConstellation, SatelliteSystem> systems;
229 std::map<GPSConstellation, std::map<int, SatelliteData>> satellites;
230 const auto includeView = [&systems](GPSConstellation constellation, uint64_t timestamp) {
231 auto& out = systems[constellation];
232 out.constellation = constellation;
233 out.inViewTimestampUs = out.inViewTimestampUs ? std::min(out.inViewTimestampUs, timestamp) : timestamp;
234 };
235 for (const auto& [system, signalReports] : _views) {
236 for (const auto& [signal, report] : signalReports) {
237 if (!report.complete())
238 continue;
239 // GN identifies coverage only through its entries, including across pages and signals.
240 if (system != GPSConstellation::Unknown) {
241 includeView(system, report.timestamp);
242 if (system == GPSConstellation::GPS)
243 includeView(GPSConstellation::SBAS, report.timestamp);
244 }
245 for (const auto& satellite : report.satellites) {
246 includeView(satellite.constellation, report.timestamp);
247 auto& view = satellites[satellite.constellation];
248 const auto existing = view.find(satellite.id);
249 if (existing == view.end() || satellite.signal.value_or(-1) > existing->second.signal.value_or(-1))
250 view[satellite.id] = satellite;
251 }
252 }
253 }
254 for (auto& [system, view] : satellites) {
255 for (auto& [id, satellite] : view)
256 systems[system].satellites.push_back(std::move(satellite));
257 }
258 for (const auto& [system, used] : _used) {
259 auto& out = systems[system];
260 out.constellation = system;
261 out.inUseTimestampUs = used.timestamp;
262 out.usedIds = used.ids;
263 }
264 SatelliteEpoch result;
265 for (auto& [system, report] : systems)
266 result.push_back(std::move(report));
267 _views.clear();
268 _used.clear();
269 return result;
270}
