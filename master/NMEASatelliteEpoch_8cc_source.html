29std::optional<GSV> gsv(const Sentence& input)
30{
31 if (input.type() != "GSV" || input.count < GSV_FIRST_SATELLITE_FIELD)
32 return {};
33 const auto& f = input.fields;
34 const auto total = number<int>(f[GSV_PAGE_COUNT_FIELD]);
35 const auto message = number<int>(f[GSV_PAGE_NUMBER_FIELD]);
36 const auto count = number<int>(f[GSV_SATELLITE_COUNT_FIELD]);
37 const auto system = satelliteConstellation(input.talker(), {}, {});
38 if (!total || !message || !count || (system == GPSConstellation::Unknown && input.talker() != "GN") || *total < 1 ||
39 *total > MAX_GSV_PAGES || *message < 1 || *message > *total || *count < 0 || *count > MAX_GSV_SATELLITES ||
40 *total != std::max(1, (*count + GSV_SATELLITES_PER_PAGE - 1) / GSV_SATELLITES_PER_PAGE))
41 return {};
42 const int entries = std::min(GSV_SATELLITES_PER_PAGE, *count - (*message - 1) * GSV_SATELLITES_PER_PAGE);
43 const size_t end = GSV_FIRST_SATELLITE_FIELD + entries * GSV_SATELLITE_FIELDS;
44 const size_t fields = input.count - GSV_FIRST_SATELLITE_FIELD;
45 const size_t slots = fields / GSV_SATELLITE_FIELDS;
46 const size_t trailingFields = fields % GSV_SATELLITE_FIELDS;
47 if (slots < static_cast<size_t>(entries) || slots > GSV_SATELLITES_PER_PAGE || trailingFields > 1)
48 return {};
49 const size_t signalIndex = GSV_FIRST_SATELLITE_FIELD + slots * GSV_SATELLITE_FIELDS;
50 // Some receivers retain all four satellite slots on short or empty pages.
51 for (size_t index = end; index < signalIndex; ++index) {
52 if (!f[index].empty())
53 return {};
54 }
55 GSV result{system, *total, *message, *count};
56 if (trailingFields == 1 && !f[signalIndex].empty()) {
57 const auto signal = number<int>(f[signalIndex], HEX_BASE);
58 if (!signal || *signal < 0 || *signal > MAX_GSV_SIGNAL_ID)
59 return {};
60 result.signal = *signal;
61 }
62 for (size_t index = GSV_FIRST_SATELLITE_FIELD; index < end; index += GSV_SATELLITE_FIELDS) {
63 const auto id = number<int>(f[index]);
64 if (!id || *id < 1 || *id > MAX_GSV_SATELLITE_ID)
65 return {};
66 SatelliteData satellite;
67 satellite.constellation = satelliteConstellation(input.talker(), {}, *id);
68 if (satellite.constellation == GPSConstellation::Unknown)
69 return {};
70 satellite.id = gpsSatelliteId(satellite.constellation, *id);
71 satellite.prn = *id;
72 if (const auto value = number<double>(f[index + GSV_ELEVATION_OFFSET]);
73 value && *value >= 0 && *value <= MAX_ELEVATION_DEGREES)
74 satellite.elevation = value;
75 if (const auto value = number<double>(f[index + GSV_AZIMUTH_OFFSET]);
76 value && *value >= 0 && *value <= MAX_AZIMUTH_DEGREES)
77 satellite.azimuth = value;
78 if (const auto value = number<int>(f[index + GSV_SIGNAL_OFFSET]);
79 value && *value >= 0 && *value <= MAX_SIGNAL_STRENGTH)
80 satellite.signal = value;
81 result.satellites.push_back(satellite);
82 }
83 return result;
84}

167SatelliteEpoch SatelliteAssembler::flush()
168{
169 std::map<GPSConstellation, SatelliteSystem> systems;
170 std::map<GPSConstellation, std::map<int, SatelliteData>> satellites;
171 const auto includeView = [&systems](GPSConstellation constellation, uint64_t timestamp) {
172 auto& out = systems[constellation];
173 out.constellation = constellation;
174 out.inViewTimestampUs = out.inViewTimestampUs ? std::min(out.inViewTimestampUs, timestamp) : timestamp;
175 };
176 for (const auto& [system, signalReports] : _views) {
177 for (const auto& [signal, report] : signalReports) {
178 if (!report.complete())
179 continue;
180 // GN identifies coverage only through its entries, including across pages and signals.
181 if (system != GPSConstellation::Unknown) {
182 includeView(system, report.timestamp);
183 if (system == GPSConstellation::GPS)
184 includeView(GPSConstellation::SBAS, report.timestamp);
185 }
186 for (const auto& satellite : report.satellites) {
187 includeView(satellite.constellation, report.timestamp);
188 auto& view = satellites[satellite.constellation];
189 const auto existing = view.find(satellite.id);
190 if (existing == view.end() || satellite.signal.value_or(-1) > existing->second.signal.value_or(-1))
191 view[satellite.id] = satellite;
192 }
193 }
194 }
195 for (auto& [system, view] : satellites) {
196 for (auto& [id, satellite] : view)
197 systems[system].satellites.push_back(std::move(satellite));
198 }
199 for (const auto& [system, used] : _used) {
200 auto& out = systems[system];
201 out.constellation = system;
202 out.inUseTimestampUs = used.timestamp;
203 out.usedIds = used.ids;
204 }
205 SatelliteEpoch result;
206 for (auto& [system, report] : systems)
207 result.push_back(std::move(report));
208 _views.clear();
209 _used.clear();
210 return result;
211}
