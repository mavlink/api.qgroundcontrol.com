198bool GPSCorrectionRouter::acceptIngress(const GPSCorrectionIngress& ingress)
199{
200 const auto& token = ingress.token();
201 if (!token.belongsTo(this) || !token.valid() ||
202 (!token.instance().isEmpty() && ingress.frame().sourceInstance != token.instance())) {
203 return false;
204 }
205 auto frame = ingress.frame();
206 if (ingress.rejection() != GPSCorrectionReason::None) {
207 recordRejectedFrame(frame, ingress.rejection());
208 return false;
209 }
210 if ((!frame.validated && frame.source != GPSCorrectionSource::Udp) ||
211 (frame.validated && !RTCM::isValidFrame(frame.data))) {
212 frame.validated = false;
213 recordRejectedFrame(frame, GPSCorrectionReason::InvalidFrame);
214 return false;
215 }
216 return acceptFrame(frame);
217}

355bool GPSCorrectionRouter::acceptFrame(GPSCorrectionFrame frame)
356{
357 const int index = _sourceIndex(frame.source);
358 if (_shutdown || _submitting || index < 0 || frame.data.isEmpty()) {
359 return false;
360 }
361 frame.deliveryId = ++_nextDelivery;
362 const auto& stats = _ledger.statistics()[index];
363 if (!stats.active || stats.session != frame.session) {
364 _ledger.recordEvent(
365 frame, GPSCorrectionStage::Dropped,
366 stats.session != frame.session ? GPSCorrectionReason::SessionMismatch : GPSCorrectionReason::InactiveSource,
367 frame.data.size());
368 return false;
369 }
370 if (frame.sourceInstance.isEmpty()) {
371 frame.sourceInstance = _configuredInstances[index];
372 } else if (!_configuredInstances[index].isEmpty() && frame.sourceInstance != _configuredInstances[index]) {
373 _ledger.recordEvent(frame, GPSCorrectionStage::Dropped, GPSCorrectionReason::SessionMismatch,
374 frame.data.size());
375 return false;
376 }
377 _ledger.received(frame);
378 const qint64 now = _clock();
379 const qint64 age = GPSCorrectionFrame::ageMs(frame.receivedAtMs, now);
380 if (age < 0) {
381 _ledger.filtered(frame);
382 _ledger.recordDrop(frame, GPSCorrectionReason::InvalidTimestamp, frame.data.size());
383 return false;
384 }
385 if (frame.validated) {
386 _ledger.validated(frame);
387 }
388 _selector.observe(frame, false, now);
389 if (frame.filtered || age >= FRESHNESS_TIMEOUT_MS) {
390 _ledger.filtered(frame);
391 _ledger.recordDrop(frame, frame.filtered ? GPSCorrectionReason::MessageFiltered : GPSCorrectionReason::Expired,
392 frame.data.size());
393 return false;
394 }
395 _selector.observe(frame, true, now);
396 if (frame.validated && frame.messageId == 0 && frame.data.size() >= 8 &&
397 static_cast<quint8>(frame.data[0]) == 0xD3) {
398 frame.messageId = (static_cast<quint8>(frame.data[3]) << 4) | (static_cast<quint8>(frame.data[4]) >> 4);
399 }
400 const bool selected = _selector.selected(frame, now);
401 if (!selected) {
402 _ledger.filtered(frame);
403 _ledger.recordDrop(frame, GPSCorrectionReason::NotSelected, frame.data.size());
404 } else {
405 _ledger.selected(frame);
406 }
407 return _submit(frame, selected);
408}

410bool GPSCorrectionRouter::_submit(const GPSCorrectionFrame& frame, bool selected)
411{
412 const QString key = GPSCorrectionSelector::key(frame.source, frame.sourceInstance);
413 const QPointer<GPSCorrectionRouter> guard(this);
414 const quint64 revision = _revision;
415 const auto sinks = _sinks;
416 _submitting = true;
417 auto finishSubmitting = qScopeGuard([guard] {
418 if (guard) {
419 guard->_submitting = false;
420 }
421 });
422 const QString submissionSource = key + QLatin1Char('#') + QString::number(frame.session);
423 if (selected && _lastSubmittedSource != submissionSource) {
424 _lastSubmittedSource = submissionSource;
425 emit sourceSelected(frame.source, frame.sourceInstance);
426 if (!guard) {
427 return false;
428 }
429 if (_shutdown || revision != _revision) {
430 return false;
431 }
432 }
433 quint64 logicalQueuedBytes = 0;
434 bool completeSubmission = false;
435 bool attempted = false;
436 GPSCorrectionReason submissionFailure = GPSCorrectionReason::DestinationUnavailable;
437 for (auto it = sinks.cbegin(); it != sinks.cend(); ++it) {
438 if (it->scope == GPSCorrectionSource::Unknown ? !selected : it->scope != frame.source) {
439 continue;
440 }
441 attempted = true;
442 if (it->completion == Completion::Reported && !_ledger.admissionAvailable()) {
443 submissionFailure = GPSCorrectionReason::DiagnosticsBackpressure;
444 _ledger.recordDrop(frame, submissionFailure, frame.data.size(), it.key(), 0, false);
445 continue;
446 }
447 _admission.emplace();
448 _admission->deliveryId = frame.deliveryId;
449 const auto finishAdmission = qScopeGuard([guard] {
450 if (guard) {
451 guard->_finishAdmission();
452 }
453 });
454 const QList<Admission> admissions = it->admit(frame);
455 if (!guard) {
456 return false;
457 }
458 // Callback retirement must not discard returned admission evidence.
459 for (const auto& admission : admissions) {
460 if (admission.destination.isEmpty()) {
461 continue;
462 }
463 const auto& submitted = admission.submission;
464 const quint64 bytes = (std::min) (submitted.queuedBytes, static_cast<quint64>(frame.data.size()));
465 const bool complete = admission.complete && bytes == static_cast<quint64>(frame.data.size());
466 if (!_ledger.admitted(frame, it.key(), admission.destination, submitted.destinationSession, bytes, complete,
467 it->completion == Completion::Reported)) {
468 continue;
469 }
470 logicalQueuedBytes = (std::max) (logicalQueuedBytes, bytes);
471 completeSubmission |= complete;
472 if (!complete) {
473 submissionFailure = submitted.reason == GPSCorrectionReason::None
474 ? GPSCorrectionReason::DestinationUnavailable
475 : submitted.reason;
476 _ledger.recordDrop(frame, submissionFailure, frame.data.size() - bytes, admission.destination,
477 submitted.destinationSession, false);
478 }
479 }
480 if (_shutdown || revision != _revision) {
481 for (const auto& admission : admissions) {
482 _deferRetirement(
483 {RetirementKind::Destination, admission.destination, admission.submission.destinationSession});
484 }
485 break;
486 }
487 }
488 _ledger.queued(frame, logicalQueuedBytes, completeSubmission);
489 if (attempted && !completeSubmission) {
490 _ledger.recordDrop(frame, submissionFailure, frame.data.size() - logicalQueuedBytes);
491 }
492 _ledger.pruneDestinationHistory();
493 _submitting = false;
494 finishSubmitting.dismiss();
495 if (selected && !_shutdown && revision == _revision) {
496 emit frameRouted(frame);
497 }
498 return selected;
499}
