120GPSSatelliteReport satellites(const satellite_info_s& source, GPSType type)
121{
122 GPSSatelliteReport result;
123 result.timestampUs = source.timestamp;
124 result.count = (std::min) (source.count, satellite_info_s::SAT_INFO_MAX_SATELLITES);
125 if (type == GPSType::septentrio) {
126 // The PX4 SBF wrapper publishes a count, not per-satellite observations.
127 return result;
128 }
129 for (uint16_t i = 0; i < result.count; ++i) {
130 auto& satellite = result.satellites[i];
131 satellite.id = source.svid[i];
132 satellite.prn = source.prn[i];
133 satellite.used = source.used[i] != 0;
134 if (satellite.id == 0) {
135 continue;
136 }
137 satellite.signalStrength = source.snr[i];
138 const int elevation = std::bit_cast<int8_t>(source.elevation[i]);
139 if (elevation >= -90 && elevation <= 90) {
140 satellite.elevationDegrees = static_cast<float>(elevation);
141 }
142 if (type == GPSType::ublox) {
143 satellite.azimuthDegrees = static_cast<float>(source.azimuth[i]) * 360.0f / 255.0f;
144 }
145 // Ashtech truncates degree azimuths to uint8_t; the original angle is unrecoverable.
146 }
147 return result;
148}
