75void VideoSettings::_offerCameras(FactMetaData* metaData, Fact* fact, QVariantList values, QStringList labels,
76 const QStringList& cameras)
77{
78 for (const QString& camera : cameras) {
79 values.append(camera);
80 labels.append(camera);
81 }
82 const QString selected = fact ? fact->rawValue().toString() : QString();
83 if (!selected.isEmpty() && !values.contains(selected)) {
84 values.append(selected);
85 labels.append(tr("%1 (not connected)").arg(selected));
86 }
87
88 if ((values == metaData->enumValues()) && (labels == metaData->enumStrings())) {
89 return;
90 }
91 if (fact) {
92 // Tells the settings page's combo box as well.
93 fact->setEnumInfo(labels, values);
94 } else {
95 metaData->setEnumInfo(labels, values);
96 }
97}

210void VideoSettings::pruneUnavailableDecoders()
211{
212 if (!hasPipelineSettings()) {
213 return;
214 }
215 static const QList<VideoDecoderOption> hardwareFamilies{
216 VideoDecoderOption::NVIDIA, VideoDecoderOption::VAAPI, VideoDecoderOption::DirectX3D,
217 VideoDecoderOption::VideoToolbox, VideoDecoderOption::Intel, VideoDecoderOption::Vulkan,
218 };
219
220 const QList<VideoDecoderOption> available = VideoBackend::instance().availableDecoderFamilies();
221 const auto metaIt = _nameToMetaDataMap.constFind(forceVideoDecoderName);
222 if (metaIt == _nameToMetaDataMap.constEnd() || !metaIt.value()) {
223 return;
224 }
225 FactMetaData* const metaData = metaIt.value();
226 bool pruned = false;
227 for (const auto family : hardwareFamilies) {
228 // removeEnumInfo() qWarns on an absent value, so skip families not in the enum; values are
229 // stored as QVariant(int), so match that representation.
230 const QVariant familyValue = static_cast<int>(family);
231 if (!available.contains(family) && metaData->enumValues().contains(familyValue)) {
232 metaData->removeEnumInfo(familyValue);
233 pruned = true;
234 }
235 }
236
237 Fact* const fact = forceVideoDecoder();
238 if (pruned) {
239 // Backend init is async — refresh any live FactComboBox bound to this fact.
240 emit fact->enumsChanged();
241 }
242 if (!metaData->enumValues().contains(fact->rawValue())) {
243 fact->setRawValue(static_cast<int>(VideoDecoderOption::Default));
244 }
