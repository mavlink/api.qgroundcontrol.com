18/// Mission path for the GeoMap: a 3D ribbon connecting every mission item in
19/// sequence order (simple items at their coordinate; complex items via their
20/// generic entryCoordinate/exitCoordinate, so the ribbon routes straight
21/// through a complex item's own footprint rather than skipping it), at true
22/// (AMSL) altitude. Stays visible in 2D mode (crossfade3D off), same as
23/// GeoMapFlightPath. Rebuilt wholesale on any mission change (missions are
24/// edited, not streamed, so the flight path's incremental append API is not
25/// needed here).
26GeoMapItem {
27 id: root
28
29 property var missionController
30 property var vehicle ///< Active vehicle, for the RTL-to-home segment
31 // DEM-vs-home-AMSL bias (see GeoMapVehicleItem), computed once by the
32 // consumer for the active vehicle and shared with GeoMapMissionItems
33 // instead of being re-sampled here independently
34 property real homeTerrainBias: 0
35 property real lineWidth: ScreenTools.defaultFontPixelHeight / 5
36 // Matches MissionLineView.qml's existing flight-path line color
37 property color lineColor: QGroundControl.globalPalette.mapMissionTrajectory
38
39 altitudeMode: GeoMapItem.Absolute
40 crossfade3D: false
41 coordinate: {
42 if (!_geometry || !_geometry.anchorCoordinate.isValid) {
43 return QtPositioning.coordinate()
44 }
45 const anchor = _geometry.anchorCoordinate
46 return QtPositioning.coordinate(anchor.latitude, anchor.longitude, anchor.altitude + root.homeTerrainBias)
47 }
48 visible: !!missionController
49
50 // The geometry lives inside the 3D delegate (created/destroyed with the
51 // scene); this bridges its anchor out for the coordinate binding above
52 property var _geometry: null
53
54 readonly property var _camera: scene ? scene.camera : null
55
56 // Pixel-to-scene-unit factor at unit distance; the shader scales it by
57 // each vertex's own camera distance
58 readonly property real _screenFactor: _camera ? _camera.unitsPerPixelAtUnitDistance : 0

60 // Ordered coordinates (with AMSL altitude) of every item the vehicle
61 // actually routes through, mirroring MissionController::_recalcFlightPathSegments:
62 // items must specifiesCoordinate && !isStandaloneCoordinate to get a
63 // segment (excludes RTL, which has no mission-encoded coordinate, and
64 // standalone items like ROI); a qualifying item contributes its
65 // entryCoordinate, plus its exitCoordinate too when that differs
66 // (exitCoordinateSameAsEntry false) — for a simple item these are always
67 // the same point, for a complex item this routes the ribbon straight
68 // through its footprint edge-to-edge. Hitting RTL stops the walk and
69 // links the last point to home instead (linkEndToHome). A land item ends
70 // the walk at its entry: its interior path (loiter exit -> touchdown) is
71 // drawn by the landing pattern visual from the slope start, not straight
72 // from the loiter center, and _recalcFlightPathSegments draws no segment
73 // after a landing item either. The walk starts at item 1 (item 0 is the
74 // MissionSettingsItem, i.e. home); home is prepended only when the mission
75 // starts from the ground — rover, or a takeoff command before the first
76 // coordinate item — mirroring linkStartToHome. Recomputes automatically on
77 // any structural or per-item change since the loop below reads every
78 // NOTIFY-backed property it depends on.
79 readonly property var _pathCoordinates: {
80 const coordinates = []
81 if (root.missionController) {
82 const items = root.missionController.visualItems
83 let linkStartToHome = root.vehicle ? root.vehicle.rover : false
84 for (let i = 1; i < items.count; ++i) {
85 const visualItem = items.get(i)
86 if (visualItem.isSimpleItem && visualItem.command === MAVLinkEnums.MAV_CMD_NAV_RETURN_TO_LAUNCH) {
87 if (coordinates.length > 0 && root.vehicle && root.vehicle.homePosition.isValid) {
88 coordinates.push(root.vehicle.homePosition)
89 }
90 break
91 }
92 if (coordinates.length === 0 && visualItem.isTakeoffItem) {
93 linkStartToHome = true
94 }
95 if (visualItem.specifiesCoordinate && !visualItem.isStandaloneCoordinate) {
96 coordinates.push(QtPositioning.coordinate(visualItem.entryCoordinate.latitude,
97 visualItem.entryCoordinate.longitude,
98 visualItem.amslEntryAlt))
99 if (visualItem.isLandCommand) {
100 break
101 }
102 if (!visualItem.exitCoordinateSameAsEntry) {
103 coordinates.push(QtPositioning.coordinate(visualItem.exitCoordinate.latitude,
104 visualItem.exitCoordinate.longitude,
105 visualItem.amslExitAlt))
106 }
107 }
108 }
109 if (linkStartToHome && coordinates.length > 0 && items.count > 0) {
110 const home = items.get(0)
111 if (home.coordinate.isValid) {
112 coordinates.unshift(QtPositioning.coordinate(home.coordinate.latitude,
113 home.coordinate.longitude,
114 home.amslEntryAlt))
115 }
116 }
117 }
118 return root._subdivided(coordinates)
119 }
120
121 // Legs longer than this are subdivided along the great circle (matching
122 // MissionLineView.qml): the geometry joins points linearly in Web
123 // Mercator, which diverges materially from the actual route over long
124 // distances
125 readonly property real _maxSegmentLengthM: 50000
126
127 function _subdivided(coordinates) {
128 const result = []
129 for (let i = 0; i < coordinates.length; ++i) {
130 const coord = coordinates[i]
131 if (i > 0) {
132 const prev = coordinates[i - 1]
133 const distance = prev.distanceTo(coord)
134 if (distance > root._maxSegmentLengthM) {
135 const segments = Math.ceil(distance / root._maxSegmentLengthM)
136 const azimuth = prev.azimuthTo(coord)
137 for (let j = 1; j < segments; ++j) {
138 const point = prev.atDistanceAndAzimuth((j * distance) / segments, azimuth)
139 result.push(QtPositioning.coordinate(point.latitude, point.longitude,
140 prev.altitude + ((coord.altitude - prev.altitude) * j) / segments))
141 }
142 }
143 }
144 result.push(coord)
145 }
146 return result
147 }
148

190 }
191 }
