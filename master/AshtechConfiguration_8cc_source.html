39{

42 "$PASHS,NME,POS,%c,ON,0.2\r\n", // reduce position updates to 5 Hz
43
44 "$PASHS,RT3,1074,%c,ON,1\r\n", // GPS observations
45 "$PASHS,RT3,1084,%c,ON,1\r\n", // GLONASS observations
46 "$PASHS,RT3,1094,%c,ON,1\r\n", // Galileo observations

52 "$PASHS,RT3,1013,%c,ON,1\r\n", // System parameters
53 "$PASHS,RT3,1029,%c,ON,1\r\n", // ASCII message

62 for (unsigned int conf_i = 0; conf_i < sizeof(rtcm_options) / sizeof(rtcm_options[0]); conf_i++) {

66 controlFailed();

72int GPSNativeAshtech::writeAckedCommand(const void* buf, int buf_length, unsigned timeout)
73{
74 if (!writeCommand({std::string(static_cast<const char*>(buf), buf_length), std::chrono::milliseconds(timeout)},
75 {static_cast<const uint8_t*>(buf), static_cast<size_t>(buf_length)})) {
76 return -1;
77 }
78
79 return waitForReply(NMEACommand::Acked, timeout);
80}

82int GPSNativeAshtech::waitForReply(NMEACommand command, const unsigned timeout)
83{
84 _command_state = NMEACommandState::waiting;
85 _waiting_for_command = command;
86 const auto clearReply = qScopeGuard([this] { _command_state = NMEACommandState::idle; });
87
88 const auto result = awaitCommand({_commandWrite.evidence.command, std::chrono::milliseconds(timeout)}, [this] {
89 return _command_state == NMEACommandState::received ? GPSCommandOutcome::Acknowledged
90 : _command_state == NMEACommandState::nack ? GPSCommandOutcome::Rejected
91 : GPSCommandOutcome::Pending;
92 });
93 return result.evidence.outcome == GPSCommandOutcome::Acknowledged ? 0 : -1;
94}

96int GPSNativeAshtech::configure(unsigned& baudrate, const GPSConfig& config)
97{
98 _configure_done = false;
99 resetIOError();
100 _survey_duration = 0;
101 _survey_in_start = 0;
102 _correction_output_activated = false;
103 _correctionSetupPending = false;
104 _rtcmActivationPending = false;
105 _got_pashr_pos_message = false;
106 _command_state = NMEACommandState::idle;
107 _rtcm_parsing.reset();
108 decodeInit();
109 if (!validateConfiguration(config)) {
110 return -1;
111 }
112 _baseConfig = config.base;
113 _output_mode = config.output_mode;
114
115 /* Try different baudrates (115200 is the default for Trimble) and request the baudrate that we want.
116 *
117 * These are Ashtech proprietary commands, we can use them for auto-detection:
118 * $PASHS for setting
119 * $PASHQ for querying
120 * $PASHR for a response
121 */
122 const unsigned baudrates_to_try[] = {9600, 38400, 19200, 57600, 115200};
123 bool success = false;
124
125 unsigned test_baudrate;
126
127 for (unsigned int baud_i = 0; !success && baud_i < sizeof(baudrates_to_try) / sizeof(baudrates_to_try[0]);
128 baud_i++) {
129 test_baudrate = baudrates_to_try[baud_i];
130
131 if (baudrate > 0 && baudrate != test_baudrate) {
132 continue; // skip to next baudrate
133 }
134
135 setBaudrate(test_baudrate);
136
137 const char port_config[] = "$PASHQ,PRT\r\n"; // ask for the current port configuration
138
139 for (int run = 0; run < 2; ++run) { // try several times
140 writeCommand({port_config, std::chrono::milliseconds(ASH_RESPONSE_TIMEOUT)},
141 {reinterpret_cast<const uint8_t*>(port_config), sizeof(port_config) - 1});
142
143 if (waitForReply(NMEACommand::PRT, ASH_RESPONSE_TIMEOUT) == 0) {
144 success = true;
145 break;
146 }
147 }
148 }
149
150 if (!success) {
151 return -1;
152 }
153
154 // We successfully got a response and know to which port we are connected. Now set the desired baudrate
155 // if it's different from the current one.
156 const unsigned desired_baudrate = 115200; // changing this requires also changing the SPD command
157
158 baudrate = test_baudrate;
159
160 if (baudrate != desired_baudrate) {
161 baudrate = desired_baudrate;
162 const char baud_config[] = "$PASHS,SPD,%c,9\r\n"; // configure baudrate to 115200
163 char baud_config_str[sizeof(baud_config)];
164 int len = snprintf(baud_config_str, sizeof(baud_config_str), baud_config, _port);
165 writeCommand({baud_config_str, std::chrono::milliseconds(ASH_RESPONSE_TIMEOUT)},
166 {reinterpret_cast<const uint8_t*>(baud_config_str), static_cast<size_t>(len)});
167 decodeInit();
168 receiveWait(200);
169 decodeInit();
170 setBaudrate(baudrate);
171
172 success = false;
173
174 for (int run = 0; run < 10; ++run) {
175 // We ask for the port config again. If we get a reply, we know that the changed settings work.
176 const char port_config[] = "$PASHQ,PRT\r\n";
177 writeCommand({port_config, std::chrono::milliseconds(ASH_RESPONSE_TIMEOUT)},
178 {reinterpret_cast<const uint8_t*>(port_config), sizeof(port_config) - 1});
179
180 if (waitForReply(NMEACommand::PRT, ASH_RESPONSE_TIMEOUT) == 0) {
181 success = true;
182 break;
183 }
184 }
185
186 if (!success) {
187 return -1;
188 }
189 }
190
191 // Additional commands that might be useful:
192 // Reading firmware version:
193 // $PASHQ,VER
194 // Reading installed firmware options:
195 // $PASHQ,OPTION
196 // The output for the Trimble MB-two is:
197 // $PASHR,OPTION,0,SERIAL NUMBER,5730C00370*3E
198 // $PASHR,OPTION,@1,GEOFENCING_WW,034017C7114ED*36
199 // $PASHR,OPTION,N,GPS,0340173F8924D*66
200 // $PASHR,OPTION,G,GLONASS,0340178A9E138*69
201 // $PASHR,OPTION,B,BEIDOU,03401434EC35A*4D
202 // $PASHR,OPTION,X,L1TRACKING,0340119C547B8*40
203 // $PASHR,OPTION,Y,L2TRACKING,034012CD03607*42
204 // $PASHR,OPTION,W,20HZ,034016B5A5225*2A
205 // $PASHR,OPTION,J,RTKROVER,034010C800693*41
206 // $PASHR,OPTION,K,RTKBASE,03401065AB099*7E
207 // $PASHR,OPTION,D,DUO,0340138851415*70
208 // $PASHR,OPTION,S,L3TRACKING,034011C7AB73D*48
209 // Reset the full configuration (however it will lead to a reboot and requires about 15s waiting time)
210 // $PASHS,RST
211
212 // get the board identification
213 const char board_identification[] = "$PASHQ,RID\r\n";
214
215 if (writeCommand({board_identification, std::chrono::milliseconds(ASH_RESPONSE_TIMEOUT)},
216 {reinterpret_cast<const uint8_t*>(board_identification), sizeof(board_identification) - 1})) {
217 if (waitForReply(NMEACommand::RID, ASH_RESPONSE_TIMEOUT) != 0) {
218 return -1;
219 }
220 }
221
222 // Now configure the messages we want
223
224 const char update_rate[] = "$PASHS,POP,20\r\n"; // set internal update rate to 20 Hz
225
226 if (writeAckedCommand(update_rate, sizeof(update_rate) - 1, ASH_RESPONSE_TIMEOUT) != 0) {
227 // for some reason we don't get a response here
228 }
229
230 // Enable dual antenna mode (2: both antennas are L1/L2 GNSS capable, flex mode, avoids the need to determine
231 // the baseline length through a prior calibration stage)
232 // Needs to be set before other commands
233 const bool use_dual_mode = _output_mode != OutputMode::RTCM && _board == AshtechBoard::trimble_mb_two;
234
235 if (use_dual_mode) {
236 const char duo_mode[] = "$PASHS,SNS,DUO,2\r\n";
237
238 if (writeAckedCommand(duo_mode, sizeof(duo_mode) - 1, ASH_RESPONSE_TIMEOUT) != 0) {
239 }
240
241 } else {
242 const char solo_mode[] = "$PASHS,SNS,SOL\r\n";
243
244 if (writeAckedCommand(solo_mode, sizeof(solo_mode) - 1, ASH_RESPONSE_TIMEOUT) != 0) {
245 }
246 }
247
248 char buffer[40];
249 const char* config_options[] = {
250 "$PASHS,NME,ALL,%c,OFF\r\n", // disable all NMEA and NMEA-Like Messages
251 "$PASHS,ATM,ALL,%c,OFF\r\n", // disable all ATM (ATOM) Messages
252 "$PASHS,OUT,%c,ON\r\n", // enable periodic output
253 "$PASHS,NME,ZDA,%c,ON,3\r\n", // enable ZDA (date & time) output every 3s
254 "$PASHS,NME,GST,%c,ON,3\r\n", // position accuracy messages
255 "$PASHS,NME,POS,%c,ON,0.05\r\n", // position & velocity (we can go up to 20Hz if FW option [W] is given and to
256 // 50Hz if [8] is given)
257 "$PASHS,NME,GSV,%c,ON,1\r\n" // satellite status
258 };
259
260 for (unsigned int conf_i = 0; conf_i < sizeof(config_options) / sizeof(config_options[0]); conf_i++) {
261 int len = snprintf(buffer, sizeof(buffer), config_options[conf_i], _port);
262
263 if (writeAckedCommand(buffer, len, ASH_RESPONSE_TIMEOUT) != 0) {
264 // some commands are not acked (e.g. GSV), so don't make this fatal
265 }
266 }
267
268 if (use_dual_mode) {
269 // enable heading output
270 const char heading_output[] = "$PASHS,NME,HDT,%c,ON,0.05\r\n";
271 int len = snprintf(buffer, sizeof(buffer), heading_output, _port);
272
273 if (writeAckedCommand(buffer, len, ASH_RESPONSE_TIMEOUT) != 0) {
274 }
275 }
276
277 if (_output_mode == OutputMode::RTCM) {
278 if (!_rtcm_parsing) {
279 _rtcm_parsing.emplace();
280 }
281
282 _rtcm_parsing->reset();
283 }
284
285 if (_output_mode == OutputMode::RTCM && _board == AshtechBoard::trimble_mb_two) {
286 GPSNativeSurveyReport status{};
287 status.latitude = status.longitude = (double) NAN;
288 status.altitude = NAN;
289 status.duration = 0;
290 status.mean_accuracy = 0;
291 const bool valid = false;
292 const bool active = true;
293 status.flags = (int) valid | ((int) active << 1);
294 surveyInStatus(status);
295 }
296
297 _configure_done = true;
298 return ioError();
299}

301void GPSNativeAshtech::activateCorrectionOutput()
302{
303 if (_correction_output_activated || _output_mode != OutputMode::RTCM) {
304 return;
305 }
306
307 char buffer[100];
308
309 if (!_baseConfig.useFixedBase) {
310 // setup the base reference: average the position over N seconds
311 const char avg_pos[] = "$PASHS,POS,AVG,%u\r\n";
312 // alternatively use the current position as reference: "$PASHS,POS,CUR\r\n"
313 int len = snprintf(buffer, sizeof(buffer), avg_pos, static_cast<unsigned>(_baseConfig.surveyInDurationSecs));
314
315 writeCommand({buffer, std::chrono::milliseconds(ASH_RESPONSE_TIMEOUT)},
316 {reinterpret_cast<const uint8_t*>(buffer), static_cast<size_t>(len)});
317
318 if (waitForReply(NMEACommand::RECEIPT, ASH_RESPONSE_TIMEOUT) != 0) {
319 controlFailed();
320 return;
321 }
322
323 const char* config_options[] = {
324 "$PASHS,ANP,OWN,TRM55971.00\r\n", // set antenna name (arbitrary)
325 "$PASHS,STI,0001\r\n" // enter a base ID
326 };
327
328 for (unsigned int conf_i = 0; conf_i < sizeof(config_options) / sizeof(config_options[0]); conf_i++) {
329 if (writeAckedCommand(config_options[conf_i], strlen(config_options[conf_i]), ASH_RESPONSE_TIMEOUT) != 0) {
330 controlFailed();
331 return;
332 }
333 }
334
335 if (!_rtcmActivationPending) {
336 _survey_duration = 0; // use it as counter how long survey-in has been active
337 _survey_in_start = nowUs();
338 sendSurveyInStatusUpdate(true, false);
339 }
340
341 } else {
342 const GPSBaseStationConfig& settings = _baseConfig;
343 char ns, ew;
344 double latitude = settings.fixedPosition.latitudeDegrees;
345
346 if (latitude < 0.) {
347 latitude = -latitude;
348 ns = 'S';
349
350 } else {
351 ns = 'N';
352 }
353
354 // convert to ddmm.mmmmmm format
355 latitude = ((int) latitude) * 100. + (latitude - ((int) latitude)) * 60.;
356
357 double longitude = settings.fixedPosition.longitudeDegrees;
358
359 if (longitude < 0.) {
360 longitude = -longitude;
361 ew = 'W';
362
363 } else {
364 ew = 'E';
365 }
366
367 // convert to ddmm.mmmmmm format
368 longitude = ((int) longitude) * 100. + (longitude - ((int) longitude)) * 60.;
369
370 int len = snprintf(buffer, sizeof(buffer), "$PASHS,POS,%.8f,%c,%.8f,%c,%.5f,PC1", latitude, ns, longitude, ew,
371 (double) settings.fixedPosition.altitudeMeters);
372
373 if (len >= 0 && len < (int) sizeof(buffer)) {
374 if (writeAckedCommand(buffer, len, ASH_RESPONSE_TIMEOUT) != 0) {
375 controlFailed();
376 return;
377 }
378
379 } else {
380 controlFailed();
381 return;
382 }
383
384 activateRTCMOutput();
385 if (ioError()) {
386 return;
387 }
388 sendSurveyInStatusUpdate(false, true, settings.fixedPosition.latitudeDegrees,
389 settings.fixedPosition.longitudeDegrees, settings.fixedPosition.altitudeMeters);
390 }
391 _correction_output_activated = true;
392}
