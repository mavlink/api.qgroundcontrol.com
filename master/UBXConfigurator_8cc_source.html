26bool Protocol::_baseStationUnsupported() const
27{
28 const auto& identity = _identity;
29 // A receiver without time mode can neither survey in nor hold a fixed position.
30 if (identity.timeModeUnsupported) {
31 return true;
32 }
33 if (identity.board == UBX::Board::u_blox8) {
34 return !identity.isM8p && !identity.model.isEmpty();
35 }
36 const auto profile = UBX::receiverProfile(identity.board);
37 return !profile.rtcmOutput && profile.baseCapabilityKnown;

40bool Protocol::configure(GPSCommandChannel& channel, GPSConfig config, unsigned& baud)
41{
42 _base = config.base;
43 _valsetAckAmbiguous = false;
44 _legacyRTCM = {};
45 _ready = false;
46 _setMode({}, channel.stream());
47 _controller = {};
48 _requests = {};
49
50 const unsigned requestedBaud = baud;
51 const auto detection = channel.detectBaud(Plan::BAUD_RATES, requestedBaud, [this, &channel]() -> GPSBaudProbe {
52 channel.stream().reset(GPSFrameKind::UBX);
53 (void) channel.receiveCycle(Plan::BAUD_PROBE_DRAIN);
54 channel.stream().reset(GPSFrameKind::UBX);
55 if (channel.failed()) {
56 return GPSBaudProbe::Stop;
57 }
58 const bool identified = _identify(channel);
59 return identified ? GPSBaudProbe::Found : GPSBaudProbe::TryNext;
60 });
61 const auto& identity = _identity;
62 // Discovery only polls identity: silence or an unsupported identity must not change receiver settings.
63 if (!detection.found) {
64 channel.failNoAnswer(detection.probed);
65 return false;
66 }
67 if (identity.board == UBX::Board::unknown) {
68 QString detail = QStringLiteral("Unsupported u-blox receiver: hardware version %1")
69 .arg(QString::fromLatin1(identity.hardware));
70 if (!identity.protocolVersion.isEmpty()) {
71 detail += QStringLiteral(", protocol version %1").arg(QString::fromLatin1(identity.protocolVersion));
72 }
73 channel.failControl(detail);
74 return false;
75 }
76 if (_baseStationUnsupported()) {
77 const QString model =
78 identity.model.isEmpty() ? QStringLiteral("This u-blox receiver") : QString::fromLatin1(identity.model);
79 channel.failControl(QStringLiteral("%1 cannot run as an RTK base station").arg(model));
80 return false;
81 }
82 const unsigned desiredBaud = requestedBaud ? requestedBaud : Plan::BASE_BAUD;
83 if (!_setUpPort(channel, detection.baud, desiredBaud)) {
84 return false;
85 }
86 baud = desiredBaud;
87 _setMode({.corrections = true}, channel.stream());
88
89 bool deviceConfigured = false;
90 if (identity.protocol27) {
91 deviceConfigured = _configureDevice(channel);
92 } else {
93 deviceConfigured = _configureLegacyDevice(channel);
94 }
95 if (!deviceConfigured) {
96 return false;
97 }
98 if (!_restartSurveyIn(channel)) {
99 return false;
100 }
101 _ready = true;
102 _mode.navigation = true;
103 return true;
104}

116bool Protocol::_setUpPort(GPSCommandChannel& channel, unsigned detectedBaud, unsigned desiredBaud)
117{
118 const bool modern = _identity.protocol27;
119 if (modern) {
120 if (!_transactValset(channel, Plan::uart1Protocols()).succeeded()) {
121 return false;
122 }
123 } else {
124 const auto ports = Plan::legacyPorts(detectedBaud);
125 if (!_sendAcknowledged(channel, Msg::CFG_PRT, ports)) {
126 return false;
127 }
128 }
129 if (desiredBaud == detectedBaud) {
130 return true;
131 }
132
133 const UBX::Board identifiedBoard = _identity.board;
134 const UBX::MessageId command = modern ? Msg::CFG_VALSET : Msg::CFG_PRT;
135 if (modern) {
136 if (!_writeValset(channel, Plan::uart1Baudrate(desiredBaud))) {
137 return false;
138 }
139 } else {
140 const auto ports = Plan::legacyPorts(desiredBaud);
141 if (!_send(channel, Msg::CFG_PRT, ports, {{}, Plan::CONFIG_TIMEOUT, false})) {
142 return false;
143 }
144 }
145 const auto result = _waitForAck(channel, command);
146 const bool acknowledged = result.succeeded();
147 if (channel.failed() || result.outcome == GPSCommandOutcome::Rejected) {
148 return false;
149 }
150 if (!channel.setBaudrate(desiredBaud)) {
151 channel.failControl(QStringLiteral("The link cannot run at %1 baud").arg(desiredBaud));
152 return false;
153 }
154 if (modern && !acknowledged) {
155 // The ACK may be lost in the UART handoff; every later VALSET needs a matching readback.
156 _controller.requireConfigurationReadback(command.value());
157 }
158 channel.stream().reset(GPSFrameKind::UBX);
159 if (!_identify(channel)) {
160 return false;
161 }
162 if (_identity.board != identifiedBoard || _identity.protocol27 != modern || _controller.lateRejection()) {
163 return false;
164 }
165 if (modern && !acknowledged) {
166 return _verifyValset(channel, {"UBX-CFG-VALSET readback", Plan::CONFIG_TIMEOUT}).succeeded();
167 }
168 return true;
169}

187{
188 // Firmware with CFG-SEC-JAMDET has detection always on and no CFG-ITFM, so a rejection asks for the older monitor.
189 // An unanswered or refused probe leaves the receiver state unknown.

309bool Protocol::_waitForSurveyStop(GPSCommandChannel& channel)
310{
311 _surveyStopped = false;
312 const uint64_t deadline = GPSDeadline::after(channel.nowUs(), Plan::SURVEY_STOP_TIMEOUT).untilUs;
313 while (!_surveyStopped && channel.nowUs() < deadline) {
314 if (!_send(channel, Msg::NAV_SVIN, {}, {"UBX-NAV-SVIN stopped", Plan::CONFIG_TIMEOUT})) {
315 return false;
316 }
317 (void) channel.receiveUntil([this] { return _surveyStopped; }, Plan::SURVEY_STOP_POLL);
318 if (channel.failed()) {
319 return false;
320 }
321 }
322 if (!_surveyStopped) {
323 qCWarning(UBXProtocolLog) << "Time mode did not stop";
324 channel.finishCommand(GPSCommandOutcome::TimedOut);
325 return false;
326 }
327 channel.finishCommand(GPSCommandOutcome::ReadbackVerified);
328 return true;

340Protocol::RTCMActivation Protocol::_activateLegacyRTCMOutput(GPSCommandChannel& channel)
341{
342 auto& progress = _legacyRTCM;
343 const auto rate = Wire::encode(Plan::LEGACY_BASE_RATE);
344 if (!_send(channel, Msg::CFG_RATE, rate, {{}, Plan::LEGACY_REPLY_TIMEOUT})) {
345 progress = {};
346 return RTCMActivation::Failed;
347 }
348 // A CFG-MSG acknowledgement does not name the message, so replies are attributed by their order: the base-rate
349 // reply names CFG-RATE and follows any late reply to an earlier activation, and each later command waits for its
350 // own. A rate left unanswered is polled; once a poll expires with a reply still to come, later replies cannot be
351 // attributed, so no more rates are written and activation depends on the outputs confirmed so far. An unanswered
352 // base rate leaves activation to a later service; a rejected one leaves RTCM at the survey rate.
353 const auto baseRate = _waitForAck(channel, Msg::CFG_RATE).outcome;
354 if (baseRate == GPSCommandOutcome::TimedOut) {
355 return RTCMActivation::Unanswered;
356 }
357 if (baseRate != GPSCommandOutcome::Acknowledged && baseRate != GPSCommandOutcome::Rejected) {
358 progress = {};
359 return RTCMActivation::Failed;
360 }
361 // The base status rates come first; their rejection is ignored, as that of an optional RTCM message.
362 const auto status = Plan::legacyBaseStatus();
363 const auto rtcm = Plan::legacyRTCMOutput(_base.compactObservations);
364 bool unanswered = false;
365 for (; progress.next < status.size() + rtcm.size(); ++progress.next) {
366 // Send a rate only with time for its poll; a late reply would be credited to the next command.
367 if (channel.operationDeadline().remaining(channel.nowUs()) <
368 Plan::CONFIG_TIMEOUT + Plan::LEGACY_REPLY_TIMEOUT) {
369 qCDebug(UBXProtocolLog) << "Service budget spent; RTCM activation resumes in the next service";
370 return RTCMActivation::Unanswered;
371 }
372 const auto output = progress.next < status.size()
373 ? Plan::RTCMOutput{status[progress.next], Plan::RTCMContent::Optional}
374 : rtcm[progress.next - status.size()];
375 const auto reply = _confirmMessageRate(channel, output.rate);
376 if (reply.outcome == GPSCommandOutcome::TimedOut) {
377 unanswered = true;
378 break;
379 }
380 if (reply.outcome != GPSCommandOutcome::Acknowledged && reply.outcome != GPSCommandOutcome::ReadbackVerified &&
381 reply.outcome != GPSCommandOutcome::Rejected) {
382 progress = {};
383 return RTCMActivation::Failed;
384 }
385 if (reply.outcome != GPSCommandOutcome::Rejected) {
386 progress.stationPosition = progress.stationPosition || output.content == Plan::RTCMContent::StationPosition;
387 progress.observations = progress.observations || output.content == Plan::RTCMContent::Observations;
388 }
389 if (!reply.settled) {
390 ++progress.next;
391 unanswered = true;
392 break;
393 }
394 }
395 const bool active = progress.stationPosition && progress.observations;
396 if (unanswered) {
397 qCWarning(UBXProtocolLog) << "Receiver did not answer a CFG-MSG rate or its poll; later RTCM messages are not"
398 " configured";
399 if (!active) {
400 return RTCMActivation::Unanswered;
401 }
402 } else if (!active) {
403 qCWarning(UBXProtocolLog) << "Receiver rejected the RTCM station position or every observation message";
404 }
405 progress = {};
406 return active ? RTCMActivation::Active : RTCMActivation::Failed;
407}

409void Protocol::serviceStreaming(GPSCommandChannel& channel)
410{
411 auto& requests = _requests;
412 if (requests.rtcmActivation) {
413 const auto activation = _activateRTCMOutput(channel);
414 // A slow receiver keeps its session and finished survey: the next service resumes activation. Survey reports
415 // decoded meanwhile do not request another.
416 requests.rtcmActivation = activation == RTCMActivation::Unanswered;
417 if (activation == RTCMActivation::Failed) {
418 channel.failControl(QStringLiteral("u-blox receiver did not start RTCM output after the survey-in"));
419 }
420 }
421 if (const uint16_t message = std::exchange(requests.disableMessage, 0)) {
422 _disableUnexpectedMessage(channel, message);
423 }

426void Protocol::_disableUnexpectedMessage(GPSCommandChannel& channel, uint16_t message)
427{
428 const uint64_t now = channel.nowUs();
429 if (now <= _lastDisableUs + static_cast<uint64_t>(Plan::DISABLE_MESSAGE_INTERVAL.count())) {
430 return;
431 }
432 if (!_identity.protocol27) {
433 _lastDisableUs = now;
434 (void) _setMessageRate(channel,
435 {{static_cast<uint8_t>(message), static_cast<uint8_t>(message >> 8)}, Plan::NO_OUTPUT});
436 return;
437 }
438 const auto output = std::ranges::find(Plan::UNEXPECTED_OUTPUT, message,
439 [](const Plan::UnexpectedOutput& entry) { return entry.message.value(); });
440 if (output != Plan::UNEXPECTED_OUTPUT.end() && _ready) {
441 _lastDisableUs = now;
442 (void) _writeValset(channel, Plan::disableMessage(output->key));
443 }
444}

446bool Protocol::_send(GPSCommandChannel& channel, UBX::MessageId message, std::span<const uint8_t> payload,
447 GPSConfigurationStep step)
448{
449 if (step.command.isEmpty()) {
450 step.command = UBX::messageName(message);
451 }
452 channel.beginCommand(std::move(step));
453 // Identity replies can take two seconds, but writing the frame keeps the shorter configuration cap.
454 const auto scope = channel.deadlineScope(Plan::CONFIG_TIMEOUT);
455 std::array<uint8_t, UBX::FRAME_OVERHEAD + Plan::VALSET_CAPACITY> buffer{};
456 const auto frame = UBX::encodeFrame(message, payload, buffer);
457 if (frame.empty()) {
458 channel.finishCommand(GPSCommandOutcome::TransportError);
459 return false;
460 }
461 return channel.write(frame);
462}

472GPSConfigurationEvidence Protocol::_waitForAck(GPSCommandChannel& channel, UBX::MessageId message)
473{
474 auto& controller = _controller;
475 const bool valset = message.value() == Msg::CFG_VALSET.value();
476 controller.beginAcknowledgement(message.value());
477 if (valset && controller.configurationReadbackRequired()) {
478 controller.finishAcknowledgement();
479 const GPSDeadline commandDeadline = channel.commandDeadline();
480 const auto scope = channel.deadlineScope(commandDeadline);
481 const auto& command = channel.currentCommand();
482 return _verifyValset(channel,
483 {"UBX-CFG-VALSET readback", commandDeadline.remaining(channel.nowUs()), command.required});
484 }
485 const auto result = channel.awaitReply([&controller] { return controller.acknowledgement(); });
486 controller.finishAcknowledgement();
487 if (valset && result.outcome == GPSCommandOutcome::TimedOut) {
488 _valsetAckAmbiguous = true;
489 }
490 return result;
491}

512Protocol::RateConfirmation Protocol::_confirmMessageRate(GPSCommandChannel& channel, UBX::Plan::MessageRate rate)
513{
514 const auto outcome = _setMessageRateReply(channel, rate);
515 if (outcome != GPSCommandOutcome::TimedOut) {
516 return RateConfirmation{outcome, true};
517 }
518 auto& controller = _controller;
519 const std::array<uint8_t, 2> poll{rate.message.cls, rate.message.id};
520 controller.beginRatePoll(rate.message.value());
521 if (!_send(channel, Msg::CFG_MSG, poll,
522 {"UBX-CFG-MSG " + UBX::messageName(rate.message) + " readback", Plan::LEGACY_REPLY_TIMEOUT})) {
523 controller.finishRatePoll();
524 return RateConfirmation{GPSCommandOutcome::TransportError, false};
525 }
526 // The 3-byte CFG-MSG sets the rate of the port it arrives on, which is UART1 or USB.
527 const auto decide = [&controller, expected = rate.rate] {
528 if (const auto rates = controller.polledRates()) {
529 return (*rates)[Plan::LEGACY_PORT_UART1] == expected || (*rates)[Plan::LEGACY_PORT_USB] == expected
530 ? GPSCommandOutcome::ReadbackVerified
531 : GPSCommandOutcome::Rejected;
532 }
533 return controller.ratePollRejected() ? GPSCommandOutcome::Rejected : GPSCommandOutcome::TimedOut;
534 };
535 const auto result = channel.awaitReply([&channel, &controller, decide] {
536 const bool expired = channel.nowUs() >= channel.commandDeadline().untilUs;
537 return controller.ratePollSettled() || expired ? decide() : GPSCommandOutcome::Pending;
538 });
539 const bool settled = controller.ratePollSettled();
540 controller.finishRatePoll();
541 return RateConfirmation{result.outcome, settled};
542}

606
607GPSConfigurationEvidence Protocol::_awaitValsetAck(GPSCommandChannel& channel)
608{
609 auto result = _waitForAck(channel, Msg::CFG_VALSET);
610 if (result.outcome != GPSCommandOutcome::TimedOut || result.required ||
611 _controller.configurationReadbackRequired()) {
612 return result;
613 }
614 // A VALSET acknowledgement does not say which VALSET it answers, so a late one would be taken for the next. The
615 // receiver answers in order: the readback reply follows any late reply to this batch, so a late ACK and matching
616 // values accept it, a late NAK, other values or a NAK of the readback reject it, and either way nothing of it is
617 // left to arrive. Required batches still fail on the timeout.
618 result = _verifyValset(channel, {"UBX-CFG-VALSET readback", Plan::CONFIG_TIMEOUT, false});
619 if (result.outcome == GPSCommandOutcome::ReadbackVerified || result.outcome == GPSCommandOutcome::Rejected) {
620 _valsetAckAmbiguous = false;
621 }
622 return result;
623}

625GPSConfigurationEvidence Protocol::_verifyValset(GPSCommandChannel& channel, GPSConfigurationStep step)
626{
627 auto& controller = _controller;
628 const auto scope = channel.deadlineScope(step.timeout);
629 UBX::ConfigurationValueCursor cursor(_valset.entries());
630 GPSConfigurationEvidence result;
631 while (!cursor.empty()) {
632 UBX::ConfigurationValues expected;
633 std::array<uint32_t, UBX::ConfigurationValues::MAX_KEYS> keys{};
634 std::array<uint8_t, 4 + UBX::ConfigurationValues::MAX_KEYS * sizeof(uint32_t)> request{};
635 while (!cursor.empty() && expected.count < expected.values.size()) {
636 const auto entry = cursor.next();
637 if (!entry) {
638 channel.beginCommand(step);
639 return channel.completeCommand(GPSCommandOutcome::Rejected);
640 }
641 (void) LittleEndian::write(request, 4 + expected.count * sizeof(uint32_t), entry->key);
642 keys[expected.count] = entry->key;
643 expected.values[expected.count++] = *entry;
644 }
645 controller.beginReadback(std::span(keys).first(expected.count));
646 if (!_send(channel, Msg::CFG_VALGET, std::span(request).first(4 + expected.count * sizeof(uint32_t)), step)) {
647 controller.finishReadback();
648 return channel.completeCommand(GPSCommandOutcome::TransportError);
649 }
650 result = channel.awaitReply([&controller, &expected] {
651 if (controller.lateRejection() || controller.readbackRejected()) {
652 return GPSCommandOutcome::Rejected;
653 }
654 if (!controller.readbackReady()) {
655 return GPSCommandOutcome::Pending;
656 }
657 return controller.readback().values == expected.values ? GPSCommandOutcome::ReadbackVerified
658 : GPSCommandOutcome::Rejected;
659 });
660 controller.finishReadback();
661 if (result.outcome != GPSCommandOutcome::ReadbackVerified) {
662 return result;
663 }
664 }
665 return result;
666}
