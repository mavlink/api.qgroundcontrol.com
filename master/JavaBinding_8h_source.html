67 Returned call(JNIEnv* env, jclass javaClass, Args... args) const
68 {
69 if constexpr (std::is_same_v<Ret, jboolean>) {
70 return env->CallStaticBooleanMethod(javaClass, id, args...);
71 } else if constexpr (std::is_same_v<Ret, jlong>) {
72 return env->CallStaticLongMethod(javaClass, id, args...);
73 } else if constexpr (std::is_same_v<Ret, jint>) {
74 return env->CallStaticIntMethod(javaClass, id, args...);
75 } else {
76 return static_cast<Returned>(env->CallStaticObjectMethod(javaClass, id, args...));
77 }
78 }
