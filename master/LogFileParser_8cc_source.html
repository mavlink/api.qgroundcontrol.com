21LogParseResult _parseFile(const QString &filePath)
22{
23 const QString suffix = QFileInfo(filePath).suffix().toLower();
24
25 if (suffix == QStringLiteral("bin") || suffix == QStringLiteral("log")) {
26 return DataFlashParser::parseFile(filePath);
27 }
28
29 if (suffix == QStringLiteral("ulg")) {
30 return ULogParser::parseFile(filePath);
31 }
32
33 const QString fileTypeDescription = suffix.isEmpty()
34 ? LogFileParser::tr("no extension")
35 : QStringLiteral(".%1").arg(suffix);
36
37 LogParseResult result;
38 result.errorMessage = LogFileParser::tr(
39 "Unsupported file type (%1) for file '%2'. Expected .bin, .log, or .ulg.")
40 .arg(fileTypeDescription, filePath);
41 return result;
42}

205QVariantList LogFileParser::fieldSamplesFiltered(const QString &fieldName, double minX, double maxX, int pixelWidth) const
206{
207 QVariantList output;
208 const auto it = _fieldSamples.constFind(fieldName);
209 if (it == _fieldSamples.cend() || pixelWidth <= 0 || maxX <= minX) { return output; }
210
211 const QVector<QPointF> &points = it.value();
212
213 // Find the slice within [minX, maxX]
214 const auto sliceBegin = std::lower_bound(points.cbegin(), points.cend(), minX,
215 [](const QPointF &p, double t) { return p.x() < t; });
216 const auto sliceEnd = std::upper_bound(sliceBegin, points.cend(), maxX,
217 [](double t, const QPointF &p) { return t < p.x(); });
218
219 const qsizetype sliceCount = std::distance(sliceBegin, sliceEnd);
220 if (sliceCount == 0) { return output; }
221
222 // If already sparse enough, return slice as-is
223 if (sliceCount <= 4 * pixelWidth) {
224 output.reserve(sliceCount);
225 for (auto jt = sliceBegin; jt != sliceEnd; ++jt) { output.append(*jt); }
226 return output;
227 }
228
229 // Screen-space min/max bucketing: one bucket per pixel column.
230 // For each column track first, min-y, max-y, last indices; flush in time order.
231 output.reserve(4 * pixelWidth);
232 const double range = maxX - minX;
233
234 auto columnOf = [&](double x) -> int {
235 return static_cast<int>((x - minX) / range * pixelWidth);
236 };
237
238 // Per-column state
239 int curCol = -1;
240 qsizetype firstIdx = -1;
241 qsizetype minIdx = -1;
242 qsizetype maxIdx = -1;
243 qsizetype lastIdx = -1;
244
245 auto flush = [&]() {
246 if (firstIdx < 0) return;
247 // Collect the up-to-4 representative indices in time order, deduplicated
248 qsizetype indices[4] = { firstIdx, minIdx, maxIdx, lastIdx };
249 std::sort(indices, indices + 4);
250 qsizetype prev = -1;
251 for (qsizetype idx : indices) {
252 if (idx != prev) {
253 output.append(*(sliceBegin + idx));
254 prev = idx;
255 }
256 }
257 };
258
259 for (qsizetype i = 0; i < sliceCount; ++i) {
260 const QPointF &p = *(sliceBegin + i);
261 const int col = std::clamp(columnOf(p.x()), 0, pixelWidth - 1);
262 if (col != curCol) {
263 flush();
264 curCol = col;
265 firstIdx = i;
266 minIdx = i;
267 maxIdx = i;
268 } else {
269 if (p.y() < (sliceBegin + minIdx)->y()) minIdx = i;
270 if (p.y() > (sliceBegin + maxIdx)->y()) maxIdx = i;
271 }
272 lastIdx = i;
273 }
274 flush();
275
276 return output;
277}

279double LogFileParser::fieldValueAt(const QString &fieldName, double timestampSeconds) const
280{
281 const auto it = _fieldSamples.constFind(fieldName);
282 if (it == _fieldSamples.cend() || it->isEmpty()) {
283 return std::numeric_limits<double>::quiet_NaN();
284 }
285 const QVector<QPointF> &points = it.value();
286 const auto lower = std::lower_bound(points.cbegin(), points.cend(), timestampSeconds,
287 [](const QPointF &p, double t) { return p.x() < t; });
288
289 if (lower == points.cbegin()) { return lower->y(); }
290 if (lower == points.cend()) { return points.constLast().y(); }
291
292 const auto prev = std::prev(lower);
293 return (std::fabs(prev->x() - timestampSeconds) <= std::fabs(lower->x() - timestampSeconds))
294 ? prev->y() : lower->y();
295}

340QVariantMap LogFileParser::gpsCoordAt(double timestampSeconds) const
341{
342 if (_gpsLatField.isEmpty() || _gpsLonField.isEmpty()) {
343 return {};
344 }
345
346 const auto latIt = _fieldSamples.constFind(_gpsLatField);
347 const auto lonIt = _fieldSamples.constFind(_gpsLonField);
348 if (latIt == _fieldSamples.cend() || lonIt == _fieldSamples.cend()) {
349 return {};
350 }
351
352 const QVector<QPointF> &latPts = latIt.value();
353 const QVector<QPointF> &lonPts = lonIt.value();
354 if (latPts.isEmpty() || lonPts.isEmpty()) {
355 return {};
356 }
357
358 // Binary search for the sample with timestamp closest to timestampSeconds.
359 int lo = 0;
360 int hi = latPts.size() - 1;
361 while (lo < hi) {
362 const int mid = (lo + hi) / 2;
363 if (latPts[mid].x() < timestampSeconds) {
364 lo = mid + 1;
365 } else {
366 hi = mid;
367 }
368 }
369 // lo is the first index >= timestampSeconds; compare with lo-1.
370 if (lo > 0) {
371 const double dPrev = timestampSeconds - latPts[lo - 1].x();
372 const double dCurr = latPts[lo].x() - timestampSeconds;
373 if (dPrev < dCurr) {
374 --lo;
375 }
376 }
377
378 const int lonIdx = std::min(lo, static_cast<int>(lonPts.size()) - 1);
379 QVariantMap coord;
380 coord[QStringLiteral("latitude")] = latPts[lo].y();
381 coord[QStringLiteral("longitude")] = lonPts[lonIdx].y();
382 return coord;
383}

385QVariantList LogFileParser::gpsPath() const
386{
387 // Candidate field-name pairs tried in priority order.
388 // All values are stored in degrees (the parsers handle any raw-unit conversion).
389 // statusField: if non-null, that field's value must be >= statusMinValue to accept the sample.
390 // This is used for APM GPS messages where Status < 3 means no valid 3D fix.
391 struct CandidatePair {
392 const char *latField;
393 const char *lonField;
394 const char *altField;
395 const char *statusField;
396 double statusMinValue;
397 };
398
399 static const CandidatePair candidates[] = {
400 // PX4 ULog — vehicle_global_position (EKF-fused position, double degrees)
401 { "vehicle_global_position.lat", "vehicle_global_position.lon", "vehicle_global_position.alt", nullptr, 0 },
402 { "vehicle_global_position.latitude_deg", "vehicle_global_position.longitude_deg", "vehicle_global_position.alt", nullptr, 0 },
403 // PX4 ULog — vehicle_gps_position / sensor_gps (newer firmware uses latitude_deg)
404 { "vehicle_gps_position.latitude_deg", "vehicle_gps_position.longitude_deg", "vehicle_gps_position.altitude_msl_m", nullptr, 0 },
405 { "vehicle_gps_position[0].latitude_deg", "vehicle_gps_position[0].longitude_deg", "vehicle_gps_position[0].altitude_msl_m", nullptr, 0 },
406 { "sensor_gps.latitude_deg", "sensor_gps.longitude_deg", "sensor_gps.altitude_msl_m", nullptr, 0 },
407 { "sensor_gps[0].latitude_deg", "sensor_gps[0].longitude_deg", "sensor_gps[0].altitude_msl_m", nullptr, 0 },
408 // APM DataFlash — GPS message (Status >= 3 = 3D fix; 'L' type already divided by 1e7)
409 { "GPS.Lat", "GPS.Lng", "GPS.Alt", "GPS.Status", 3 },
410 { "GPS2.Lat", "GPS2.Lng", "GPS2.Alt", "GPS2.Status", 3 },
411 // APM DataFlash — POS message (EKF-fused; no status field, 'L' type already divided by 1e7)
412 { "POS.Lat", "POS.Lng", "POS.Alt", nullptr, 0 },
413 };
414
415 for (const auto &c : candidates) {
416 const auto latIt = _fieldSamples.constFind(QLatin1String(c.latField));
417 const auto lonIt = _fieldSamples.constFind(QLatin1String(c.lonField));
418 if (latIt == _fieldSamples.cend() || lonIt == _fieldSamples.cend()) {
419 continue;
420 }
421
422 const QVector<QPointF> &latPts = latIt.value();
423 const QVector<QPointF> &lonPts = lonIt.value();
424 if (latPts.isEmpty() || lonPts.isEmpty()) {
425 continue;
426 }
427
428 // Resolve optional status field (same message, same sample count as lat/lon).
429 const QVector<QPointF> *statusPts = nullptr;
430 if (c.statusField) {
431 const auto statusIt = _fieldSamples.constFind(QLatin1String(c.statusField));
432 if (statusIt != _fieldSamples.cend() && !statusIt.value().isEmpty()) {
433 statusPts = &statusIt.value();
434 }
435 }
436
437 qCDebug(LogFileParserLog) << "gpsPath: found candidate" << c.latField
438 << "samples:" << latPts.size()
439 << "first lat:" << latPts.first().y()
440 << "first lon:" << lonPts.first().y();
441
442 QVariantList path;
443 const int n = std::min(latPts.size(), lonPts.size());
444 path.reserve(n);
445
446 for (int i = 0; i < n; i++) {
447 // Skip samples that don't have a valid GPS fix.
448 if (statusPts && i < statusPts->size() && (*statusPts)[i].y() < c.statusMinValue) {
449 continue;
450 }
451
452 const double lat = latPts[i].y();
453 const double lon = lonPts[i].y();
454
455 if (lat < -90.0 || lat > 90.0 || lon < -180.0 || lon > 180.0
456 || (qFuzzyIsNull(lat) && qFuzzyIsNull(lon))) {
457 continue;
458 }
459 QVariantMap coord;
460 coord[QStringLiteral("latitude")] = lat;
461 coord[QStringLiteral("longitude")] = lon;
462 path.append(coord);
463 }
464
465 qCDebug(LogFileParserLog) << "gpsPath: valid points after filter:" << path.size();
466
467 if (!path.isEmpty()) {
468 _gpsLatField = QLatin1String(c.latField);
469 _gpsLonField = QLatin1String(c.lonField);
470 // Only cache the alt field if it actually exists and has samples;
471 // otherwise the altitude chart would be shown with no data.
472 const QLatin1String altField(c.altField);
473 const auto altIt = _fieldSamples.constFind(altField);
474 _gpsAltField = (altIt != _fieldSamples.cend() && !altIt.value().isEmpty()) ? altField : QLatin1String{};
475 return path;
476 }
477
478 qCDebug(LogFileParserLog) << "gpsPath: all" << n << "points filtered out for candidate" << c.latField;
479 }
480
481 qCDebug(LogFileParserLog) << "gpsPath: no GPS data found; available fields containing 'lat' or 'lon':";
482 for (auto it = _fieldSamples.cbegin(); it != _fieldSamples.cend(); ++it) {
483 const QString &fn = it.key();
484 if (fn.contains(QLatin1String("lat"), Qt::CaseInsensitive) || fn.contains(QLatin1String("lon"), Qt::CaseInsensitive)) {
485 qCDebug(LogFileParserLog) << " " << fn << "samples:" << it.value().size()
486 << (it.value().isEmpty() ? 0.0 : it.value().first().y());
487 }
488 }
489 return {};
490}
