48FixedPositionWire fixedPositionWire(const GPSEllipsoidPosition& position)
49{
50 const auto latitude = static_cast<int64_t>(position.latitudeDegrees * 1e9);
51 const auto longitude = static_cast<int64_t>(position.longitudeDegrees * 1e9);

213std::array<uint8_t, 2 * WIRE_SIZE<CfgPrt>> legacyPorts(uint32_t baudrate)
214{
215 std::array<uint8_t, 2 * WIRE_SIZE<CfgPrt>> payload{};
216 auto next = payload.begin();
217 for (const uint16_t id : {LEGACY_PORT_UART1, LEGACY_PORT_USB}) {
218 const uint16_t nmea = id == LEGACY_PORT_USB ? LEGACY_PROTOCOL_NMEA : 0;
219 const auto port = Wire::encode(
220 CfgPrt{.portID = static_cast<uint8_t>(id),
221 .mode = LEGACY_PORT_MODE_8N1,
222 .baudRate = baudrate,
223 .inProtoMask = LEGACY_PROTOCOL_UBX,
224 .outProtoMask = static_cast<uint16_t>(LEGACY_PROTOCOL_UBX | nmea | LEGACY_PROTOCOL_RTCM3)});
225 next = std::copy(port.begin(), port.end(), next);
226 }
227 return payload;
228}
