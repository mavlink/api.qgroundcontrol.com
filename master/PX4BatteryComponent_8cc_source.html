30bool PX4BatteryComponent::setupComplete() const
31{
32 ParameterManager* const parameterManager = _vehicle->parameterManager();
33 const int componentId = ParameterManager::defaultComponentId;
34 if (!parameterManager->parameterExists(componentId, "BAT1_SOURCE") ||
35 !parameterManager->parameterExists(componentId, "BAT1_V_CHARGED") ||
36 !parameterManager->parameterExists(componentId, "BAT1_V_EMPTY") ||
37 !parameterManager->parameterExists(componentId, "BAT1_N_CELLS")) {
38 return true;
39 }
40
41 // Cells and voltages are only used, and only shown on the page, when PX4 estimates the battery itself.
42 // Keep in sync with PX4BatteryRows.batteryInputs().
43 const int source = parameterManager->getParameter(componentId, "BAT1_SOURCE")->rawValue().toInt();
44 const bool estimated = source == 0 || source == 2 || _canFiltered();
45 if (!estimated) {
46 return true;
47 }
48
49 return parameterManager->getParameter(componentId, "BAT1_V_CHARGED")->rawValue().toFloat() != 0.0f &&
50 parameterManager->getParameter(componentId, "BAT1_V_EMPTY")->rawValue().toFloat() != 0.0f &&
51 parameterManager->getParameter(componentId, "BAT1_N_CELLS")->rawValue().toInt() != 0;
52}

76void PX4BatteryComponent::setupTriggerSignals()
77{
78 VehicleComponent::setupTriggerSignals();
79
80 // PX4 only streams the parameters of batteries it uses, and the rest can arrive later
81 (void) connect(_vehicle->parameterManager(), &ParameterManager::factAdded, this,
82 [this](int componentId, Fact* fact) {
83 if (componentId != _vehicle->defaultComponentId() || !fact->name().startsWith("BAT")) {
84 return;
85 }
86
87 if (setupCompleteChangedTriggerList().contains(fact->name())) {
88 (void) connect(fact, &Fact::valueChanged, this, &PX4BatteryComponent::_triggerUpdated);
89 emit setupCompleteChanged();

97QList<VehicleSummaryItem> PX4BatteryComponent::summaryItems() const
98{
99 int batteryCount = 0;
100 while (_parameterFact(_batteryParam(batteryCount + 1, "SOURCE"))) {
101 batteryCount++;
102 }
103
104 // Only battery 1 is checked by setupComplete
105 const bool battery1Incomplete = !setupComplete();
106
107 // Battery 1 is always listed, the others unless they are disabled (BATn_SOURCE -1), which still leaves the
108 // DroneCAN battery estimate running
109 QList<int> shownBatteries;
110 for (int index = 1; index <= batteryCount; index++) {
111 if (index == 1 || _parameterFact(_batteryParam(index, "SOURCE"))->rawValue().toInt() != -1 ||
112 _canFiltered()) {
113 shownBatteries.append(index);
114 }
115 }
116
117 // A single battery needs no box of its own
118 const bool grouped = shownBatteries.size() > 1;
119
120 QList<VehicleSummaryItem> items;
121 for (const int index : shownBatteries) {
122 const QString group = grouped ? tr("Battery %1").arg(index) : QString();
123 const auto caption = [grouped](const QString& single, const QString& inGroup) {
124 return grouped ? inGroup : single;
125 };
126 const auto addValue = [this, &items, &group, index, battery1Incomplete](const char* suffix,
127 const QString& text, bool isCount) {
128 const Fact* const fact = _parameterFact(_batteryParam(index, suffix));
129 if (!fact) {
130 items.append({text, tr("N/A"), VehicleSummaryItem::Muted, group});
131 return;
132 }
133 const bool unset = fact->rawValue().toFloat() == 0.0f;
134 items.append({text, isCount ? fact->cookedValueString() : _valueWithUnits(fact),
135 (index == 1 && battery1Incomplete && unset) ? VehicleSummaryItem::Attention
136 : VehicleSummaryItem::Normal,
137 group});
138 };
139
140 Fact* const source = _parameterFact(_batteryParam(index, "SOURCE"));
141 items.append({caption(tr("Battery Source"), tr("Source")), source->enumStringValue(),
142 VehicleSummaryItem::Normal, group});
143 addValue("V_CHARGED", caption(tr("Battery Full"), tr("Full")), false);
144 addValue("V_EMPTY", caption(tr("Battery Empty"), tr("Empty")), false);
145 addValue("N_CELLS", caption(tr("Number of Cells"), tr("Number of Cells")), true);
146 }
147
148 return items;
149}
