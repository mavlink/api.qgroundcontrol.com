19int formatCharSize(char c)
20{
21 switch (c) {

32 case 'N':

63QVariant parseValue(const char* data, qint64 size, char formatChar)
64{
65 const int requiredSize = formatCharSize(formatChar);
66 if (requiredSize == 0) {
67 qCWarning(APMDataFlashUtilityLog) << "Unsupported DataFlash format character:" << formatChar;
68 return {};
69 }
70 if (!data || size < requiredSize) {
71 qCWarning(APMDataFlashUtilityLog) << "Missing or truncated DataFlash value for format:" << formatChar;
72 return {};
73 }
74
75 const std::span<const uint8_t> bytes{reinterpret_cast<const uint8_t*>(data),
76 static_cast<std::size_t>(requiredSize)};
77 switch (formatChar) {
78 case 'b':
79 return *LittleEndian::read<int8_t>(bytes);
80 case 'B':
81 case 'M':
82 return *LittleEndian::read<uint8_t>(bytes);
83 case 'h': {
84 const auto val = *LittleEndian::read<int16_t>(bytes);
85 return val;
86 }
87 case 'H': {
88 const auto val = *LittleEndian::read<uint16_t>(bytes);
89 return val;
90 }
91 case 'c': {
92 const auto val = *LittleEndian::read<int16_t>(bytes);
93 return val / 100.0;
94 }
95 case 'C': {
96 const auto val = *LittleEndian::read<uint16_t>(bytes);
97 return val / 100.0;
98 }
99 case 'i': {
100 const auto val = *LittleEndian::read<int32_t>(bytes);
101 return val;
102 }
103 case 'I': {
104 const auto val = *LittleEndian::read<uint32_t>(bytes);
105 return val;
106 }
107 case 'e': {
108 const auto val = *LittleEndian::read<int32_t>(bytes);
109 return val / 100.0;
110 }
111 case 'E': {
112 const auto val = *LittleEndian::read<uint32_t>(bytes);
113 return val / 100.0;
114 }
115 case 'L': {
116 const auto val = *LittleEndian::read<int32_t>(bytes);
117 return val / 1.0e7; // Latitude/longitude in degrees
118 }
119 case 'f': {
120 const auto val = *LittleEndian::read<float>(bytes);
121 return static_cast<double>(val);
122 }
123 case 'd': {
124 const auto val = *LittleEndian::read<double>(bytes);
125 return val;
126 }
127 case 'q': {
128 const auto val = *LittleEndian::read<int64_t>(bytes);
129 return static_cast<qlonglong>(val);
130 }
131 case 'Q': {
132 const auto val = *LittleEndian::read<uint64_t>(bytes);
133 return static_cast<qulonglong>(val);
134 }
135 case 'g': {
136 const auto bits = *LittleEndian::read<uint16_t>(bytes);
137 return static_cast<double>(halfToFloat(bits));
138 }
139 case 'n':
140 return QString::fromLatin1(data, qstrnlen(data, 4));
141 case 'N':
142 return QString::fromLatin1(data, qstrnlen(data, 16));
143 case 'Z':
144 return QString::fromLatin1(data, qstrnlen(data, 64));
145 case 'a':
146 // 64-byte array (32 int16 values) - return as raw bytes
147 return QByteArray(data, 64);
148 default:
149 return QVariant();
150 }
151}

153QMap<QString, QVariant> parseMessage(const char* data, qint64 size, const MessageFormat& fmt)
154{
155 if (!data || size < 0) {
156 qCWarning(APMDataFlashUtilityLog) << "Invalid DataFlash message payload";
157 return {};
158 }
159
160 QMap<QString, QVariant> result;
161 qint64 offset = 0;
162
163 for (qsizetype i = 0; i < fmt.format.size(); ++i) {
164 const char formatChar = fmt.format.at(i).toLatin1();
165
166 const int fieldSize = formatCharSize(formatChar);
167 const QVariant value = parseValue(data + offset, size - offset, formatChar);
168 if (!value.isValid()) {
169 return {};
170 }
171 if (i < fmt.columns.size()) {
172 result[fmt.columns.at(i)] = value;
173 }
174 offset += fieldSize;
175 }
176
177 return result;
178}

193qint64 findNextHeader(const char *data, qint64 size, qint64 offset)
194{
195 while (offset + 2 < size) {

201 }

209MessageFormat parseFmtPayload(const char *data)
210{
211 MessageFormat fmt;
212 fmt.type = static_cast<uint8_t>(data[0]);
213 fmt.length = static_cast<uint8_t>(data[1]);
214 fmt.name = QString::fromLatin1(data + 2, qstrnlen(data + 2, 4));
215 fmt.format = QString::fromLatin1(data + 6, qstrnlen(data + 6, 16));
216 const QString columnsStr = QString::fromLatin1(data + 22, qstrnlen(data + 22, 64));
217 fmt.columns = columnsStr.trimmed().split(',');
218 return fmt;
219}

221bool parseFmtMessages(const char *data, qint64 size, QMap<uint8_t, MessageFormat> &formats)
222{
223 formats.clear();
224
225 if (!isValidHeader(data, size)) {
226 return false;
227 }
228
229 qint64 pos = 0;
230
231 while (pos + 3 <= size) {
232 // Find next message header
233 if (static_cast<uint8_t>(data[pos]) != kHeaderByte1 ||
234 static_cast<uint8_t>(data[pos + 1]) != kHeaderByte2) {
235 ++pos;
236 continue;
237 }
238
239 const uint8_t msgType = static_cast<uint8_t>(data[pos + 2]);
240 pos += 3;
241
242 if (msgType == kFmtMessageType) {
243 if (pos + kFmtPayloadSize > size) {
244 break;
245 }
246
247 const MessageFormat fmt = parseFmtPayload(data + pos);
248 formats[fmt.type] = fmt;
249 pos += kFmtPayloadSize;
250 } else {
251 // Skip message if we know its length
252 if (formats.contains(msgType)) {
253 pos += formats[msgType].length - 3; // -3 for header already consumed
254 } else {
255 // Unknown format, try to find next header
256 ++pos;
257 }
258 }
259 }
260
261 return !formats.isEmpty();
262}

269 const QMap<uint8_t, MessageFormat> &formats,
270 const MessageCallback &callback,

272{
273 int count = 0;
274 qint64 pos = 0;
275
276 while (pos + 3 <= size) {
277 // Find next message header
278 if (static_cast<uint8_t>(data[pos]) != kHeaderByte1 ||
279 static_cast<uint8_t>(data[pos + 1]) != kHeaderByte2) {
280 ++pos;
281 continue;
282 }
283
284 const uint8_t msgType = static_cast<uint8_t>(data[pos + 2]);
285 pos += 3;
286
287 if (!formats.contains(msgType)) {
288 continue;
289 }
290
291 const MessageFormat &fmt = formats[msgType];
292 const int payloadSize = fmt.length - 3;
293
294 if (pos + payloadSize > size) {
295 break;
296 }
297
298 ++count;
299 if (!callback(msgType, data + pos, payloadSize, fmt)) {
300 break;
301 }
302
303 pos += payloadSize;
304
305 if (progressCallback && (count % 1000 == 0)) {
306 progressCallback(static_cast<float>(pos) / static_cast<float>(size));
307 }
308 }
309
310 return count;
311}
