33// Vulkan handles snapshotted render-side by QGCRhiCapture::populateSnapshot(); primeLocked() runs on the bus-sync
34// streaming thread, so it must never dereference the cached QRhi* (races render-thread QRhi teardown). Desktop Linux
35// pins GL in Platform::initialize (Vulkan import is dormant), so on Linux this is the expected no-op path.
36struct SnapshotHandles
37{
38 VkPhysicalDevice physDev = VK_NULL_HANDLE;
39 quint32 queueFamilyIdx = 0;
40 quint32 queueIdx = 0;
41};
42
43bool readSnapshotHandles(SnapshotHandles& out)
44{
45 auto& snap = QGCRhiCapture::deviceSnapshot();
46 // backend is published last (release); acquire-loading it first guarantees the handle fields are visible.
47 if (snap.backend.load(std::memory_order_acquire) != static_cast<int>(QRhi::Vulkan)) {
48 return false;
49 }
50 out.physDev = static_cast<VkPhysicalDevice>(snap.vkPhysicalDevice.load(std::memory_order_acquire));
51 out.queueFamilyIdx = snap.vkQueueFamilyIdx.load(std::memory_order_acquire);
52 out.queueIdx = snap.vkQueueIdx.load(std::memory_order_acquire);
53 return out.physDev != VK_NULL_HANDLE;
54}

56// gst-vulkan 1.24 exposes no wrapped-instance/device constructor, so we cannot hand vulkanh26xdec QRhi's *own*
57// VkInstance/VkDevice. We build a gst instance, then bind a GstVulkanDevice to the GstVulkanPhysicalDevice matching
58// QRhi's physDev. That yields same-physical-device but a distinct VkDevice; the importer's device-match guard then
59// routes those (foreign-VkDevice) frames to CPU. True same-VkDevice zero-copy needs a wrap API not present here.
60using PhysicalDevicePtr = std::unique_ptr<GstVulkanPhysicalDevice, GStreamer::GstObjectDeleter>;
61
62PhysicalDevicePtr matchPhysicalDevice(GstVulkanInstance* instance, VkPhysicalDevice want)
63{
64 const guint n = instance->n_physical_devices;
65 for (guint i = 0; i < n; ++i) {
66 PhysicalDevicePtr phys{gst_vulkan_physical_device_new(instance, i)};
67 if (phys && (phys->device == want)) {
68 return phys;
69 }
70 }
71 return {};
72}

100bool VulkanContextBridge::primeLocked()
101{
102 SnapshotHandles nh;
103 if (!readSnapshotHandles(nh)) {
104 // Expected on the default GL RHI build; retry in case the Vulkan RHI initializes later.
105 if (!retryWithinBudget() && retriesJustExhausted()) {
106 qCInfo(GstVulkanBridgeLog) << "active RHI is not Vulkan after" << kMaxPrimeRetries
107 << "retries; Vulkan bridge inactive";
108 }
109 return false;
110 }
111
112 auto bail = [this](const char* what) -> bool {
113 qCWarning(GstVulkanBridgeLog) << "Vulkan bridge prime failed:" << what;
114 clearLocked();
115 return retryLater();
116 };
117
118 _instance = gst_vulkan_instance_new();
119 if (!_instance || !gst_vulkan_instance_open(_instance, nullptr)) {
120 return bail("gst_vulkan_instance_open");
121 }
122
123 const PhysicalDevicePtr phys = matchPhysicalDevice(_instance, nh.physDev);
124 if (!phys) {
125 return bail("no GstVulkanPhysicalDevice matches QRhi physDev");
126 }
127 _device = gst_vulkan_device_new(phys.get());
128 if (!_device || !gst_vulkan_device_open(_device, nullptr)) {
129 return bail("gst_vulkan_device_open");
130 }
131 _queue = gst_vulkan_device_get_queue(_device, nh.queueFamilyIdx, nh.queueIdx);
132 if (!_queue) {
133 return bail("gst_vulkan_device_get_queue");
134 }
135
136 // Distinct-VkDevice limitation is by design until a wrap API lands; flag it once so the CPU fallback isn't mistaken
137 // for a bug.
138 qCInfo(GstVulkanBridgeLog) << "Vulkan bridge primed on QRhi physical device — note: gst VkDevice differs from "
139 "QRhi VkDevice, so import falls back to CPU until same-device wrapping is available";
140 return true;
141}
