47{
48 auto casterConfig = config;
49 casterConfig.mountpoint.clear();
50 const bool sameCaster = casterConfig == _lastFetchConfig;
51 const QString error = config.validationError();
52
53 if (error.isEmpty() && _attempt && sameCaster) {
54 _sortCoord = sortCoord;
55 return;
56 }
57
58 _abortFetch();
59 if (!error.isEmpty()) {
60 _onFetchError(error);
61 return;
62 }
63
64 if (_model->count() > 0 && _cacheStoredAtUs && sameCaster) {
65 const auto nowUs = _scheduler->nowUs();
66 if (MonotonicClock::fresh(*_cacheStoredAtUs, nowUs, CACHE_TTL)) {
67 qCDebug(NTRIPSourceTableControllerLog) << "Source table cache hit, age:"
68 << std::chrono::duration_cast<std::chrono::milliseconds>(
69 *MonotonicClock::age(*_cacheStoredAtUs, nowUs));
70 _sortCoord = sortCoord;
71 _model->updateDistances(_sortCoord);
72 _setFetchState(FetchStatus::Success, QString());
73 return;
74 }

76
77 _cacheStoredAtUs.reset();

80 _lastFetchConfig = casterConfig;

95void NTRIPSourceTableController::_startFetch()
96{
97 _attempt = std::make_unique<FetchAttempt>(_scheduler, this);
98 auto* session = new NTRIPHttpSession(this, _scheduler, NTRIPHttpPurpose::SourceTable);
99 _attempt->session = session;
100 // _abortFetch() retires the session, which disconnects it, so these run only for the live attempt.
101 connect(session, &NTRIPHttpSession::certificatePinned, this, &NTRIPSourceTableController::certificatePinned);
102 connect(session, &NTRIPHttpSession::responseStarted, this, &NTRIPSourceTableController::_armFetchTimeout);
103 connect(session, &NTRIPHttpSession::bodyReceived, this, &NTRIPSourceTableController::_readReply);
104 connect(session, &NTRIPHttpSession::failed, this,
105 [this](const NTRIPFailure& failure) { _finishFetch(failure.detail); });
106 connect(session, &NTRIPHttpSession::finished, this,
107 [this]() { _finishFetch(tr("Response does not contain a valid source table")); });
108 _armFetchTimeout();
109 if (const QString openError = session->open(_lastFetchConfig); !openError.isEmpty()) {
110 _finishFetch(openError);
111 }
112}

114void NTRIPSourceTableController::_armFetchTimeout()
115{
116 // An inactivity timeout: a large table on a slow link keeps the fetch alive while bytes arrive.
117 _attempt->timeout.schedule(FETCH_TIMEOUT, [this]() {

125void NTRIPSourceTableController::_readReply(const QByteArray& bytes)
126{
127 _armFetchTimeout();
128 if (_attempt->body.size() + bytes.size() >= MAX_SOURCE_TABLE_BYTES) {
129 _finishFetch(tr("Source table too large (exceeds %1 MB)").arg(MAX_SOURCE_TABLE_BYTES / (1024 * 1024)));
130 return;
131 }
132 const qsizetype checkedSize = _attempt->body.size();
133 _attempt->body += bytes;
