192 duration = std::min(duration, std::chrono::microseconds(std::min<uint64_t>(remaining, INT64_MAX)));

233 int read(uint8_t* buf, int buf_length, int timeout)
234 {
235 if (_io_error) {
236 return _io_error;
237 }
238 if (!buf || buf_length <= 0) {
239 return 0;
240 }
241 GPSDeadline deadline{std::min(_operationDeadline.untilUs, nowUs() + uint64_t(std::max(timeout, 0)) * 1000)};
242 const auto result =
243 _io.read ? _io.read({buf, static_cast<size_t>(buf_length)}, deadline) : GPSReadResult{GPSReadStatus::Error};
244 _ioErrorDetail = result.detail;
245 if (result.status == GPSReadStatus::Data && result.bytesRead >= 0 && result.bytesRead <= buf_length) {
246 return result.bytesRead;
247 }
248 if (result.status == GPSReadStatus::TimedOut && result.bytesRead == 0) {
249 return 0;
250 }
251 _io_error = result.status == GPSReadStatus::Cancelled ? ReadCancelled : -EIO;
252 if (_io_error != ReadCancelled && _io.log) {
253 _io.log(GPSProtocolLogLevel::Warning, QStringLiteral("Receiver read failed (status %1, code %2): %3")
254 .arg(static_cast<int>(result.status))
255 .arg(_io_error)
256 .arg(_ioErrorDetail));
257 }
258 return _io_error;
259 }
260
267 int write(const void* buf, int buf_length)
268 {
269 if (_io_error) {
270 return _io_error;
271 }
272 if (!buf || buf_length < 0) {
273 _ioErrorDetail.clear();
274 return _io_error = -EINVAL;
275 }
276 const auto result = _io.write ? _io.write({static_cast<const uint8_t*>(buf), static_cast<size_t>(buf_length)},
277 _operationDeadline)
278 : GPSWriteResult{};
279 _ioErrorDetail = result.detail;
280 _commandWrite.evidence.acceptedBytes += result.acceptedBytes;
281 _commandWrite.evidence.writtenBytes += result.writtenBytes;
282 _commandWrite.evidence.uncertainBytes += result.uncertainBytes();
283 if (result.status == GPSWriteStatus::Completed && result.acceptedBytes == buf_length &&
284 result.writtenBytes == buf_length && result.uncertainBytes() == 0) {
285 return result.writtenBytes;
286 }
287 if (result.status == GPSWriteStatus::Unsupported && result.acceptedBytes == 0 && result.writtenBytes == 0) {
288 failCommandWrite(GPSCommandOutcome::TransportError);
289 return -1;
290 }
291 _io_error = result.status == GPSWriteStatus::Cancelled ? ReadCancelled : -EIO;
292 failCommandWrite(result.status == GPSWriteStatus::Cancelled ? GPSCommandOutcome::Cancelled
293 : GPSCommandOutcome::TransportError);
294 return _io_error;
295 }
