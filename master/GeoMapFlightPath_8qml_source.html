18/// Vehicle flight-path trail for the GeoMap: a red 3D ribbon in the scene at
19/// true flight altitude, depth-tested so terrain occludes it. Width stays
20/// constant in screen pixels via vertex-shader expansion. Stays visible in 2D
21/// mode (crossfade3D off): the altitude offsets flatten with terrainScale so
22/// it renders as the classic flat trail line. Altitudes get the same
23/// DEM-vs-home-AMSL bias correction as GeoMapVehicleItem, applied at the
24/// anchor so the whole trail shifts with it.
25GeoMapItem {
26 id: root
27
28 property var vehicle
29 property real lineWidth: ScreenTools.defaultFontPixelHeight / 5
30 property color lineColor: "red"
31
32 altitudeMode: GeoMapItem.Absolute
33 crossfade3D: false
34 coordinate: {
35 if (!_geometry || !_geometry.anchorCoordinate.isValid) {
36 return QtPositioning.coordinate()
37 }
38 const anchor = _geometry.anchorCoordinate
39 return QtPositioning.coordinate(anchor.latitude, anchor.longitude, anchor.altitude + _homeTerrainBias)
40 }
41 visible: !!vehicle
42
43 // The geometry lives inside the 3D delegate (created/destroyed with the
44 // scene); this bridges its anchor out for the coordinate binding above
45 property var _geometry: null
46
47 readonly property var _camera: scene ? scene.camera : null
48
49 // Pixel-to-scene-unit factor at unit distance; the shader scales it by
50 // each vertex's own camera distance
51 readonly property real _screenFactor: _camera ? _camera.unitsPerPixelAtUnitDistance : 0
52
53 // DEM height minus vehicle-reported AMSL, sampled at home (same
54 // correction as GeoMapVehicleItem so trail and marker stay aligned)
55 property real _homeTerrainBias: 0
56
57 function _updateHomeTerrainBias() {
58 if (surfaceModel && vehicle && vehicle.homePosition.isValid && !isNaN(vehicle.homePosition.altitude)) {
59 _homeTerrainBias = surfaceModel.terrainHeightAt(vehicle.homePosition) - vehicle.homePosition.altitude
60 } else {
61 _homeTerrainBias = 0
62 }
63 }
64
65 function _reloadPath() {
66 if (_geometry) {
67 _geometry.setPath(vehicle ? vehicle.trajectoryPoints.list() : [])
68 }
69 }
70
71 onVehicleChanged: {
72 _updateHomeTerrainBias()
73 _reloadPath()
74 }
75 onSurfaceModelChanged: _updateHomeTerrainBias()
76
77 delegate3D: Component {
78 // z offsets are relative altitudes; flatten them with the terrain
79 // during the 2D<->3D transition (anchor z already tracks terrainScale
80 // through GeoMapItem)
81 Node {
82 scale: Qt.vector3d(1, 1, root.scene ? root.scene.terrainScale : 1)
83
84 Model {
85 geometry: FlightPathGeometry {
86 id: pathGeometry
87
88 scene: root.scene
89 }
90
91 materials: CustomMaterial {
92 shadingMode: CustomMaterial.Unshaded
93 cullMode: Material.NoCulling
94 vertexShader: "shaders/flightpath.vert"
95 fragmentShader: "shaders/flightpath.frag"
96
97 property real lineWidth: root.lineWidth
98 property real screenFactor: root._screenFactor
99 property color pathColor: root.lineColor
100 // Ramps in as the terrain flattens, effectively disabling
101 // depth testing in 2D mode (see flightpath.vert)
102 property real depthPull: root.scene ? 0.5 * (1.0 - root.scene.terrainScale) : 0.0
103 }
104
105 Component.onCompleted: {
106 root._geometry = pathGeometry
107 root._reloadPath()
108 }
109 Component.onDestruction: root._geometry = null
110 }
111
112 Connections {
113 target: root.vehicle ? root.vehicle.trajectoryPoints : null
114
115 function onPointAdded(coordinate) { pathGeometry.appendPoint(coordinate) }
116 function onUpdateLastPoint(coordinate) { pathGeometry.updateLastPoint(coordinate) }
117 function onPointsCleared() { pathGeometry.clearPath() }
118 }
119 }
120 }
121
122 Connections {
123 target: root.surfaceModel
124
125 function onTerrainHeightsChanged() {
126 root._updateHomeTerrainBias()
127 }
128 }
129
130 Connections {
131 target: root.vehicle

135 }
136 }
