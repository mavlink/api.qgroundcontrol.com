67GPSWriteResult GPSDeviceTransport::writeData(QByteArrayView bytes, QDeadlineTimer deadline)
68{
69 QIODevice* const io = device();
70 if (fatalError() || !io || io->bytesToWrite() != 0) {
71 return {GPSWriteStatus::Error, 0, 0, readFailure().detail};
72 }
73 qint64 drained = 0;
74 const QMetaObject::Connection counter = QObject::connect(
75 io, &QIODevice::bytesWritten, io, [&drained](qint64 count) { drained += count; }, Qt::DirectConnection);
76 const int length = static_cast<int>(bytes.size());
77 int accepted = 0;
78 GPSWriteStatus status = GPSWriteStatus::Completed;
79 while (accepted < length || io->bytesToWrite() > 0) {
80 if (isCancelled() || fatalError() || deadline.hasExpired()) {
81 break;
82 }
83 if (accepted < length && io->bytesToWrite() < WRITE_BUFFER_BYTES) {
84 const qint64 count =
85 io->write(bytes.data() + accepted,
86 (std::min) (qint64(length - accepted), WRITE_BUFFER_BYTES - io->bytesToWrite()));
87 if (count < 0) {
88 status = GPSWriteStatus::Error;
89 break;
90 }
91 accepted += static_cast<int>(count);
92 }
93 if (io->bytesToWrite() > 0) {
94 waitWritten(deadline);
95 }
96 }
97 QObject::disconnect(counter);
98 const int written = static_cast<int>(std::clamp(confirmedWritten(accepted, drained), qint64(0), qint64(accepted)));
99 // A write confirmed in full succeeded, even if the deadline, a stop or a link failure followed it.
100 if (written < length) {
101 if (isCancelled()) {
102 status = GPSWriteStatus::Cancelled;
103 } else if (fatalError()) {
104 status = GPSWriteStatus::Error;
105 } else if (deadline.hasExpired()) {
106 status = GPSWriteStatus::TimedOut;
107 }
108 }
109 const GPSWriteResult result{status, accepted, written,
110 status == GPSWriteStatus::Error ? readFailure().detail : QString()};
111 if (status != GPSWriteStatus::Completed && accepted > 0) {
112 retire();
113 }
114 return result;
115}

135{

141 loop.quit();
142 }
143 };
144 if (QIODevice* const io = device()) {

148 (void) QObject::connect(socket, &QAbstractSocket::errorOccurred, &loop, check);

150 }
151#ifndef QGC_NO_SERIAL_LINK
152 if (auto* const port = qobject_cast<QSerialPort*>(io)) {
153 (void) QObject::connect(port, &QSerialPort::errorOccurred, &loop, check);
154 }
155#endif
156 }

158 timeout.setSingleShot(true);
159 timeout.setTimerType(Qt::PreciseTimer);
160 if (!deadline.isForever()) {
161 timeout.start(
162 static_cast<int>((std::min) (deadline.remainingTime(), qint64((std::numeric_limits<int>::max)()))));
163 }
164 // cancel() runs this on the cancelling thread, so it only posts. Declared after the loop so it is destroyed
165 // first: the destructor waits for a running invocation, and the loop's destructor discards an unhandled post.
166 const GPSCancelCallback wake(
167 cancelToken(), [&loop]() { QMetaObject::invokeMethod(&loop, &QEventLoop::quit, Qt::QueuedConnection); });
168 loop.exec();
169 }
170 return !isCancelled() && done();
171}
