58VehicleReadiness::Result VehicleReadiness::_evaluate(const Inputs& inputs)
59{
60 Result result;
61
62 if (inputs.healthAndArmingChecksReported) {
63 const auto modeGroupSet = [](quint64 modeGroups, int modeGroup) {
64 if ((modeGroup < 0) || (modeGroup >= HealthAndArmingCheckReport::MaxModeGroups)) {
65 return false;
66 }
67 return (modeGroups & (1ull << modeGroup)) != 0;
68 };
69 const auto actionReady = [&inputs, &modeGroupSet](const ActionModeGroups& modeGroups) {
70 const int armModeGroup = (modeGroups.armModeGroup < 0) ? inputs.currentModeGroup : modeGroups.armModeGroup;
71 const int targetModeGroup =
72 (modeGroups.targetModeGroup < 0) ? inputs.currentModeGroup : modeGroups.targetModeGroup;
73
74 // An armed vehicle only switches into the target mode
75 if (inputs.armed) {
76 return modeGroupSet(inputs.canRunModeGroups, targetModeGroup);
77 }
78
79 // An action which switches mode after arming also needs to be able to run in the target mode. libevents
80 // only promises can_run is meaningful while armed, so this part is best effort.
81 const bool canArmModeGroup = modeGroupSet(inputs.canArmModeGroups, armModeGroup);
82 if (armModeGroup == targetModeGroup) {
83 return canArmModeGroup;
84 }
85 return canArmModeGroup && modeGroupSet(inputs.canRunModeGroups, targetModeGroup);
86 };
87
88 result.canArm = actionReady(ActionModeGroups{});
89 result.canTakeoff = actionReady(inputs.takeoff);
90 result.canStartMission = actionReady(inputs.startMission);
91 if (inputs.currentModeGroup < 0) {
92 // The current mode is not in the vehicle's metadata, so its readiness is unknown
93 result.status = Status::Unknown;
94 } else {
95 result.status = result.canArm ? Status::Ready : Status::NotReady;
96 }
97 } else if (!inputs.healthReportWaitExpired && inputs.healthAndArmingChecksSupported) {
98 result.status = Status::Unknown;
99 } else if (inputs.prearmCheckAvailable) {
100 result.canArm = inputs.prearmCheckPassed;
101 result.canTakeoff = inputs.prearmCheckPassed;
102 result.canStartMission = inputs.prearmCheckPassed;
103 result.status = inputs.prearmCheckPassed ? Status::Ready : Status::NotReady;
104 } else {
105 result.canArm = true;
106 result.canTakeoff = true;
107 result.canStartMission = true;
108 // Sensor health and setup are pre-flight hints, so they never mark an armed vehicle not ready
109 result.status =
110 (inputs.armed || (inputs.sensorsHealthy && inputs.setupComplete)) ? Status::Ready : Status::NotReady;
111 }
112
113 return result;
114}

116void VehicleReadiness::_applyLockouts(Result& result, const Inputs& inputs)
117{
118 // Vehicle state is incomplete until initial connect finishes, so nothing pre-flight is allowed yet
119 if (!inputs.initialConnectComplete) {
120 result = Result{.status = Status::Syncing};
121 return;
122 }
123
124 // The checklist is pre-flight, so it only blocks a disarmed vehicle
125 if (!inputs.armed && inputs.checklistBlocksArming) {
126 result.canArm = false;
127 result.canTakeoff = false;
128 result.canStartMission = false;
129 result.status = Status::NotReady;
130 }
131}

133VehicleReadiness::Inputs VehicleReadiness::_gatherInputs() const
134{
135 Inputs inputs;
136
137 inputs.armed = _vehicle->armed();
138 inputs.initialConnectComplete = _vehicle->isInitialConnectComplete();
139 inputs.healthReportWaitExpired = _healthReportWaitExpired;
140
141 AppSettings* const appSettings = SettingsManager::instance()->appSettings();
142 inputs.checklistBlocksArming = appSettings->useChecklist()->rawValue().toBool() &&
143 appSettings->enforceChecklist()->rawValue().toBool() &&
144 (_vehicle->checkListState() != Vehicle::CheckListPassed);
145
146 inputs.healthAndArmingChecksSupported = _vehicle->healthAndArmingChecksSupported(MAV_COMP_ID_AUTOPILOT1);
147 const HealthAndArmingCheckReport* const report = _vehicle->healthAndArmingCheckReport();
148 if (report && report->supported()) {
149 const HealthAndArmingCheckReport::ModeGroupResults& modeGroupResults = report->modeGroupResults();
150 inputs.healthAndArmingChecksReported = true;
151 inputs.currentModeGroup = modeGroupResults.currentModeGroup;
152 inputs.canArmModeGroups = modeGroupResults.canArm;
153 inputs.canRunModeGroups = modeGroupResults.canRun;
154 inputs.takeoff = _actionModeGroups(GuidedAction::Takeoff);
155 inputs.startMission = _actionModeGroups(GuidedAction::StartMission);
156 }
157
158 inputs.prearmCheckAvailable = _vehicle->readyToFlyAvailable();
159 inputs.prearmCheckPassed = _vehicle->readyToFly();
160
161 inputs.sensorsHealthy = _vehicle->allSensorsHealthy();
162 const AutoPilotPlugin* const autopilotPlugin = _vehicle->autopilotPlugin();
163 inputs.setupComplete = autopilotPlugin && autopilotPlugin->setupComplete();
164
165 return inputs;
166}

189void VehicleReadiness::update()
190{
191 const Inputs inputs = _gatherInputs();
192 qCDebug(VehicleReadinessLog) << "armed:" << inputs.armed
193 << "initialConnectComplete:" << inputs.initialConnectComplete
194 << "checklistBlocksArming:" << inputs.checklistBlocksArming
195 << "healthChecks supported/reported:" << inputs.healthAndArmingChecksSupported
196 << inputs.healthAndArmingChecksReported
197 << "currentModeGroup:" << inputs.currentModeGroup
198 << "takeoff arm/target:" << inputs.takeoff.armModeGroup
199 << inputs.takeoff.targetModeGroup
200 << "startMission arm/target:" << inputs.startMission.armModeGroup
201 << inputs.startMission.targetModeGroup << "canArm:" << Qt::hex
202 << inputs.canArmModeGroups << "canRun:" << inputs.canRunModeGroups << Qt::dec
203 << "prearm available/passed:" << inputs.prearmCheckAvailable
204 << inputs.prearmCheckPassed << "sensorsHealthy:" << inputs.sensorsHealthy
205 << "setupComplete:" << inputs.setupComplete;
206
207 const Result generic = _evaluate(inputs);
208 Result result{.canArm = _canArm(generic.canArm),
209 .canTakeoff = _canTakeoff(generic.canTakeoff),
210 .canStartMission = _canStartMission(generic.canStartMission),
211 .status = _status(generic.status)};
212 _applyLockouts(result, inputs);
213
214 if (result == _result) {
215 return;
216 }
217
218 _result = result;
219 qCDebug(VehicleReadinessLog) << "canArm:" << _result.canArm << "canTakeoff:" << _result.canTakeoff
220 << "canStartMission:" << _result.canStartMission << "status:" << _result.status;
221 emit readinessChanged();
222}
