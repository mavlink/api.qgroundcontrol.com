23 return left.size() == right.size() &&

37bool validChecksum(std::string_view line, bool crc32)
38{
39 const auto star = line.find('*');
40 const size_t digits = crc32 ? 8 : 2;
41 if (star == std::string_view::npos || star + digits + 1 != line.size()) {
42 return false;
43 }
44 for (const auto ch : line.substr(star + 1)) {
45 if (!((ch >= '0' && ch <= '9') || (ch >= 'a' && ch <= 'f') || (ch >= 'A' && ch <= 'F'))) {
46 return false;
47 }
48 }
49 const auto expected = NMEA::number<uint32_t>(line.substr(star + 1), 16);
50 if (!expected) {
51 return false;
52 }
53 uint32_t checksum = 0;
54 if (crc32) {
55 // Reflected CRC-32, initial value zero and no final XOR; '#' is excluded.
56 checksum = QGC::crc32Update({reinterpret_cast<const uint8_t*>(line.data() + 1), star - 1});
57 } else {
58 // MODE and command acknowledgments include their leading '#' / '$', unlike NMEA.
59 checksum = NMEA::checksum(line.substr(0, star));
60 }
61 return checksum == *expected;
62}

83bool GPSNativeUnicore::_execute(std::string command, Reply reply)
84{
85 _command = std::move(command);
86 _expectedReply = reply;
87 _replyOutcome = GPSCommandOutcome::Pending;
88 _configurationDetail.clear();
89 _commandActive = true;
90 const auto clearReply = qScopeGuard([this] { _commandActive = false; });
91 const GPSConfigurationStep step{_command, std::chrono::milliseconds(COMMAND_TIMEOUT_MS)};
92 const auto wire = _command + "\r\n";
93 if (!writeCommand(step, {reinterpret_cast<const uint8_t*>(wire.data()), wire.size()})) {
94 _configurationDetail =
95 QStringLiteral("Unicore command '%1' could not be written").arg(QString::fromStdString(_command));
96 return false;
97 }
98 const auto result = awaitCommand(step, [this] { return _replyOutcome; });
99 _commandActive = false;
100 if (result.evidence.outcome != GPSCommandOutcome::Acknowledged &&
101 result.evidence.outcome != GPSCommandOutcome::ReadbackVerified) {
102 if (result.evidence.outcome != GPSCommandOutcome::Cancelled) {
103 if (_configurationDetail.isEmpty()) {
104 const auto failure = result.evidence.outcome == GPSCommandOutcome::TimedOut
105 ? QStringLiteral("timed out")
106 : result.evidence.outcome == GPSCommandOutcome::Rejected
107 ? QStringLiteral("was rejected or its readback did not match")
108 : QStringLiteral("failed");
109 _configurationDetail =
110 QStringLiteral("Unicore command '%1' %2").arg(QString::fromStdString(_command), failure);
111 }
112 log(GPSProtocolLogLevel::Warning, "Unicore command failed (%d): %s",
113 static_cast<int>(result.evidence.outcome), _command.c_str());
114 }
115 return false;
116 }
117 return true;

217 if (!_averaging && !_execute("BESTNAVXYZA", Reply::FixedPosition)) {
218 return _configurationFailed();

221 return _configurationFailed();

297void GPSNativeUnicore::_handlePosition(std::string_view body)
298{
299 if (!_monitorBase) {
300 return;
301 }
302 std::array<std::string_view, 28> fields{};
303 if (NMEA::splitFields(body, fields) != fields.size()) {
304 return;
305 }
306 const auto x = NMEA::number<double>(fields[2]);
307 const auto y = NMEA::number<double>(fields[3]);
308 const auto z = NMEA::number<double>(fields[4]);
309 if (!x || !y || !z) {
310 return;
311 }
312 const EcefMeters coordinates{*x, *y, *z};
313 const auto samePosition = [](const EcefMeters& left, const EcefMeters& right) {
314 // ECEF command/readback values have four decimal places; this is not survey accuracy.
315 return std::hypot(left.x - right.x, left.y - right.y, left.z - right.z) < 0.02;
316 };
317 const auto radius = std::hypot(*x, *y, *z);
318 const bool fixed = fields[0] == "SOL_COMPUTED" && fields[1] == "FIXEDPOS" && radius > 6000000 && radius < 7000000;
319 const bool matches = _averaging || samePosition(coordinates, _fixedECEF);
320 if (_commandActive && _expectedReply == Reply::FixedPosition && fixed &&
321 _replyOutcome == GPSCommandOutcome::Pending) {
322 _replyOutcome = matches ? GPSCommandOutcome::ReadbackVerified : GPSCommandOutcome::Rejected;
323 }
324 if (_ready && _baseValid && (!fixed || !matches || !samePosition(coordinates, _baseECEF))) {
325 _invalidateBase();
326 return;
327 }
328 _lastBaseStatus = nowUs();
329 _baseValid = fixed && matches;
330 if (_baseValid) {
331 _baseECEF = coordinates;
332 }
333 setRTCMEnabled(_ready && _baseValid);
334 _publishBase(_baseValid, _averaging && !_baseValid);
335}

391int GPSNativeUnicore::handleReceiverLine(std::string_view line)
392{
393 if (line.starts_with("$command,")) {
394 if (!_commandActive || !validChecksum(line, false)) {
395 return 0;
396 }
397 const auto response = line.find(",response: ", 9);
398 if (response == std::string_view::npos || !equalCommand(line.substr(9, response - 9), _command)) {
399 return 0;
400 }
401 const auto status = line.substr(response + 11, line.find('*') - response - 11);
402 if (_replyOutcome == GPSCommandOutcome::Pending) {
403 if (status != "OK") {
404 _replyOutcome = GPSCommandOutcome::Rejected;
405 } else if (_expectedReply == Reply::Acknowledgment) {
406 _replyOutcome = GPSCommandOutcome::Acknowledged;
407 }
408 }
409 return GPSDecodedBatch::PROTOCOL_ACTIVITY;
410 }
411 if (!line.starts_with('#')) {
412 return 0;
413 }
414 const auto comma = line.find(',');
415 const auto name = line.substr(1, comma - 1);
416 if (name != "VERSIONA" && name != "MODE" && name != "BESTNAVXYZA") {
417 return 0;
418 }
419 if (!validChecksum(line, name != "MODE")) {
420 return 0;
421 }
422 const auto semicolon = line.find(';');
423 if (semicolon == std::string_view::npos || semicolon >= line.find('*')) {
424 return 0;
425 }
426 std::array<std::string_view, 10> header{};
427 if (NMEA::splitFields(line.substr(1, semicolon - 1), header) != header.size()) {
428 return 0;
429 }
430 const auto body = line.substr(semicolon + 1, line.find('*') - semicolon - 1);
431 if (name == "VERSIONA") {
432 _handleVersion(body);
433 } else if (name == "MODE") {
434 _handleMode(body);
435 } else {
436 if (!_monitorBase) {
437 return GPSDecodedBatch::PROTOCOL_ACTIVITY;
438 }
439 const auto week = NMEA::number<uint16_t>(header[4]);
440 const auto milliseconds = NMEA::number<uint32_t>(header[5]);
441 if (header[2] != "GPS" || header[3] != "FINE" || !week || !milliseconds || *milliseconds >= 604800000) {
442 return 0;
443 }
444 const uint64_t epoch = uint64_t(*week) * 604800000 + *milliseconds;
445 if (_lastBaseEpoch && epoch < *_lastBaseEpoch) {
446 _invalidateBase();
447 return GPSDecodedBatch::PROTOCOL_ACTIVITY;
448 }
449 if (_lastBaseEpoch && epoch == *_lastBaseEpoch) {
450 return GPSDecodedBatch::PROTOCOL_ACTIVITY;
451 }
452 _lastBaseEpoch = epoch;
453 _handlePosition(body);
454 }
455 return GPSDecodedBatch::PROTOCOL_ACTIVITY;
