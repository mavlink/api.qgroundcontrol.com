51bool Protocol::configure(GPSCommandChannel& channel, GPSConfig config, unsigned& baud)
52{
53 _ready = false;
54 _rtcmOutputActive = false;
55 _rtcmActivationPending = false;
56 _surveyClock = {};
57 // Corrections are only framed from a receiver that answered.
58 GPSStreamDemux& stream = channel.stream();
59 stream.setEnabled(GPSFrameKind::RTCM3, false);
60 _nmea.setRTCMEnabled(false);
61 _nmea.reset(stream);
62 _base = config.base;
63 // No command depends on the rate, so the link keeps the one selected or detected; otherwise it runs at Plan::BAUD.
64 baud = config.linkBaud(baud, Femto::Plan::BAUD);
65 (void) channel.setBaudrate(baud);
66 auto identity = channel.runSequence(Femto::Plan::sequence(Femto::Plan::IDENTIFY));
67 for (unsigned round = 1; round < Femto::Plan::IDENTIFY_ROUNDS && !identity.succeeded(); ++round) {
68 identity = channel.runSequence(Femto::Plan::sequence(Femto::Plan::IDENTIFY));
69 }
70 if (!identity.succeeded()) {
71 if (identity.outcome == GPSCommandOutcome::Rejected) {
72 channel.failSequence(identity);
73 } else {
74 channel.failNoAnswer({&baud, 1});
75 }
76 return false;
77 }
78
79 stream.reset(GPSFrameKind::NMEASentence);
80 stream.setEnabled(GPSFrameKind::RTCM3, true);
81 _nmea.setRTCMEnabled(true);
82 _setUpBase(channel);
83 _ready = !channel.failed();
84 return _ready;
85}

95void Protocol::_setUpBase(GPSCommandChannel& channel)
96{
97 GPSDecodeContext& context = channel.context();
98 const auto* fixed = std::get_if<GPSBaseStationConfig::Fixed>(&_base.mode);
99 if (!fixed) {
100 if (!channel.runRequired(Femto::Plan::sequence(Femto::Plan::SURVEY_IN))) {
101 return;
102 }
103 _surveyClock.start(context);
104 return;
105 }
106 if (!channel.runRequired(Femto::Plan::fixedBase(fixed->position))) {
107 return;
108 }
109 _activateRTCMOutput(channel);
110 if (_rtcmOutputActive) {
111 context.publishSurvey(false, true, {}, fixed->position);
112 }
113}

122GPSReceiveUpdates Protocol::_decodeLine(std::string_view line, GPSDecodeContext& context)
123{
124 if (line.size() >= 6 && line.substr(3, 3) == "GGA") {
125 const auto sentence = NMEA::sentence(line);
126 const auto fix = sentence ? NMEA::gga(*sentence) : std::nullopt;
127 if (!fix) {
128 return {};
129 }
130 // Only a survey started by this configuration may complete it; earlier GGA output is stale.
131 if (!_rtcmOutputActive && _surveyClock.running() && fix->quality == Femto::Plan::AVERAGED_POSITION_QUALITY) {
132 _surveyClock.finish(context, true,
133 {.latitudeDegrees = fix->latitude,
134 .longitudeDegrees = fix->longitude,
135 .altitudeMeters = fix->altitude + fix->geoidSeparation});
136 _rtcmActivationPending = true;
137 }
138 }
139 _surveyClock.publishProgress(context);
140 return {};
141}

143bool portName(std::string_view field)
144{
145 const size_t digits = field.find_first_of("0123456789");
146 if (digits == 0 || digits == std::string_view::npos) {
147 return false;
148 }
149 return std::ranges::all_of(field.substr(0, digits), NMEA::isAsciiUpper) &&
150 std::ranges::all_of(field.substr(digits), NMEA::isAsciiDigit);
151}

155QLatin1StringView signature(const GPSFrame& frame)
156{
157 const std::string_view text = frame.text();
158 if (frame.kind != GPSFrameKind::ASCIILine) {
159 return {};
160 }
161 if (text.starts_with('<') && (text.ends_with(" OK") || text.starts_with(Femto::Plan::REJECTED))) {
162 return QLatin1StringView("abbreviated ASCII replies");
163 }
164 const auto log = gpsASCIILog(text);
165 return log && portName(log->source) ? QLatin1StringView("NovAtel-format ASCII logs") : QLatin1StringView();
166}

169bool Protocol::probe(GPSCommandChannel& channel)
170{
171 bool identified = false;
172 for (unsigned round = 0; round < Femto::Plan::IDENTIFY_ROUNDS && !identified && !channel.failed(); ++round) {
173 identified = channel.runSequence(Femto::Plan::sequence({&Femto::Plan::VERSION, 1})).succeeded();
174 }
175 return identified;
176}
