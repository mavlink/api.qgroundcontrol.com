15/// Mission-item overlay for the GeoMap: instantiates the GeoMap-native visual
16/// declared by each item's VisualMissionItem.geoMapVisualQML (mirrors the
17/// existing mapVisualQML dispatch consumed by src/PlanView/MissionItemMapVisual.qml
18/// — every item type states its own visual, simple items included, rather
19/// than special-casing GeoMapWaypointItem outside the dispatch). Items with
20/// no GeoMap-native visual render nothing: the home item (already shown via
21/// its own GeoMapPin) and the Survey/Corridor/Structure complex items, which
22/// never appear in the fly-view controller (vehicle downloads reconstruct
23/// only simple items and landing patterns).
24/// Overlapping waypoint markers are grouped (MissionItemIndicatorGroup parity): a
25/// click reports the whole group in either mode, and settled top-down 2D also
26/// collapses each group into one badged marker.
27Item {
28 id: root

30 property var missionController
31 property var scene
32 property var surfaceModel
33 property real homeTerrainBias: 0 ///< See GeoMapWaypointItem.homeTerrainBias
34
35 /// A marker was clicked; item is its VisualMissionItem (mirrors
36 /// MissionItemMapVisual.clicked as consumed by PlanMapItems)
37 signal itemClicked(var item)
38
39 /// A marker overlapping others was clicked; items are the group's
40 /// VisualMissionItems by sequence number, marker the clicked GeoMapWaypointItem
41 signal groupClicked(var items, var marker)
42
43 /// Markers moved on screen (zoom, resize, 2D/3D), so an open group picker is stale.
44 /// Center moves don't count: vehicle follow recenters on every position update.
45 signal groupsInvalidated()

47 readonly property var _camera: scene ? scene.camera : null
48 // Top-down 2D only pans, rotates and zooms, so groups change only with the zoom
49 readonly property bool _collapse: _camera !== null && _camera.mode === GeoMapCamera.Mode2D && _camera.isTopDown
50 && scene.terrainScale === 0

52 // Re-evaluates when a marker moves or the current item changes
53 readonly property var _groupingState: {
54 const state = []
55 const items = root.missionController ? root.missionController.visualItems : null
56 for (let i = 0; items && i < items.count; ++i) {
57 const item = items.get(i)
58 if (item.isSimpleItem && item.specifiesCoordinate) {
59 state.push([item.coordinate, item.isCurrentItem])
60 }
61 }
62 return state
63 }

65 function _waypointMarkers() {
66 const markers = []
67 for (let i = 0; i < repeater.count; ++i) {
68 const loader = repeater.itemAt(i)
69 if (loader && loader.item && loader.item.hasOwnProperty("collapsed")) {
70 markers.push(loader.item)
71 }
72 }
73 return markers
74 }

76 // Groups of on-screen waypoint markers, each led by its representative: the
77 // current item, else the lowest sequence number
78 function _markerGroups() {
79 const markers = root._waypointMarkers().filter(marker => marker.hasMarker && marker.projected)
80 markers.sort((first, second) => {
81 const firstCurrent = first.item.isCurrentItem || first.item.hasCurrentChildItem
82 const secondCurrent = second.item.isCurrentItem || second.item.hasCurrentChildItem
83 if (firstCurrent !== secondCurrent) {
84 return firstCurrent ? -1 : 1
85 }
86 return first.item.sequenceNumber - second.item.sequenceNumber
87 })
88 if (markers.length === 0) {
89 return []
90 }
91 const points = markers.map(marker => marker.mapToItem(root, marker.anchorPoint.x, marker.anchorPoint.y))
92 // A small marker's radius, as MissionItemIndicatorGroup
93 const representatives = MapMarkerGrouping.representatives(points, markers[0].smallIndicatorSize / 2)
94 const groups = []
95 const groupsByRepresentative = {}
96 for (let i = 0; i < markers.length; ++i) {
97 const representative = representatives[i]
98 if (!groupsByRepresentative[representative]) {
99 groupsByRepresentative[representative] = []
100 groups.push(groupsByRepresentative[representative])
101 }
102 groupsByRepresentative[representative].push(markers[i])
103 }
104 return groups
105 }

107 function _regroup() {
108 const markers = root._waypointMarkers()
109 for (const marker of markers) {
110 marker.collapsed = false
111 marker.grouped = false
112 }
113 if (!root._collapse) {
114 return
115 }
116 for (const group of root._markerGroups()) {
117 if (group.length > 1) {
118 group[0].grouped = true
119 for (let i = 1; i < group.length; ++i) {
120 group[i].collapsed = true
121 }
122 }
123 }
124 }
125
126 function _scheduleRegroup() {
127 Qt.callLater(root._regroup)
128 }
129
130 function _markerClicked(marker) {
131 const group = root._markerGroups().find(members => members.includes(marker))
132 if (group && group.length > 1) {
133 const items = group.map(member => member.item)
134 items.sort((first, second) => first.sequenceNumber - second.sequenceNumber)
135 root.groupClicked(items, marker)
136 } else {
137 root.itemClicked(marker.item)
138 }
139 }

191 } else if (visual.clicked !== undefined) {
192 visual.clicked.connect(() => root.itemClicked(object))
