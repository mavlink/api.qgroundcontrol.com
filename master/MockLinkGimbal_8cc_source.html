36void MockLinkGimbal::run1HzTasks()
37{
38 // Runs every 1s (1Hz on worker thread). Reads intervals and gimbal state main thread modifies.
39 // Must serialize access to prevent:
40 // - Auto-movement overwriting manual control commands
41 // - Reading stale/inconsistent interval values for status transmission
42 QMutexLocker locker(&_stateMutex);
43 const qint64 nowMs = QDateTime::currentMSecsSinceEpoch();
44
45 // Send GIMBAL_MANAGER_STATUS if interval is set
46 if (_managerStatusIntervalUs > 0) {
47 const qint64 intervalMs = _managerStatusIntervalUs / 1000;
48 if ((nowMs - _managerStatusLastSentMs) >= intervalMs) {
49 _sendGimbalManagerStatus();
50 _managerStatusLastSentMs = nowMs;
51 }
52 }
53
54 // Send GIMBAL_DEVICE_ATTITUDE_STATUS if interval is set
55 if (_deviceAttitudeStatusIntervalUs > 0) {
56 const qint64 intervalMs = _deviceAttitudeStatusIntervalUs / 1000;
57 if ((nowMs - _deviceAttitudeStatusLastSentMs) >= intervalMs) {
58 // Only simulate automatic movement if not under manual control
59 if (!_manualControl) {
60 // Simulate simple gimbal movement (slow sine wave)
61 const qint64 secondsSinceEpoch = nowMs / 1000;
62 _pitch = 10.0f * qSin(secondsSinceEpoch * 0.1); // ±10° pitch oscillation
63 _yaw = 15.0f * qCos(secondsSinceEpoch * 0.15); // ±15° yaw oscillation
64 }
65
66 _sendGimbalDeviceAttitudeStatus();
67 _deviceAttitudeStatusLastSentMs = nowMs;
68 }
69 }
70}

72bool MockLinkGimbal::handleMavlinkMessage(const mavlink_message_t &msg)
73{
74 if (msg.msgid == MAVLINK_MSG_ID_GIMBAL_MANAGER_SET_ATTITUDE) {
75 _handleGimbalManagerSetAttitude(msg);
76 return true;
77 }
78
79 if (msg.msgid != MAVLINK_MSG_ID_COMMAND_LONG) {
80 return false;
81 }
82
83 mavlink_command_long_t request{};
84 mavlink_msg_command_long_decode(&msg, &request);
85
86 // Handle gimbal-specific commands
87 switch (request.command) {
88 case MAV_CMD_SET_MESSAGE_INTERVAL:
89 if (_handleSetMessageInterval(request)) {
90 _sendCommandAck(request.command, MAV_RESULT_ACCEPTED, request.target_component);
91 return true;
92 }
93 return false;
94
95 case MAV_CMD_REQUEST_MESSAGE: {
96 if (static_cast<int>(request.param1) != MAVLINK_MSG_ID_GIMBAL_MANAGER_INFORMATION) {
97 return false;
98 }
99 InformationResponse response;
100 {
101 QMutexLocker locker(&_stateMutex);
102 response = _informationResponse;
103 }
104 switch (response) {
105 case InformationResponse::Accept:
106 _sendCommandAck(request.command, MAV_RESULT_ACCEPTED, request.target_component);
107 _handleRequestMessage(request);
108 break;
109 case InformationResponse::Nack:
110 _sendCommandAck(request.command, MAV_RESULT_UNSUPPORTED, request.target_component);
111 break;
112 case InformationResponse::Silent:
113 break;
114 }
115 return true;
116 }
117
118 case MAV_CMD_DO_GIMBAL_MANAGER_PITCHYAW:
119 if (_handleGimbalManagerPitchYaw(request)) {
120 _sendCommandAck(request.command, MAV_RESULT_ACCEPTED, request.target_component);
121 return true;
122 }
123 _sendCommandAck(request.command, MAV_RESULT_DENIED, request.target_component);
124 return true;
125
126 case MAV_CMD_DO_GIMBAL_MANAGER_CONFIGURE:
127 if (_handleGimbalManagerConfigure(request, msg.sysid, msg.compid)) {
128 _sendCommandAck(request.command, MAV_RESULT_ACCEPTED, request.target_component);
129 return true;
130 }
131 _sendCommandAck(request.command, MAV_RESULT_DENIED, request.target_component);
132 return true;
133
134 default:
135 return false;
136 }

160bool MockLinkGimbal::_handleSetMessageInterval(const mavlink_command_long_t &request)
161{
162 const int msgId = static_cast<int>(request.param1);
163 const int intervalUs = static_cast<int>(request.param2);
164 const mavlink_message_info_t* info = mavlink_get_message_info_by_id(static_cast<uint32_t>(msgId));
165 QString msgName = info ? info->name : QString::number(msgId);
166
167 qCDebug(MockLinkGimbalLog) << "SET_MESSAGE_INTERVAL -" << QString("%1(%2)").arg(msgName).arg(msgId) << "intervalUs:" << intervalUs;
168
169 // Handle interval values per MAVLink spec:
170 // -1 = disable
171 // 0 = default rate
172 // >0 = interval in microseconds
173 int effectiveInterval = intervalUs;
174 if (intervalUs == 0) {
175 effectiveInterval = kDefaultIntervalUs;
176 } else if (intervalUs < -1) {
177 // Invalid interval
178 return false;
179 }
180
181 // Thread-safe access: Main thread writing interval while worker thread reads every 1s.
182 // Serialize to avoid worker using stale interval for message transmission logic.
183 QMutexLocker locker(&_stateMutex);
184 if (msgId == MAVLINK_MSG_ID_GIMBAL_MANAGER_STATUS) {
185 _statusIntervalRequests.append(intervalUs);
186 if (_ignoreStatusIntervalRequests > 0) {
187 --_ignoreStatusIntervalRequests;
188 qCDebug(MockLinkGimbalLog) << msgName << "interval request ignored, remaining to ignore:" << _ignoreStatusIntervalRequests;
189 return true;
190 }
191 _managerStatusIntervalUs = effectiveInterval;
192 qCDebug(MockLinkGimbalLog) << msgName << "interval set to" << effectiveInterval << "us";
193 return true;
194 } else if (msgId == MAVLINK_MSG_ID_GIMBAL_DEVICE_ATTITUDE_STATUS) {
195 _deviceAttitudeStatusIntervalUs = effectiveInterval;
196 qCDebug(MockLinkGimbalLog) << msgName << "interval set to" << effectiveInterval << "us";
197 return true;
198 }
199 return false;
200}

217bool MockLinkGimbal::_handleGimbalManagerPitchYaw(const mavlink_command_long_t &request)
218{
219 // param1: pitch angle (degrees, positive: down, negative: up)
220 // param2: yaw angle (degrees, positive: to the right, negative: to the left)
221 // param3: pitch rate (deg/s)
222 // param4: yaw rate (deg/s)
223 // param5: flags (GIMBAL_MANAGER_FLAGS)
224 // param7: gimbal device id
225
226 const float requestedPitch = request.param1;
227 const float requestedYaw = request.param2;
228 const uint32_t flags = static_cast<uint32_t>(request.param5);
229
230 qCDebug(MockLinkGimbalLog) << "DO_GIMBAL_MANAGER_PITCHYAW - pitch:" << requestedPitch << "yaw:" << requestedYaw << "flags:" << flags;
231
232 QMutexLocker locker(&_stateMutex);
233 _lastPitchYawCommand.pitchDeg = requestedPitch;
234 _lastPitchYawCommand.yawDeg = requestedYaw;
235 _lastPitchYawCommand.pitchRateDegS = request.param3;
236 _lastPitchYawCommand.yawRateDegS = request.param4;
237 _lastPitchYawCommand.flags = flags;
238 _lastPitchYawCommand.deviceId = static_cast<uint8_t>(request.param7);
239 ++_lastPitchYawCommand.count;
240
241 // Flag-only commands (yaw lock / retract toggles) carry the current angles, so flags always apply
242 _yawLock = (flags & GIMBAL_MANAGER_FLAGS_YAW_LOCK) != 0;
243
244 // Check if this is a valid request (NaN means no change requested)
245 const bool updatePitch = !qIsNaN(requestedPitch);
246 const bool updateYaw = !qIsNaN(requestedYaw);
247 const bool updateRates = !qIsNaN(request.param3) || !qIsNaN(request.param4);
248
249 if (!updatePitch && !updateYaw && !updateRates) {
250 qCDebug(MockLinkGimbalLog) << "DO_GIMBAL_MANAGER_PITCHYAW - no valid pitch/yaw requested (both NaN)";
251 return false; // Nothing to do
252 }
253
254 // Apply limits based on gimbal capabilities
255 if (updatePitch && _hasPitchAxis) {
256 _pitch = qBound(-45.0f, requestedPitch, 45.0f);
257 _manualControl = true; // Switch to manual control mode
258 }
259
260 if (updateYaw && _hasYawAxis) {
261 // Same frame rules as GIMBAL_DEVICE_ATTITUDE_STATUS: explicit frame flag, else YAW_LOCK means earth frame
262 bool earthFrame = false;
263 if (flags & GIMBAL_MANAGER_FLAGS_YAW_IN_EARTH_FRAME) {
264 earthFrame = true;
265 } else if (!(flags & GIMBAL_MANAGER_FLAGS_YAW_IN_VEHICLE_FRAME)) {
266 earthFrame = _yawLock;
267 }
268 _yawInEarthFrame = earthFrame;
269
270 // _yaw is always body yaw; an earth-frame command is convertible whenever the mock has its own heading
271 float bodyYaw = qBound(-180.0f, requestedYaw, 180.0f);
272 if (earthFrame && _deltaYawMode != DeltaYawMode::Legacy) {
273 bodyYaw = std::remainder(requestedYaw - _deltaYawDeg, 360.0f);
274 } else if (earthFrame) {
275 qCDebug(MockLinkGimbalLog) << "earth-frame yaw command stored as body yaw: mock has no heading in Legacy mode";
276 }
277 _yaw = bodyYaw;
278 _manualControl = true; // Switch to manual control mode
279 }
280
281 qCDebug(MockLinkGimbalLog) << "Gimbal commanded to pitch:" << _pitch << "yaw:" << _yaw << "- switched to manual control";
282 return true;
283}

476void MockLinkGimbal::_sendGimbalDeviceAttitudeStatus(uint8_t sourceCompid, uint8_t deviceIdField)
477{
478 qCDebug(MockLinkGimbalLog) << "Sending GIMBAL_DEVICE_ATTITUDE_STATUS - pitch:" << _pitch << "yaw:" << _yaw
479 << "manual:" << _manualControl << "deltaYawMode:" << static_cast<int>(_deltaYawMode)
480 << "deltaYaw:" << _deltaYawDeg << "earthFrame:" << _yawInEarthFrame
481 << "from:" << sourceCompid << "deviceId:" << deviceIdField;
482
483 uint16_t flags = GIMBAL_DEVICE_FLAGS_NEUTRAL | GIMBAL_DEVICE_FLAGS_ROLL_LOCK | GIMBAL_DEVICE_FLAGS_PITCH_LOCK;
484 if (_yawLock) {
485 flags |= GIMBAL_DEVICE_FLAGS_YAW_LOCK;
486 }
487
488 float deltaYawRad = 0.0f;
489 switch (_deltaYawMode) {
490 case DeltaYawMode::Legacy:
491 break;
492 case DeltaYawMode::Unknown:
493 flags |= _yawInEarthFrame ? GIMBAL_DEVICE_FLAGS_YAW_IN_EARTH_FRAME : GIMBAL_DEVICE_FLAGS_YAW_IN_VEHICLE_FRAME;
494 deltaYawRad = NAN;
495 break;
496 case DeltaYawMode::Explicit:
497 flags |= _yawInEarthFrame ? GIMBAL_DEVICE_FLAGS_YAW_IN_EARTH_FRAME : GIMBAL_DEVICE_FLAGS_YAW_IN_VEHICLE_FRAME;
498 deltaYawRad = qDegreesToRadians(_deltaYawDeg);
499 break;
500 }
501
502 // _yaw is body yaw; earth-frame reports q_earth = q_delta_yaw * q_body per the message spec. _deltaYawDeg is
503 // the mock's true heading even in Unknown mode where it is transmitted as NaN.
504 const float reportedYawDeg = (_yawInEarthFrame && _deltaYawMode != DeltaYawMode::Legacy) ? _yaw + _deltaYawDeg : _yaw;
505
506 // Convert Euler angles (degrees) to quaternion
507 const float rollRad = qDegreesToRadians(_roll);
508 const float pitchRad = qDegreesToRadians(_pitch);
509 const float yawRad = qDegreesToRadians(reportedYawDeg);
510
511 const float cy = qCos(yawRad * 0.5f);
512 const float sy = qSin(yawRad * 0.5f);
513 const float cp = qCos(pitchRad * 0.5f);
514 const float sp = qSin(pitchRad * 0.5f);
515 const float cr = qCos(rollRad * 0.5f);
516 const float sr = qSin(rollRad * 0.5f);
517
518 float q[4];
519 q[0] = cr * cp * cy + sr * sp * sy; // w
520 q[1] = sr * cp * cy - cr * sp * sy; // x
521 q[2] = cr * sp * cy + sr * cp * sy; // y
522 q[3] = cr * cp * sy - sr * sp * cy; // z
523
524 mavlink_message_t msg{};
525 (void) mavlink_msg_gimbal_device_attitude_status_pack_chan(
526 _mockLink->vehicleId(),
527 sourceCompid,
528 _mockLink->outgoingMavlinkChannel(),
529 &msg,
530 0, 0, // target system, component
531 0, // time_boot_ms
532 flags,
533 q,
534 0.0f, 0.0f, 0.0f, // angular_velocity_x, y, z
535 0, // failure flags
536 deltaYawRad,
537 NAN, // delta_yaw_velocity unknown
538 deviceIdField);
539 _mockLink->respondWithMavlinkMessage(msg);
540}
