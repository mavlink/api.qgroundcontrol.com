46void VideoStreamController::setReceiver(VideoReceiver* receiver)
47{
48 if (receiver == _receiver) {
49 return;
50 }
51
52 const bool wasStreaming = streaming();
53 const bool wasDecoding = decoding();
54 const bool wasRecording = recording();
55 const QSize oldVideoSize = videoSize();
56 const VideoStreamStats oldStats = stats();
57
58 // Detach the old receiver first: what it emits while being destroyed must not reach a half-destroyed object.
59 VideoReceiver* const oldReceiver = std::exchange(_receiver, receiver);
60 if (oldReceiver) {
61 (void) disconnect(oldReceiver, nullptr, this, nullptr);
62 delete oldReceiver;
63 }
64 _decoded = false;
65
66 if (_receiver) {
67 _receiver->setParent(this);
68 _receiver->setName(name());
69 (void) connect(_receiver, &VideoReceiver::streamingChanged, this, &VideoStreamController::streamingChanged);
70 (void) connect(_receiver, &VideoReceiver::decodingChanged, this, [this] {
71 // Updated first: a stream that loses its decoder is reconnecting by the time decodingChanged arrives, so
72 // QML never sees it neither decoding nor reconnecting, which would clear the frame it keeps.
73 if (decoding()) {
74 _decoded = true;
75 }
76 _updateReconnecting();
77 emit decodingChanged();
78 });

83 emit videoSizeChanged();

171void VideoStreamController::_apply()
172{
173 _applyPending = false;
174 const std::optional<VideoReceiver::Config> config = _resolvedConfig();
175 if (const QString uri = config ? config->uri : QString(); uri != _uri) {
176 _uri = uri;
177 _forgetDecoded();
178 emit uriChanged();
179 }
180 if (!_receiver) {
181 return;
182 }
183
184 const bool active = _enabled && config.has_value();
185 if (!active) {
186 _forgetDecoded();
187 }
188 qCDebug(VideoStreamControllerLog) << name() << "active:" << active << "uri:"
189 << (config ? QGCNetworkHelper::redactedUrlForLogging(config->uri) : QString());
190
191 // Configure before activating and deactivate before configuring, so the receiver never starts a stream that is
192 // about to change.
193 if (active) {
194 _receiver->setConfig(*config);
195 _receiver->setActive(true);
196 } else {
197 _receiver->setActive(false);
198 if (config) {
199 _receiver->setConfig(*config);
200 }
201 }
202}
