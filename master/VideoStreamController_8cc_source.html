149void VideoStreamController::_apply()
150{
151 _applyPending = false;
152 if (!_receiver) {
153 return;
154 }
155
156 const std::optional<VideoReceiver::Config> config = _resolvedConfig();
157 const bool active = _enabled && config.has_value();
158 qCDebug(VideoStreamControllerLog) << name() << "active:" << active << "uri:"
159 << (config ? QGCNetworkHelper::redactedUrlForLogging(config->uri) : QString());
160
161 // Configure before activating and deactivate before configuring, so the receiver never starts a stream that is
162 // about to change.
163 if (active) {
164 _receiver->setConfig(*config);
165 _receiver->setActive(true);
166 } else {
167 _receiver->setActive(false);
168 if (config) {
169 _receiver->setConfig(*config);
170 }
171 }
172}
