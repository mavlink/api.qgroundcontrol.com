113void VideoStreamController::setReceiver(VideoReceiver* receiver)
114{
115 if (_hasOwnReceiver || (receiver == _receiver)) {
116 return;
117 }
118
119 const bool wasDecoding = decoding();
120 const bool wasRecording = recording();
121 const bool wasEncoding = encodesRecordings();
122 const QSize oldVideoSize = videoSize();
123 const VideoStreamStats oldStats = stats();
124
125 // Detach the old receiver first: what it emits while being destroyed must not reach a half-destroyed object.
126 // No receiver is set while it goes: one that finishes a recording runs the event loop in its destructor, from
127 // where this function may be called again.
128 _decoded = false;
129 if (VideoReceiver* const oldReceiver = std::exchange(_receiver, nullptr)) {
130 (void) disconnect(oldReceiver, nullptr, this, nullptr);
131 delete oldReceiver;
132 }
133
134 if (_receiver) {
135 // Set by such a call, which came after this one: its receiver stays.
136 if (receiver != _receiver) {
137 delete receiver;
138 }
139 } else if (receiver) {
140 _receiver = receiver;
141 _receiver->setParent(this);
142 _receiver->setName(name());
143 (void) connect(_receiver, &VideoReceiver::decodingChanged, this, [this] {
144 // Updated first: a stream that loses its decoder is reconnecting by the time decodingChanged arrives, so
145 // QML never sees it neither decoding nor reconnecting, which would clear the frame it keeps.
146 if (decoding()) {
147 _decoded = true;
148 }
149 _updateReconnecting();
150 emit decodingChanged();
151 });
152 (void) connect(_receiver, &VideoReceiver::stateChanged, this, &VideoStreamController::_onReceiverStateChanged);
153 (void) connect(_receiver, &VideoReceiver::recordingChanged, this, &VideoStreamController::recordingChanged);
154 (void) connect(_receiver, &VideoReceiver::videoSizeChanged, this, [this](QSize size) {
155 qCDebug(VideoStreamControllerLog) << name() << "video size" << size;
156 emit videoSizeChanged();
157 emit aspectRatioChanged();
158 });
159 (void) connect(_receiver, &VideoReceiver::statsChanged, this, &VideoStreamController::statsChanged);
160 (void) connect(_receiver, &VideoReceiver::recordingStarted, this, &VideoStreamController::recordingStarted);
161 (void) connect(_receiver, &VideoReceiver::recordingFinished, this, &VideoStreamController::recordingFinished);
162 (void) connect(_receiver, &VideoReceiver::errorOccurred, this, &VideoStreamController::errorOccurred);
163 _receiver->setDisplayConfig(_resolvedDisplayConfig());
164 _receiver->setRecordingEncoding(_recordingEncoding);
165 _receiver->setVideoOutput(_displaySuspended ? nullptr : _videoOutput.data());
166 _apply();
167 }
168
169 if (decoding() != wasDecoding) {
170 emit decodingChanged();
171 }
172 if (recording() != wasRecording) {
173 emit recordingChanged(recording());
174 }
175 if (encodesRecordings() != wasEncoding) {
176 emit encodesRecordingsChanged();
177 }
178 if (videoSize() != oldVideoSize) {
179 emit videoSizeChanged();
180 emit aspectRatioChanged();

267void VideoStreamController::_apply()
268{
269 _applyPending = false;
270 const std::optional<VideoReceiver::Config> config = _resolvedConfig();
271 if (const QString uri = config ? config->uri : QString(); uri != _uri) {
272 _uri = uri;
273 _forgetDecoded();
274 emit uriChanged();
275 }
276 if (!_receiver) {
277 return;
278 }
279
280 const bool active = _enabled && config.has_value();
281 if (!active) {
282 _forgetDecoded();
283 }
284 qCDebug(VideoStreamControllerLog) << name() << "active:" << active << "uri:" << uri();
285
286 // Configure before activating and deactivate before configuring, so the receiver never starts a stream that is
287 // about to change.
288 if (active) {
289 _receiver->setConfig(*config);
290 _receiver->setActive(true);
291 } else {
292 _receiver->setActive(false);
293 if (config) {
294 _receiver->setConfig(*config);
295 }
296 }

465double VideoStreamController::aspectRatio() const
466{

469 return static_cast<double>(size.width()) / size.height();

471 if (_streamInfo) {
