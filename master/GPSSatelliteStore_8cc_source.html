77bool GPSSatelliteStore::_accept(quint64 receipt, quint64 current, quint64& retired, quint64 nowUs) const
78{
79 if (!receipt || receipt <= _clearedThroughUs || receipt <= retired || receipt < current || receipt > nowUs) {
80 return false;
81 }
82 if (MonotonicClock::remaining(receipt, nowUs, std::chrono::milliseconds(_freshnessTimeoutMs)).count() == 0) {
83 retired = std::max(retired, receipt);

204void GPSSatelliteStore::_publish()
205{
206 const quint64 nowUs = _scheduler ? _scheduler->nowUs() : 0;
207 _expire(nowUs);
208 _observation.satellites.clear();
209 _observation.provenance.clear();
210 _observation.monotonicTimestampUs = 0;
211 auto remaining = std::chrono::microseconds::max();
212 for (const auto& [constellation, state] : _constellations) {
213 if (!state.viewReceiptUs && !state.useReceiptUs) {
214 continue;
215 }
216 _observation.provenance.append(
217 {constellation, state.viewReceiptUs, state.useReceiptUs, state.usedCount, state.usedIds});
218 for (auto satellite : state.satellites) {
219 const auto used = state.used.find({satellite.id, satellite.prn});
220 satellite.used = std::nullopt;
221 if (state.useReceiptUs) {
222 if (state.usedIds) {
223 satellite.used = state.usedIds->contains(satellite.id);
224 } else if (used != state.used.cend()) {
225 satellite.used = used->second;
226 }
227 }
228 _observation.satellites.append(satellite);
229 }
230 for (const quint64 receipt : {state.viewReceiptUs, state.useReceiptUs}) {
231 if (receipt) {
232 _observation.monotonicTimestampUs = std::max(_observation.monotonicTimestampUs, receipt);
233 remaining = std::min(remaining, MonotonicClock::remaining(
234 receipt, nowUs, std::chrono::milliseconds(_freshnessTimeoutMs)));
235 }
236 }
237 }
238 _expiryTask.cancel();
239 if (remaining != std::chrono::microseconds::max()) {
240 _expiryTask.schedule(remaining, [this]() { _publish(); });
241 }
242 _observation.revision = ++_revision;
