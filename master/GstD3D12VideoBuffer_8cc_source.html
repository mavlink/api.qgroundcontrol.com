143 std::shared_ptr<StagingSlotLease> acquire(ID3D12Device* dev, D3D12_COMMAND_LIST_TYPE queueType,
144 const StagingKey& key, const D3D12_RESOURCE_DESC& dstDesc,
145 const D3D12_HEAP_PROPERTIES& heapProps, int planeIdx, guint subIdx)
146 {
147 QMutexLocker lock(&_mutex);
148 StagingRing* ring = _entries.find(key).resource;
149 if (!ring) {
150 ring = _entries.insert(key, StagingRing{}).resource;
151 }
152 for (std::size_t probe = 0; probe < kStagingRingSize; ++probe) {
153 const std::size_t slot = (ring->next + probe) % kStagingRingSize;
154 if ((*ring->leased)[slot].exchange(true, std::memory_order_acq_rel)) {
155 continue;
156 }
157 StagingEntry& e = ring->entries[slot];
158 if (e.resource) {
159 GstHwPathTelemetry::record(HwVideoBufferPath::D3D12, GstHwPathTelemetry::Counter::ImageCacheHits);
160 } else if (createEntry(dev, queueType, dstDesc, heapProps, planeIdx, subIdx, e)) {
161 GstHwPathTelemetry::record(HwVideoBufferPath::D3D12, GstHwPathTelemetry::Counter::ImageCacheMisses);
162 } else {
163 (*ring->leased)[slot].store(false, std::memory_order_release);
164 return nullptr;
165 }
166 ring->next = (slot + 1) % kStagingRingSize;
167 return std::make_shared<StagingSlotLease>(ring->leased, slot, addRefEntry(e));
168 }
169 QGC_HW_WARN_ONCE(GstD3D12Log, s_diag.loggedTextureCreateFail,
170 "resolve: every staging slot is in use (plane=" << planeIdx << ") — CPU fallback");
171 return nullptr;
172 }

174 void clear()
175 {
176 QMutexLocker lock(&_mutex);
177 _entries.clear();
178 }

188 static bool createEntry(ID3D12Device* dev, D3D12_COMMAND_LIST_TYPE queueType, const D3D12_RESOURCE_DESC& dstDesc,
189 const D3D12_HEAP_PROPERTIES& heapProps, int planeIdx, guint subIdx, StagingEntry& e)
190 {
191 if (FAILED(dev->CreateCommittedResource(&heapProps, D3D12_HEAP_FLAG_NONE, &dstDesc, D3D12_RESOURCE_STATE_COMMON,
192 nullptr, IID_PPV_ARGS(&e.resource)))) {
193 QGC_HW_WARN_ONCE(GstD3D12Log, s_diag.loggedTextureCreateFail,
194 "mapTextures: CreateCommittedResource for slice copy failed (plane="
195 << planeIdx << " subresource=" << subIdx << ")");
196 e = {};
197 return false;
198 }
199 if (FAILED(dev->CreateCommandAllocator(queueType, IID_PPV_ARGS(&e.allocator))) ||
200 FAILED(dev->CreateCommandList(0, queueType, e.allocator, nullptr, IID_PPV_ARGS(&e.list)))) {
201 QGC_HW_WARN_ONCE(GstD3D12Log, s_diag.loggedTextureCreateFail,
202 "mapTextures: command allocator/list create failed (plane=" << planeIdx << ")");
203 releaseEntryResources(e);
204 e = {};
205 return false;
206 }
207 // Freshly created lists are open; close so the per-frame Reset() path is uniform.
208 e.list->Close();
209 return true;
210 }

229ID3D12Resource* copySliceToStaging(ID3D12Resource* resource, guint subIdx, int planeIdx,
230 const D3D12_RESOURCE_DESC& srcDesc, ID3D12Device* d3dDev, GstD3D12CmdQueue* cmdQueue,
231 D3D12_COMMAND_LIST_TYPE queueType, guint64& maxFenceValue,
232 std::shared_ptr<StagingSlotLease>& lease)
233{
234 D3D12_RESOURCE_DESC dstDesc = srcDesc;
235 dstDesc.DepthOrArraySize = 1;
236 dstDesc.MipLevels = 1;
237
238 D3D12_HEAP_PROPERTIES heapProps = {};
239 heapProps.Type = D3D12_HEAP_TYPE_DEFAULT;
240
241 const StagingKey key{srcDesc.Width, srcDesc.Height, UINT(srcDesc.Format), planeIdx};
242 std::shared_ptr<StagingSlotLease> slotLease =
243 StagingResourcePool::instance().acquire(d3dDev, queueType, key, dstDesc, heapProps, planeIdx, subIdx);
244 if (!slotLease || !slotLease->entry().valid()) {
245 return nullptr;
246 }
247 const StagingEntryRef& entry = slotLease->entry();
248
249 if (FAILED(entry.allocator()->Reset()) || FAILED(entry.list()->Reset(entry.allocator(), nullptr))) {
250 QGC_HW_WARN_ONCE(GstD3D12Log, s_diag.loggedTextureCreateFail,
251 "mapTextures: command allocator/list reset failed (plane=" << planeIdx << ")");
252 return nullptr;
253 }
254 ID3D12GraphicsCommandList* cmdList = entry.list();
255
256 D3D12_TEXTURE_COPY_LOCATION srcLoc = {};
257 srcLoc.pResource = resource;
258 srcLoc.Type = D3D12_TEXTURE_COPY_TYPE_SUBRESOURCE_INDEX;
259 srcLoc.SubresourceIndex = subIdx;
260
261 D3D12_TEXTURE_COPY_LOCATION dstLoc = {};
262 dstLoc.pResource = entry.resource();
263 dstLoc.Type = D3D12_TEXTURE_COPY_TYPE_SUBRESOURCE_INDEX;
264 dstLoc.SubresourceIndex = 0;
265
266 // Decoder output is parked in COMMON by gst_d3d12_memory_sync; gst-d3d12 exposes no per-resource state getter, so
267 // COMMON is the only valid before-state.
268 D3D12_RESOURCE_BARRIER toCopySrc = {};
269 toCopySrc.Type = D3D12_RESOURCE_BARRIER_TYPE_TRANSITION;
270 toCopySrc.Transition.pResource = resource;
271 toCopySrc.Transition.Subresource = subIdx;
272 toCopySrc.Transition.StateBefore = D3D12_RESOURCE_STATE_COMMON;
273 toCopySrc.Transition.StateAfter = D3D12_RESOURCE_STATE_COPY_SOURCE;
274 cmdList->ResourceBarrier(1, &toCopySrc);
275
276 cmdList->CopyTextureRegion(&dstLoc, 0, 0, 0, &srcLoc, nullptr);
277
278 D3D12_RESOURCE_BARRIER toCommon = toCopySrc;
279 toCommon.Transition.StateBefore = D3D12_RESOURCE_STATE_COPY_SOURCE;
280 toCommon.Transition.StateAfter = D3D12_RESOURCE_STATE_COMMON;
281 cmdList->ResourceBarrier(1, &toCommon);
282
283 cmdList->Close();
284
285 ID3D12CommandList* lists[] = {cmdList};
286 guint64 fenceValue = 0;
287 // Wrapper execute takes gst-d3d12's queue lock, serializing the copy against decoder-thread submits.
288 if (FAILED(gst_d3d12_cmd_queue_execute_command_lists(cmdQueue, 1, lists, &fenceValue))) {
289 // The caller skips the fence wait on failure, so the pooled allocator's GPU work may still be in flight; drop
290 // the pool so the next frame rebuilds clean command objects instead of resetting a busy allocator.
291 StagingResourcePool::instance().clear();
292 return nullptr;
293 }
294 if (fenceValue > maxFenceValue) {
295 maxFenceValue = fenceValue;
296 }
297 ID3D12Resource* stagingResource = entry.resource();
298 stagingResource->AddRef();
299 lease = std::move(slotLease);
300 return stagingResource;
301}
