16QImage frameToImage(const QVideoFrame& frame)
17{
18 if (!frame.isValid()) {
19 return {};
20 }
21
22 // toImage() applies only the surface format's own transformation: not the frame's rotation or mirroring, nor the
23 // viewport that crops decoder padding.
24 QImage image = frame.toImage();
25 if (image.isNull()) {
26 return image;
27 }
28
29 const QRect viewport = frame.surfaceFormat().viewport();
30 if (viewport.isValid() && (image.size() == frame.size()) && (viewport != image.rect())) {
31 image = image.copy(viewport.intersected(image.rect()));
32 }
33
34 // Same order as Qt's presentation (qNormalizedFrameTransformation): rotate, then mirror horizontally.
35 const int degrees = static_cast<int>(frame.rotation());
36 if (degrees != 0) {
37 image = image.transformed(QTransform().rotate(degrees));
38 }
39 if (frame.mirrored()) {
40 image = image.flipped(Qt::Horizontal);
41 }
42 return image;
43}

71QSize VideoReceiver::presentedSize(const QVideoFrame& frame)
72{
73 if (!frame.isValid()) {
74 return {};
75 }
76 const QVideoFrameFormat format = frame.surfaceFormat();
77 const QRect viewport = format.viewport();
78 const QSize size = viewport.isValid() ? viewport.size() : frame.size();
79 // As Qt's qRotatedFramePresentationSize(): the frame's and the format's rotation add up; mirroring keeps the size.
80 const int degrees = static_cast<int>(frame.rotation()) + static_cast<int>(format.rotation());
81 return ((degrees % 180) != 0) ? size.transposed() : size;
82}
