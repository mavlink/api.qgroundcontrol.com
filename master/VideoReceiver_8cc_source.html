42QSize VideoReceiver::presentedSize(const QVideoFrame& frame)
43{
44 if (!frame.isValid()) {
45 return {};
46 }
47 const QVideoFrameFormat format = frame.surfaceFormat();
48 const QRect viewport = format.viewport();
49 const QSize size = viewport.isValid() ? viewport.size() : frame.size();
50 // As Qt's qRotatedFramePresentationSize(): the frame's and the format's rotation add up; mirroring keeps the size.
51 const int degrees = static_cast<int>(frame.rotation()) + static_cast<int>(format.rotation());
52 return ((degrees % 180) != 0) ? size.transposed() : size;
53}
