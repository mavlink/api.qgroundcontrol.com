153void RallyPoint::_requestTerrainAltitude(void)
154{
155 const QGeoCoordinate coord = coordinate();

157 return;
158 }

172void RallyPoint::_terrainDataReceived(bool success, const QList<double>& heights)
173{
174 _terrainQuery = nullptr;
175
176 const QGeoCoordinate coord = coordinate();
177 if (coord.latitude() != _lastTerrainQueryLatitude || coord.longitude() != _lastTerrainQueryLongitude) {
178 // The point moved while the request was in flight, the answer is for the old position
179 _requestTerrainAltitude();
180 return;
181 }
182
183 const bool switchToAbsoluteOnFailure = _switchToAbsoluteOnTerrainFailure;
184 _switchToAbsoluteOnTerrainFailure = false;
185
186 if (!success || heights.isEmpty()) {
187 if (switchToAbsoluteOnFailure) {
188 setAltitudeFrame(QGroundControlQmlGlobal::AltitudeFrameAbsolute);
189 }
190 _setTerrainQueryFailed(true);
191 return;
192 }
193
194 _setTerrainQueryFailed(false);
195 _followTerrain = true;
196 _setAltitudeFromTerrain(heights.first());
197}
