70size_t splitFields(std::string_view text, std::span<std::string_view> fields)
71{
72 size_t count = 0;
73 for (;;) {
74 if (count == fields.size()) {
75 return 0;
76 }
77 const auto comma = text.find(',');
78 fields[count++] = text.substr(0, comma);
79 if (comma == std::string_view::npos) {
80 return count;
81 }
82 text.remove_prefix(comma + 1);
83 }
84}

91std::optional<Frame> frame(std::string_view text)
92{
93 if (text.ends_with('\n')) {
94 text.remove_suffix(1);
95 if (text.ends_with('\r')) {
96 text.remove_suffix(1);
97 }
98 }
99 if (text.empty() || text.front() != '$') {
100 return {};
101 }
102 text.remove_prefix(Sentence::PREFIX_LENGTH);
103 const auto star = text.find('*');
104 const auto body = text.substr(0, star);
105 if (body.empty()) {
106 return {};
107 }
108 for (char byte : body) {
109 if (byte < ' ' || byte > '~' || byte == '$') {
110 return {};
111 }
112 }
113 return Frame{body, star == std::string_view::npos ? std::string_view() : text.substr(star + 1)};
114}

122std::optional<Sentence> sentence(std::string_view text)
123{

137double degreesFromDegreesMinutes(double ddmm)
138{
139 if (!std::isfinite(ddmm) || std::abs(ddmm) > MAX_LONGITUDE_DEGREES * DEGREE_MINUTE_SCALE) {
140 return NAN;
141 }
142 const double degrees = std::trunc(ddmm / DEGREE_MINUTE_SCALE);
143 const double minutes = ddmm - degrees * DEGREE_MINUTE_SCALE;
144 return std::abs(minutes) < MINUTES_PER_DEGREE ? degrees + minutes / MINUTES_PER_DEGREE : NAN;
145}

147std::optional<double> coordinate(std::string_view field, std::string_view hemisphere, bool latitude)
148{
149 const auto value = number<double>(field);
150 if (!value || *value < 0 || hemisphere.size() != 1) {
151 return {};
152 }
153 const double degrees = std::trunc(*value / DEGREE_MINUTE_SCALE);
154 const double minutes = *value - degrees * DEGREE_MINUTE_SCALE;
155 const double limit = latitude ? MAX_LATITUDE_DEGREES : MAX_LONGITUDE_DEGREES;
156 if (minutes >= MINUTES_PER_DEGREE || degrees + minutes / MINUTES_PER_DEGREE > limit) {
157 return {};
158 }
159 const char positive = latitude ? 'N' : 'E';
160 const char negative = latitude ? 'S' : 'W';
161 if (hemisphere[0] != positive && hemisphere[0] != negative) {
162 return {};
163 }
164 return (degrees + minutes / MINUTES_PER_DEGREE) * (hemisphere[0] == negative ? -1 : 1);
165}

206std::optional<int> utcMilliseconds(std::string_view field)
207{
208 if (field.size() < UTC_INTEGER_DIGITS ||
209 (field.size() > UTC_INTEGER_DIGITS &&
210 (field[UTC_INTEGER_DIGITS] != '.' || field.size() == UTC_FRACTION_OFFSET))) {
211 return {};
212 }
213 for (size_t index = 0; index < field.size(); ++index) {
214 if (index != UTC_INTEGER_DIGITS && (field[index] < '0' || field[index] > '9')) {
215 return {};
216 }
217 }
218 const auto hours = number<unsigned>(field.substr(0, UTC_COMPONENT_DIGITS));
219 const auto minutes = number<unsigned>(field.substr(UTC_COMPONENT_DIGITS, UTC_COMPONENT_DIGITS));
220 const auto seconds = number<unsigned>(field.substr(2 * UTC_COMPONENT_DIGITS, UTC_COMPONENT_DIGITS));
221 if (!hours || !minutes || !seconds || *hours >= HOURS_PER_DAY || *minutes >= MINUTES_PER_HOUR ||
222 *seconds >= SECONDS_PER_MINUTE) {
223 return {};
224 }
225 int milliseconds = 0;
226 // Truncate sub-millisecond precision without floating-point rounding into the preceding epoch.
227 for (size_t index = UTC_FRACTION_OFFSET, scale = FIRST_MILLISECOND_DIGIT_WEIGHT; index < field.size() && scale > 0;
228 ++index, scale /= DECIMAL_BASE) {
229 milliseconds += static_cast<int>((field[index] - '0') * scale);
230 }
231 const auto wholeSeconds =
232 std::chrono::hours(*hours) + std::chrono::minutes(*minutes) + std::chrono::seconds(*seconds);
233 return static_cast<int>(std::chrono::duration_cast<std::chrono::milliseconds>(wholeSeconds).count()) + milliseconds;
234}

343std::optional<NavigationStatus> navigationStatus(const Sentence& input)
344{
345 const auto type = input.type();
346 const auto& fields = input.fields;
347 if (type == "GGA" && input.count > Field::GGA_QUALITY) {
348 const auto quality = number<unsigned>(fields[Field::GGA_QUALITY]);
349 if (quality && *quality <= GgaQuality::MAX_VALUE) {
350 return NavigationStatus{*quality != GgaQuality::INVALID, utcMilliseconds(fields[Field::UTC_TIME])};
351 }
352 } else if (type == "GSA" && input.count > Field::GSA_DIMENSION) {
353 const auto dimension = number<unsigned>(fields[Field::GSA_DIMENSION]);
354 if (dimension && *dimension >= FixDimension::NO_FIX && *dimension <= FixDimension::THREE_D) {
355 return NavigationStatus{*dimension != FixDimension::NO_FIX, std::nullopt};
356 }
357 } else if (type == "RMC" || type == "GLL") {
358 const auto status = type == "RMC" ? Field::RMC_STATUS : Field::GLL_STATUS;
359 const auto time = type == "RMC" ? Field::UTC_TIME : Field::GLL_TIME;
360 if (input.count > status && (fields[status] == "A" || fields[status] == "V")) {
361 return NavigationStatus{fields[status] == "A", utcMilliseconds(fields[time])};
362 }
363 }
364 return std::nullopt;
365}

396QByteArray repairChecksum(const QByteArray& sentence)
397{
398 const auto wire = NMEA::frame(QByteArrayView(sentence));
399 if (wire && wire->body.size() + NMEA::Sentence::PREFIX_LENGTH >= MIN_REPAIRABLE_SENTENCE_LENGTH) {
400 return frame(QByteArrayView(wire->body));
401 }
402 QByteArray line = sentence;
403 if (line.endsWith('\n')) {

406 line.chop(1);

409 return line + "\r\n";

412QByteArray makeGGA(const NMEA::GGA& fix, const QTime& utc)
413{
414 if (!utc.isValid() || !std::isfinite(fix.latitude) || fix.latitude < -90.0 || fix.latitude > 90.0 ||
415 !std::isfinite(fix.longitude) || fix.longitude < -180.0 || fix.longitude > 180.0 || std::isinf(fix.altitude) ||
416 std::isinf(fix.geoidSeparation) || std::isinf(fix.hdop) || fix.hdop < 0.0 ||
417 fix.quality > NMEA::GgaQuality::MAX_VALUE) {
418 return {};
419 }
420 const QByteArray hhmmss = utc.toString(u"hhmmss").toLatin1();
421
422 auto dmm = [](double deg, bool lat) -> QByteArray {
423 const double a = qFabs(deg);
424 int d = static_cast<int>(a);
425 double m = (a - d) * NMEA::MINUTES_PER_DEGREE;
426
427 const int scaledMinutes = static_cast<int>(m * MINUTE_FRACTION_SCALE + 0.5);
428 double mRounded = scaledMinutes / MINUTE_FRACTION_SCALE;
429 if (mRounded >= NMEA::MINUTES_PER_DEGREE) {
430 mRounded -= NMEA::MINUTES_PER_DEGREE;
431 d += 1;
432 }
433
434 QByteArray mm = QByteArray::number(mRounded, 'f', MINUTE_DECIMAL_PLACES);
435 if (mRounded < NMEA::DECIMAL_BASE) {
436 mm.prepend('0');
437 }
438
439 const int dWidth = lat ? LATITUDE_DEGREE_DIGITS : LONGITUDE_DEGREE_DIGITS;
440 return QByteArray::number(d).rightJustified(dWidth, '0') + mm;
441 };
442
443 const bool latNorth = fix.latitude >= 0.0;
444 const bool lonEast = fix.longitude >= 0.0;
445
446 const QByteArray latField = dmm(fix.latitude, true);
447 const QByteArray lonField = dmm(fix.longitude, false);
448 const auto measurement = [](double value, int decimalPlaces) {
449 return std::isnan(value) ? QByteArray() : QByteArray::number(value, 'f', decimalPlaces);
450 };
451
452 QByteArray core;
453 core += "GPGGA,";
454 core += hhmmss + ',';
455 core += latField + ',';
456 core += (latNorth ? "N" : "S");
457 core += ',';
458 core += lonField + ',';
459 core += (lonEast ? "E" : "W");
460 core += ',' + QByteArray::number(fix.quality) + ',';
461 if (fix.satellitesUsed) {
462 core += QByteArray::number(*fix.satellitesUsed);
463 }
464 core += ',' + measurement(fix.hdop, DOP_DECIMAL_PLACES) + ',';
465 core += measurement(fix.altitude, ALTITUDE_DECIMAL_PLACES);
466 core += ",M," + measurement(fix.geoidSeparation, ALTITUDE_DECIMAL_PLACES) + ",M,,";
467 return frame(core);
468}
