30bool Frame::hasValidChecksum() const
31{
32 if (checksum.size() != CHECKSUM_DIGITS)

40std::optional<Frame> frame(std::string_view text)
41{
42 if (text.ends_with('\n')) {
43 text.remove_suffix(1);
44 if (text.ends_with('\r'))
45 text.remove_suffix(1);
46 }
47 if (text.empty() || text.front() != '$')
48 return {};
49 text.remove_prefix(Sentence::PREFIX_LENGTH);
50 const auto star = text.find('*');
51 const auto body = text.substr(0, star);
52 if (body.empty())
53 return {};
54 for (char byte : body) {
55 if (byte < ' ' || byte > '~' || byte == '$')
56 return {};
57 }
58 return Frame{body, star == std::string_view::npos ? std::string_view() : text.substr(star + 1)};
59}

61std::optional<Sentence> sentence(std::string_view text)
62{
63 const auto wire = frame(text);
64 if (!wire || !wire->hasValidChecksum())
65 return {};
66 Sentence result;
67 text = text.substr(0, Sentence::PREFIX_LENGTH + wire->body.size());
68 do {
69 if (result.count == result.fields.size())
70 return {};
71 const auto comma = text.find(',');
72 result.fields[result.count++] = text.substr(0, comma);
73 if (comma == std::string_view::npos)
74 break;
75 text.remove_prefix(comma + 1);
76 } while (true);
77 if (result.fields[0].size() != Sentence::HEADER_LENGTH)
78 return {};
79 return result;
80}

82std::optional<double> coordinate(std::string_view field, std::string_view hemisphere, bool latitude)
83{
84 const auto value = number<double>(field);
85 if (!value || *value < 0 || hemisphere.size() != 1)
86 return {};
87 const double degrees = std::trunc(*value / DEGREE_MINUTE_SCALE);
88 const double minutes = *value - degrees * DEGREE_MINUTE_SCALE;
89 const double limit = latitude ? MAX_LATITUDE_DEGREES : MAX_LONGITUDE_DEGREES;
90 if (minutes >= MINUTES_PER_DEGREE || degrees + minutes / MINUTES_PER_DEGREE > limit)
91 return {};
92 const char positive = latitude ? 'N' : 'E';
93 const char negative = latitude ? 'S' : 'W';
94 if (hemisphere[0] != positive && hemisphere[0] != negative)
95 return {};
96 return (degrees + minutes / MINUTES_PER_DEGREE) * (hemisphere[0] == negative ? -1 : 1);
97}

122std::optional<int> utcMilliseconds(std::string_view field)
123{
124 if (field.size() < UTC_INTEGER_DIGITS ||
125 (field.size() > UTC_INTEGER_DIGITS &&
126 (field[UTC_INTEGER_DIGITS] != '.' || field.size() == UTC_FRACTION_OFFSET)))
127 return {};
128 for (size_t index = 0; index < field.size(); ++index) {
129 if (index != UTC_INTEGER_DIGITS && (field[index] < '0' || field[index] > '9'))
130 return {};
131 }
132 const auto hours = number<unsigned>(field.substr(0, UTC_COMPONENT_DIGITS));
133 const auto minutes = number<unsigned>(field.substr(UTC_COMPONENT_DIGITS, UTC_COMPONENT_DIGITS));
134 const auto seconds = number<unsigned>(field.substr(2 * UTC_COMPONENT_DIGITS, UTC_COMPONENT_DIGITS));
135 if (!hours || !minutes || !seconds || *hours >= HOURS_PER_DAY || *minutes >= MINUTES_PER_HOUR ||
136 *seconds >= SECONDS_PER_MINUTE)
137 return {};
138 int milliseconds = 0;
139 // Truncate sub-millisecond precision without floating-point rounding into the preceding epoch.
140 for (size_t index = UTC_FRACTION_OFFSET, scale = FIRST_MILLISECOND_DIGIT_WEIGHT; index < field.size() && scale > 0;
141 ++index, scale /= DECIMAL_BASE)
142 milliseconds += static_cast<int>((field[index] - '0') * scale);
143 const auto wholeSeconds =
144 std::chrono::hours(*hours) + std::chrono::minutes(*minutes) + std::chrono::seconds(*seconds);
145 return static_cast<int>(std::chrono::duration_cast<std::chrono::milliseconds>(wholeSeconds).count()) + milliseconds;
146}

149{

151 return {};
