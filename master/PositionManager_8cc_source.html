209void PositionManager::_setDeviceActive(bool active)
210{
211 active = active && _device;
212 if (_deviceActive == active) {
213 return;
214 }
215 _deviceActive = active;
216 if (active) {
217 _device->setPreferredPositioningMethods(QGeoPositionInfoSource::SatellitePositioningMethods);
218#if !defined(Q_OS_DARWIN)
219 _device->setUpdateInterval(_device->minimumUpdateInterval());
220#endif
221 _device->startUpdates();
222 return;
223 }
224 if (_device) {
225 _device->stopUpdates();
226 }
227 // Reactivation must not publish a fix from before the stop. An inactive device is never consulted, so this
228 // reset needs no re-evaluation, and it must not notify while the service is being destroyed.

233void PositionManager::_devicePositionUpdated(const QGeoPositionInfo& position)
234{
235 // A stopped backend may still deliver a report it queued before stopUpdates().
236 if (!_deviceActive) {
237 return;
238 }
239 _platformStatus = SourceStatus::WaitingForFix;
240 if (_selectedKind == SelectedSource::Internal) {
241 _setPositioningError(QGeoPositionInfoSource::NoError);
242 }
243 GPSObservation observation;
244 observation.position = position;
245 observation.monotonicTimestampUs = _scheduler->nowUs();
246 // An altitude without a vertical accuracy may be a placeholder, such as Core Location's for a 2D fix.
247 if (position.hasAttribute(QGeoPositionInfo::VerticalAccuracy) && qIsFinite(position.coordinate().altitude())) {
248 observation.altitudeDatum = _deviceAltitudeDatum;
249 }
250 _deviceHealth->updateObservation(observation);
251}

253void PositionManager::_deviceErrorOccurred(QGeoPositionInfoSource::Error error)
254{
255 if (!_deviceActive) {
256 return;
257 }
258 if (error == QGeoPositionInfoSource::AccessError) {
259 _platformStatus = SourceStatus::PermissionDenied;
260 } else if (error == QGeoPositionInfoSource::ClosedError || error == QGeoPositionInfoSource::UnknownSourceError) {
261 _platformStatus = SourceStatus::BackendUnavailable;
262 }
263 if (_selectedKind == SelectedSource::Internal) {
264 _setPositioningError(error);
265 }
266 // A timeout means no new fix, which freshness already expires; only real failures invalidate the fix.
267 if (error != QGeoPositionInfoSource::NoError && error != QGeoPositionInfoSource::UpdateTimeoutError) {
268 _deviceHealth->invalidatePosition();
269 }
270 _update();
271}

280void PositionManager::_update(SelectedSource reporter)
281{
282 const SelectedSource chosen = _choose();
283 GPSSourceHealth* const health = _sourceHealth(chosen);
284 bool republish = reporter == chosen;
285 if (_reselect || chosen != _selectedKind || _currentHealth != health) {
286 qCDebug(PositionManagerLog) << "Ground-station position source changed"
287 << "source:" << chosen << "selected:" << health;
288 _reselect = false;
289 _selectedKind = chosen;
290 _currentHealth = health;
291 _selectionRevision = health ? health->observationRevision() : 0;
292 _authorized = false;
293 _published = {};
294 _gcsPositioningError = QGeoPositionInfoSource::NoError;
295 republish = _configuration.sourceMode == SourceMode::Automatic;
296 }
297 // Standby deadlines can run before the selected source's deadline.
298 if (republish || (_published.position.isValid() && !_accepted(_selectedKind))) {
299 _republish();
300 }
301 _updateStatus();
302 _emitChanges();
303}

305PositionManager::SelectedSource PositionManager::_choose()
306{
307 if (_configuration.sourceMode == SourceMode::ReceiverOnly) {
308 return SelectedSource::Receiver;
309 }
310 if (_configuration.sourceMode == SourceMode::InternalOnly) {
311 return SelectedSource::Internal;
312 }
313 const bool receiverUsable = _accepted(SelectedSource::Receiver).has_value();
314 const bool deviceUsable = _accepted(SelectedSource::Internal).has_value();
315 // A receiver that recovers while the device works takes over only after staying usable for RECOVERY_DELAY.
316 if (receiverUsable && deviceUsable && _selectedKind == SelectedSource::Internal) {
317 const qint64 nowMs = _scheduler->nowMs();
318 if (!_receiverUsableSinceMs) {
319 _receiverUsableSinceMs = nowMs;
320 }
321 const auto remaining = RECOVERY_DELAY - std::chrono::milliseconds(nowMs - *_receiverUsableSinceMs);
322 if (remaining > std::chrono::milliseconds::zero()) {
323 _recoveryTask.schedule(remaining, [this]() { _update(); });
324 return SelectedSource::Internal;
325 }
326 }
327 _receiverUsableSinceMs.reset();
328 _recoveryTask.cancel();
329 if (receiverUsable) {
330 return SelectedSource::Receiver;
331 }
332 if (deviceUsable) {
333 return SelectedSource::Internal;
334 }
335 if (_sourceHealth(_selectedKind)) {
336 return _selectedKind;
337 }
338 return _receiver ? SelectedSource::Receiver : SelectedSource::Internal;
339}

341void PositionManager::_republish()
342{
343 _published = {};
344 if (!_currentHealth) {
345 return;
346 }
347 // Pinned selections need a new observation, not a timeout change.
348 _authorized = _configuration.sourceMode == SourceMode::Automatic ||
349 _currentHealth->observationRevision() != _selectionRevision;
350 const auto accepted = acceptedObservation();
351 if (!accepted) {
352 if (_currentHealth->state() != GPSSourceHealth::State::NoData &&
353 _gcsPositioningError == QGeoPositionInfoSource::NoError) {
354 _setPositioningError(QGeoPositionInfoSource::UpdateTimeoutError);
355 }
356 return;
357 }
358 _setPositioningError(QGeoPositionInfoSource::NoError);
359 _published = {.position = accepted->position.coordinate(),
360 .heading = accepted->heading(),
361 .horizontalAccuracy = accepted->position.attribute(QGeoPositionInfo::HorizontalAccuracy)};
362}

364void PositionManager::_updateStatus()
365{
366 _selectedSource = _currentHealth ? _selectedKind : SelectedSource::None;
367 SourceStatus status = SourceStatus::NoSource;
368 if (_currentHealth) {
369 switch (_currentHealth->state()) {
370 case GPSSourceHealth::State::NoData:
371 status = SourceStatus::WaitingForFix;
372 break;
373 case GPSSourceHealth::State::Usable:
374 status = !_accepted(_selectedKind) ? SourceStatus::Stale
375 : _published.position.isValid() ? SourceStatus::Active
376 : SourceStatus::WaitingForFix;
377 break;
378 case GPSSourceHealth::State::Stale:
379 status = SourceStatus::Stale;
380 break;
381 case GPSSourceHealth::State::Invalid:
382 status = SourceStatus::InvalidFix;
383 break;
384 }
385 } else if (_configuration.sourceMode != SourceMode::ReceiverOnly) {
386 status = _platformStatus;
387 }
388 if (_selectedSource == SelectedSource::Internal &&
389 (_platformStatus == SourceStatus::PermissionDenied || _platformStatus == SourceStatus::BackendUnavailable)) {
390 status = _platformStatus;
391 }
392 _sourceStatus = status;
393}

395void PositionManager::_emitChanges()
396{
397 // Each notified value is recorded before its signal, so a nested update never repeats an older value.
398 const Notified::Selection selection{_selectedSource, _sourceStatus, selectedSourceName()};
399 if (std::exchange(_notified.selection, selection) != selection) {
400 emit selectionChanged();
401 }
402 const qreal accuracy = _published.horizontalAccuracy;
403 if (std::exchange(_notified.publication.horizontalAccuracy, accuracy) != accuracy) {
404 emit gcsPositionHorizontalAccuracyChanged(accuracy);
405 }
406 const qreal heading = _published.heading;
407 const qreal notifiedHeading = std::exchange(_notified.publication.heading, heading);
408 if (notifiedHeading != heading && !(qIsNaN(notifiedHeading) && qIsNaN(heading))) {
409 emit gcsHeadingChanged(heading);
410 }
411 const QGeoCoordinate position = _published.position;
412 if (std::exchange(_notified.publication.position, position) != position) {
413 emit gcsPositionChanged(position);
414 }
415}
