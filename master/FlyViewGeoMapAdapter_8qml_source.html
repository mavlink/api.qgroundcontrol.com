17/// GeoMap-engine drop-in for FlyViewMap: hosts a FlyViewGeoMap and implements
18/// the mapControl contract the Fly View overlays consume (pipState,
19/// isSatelliteMap, toCoordinate/fromCoordinate, zoomLevel, ...). Interactive
20/// map editing is 2D-only; 3D is view-only (see issue #14901).

22 id: root

30 property var toolInsets // Insets for the center viewport area
31 property string mapName

34 // the camera; the camera Connections below keeps it tracking afterwards

36 readonly property real zoomLevel: {
37 // Viewport/FOV referenced so the invokable re-evaluates on resize
38 geoMapControl.camera.viewportSize
39 geoMapControl.camera.verticalFieldOfView
40 return geoMapControl.camera.zoomLevelForDistance(geoMapControl.camera.distance)
41 }
42 readonly property bool isSatelliteMap: _mapTypeSetting.indexOf("Satellite") > -1 || _mapTypeSetting.indexOf("Hybrid") > -1
43 property alias pinchZoomDisabledByVirtualJoysticks: geoMapControl.pinchZoomDisabledByVirtualJoysticks

52 readonly property string _mapTypeSetting: QGroundControl.settingsManager.flightMapSettings.mapType.rawValue

54 // PiP analog of FlyViewMap._adjustMapZoomForPipMode: pull the camera back
55 // 3 zoom levels in the small window for situational context, restore the
56 // full-view distance on swap back. PiP is also too small for a useful 3D
57 // view, so force 2D and restore the previous camera mode with it.
58 property real _fullViewDistance: NaN
59 property int _fullViewCameraMode: GeoMapCamera.Mode2D
60
61 onPipModeChanged: {
62 const camera = geoMapControl.camera
63 if (pipMode) {
64 _fullViewDistance = camera.distance
65 _fullViewCameraMode = camera.mode
66 camera.mode = GeoMapCamera.Mode2D
67 camera.distance = camera.distanceForZoomLevel(camera.zoomLevelForDistance(camera.distance) - 3)
68 } else if (!isNaN(_fullViewDistance)) {
69 camera.mode = _fullViewCameraMode
70 camera.distance = _fullViewDistance
71 _fullViewDistance = NaN
72 }
73 }
74
75 // Breaks the center <-> camera.center cycle: camera moves echo into
76 // center via the Connections below and must not be written back
77 property bool _echoingCameraCenter: false
78
79 onCenterChanged: {
80 if (_echoingCameraCenter || !center || !center.isValid) {
81 return
82 }
83 geoMapControl.camera.center = center
84 }

86 // Screen point -> ground coordinate through the camera pick ray
87 // (clipToViewPort accepted for signature parity, screen points are
88 // always inside the viewport for the fly view use cases)
89 function toCoordinate(point, clipToViewPort) {
90 return geoMapControl.camera.coordinateAtScreenPoint(point)
91 }
92
93 // Geographic coordinate -> screen point; invalid/behind-camera projects
94 // to an off-screen point so distance math degrades instead of throwing
95 function fromCoordinate(coordinate, clipToViewPort) {
96 const screenPos = geoMapControl.scene.screenPositionFor(coordinate)
97 return (screenPos === undefined) ? Qt.point(-1, -1) : screenPos
98 }

100 FlyViewGeoMap {
101 id: geoMapControl
102 anchors.fill: parent
103
104 // The PiP window is too small for free camera panning to be useful
105 keepVehicleCentered: root.pipMode || QGroundControl.settingsManager.flyViewSettings.keepMapCenteredOnVehicle.rawValue
106
107 // Guided-action popup on click (FlyViewMap.onMapClicked parity).
108 // 2D-only: interactive map editing is view-only in 3D (issue #14901),
109 // and a tilted-camera pick can land kilometers from the visual target.
110 // isTopDown: the mode flips to 2D before the tilt animation finishes.
111 onMapClicked: (position) => {
112 if (root.pipMode || geoMapControl.camera.mode !== GeoMapCamera.Mode2D || !geoMapControl.camera.isTopDown) {
113 return
114 }
115 if (!globals.guidedControllerFlyView.guidedUIVisible &&
116 (globals.guidedControllerFlyView.showGotoLocation || globals.guidedControllerFlyView.showOrbit ||
117 globals.guidedControllerFlyView.showROI || globals.guidedControllerFlyView.showSetHome ||
118 globals.guidedControllerFlyView.showSetEstimatorOrigin)) {
119
120 const clickCoord = root.toCoordinate(Qt.point(position.x, position.y), false /* clipToViewPort */)
121 if (!clickCoord.isValid) {
122 return
123 }
124 const windowPosition = root.mapToItem(globals.parent, position.x, position.y)
125 const dropPanel = mapClickDropPanelComponent.createObject(mainWindow, { mapClickCoord: clickCoord, clickRect: Qt.rect(windowPosition.x, windowPosition.y, 0, 0) })
126 dropPanel.open()
127 }
128 }
129 }
