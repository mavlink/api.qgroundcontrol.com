54void setRtspTransport(bool overTcp)
55{
56 static const QByteArray userTransport = qgetenv(kRtspTransportEnv);
57 const QByteArray transport = overTcp ? QByteArrayLiteral("tcp") : userTransport;

70QByteArray rtpSdp(const QString& host, int port, QByteArrayView encoding)

115std::optional<QtMultimediaVideoReceiver::Source> QtMultimediaVideoReceiver::playerSource(const QString& uri)
116{
117 const QUrl url(uri);
118 if (uri.isEmpty() || !url.isValid() || url.scheme().isEmpty()) {
119 return std::nullopt;
120 }
121
122 using VideoStreamUri::Transport;
123 const std::optional<Transport> transport = VideoStreamUri::transportFor(url);
124 if ((transport == Transport::RtpUdpH264) || (transport == Transport::RtpUdpH265)) {
125 if (url.port() <= 0) {
126 return std::nullopt;
127 }
128 const QByteArrayView encoding = (transport == Transport::RtpUdpH265) ? "H265" : "H264";
129 return Source{.url = QUrl(QStringLiteral("stream.sdp")),
130 .sdp = rtpSdp(url.host(), url.port(), encoding),
131 .streamsDescribed = true};
132 }
133 if (transport == Transport::MpegTsUdp) {
134 QUrl udp(url);
135 udp.setScheme(QStringLiteral("udp"));
136 return Source{udp, {}};
137 }
138 if ((transport == Transport::Rtsp) && (url.scheme().compare(QStringLiteral("rtspt"), Qt::CaseInsensitive) == 0)) {
139 QUrl rtsp(url);
140 rtsp.setScheme(QStringLiteral("rtsp"));
141 return Source{.url = rtsp, .sdp = {}, .rtspOverTcp = true, .streamsDescribed = true};
142 }
143 return Source{.url = url, .sdp = {}, .streamsDescribed = (transport == Transport::Rtsp)};
144}

239void QtMultimediaVideoReceiver::_open()
240{
241 _openPending = false;
242 // Stopped or restarted meanwhile: that _start() reported or opens it.
243 const std::optional<Source> source = playerSource(config().uri);
244 if (_player || !active() || !source || _closing) {
245 return;
246 }
247
248 _sessionConfig = config();
249 // Before the player opens its sockets, so its own timeout can't run out sooner than this one.
250 _sinceOpened.start();

252 QPlaybackOptions options;
253 options.setNetworkTimeout(config().timeout);
254 if (config().lowLatency) {
255 options.setPlaybackIntent(QPlaybackOptions::PlaybackIntent::LowLatencyStreaming);
256 }
257 if (kShortProbe && source->streamsDescribed) {
258 options.setProbeSize(kDescribedStreamProbeBytes);
259 }
260 _player->setPlaybackOptions(options);

262 (void) connect(_player, &QMediaPlayer::errorOccurred, this,

264 _bindVideoSink();

266 setRtspTransport(source->rtspOverTcp);

283{

306bool QtMultimediaVideoReceiver::_awaitingSender() const
307{
308 const std::optional<VideoStreamUri::Transport> transport = VideoStreamUri::transportFor(QUrl(_sessionConfig.uri));
309 // A session that ends sooner than its timeout didn't run out of patience: its port is taken, or it can't play.
310 return transport && VideoStreamUri::listens(*transport) &&

315void QtMultimediaVideoReceiver::_fail(const QString& reason)
316{
317 if (!_player) {
318 return;
319 }
320 if (_awaitingSender()) {
321 // FFmpeg gives up on a silent port; a backoff would leave it unbound when the stream starts.
322 qCDebug(QtMultimediaVideoReceiverLog) << name() << "no stream yet, listening on:" << reason;
323 _start();
324 return;
325 }
326 _stopSession();
327 const StreamSupervisor::FailureOutcome outcome =
328 _supervisor->sessionFailed(reason, StreamSupervisor::Failure::Transient, config().autoReconnect);
329 if (outcome.halted) {
330 qCDebug(QtMultimediaVideoReceiverLog) << name() << "failed:" << reason;
331 emit errorOccurred(tr("Video stream failed: %1").arg(reason));
332 } else {
333 qCDebug(QtMultimediaVideoReceiverLog)
334 << name() << "failed, reconnecting in" << outcome.reconnectDelay << ":" << reason;
335 }

372{

386void QtMultimediaVideoReceiver::_recordFrame(const QVideoFrame& frame)
387{
388 if (!_frameInput || !_recorder->active()) {
389 return;
390 }
391 if (!_sinceRecordingStarted.isValid()) {
392 _sinceRecordingStarted.start();
393 }
394 // A frame of its own: its times are the recording's, not the stream's, and copies of a frame share them.
395 QVideoFrame recorded = SharedVideoFrame::withOwnMetadata(frame);
396 const qint64 recordedUs = recordingTimeUs(_lastRecordedUs, _lastStreamUs, frame.startTime(),
397 _sinceRecordingStarted.nsecsElapsed() / 1000);
398 recorded.setStartTime(recordedUs);
399 recorded.setEndTime(-1);
400 // One the encoder can't take yet is dropped, as the display drops a frame it can't show in time.
401 if (_frameInput->sendVideoFrame(recorded)) {
402 _lastRecordedUs = recordedUs;
403 _lastStreamUs = frame.startTime();
404 }
405}

407qint64 QtMultimediaVideoReceiver::recordingTimeUs(qint64 lastRecordedUs, qint64 lastStreamUs, qint64 streamUs,
408 qint64 arrivalUs)
409{
410 if (lastRecordedUs < 0) {
411 return arrivalUs;
412 }
413 // Frames queued while the GUI thread was busy arrive together, so they take the stream's spacing.
414 const qint64 streamStepUs = ((streamUs >= 0) && (lastStreamUs >= 0)) ? (streamUs - lastStreamUs) : -1;
415 const qint64 timeUs =
416 ((streamStepUs > 0) && (streamStepUs < kMaxStreamStepUs)) ? (lastRecordedUs + streamStepUs) : arrivalUs;
417 return std::max(timeUs, lastRecordedUs + kMinFrameStepUs);
418}

420void QtMultimediaVideoReceiver::_sample()
421{
422 if (!_player) {
423 return;
424 }
425 const qint64 timeoutMs = std::chrono::duration_cast<std::chrono::milliseconds>(config().timeout).count() +
426 ((_frames == 0) ? kFirstFrameGraceMs : 0);
427 if (videoSink() && (_sinceFrame.elapsed() > timeoutMs)) {
428 _fail(tr("no video for %1 s").arg(timeoutMs / 1000));
429 return;
430 }
431 if (_frames == 0) {
432 return;
433 }
434
435 VideoStreamStats stats = _statsSampler.stats();
436 stats.decoder = QStringLiteral("Qt Multimedia");
437 // Qt's FFmpeg backend hands over GPU frames only from a hardware decoder; a hardware decoder's frames copied to
438 // memory read as software.
439 stats.hardwareDecoder = _gpuFrames;
440 stats.gpuPath = _gpuFrames ? QStringLiteral("GPU") : QStringLiteral("CPU");
441 const auto codec = _player->metaData().value(QMediaMetaData::VideoCodec).value<QMediaFormat::VideoCodec>();
442 stats.codec =
443 (codec == QMediaFormat::VideoCodec::Unspecified) ? QString() : QMediaFormat::videoCodecDescription(codec);
444 // The player doesn't tell how many bytes it received, so there is no bitrate.
445 VideoStreamStatsMeter::Counters counters;
446 counters.framesDisplayed = _frames;
447 if (_statsSampler.sample(counters, stats, nullptr)) {
448 setStats(_statsSampler.stats());
449 }
