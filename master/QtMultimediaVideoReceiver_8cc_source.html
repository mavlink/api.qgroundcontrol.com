51QByteArray rtpSdp(const QString& host, int port, QByteArrayView encoding)
52{
53 const QByteArray address = (host.isEmpty() ? QStringLiteral("0.0.0.0") : host).toLatin1();
54 return QByteArrayLiteral("v=0\r\n") + "o=- 0 0 IN IP4 " + address + "\r\ns=QGroundControl\r\nc=IN IP4 " + address +

96std::optional<QtMultimediaVideoReceiver::Source> QtMultimediaVideoReceiver::playerSource(const QString& uri)
97{
98 const QUrl url(uri);
99 if (uri.isEmpty() || !url.isValid() || url.scheme().isEmpty()) {
100 return std::nullopt;
101 }
102 const QString scheme = url.scheme().toLower();
103 if ((scheme == QStringLiteral("udp")) || (scheme == QStringLiteral("udp265"))) {
104 if (url.port() <= 0) {
105 return std::nullopt;
106 }
107 const QByteArrayView encoding = (scheme == QStringLiteral("udp265")) ? "H265" : "H264";
108 return Source{QUrl(QStringLiteral("stream.sdp")), rtpSdp(url.host(), url.port(), encoding)};
109 }
110 if (scheme == QStringLiteral("mpegts")) {
111 QUrl udp(url);
112 udp.setScheme(QStringLiteral("udp"));
113 return Source{udp, {}};
114 }
115 if (scheme == QStringLiteral("rtspt")) {
116 QUrl rtsp(url);
117 rtsp.setScheme(QStringLiteral("rtsp"));
118 return Source{rtsp, {}, true};
119 }
120 return Source{url, {}};
121}

189void QtMultimediaVideoReceiver::_start()
190{
191 _stopSession();
192 if (!active() || config().uri.isEmpty()) {
193 _supervisor->stop();
194 return;
195 }
196 if (!playerSource(config().uri)) {
197 qCWarning(QtMultimediaVideoReceiverLog) << name() << "can't play" << config().uri;

200 return;

211void QtMultimediaVideoReceiver::_open()
212{
213 _openPending = false;
214 // Stopped or restarted meanwhile: that _start() reported or opens it.
215 const std::optional<Source> source = playerSource(config().uri);
216 if (_player || !active() || !source || _closing) {
217 return;
218 }
219
220 _sessionConfig = config();
221 _player = new QMediaPlayer(this);
222 QPlaybackOptions options;
223 options.setNetworkTimeout(config().timeout);
224 if (config().lowLatency) {
225 options.setPlaybackIntent(QPlaybackOptions::PlaybackIntent::LowLatencyStreaming);
226 }
227 _player->setPlaybackOptions(options);
228 (void) connect(_player, &QMediaPlayer::mediaStatusChanged, this, &QtMultimediaVideoReceiver::_onMediaStatusChanged);
229 (void) connect(_player, &QMediaPlayer::errorOccurred, this,
230 [this](QMediaPlayer::Error, const QString& errorString) { _fail(errorString); });
231 _bindVideoSink();
232 if (source->url.scheme().compare(QStringLiteral("rtsp"), Qt::CaseInsensitive) == 0) {
233 setRtspTransport(source->rtspOverTcp);
234 }
235 if (source->sdp.isEmpty()) {
236 _player->setSource(source->url);
237 } else {
238 auto* const sdp = new QBuffer(_player);
239 sdp->setData(source->sdp);
240 (void) sdp->open(QIODevice::ReadOnly);
241 _player->setSourceDevice(sdp, source->url);
242 }
243 qCDebug(QtMultimediaVideoReceiverLog) << name() << "playing" << config().uri;
244 _sinceFrame.start();
245 _sampleTimer.start();
246 _player->play();
247}

249void QtMultimediaVideoReceiver::_stopSession()
250{
251 _sampleTimer.stop();
252 if (_player) {
253 QMediaPlayer* const player = std::exchange(_player, nullptr);
254 (void) disconnect(player, nullptr, this, nullptr);
255 // Unloaded before it goes: that cancels a network read first, which stopping it or detaching its sink would
256 // wait out on this thread, for up to the network timeout per socket. Later, as this may run in one of its
257 // signals. The output keeps the last frame (VideoOutput.KeepLastFrame).
258 QMetaObject::invokeMethod(
259 player,
260 [player] {
261 player->setSource(QUrl());
262 delete player;
263 },
264 Qt::QueuedConnection);
265 }
266 (void) _statsSampler.reset();
267 _frames = 0;
268 _gpuFrames = false;
269 setDecoding(false);
270 setStats(_statsSampler.stats());

326void QtMultimediaVideoReceiver::_onVideoFrame(const QVideoFrame& frame)
327{
328 // An invalid frame clears the display; frames can also arrive queued from a session that ended.
329 if (!_player || !frame.isValid()) {

331 }

333 ++_frames;

336 _supervisor->sessionStreaming();

341void QtMultimediaVideoReceiver::_recordFrame(const QVideoFrame& frame)
342{
343 if (!_frameInput || !_recorder->active()) {
344 return;
345 }
346 if (!_sinceRecordingStarted.isValid()) {
347 _sinceRecordingStarted.start();
348 }
349 // A frame of its own: its times are the recording's, not the stream's, and copies of a frame share them.
350 QVideoFrame recorded = SharedVideoFrame::withOwnMetadata(frame);
351 const qint64 recordedUs = recordingTimeUs(_lastRecordedUs, _lastStreamUs, frame.startTime(),
352 _sinceRecordingStarted.nsecsElapsed() / 1000);
353 recorded.setStartTime(recordedUs);
354 recorded.setEndTime(-1);
355 // One the encoder can't take yet is dropped, as the display drops a frame it can't show in time.
356 if (_frameInput->sendVideoFrame(recorded)) {
357 _lastRecordedUs = recordedUs;
358 _lastStreamUs = frame.startTime();
359 }
360}

362qint64 QtMultimediaVideoReceiver::recordingTimeUs(qint64 lastRecordedUs, qint64 lastStreamUs, qint64 streamUs,
363 qint64 arrivalUs)
364{
365 if (lastRecordedUs < 0) {
366 return arrivalUs;
367 }
368 // Frames queued while the GUI thread was busy arrive together, so they take the stream's spacing.
369 const qint64 streamStepUs = ((streamUs >= 0) && (lastStreamUs >= 0)) ? (streamUs - lastStreamUs) : -1;
370 const qint64 timeUs =
371 ((streamStepUs > 0) && (streamStepUs < kMaxStreamStepUs)) ? (lastRecordedUs + streamStepUs) : arrivalUs;
372 return std::max(timeUs, lastRecordedUs + kMinFrameStepUs);
373}

375void QtMultimediaVideoReceiver::_sample()
376{
377 if (!_player) {
378 return;
379 }
380 const qint64 timeoutMs = std::chrono::duration_cast<std::chrono::milliseconds>(config().timeout).count() +
381 ((_frames == 0) ? kFirstFrameGraceMs : 0);
382 if (_videoSink && (_sinceFrame.elapsed() > timeoutMs)) {
383 _fail(tr("no video for %1 s").arg(timeoutMs / 1000));
384 return;
385 }
386 if (_frames == 0) {
387 return;
388 }
389
390 VideoStreamStats stats = _statsSampler.stats();
391 stats.decoder = QStringLiteral("Qt Multimedia");
392 // Qt's FFmpeg backend hands over GPU frames only from a hardware decoder; a hardware decoder's frames copied to
393 // memory read as software.
394 stats.hardwareDecoder = _gpuFrames;
395 stats.gpuPath = _gpuFrames ? QStringLiteral("GPU") : QStringLiteral("CPU");
396 const auto codec = _player->metaData().value(QMediaMetaData::VideoCodec).value<QMediaFormat::VideoCodec>();
397 stats.codec =
398 (codec == QMediaFormat::VideoCodec::Unspecified) ? QString() : QMediaFormat::videoCodecDescription(codec);
399 // The player doesn't tell how many bytes it received, so there is no bitrate.
400 VideoStreamStatsMeter::Counters counters;
401 counters.framesDisplayed = _frames;
402 if (_statsSampler.sample(counters, stats, nullptr)) {
403 setStats(_statsSampler.stats());
404 }
