40GPSPositionReport::FixType navigationFix(uint8_t wireFix, uint8_t flags, bool hasCarrierFlags)
41{
42 using Fix = GPSPositionReport::FixType;
43 if (!(flags & UBX_RX_NAV_PVT_FLAGS_GNSSFIXOK)) {
44 return Fix::NoFix;
45 }
46 switch (wireFix) {
47 case 0:
48 case 5:
49 return Fix::NoFix;
50 case 1:
51 return Fix::Extrapolated;
52 case 2:
53 // Correction flags cannot establish a three-dimensional navigation solution.
54 return Fix::Fix2D;
55 case 3:
56 case 4:
57 break;
58 default:
59 return Fix::Unknown;
60 }
61 if (hasCarrierFlags) {
62 const uint8_t carrier = flags >> 6;
63 if (carrier == 1) {
64 return Fix::RTKFloat;
65 }
66 if (carrier == 2) {
67 return Fix::RTKFixed;
68 }
69 }
70 return flags & UBX_RX_NAV_PVT_FLAGS_DIFFSOLN ? Fix::Differential : Fix::Fix3D;
71}

79int GPSNativeUBX::parseChar(uint8_t byte)
80{
81 if (_rtcm_parsing && (_frameDecoder.idle() || _rtcm_parsing->hasPartialFrame())) {
82 if (_rtcm_parsing->addByte(byte)) {
83 if (_rtcm_parsing->valid()) {
84 gotRTCMMessage(_rtcm_parsing->frame().data(), _rtcm_parsing->frame().size());
85 }
86 _rtcm_parsing->reset();
87 return 0;
88 }
89 if (_rtcm_parsing->hasPartialFrame()) {
90 return 0;
91 }
92 }
93 // Keep the completed frame local: synchronous logging can reenter or reset the parser.
94 const auto frame = _frameDecoder.consume(byte);
95 if (!frame) {
96 return 0;
97 }
98 if (_rtcm_parsing) {
99 _rtcm_parsing->reset();
100 }
101 return decodeValidatedPayload(frame->message, std::span(frame->payload).first(frame->length));
102}

104bool GPSNativeUBX::payloadRxInit(uint16_t message, std::span<const uint8_t> payload)
105{
106 auto state = UBX_RXMSG_HANDLE;
107
108 switch (message) {
109 case UBX_MSG_CFG_TMODE3:
110 if (!_timeModeReadbackPending) {
111 state = UBX_RXMSG_IGNORE;
112 }
113 break;
114 case UBX_MSG_CFG_VALGET:
115 if (!_controller.readbackPending()) {
116 state = UBX_RXMSG_IGNORE;
117 } else if (payload.size() < 4 || payload.size() > UBX::MAX_CONTROL_PAYLOAD_SIZE) {
118 state = UBX_RXMSG_ERROR_LENGTH;
119 }
120 break;
121 case UBX_MSG_MON_COMMS:
122 break;
123
124 case UBX_MSG_NAV_PVT:
125 if (!_decodeNavigation) {
126 state = UBX_RXMSG_IGNORE; // ignore if not _decodeNavigation
127
128 } else if (!_use_nav_pvt) {
129 state = UBX_RXMSG_DISABLE; // disable if not using NAV-PVT
130 }
131
132 break;
133
134 case UBX_MSG_INF_DEBUG:
135 case UBX_MSG_INF_ERROR:
136 case UBX_MSG_INF_NOTICE:
137 case UBX_MSG_INF_WARNING:
138 if (payload.size() >= UBX::MAX_CONTROL_PAYLOAD_SIZE) {
139 state = UBX_RXMSG_ERROR_LENGTH;
140 }
141
142 break;
143
144 case UBX_MSG_NAV_POSLLH:
145 if (!_decodeNavigation) {
146 state = UBX_RXMSG_IGNORE; // ignore if not _decodeNavigation
147
148 } else if (_use_nav_pvt) {
149 state = UBX_RXMSG_DISABLE; // disable if using NAV-PVT instead
150 }
151
152 break;
153
154 case UBX_MSG_NAV_SOL:
155 if (!_decodeNavigation) {
156 state = UBX_RXMSG_IGNORE; // ignore if not _decodeNavigation
157
158 } else if (_use_nav_pvt) {
159 state = UBX_RXMSG_DISABLE; // disable if using NAV-PVT instead
160 }
161
162 break;
163
164 case UBX_MSG_NAV_STATUS:
165 if (!_decodeNavigation) {
166 state = UBX_RXMSG_IGNORE; // ignore if not _decodeNavigation
167 }
168
169 break;
170
171 case UBX_MSG_NAV_DOP:
172 if (!_decodeNavigation) {
173 state = UBX_RXMSG_IGNORE; // ignore if not _decodeNavigation
174 }
175
176 break;
177
178 case UBX_MSG_NAV_RELPOSNED:
179 if (!_decodeNavigation) {
180 state = UBX_RXMSG_IGNORE; // ignore if not _decodeNavigation
181 }
182
183 break;
184
185 case UBX_MSG_NAV_DAHEADING:
186 if (!_decodeNavigation) {
187 state = UBX_RXMSG_IGNORE; // ignore if not _decodeNavigation
188 }
189
190 break;
191
192 case UBX_MSG_NAV_HPPOSLLH:
193 if (!_decodeNavigation) {
194 state = UBX_RXMSG_IGNORE; // ignore if not _decodeNavigation
195 }
196
197 break;
198
199 case UBX_MSG_NAV_TIMEUTC:
200 if (!_decodeNavigation) {
201 state = UBX_RXMSG_IGNORE; // ignore if not _decodeNavigation
202
203 } else if (_use_nav_pvt) {
204 state = UBX_RXMSG_DISABLE; // disable if using NAV-PVT instead
205 }
206
207 break;
208
209 case UBX_MSG_NAV_SAT:
210 case UBX_MSG_NAV_SVINFO:
211 if (_satellite_info == nullptr) {
212 state = UBX_RXMSG_DISABLE; // disable if sat info not requested
213
214 } else if (!_decodeNavigation) {
215 state = UBX_RXMSG_IGNORE; // ignore if not _decodeNavigation
216
217 } else {
218 *_satellite_info = {}; // initialize sat info
219 }
220
221 break;
222
223 case UBX_MSG_NAV_SVIN:
224 break;
225
226 case UBX_MSG_NAV_VELNED:
227 if (!_decodeNavigation) {
228 state = UBX_RXMSG_IGNORE; // ignore if not _decodeNavigation
229
230 } else if (_use_nav_pvt) {
231 state = UBX_RXMSG_DISABLE; // disable if using NAV-PVT instead
232 }
233
234 break;
235
236 case UBX_MSG_MON_VER:
237 break; // unconditionally handle this message
238
239 case UBX_MSG_MON_HW:
240 if (!_decodeNavigation) {
241 state = UBX_RXMSG_IGNORE; // ignore if not _decodeNavigation
242 }
243
244 break;
245
246 case UBX_MSG_MON_RF:
247 if (!_decodeNavigation) {
248 state = UBX_RXMSG_IGNORE; // ignore if not _decodeNavigation
249 }
250
251 break;
252
253 case UBX_MSG_SEC_SIG:
254 if (!_decodeNavigation) {
255 state = UBX_RXMSG_IGNORE;
256 }
257
258 break;
259
260 case UBX_MSG_RXM_RTCM:
261 if (!_decodeNavigation) {
262 state = UBX_RXMSG_IGNORE; // ignore if not _decodeNavigation
263 }
264
265 break;
266
267 case UBX_MSG_RXM_COR:
268 if (!_decodeNavigation) {
269 state = UBX_RXMSG_IGNORE;
270 }
271
272 break;
273
274 case UBX_MSG_ACK_ACK:
275 if (!_controller.awaitingAcknowledgement()) {
276 state = UBX_RXMSG_IGNORE; // No command is awaiting acknowledgement.
277 }
278
279 break;
280
281 case UBX_MSG_ACK_NAK:
282 if (!_controller.awaitingAcknowledgement() && !_controller.readbackPending() && !_timeModeReadbackPending) {
283 state = UBX_RXMSG_IGNORE; // No command is awaiting acknowledgement.
284 }
285
286 break;
287
288 default:
289 state = UBX_RXMSG_DISABLE; // disable all other messages
290 break;
291 }
292
293 if (state == UBX_RXMSG_DISABLE) {
294 _pendingDisableMessage = message;
295 }
296 return state == UBX_RXMSG_HANDLE;
297}

299void GPSNativeUBX::decodeNavSat(std::span<const uint8_t> payload)
300{
301 auto decoded_header = UBX::MessageCodec<ubx_payload_rx_nav_sat_part1_t>::block(payload);
302 if (!decoded_header) {
303 return;
304 }
305 const auto header = *decoded_header;
306 _satellite_info->count = std::min<size_t>(header.numSvs, _satellite_info->entries.size());
307 constexpr GPSConstellation systems[] = {
308 GPSConstellation::GPS, GPSConstellation::SBAS, GPSConstellation::Galileo, GPSConstellation::BeiDou,
309 GPSConstellation::Unknown, GPSConstellation::QZSS, GPSConstellation::GLONASS, GPSConstellation::NavIC};
310 for (size_t index = 0; index < _satellite_info->count; ++index) {
311 auto decoded_wire = UBX::MessageCodec<ubx_payload_rx_nav_sat_part2_t>::block(
312 payload, UBX::WIRE_SIZE<std::remove_cv_t<decltype(header)>> +
313 index * UBX::WIRE_SIZE<ubx_payload_rx_nav_sat_part2_t>);
314 if (!decoded_wire) {
315 return;
316 }
317 const auto wire = *decoded_wire;
318 auto& satellite = _satellite_info->entries[index];
319 satellite.constellation = wire.gnssId < std::size(systems) ? systems[wire.gnssId] : GPSConstellation::Unknown;
320 satellite.id = satellite.prn = wire.svId;
321 satellite.used = (wire.flags & 8) != 0;
322 satellite.elevation = wire.elev;
323 satellite.azimuth = wire.azim;
324 satellite.signal = wire.cno;
325 }

353void GPSNativeUBX::decodeMonVer(std::span<const uint8_t> payload)
354{
355 _board = Board::unknown;
356 _is_m8p = false;
357 _proto_ver_27_or_higher = false;
358 _model_name[0] = '\0';
359 _firmware_version[0] = '\0';
360 auto decoded_payload_rx_mon_ver_part1 = UBX::MessageCodec<ubx_payload_rx_mon_ver_part1_t>::block(payload);
361 if (!decoded_payload_rx_mon_ver_part1) {
362 return;
363 }
364 auto payload_rx_mon_ver_part1 = *decoded_payload_rx_mon_ver_part1;
365 // The protocol specifies these as nul-terminated strings, but the terminator comes
366 // from the device, so enforce it before anything walks the field.
367 payload_rx_mon_ver_part1.swVersion[sizeof(payload_rx_mon_ver_part1.swVersion) - 1] = 0;
368 payload_rx_mon_ver_part1.hwVersion[sizeof(payload_rx_mon_ver_part1.hwVersion) - 1] = 0;
369 memcpy(_firmware_version, payload_rx_mon_ver_part1.swVersion, sizeof(_firmware_version));
370
371 // Device detection (See
372 // https://forum.u-blox.com/index.php/9432/need-help-decoding-ubx-mon-ver-hardware-string)
373 static constexpr struct
374 {
375 char hw_version[9];
376 Board board;
377 } known_boards[] = {
378 {"00040005", Board::u_blox5}, {"00040007", Board::u_blox6}, {"00070000", Board::u_blox7},
379 {"00080000", Board::u_blox8}, {"00190000", Board::u_blox9}, {"000A0000", Board::u_blox10},
380 {"000B0000", Board::u_blox_X20},
381 };
382
383 bool known = false;
384
385 for (const auto& known_board : known_boards) {
386 if (strncmp((const char*) payload_rx_mon_ver_part1.hwVersion, known_board.hw_version,
387 sizeof(payload_rx_mon_ver_part1.hwVersion)) == 0) {
388 _board = known_board.board;
389 known = true;
390 break;
391 }
392 }
393
394 if (!known) {
395 log(GPSProtocolLogLevel::Warning, "unknown board hw: %s", payload_rx_mon_ver_part1.hwVersion);
396 }
397 _proto_ver_27_or_higher = _board == Board::u_blox9 || _board == Board::u_blox10 || _board == Board::u_blox_X20;
398 _timeModeUnsupported =
399 _board == Board::u_blox5 || _board == Board::u_blox6 || _board == Board::u_blox7 || _board == Board::u_blox10;
400 for (size_t offset = UBX::WIRE_SIZE<ubx_payload_rx_mon_ver_part1_t>; offset < payload.size();
401 offset += UBX::WIRE_SIZE<ubx_payload_rx_mon_ver_part2_t>) {
402 auto decoded_payload_rx_mon_ver_part2 =
403 UBX::MessageCodec<ubx_payload_rx_mon_ver_part2_t>::block(payload, offset);
404 if (!decoded_payload_rx_mon_ver_part2) {
405 return;
406 }
407 auto payload_rx_mon_ver_part2 = *decoded_payload_rx_mon_ver_part2;
408 // Part 2 complete: decode Part 2 buffer
409 // Same as above: the protocol specifies a nul-terminated string, the device provides
410 // the terminator, so enforce it before strstr() walks the field.
411 payload_rx_mon_ver_part2.extension[sizeof(payload_rx_mon_ver_part2.extension) - 1] = 0;
412
413 // "FWVER=" Firmware of product category and version
414 const char* fwver_str = strstr((const char*) payload_rx_mon_ver_part2.extension, "FWVER=");
415
416 if (fwver_str != nullptr) {
417 if (strncmp(fwver_str, "FWVER=SPG ", 10) == 0 || strncmp(fwver_str, "FWVER=HPS ", 10) == 0) {
418 _timeModeUnsupported = true;
419 }
420 strncpy(_firmware_version, fwver_str + strlen("FWVER="), sizeof(_firmware_version) - 1);
421 _firmware_version[sizeof(_firmware_version) - 1] = '\0';
422 log(GPSProtocolLogLevel::Debug, "u-blox firmware version: %s", fwver_str + strlen("FWVER="));
423
424 // Check if its a ZED-F9P-15B
425 if ((_board == Board::u_blox9) && strstr(fwver_str, "HPGL1L5")) {
426 _board = Board::u_blox9_F9P_L1L5;
427 }
428 }
429
430 // "PROTVER=" Supported protocol version.
431 const char* protver_str = strstr((const char*) payload_rx_mon_ver_part2.extension, "PROTVER=");
432
433 if (protver_str != nullptr) {
434 log(GPSProtocolLogLevel::Debug, "u-blox protocol version: %s", protver_str + strlen("PROTVER="));
435 const std::string_view version(protver_str + strlen("PROTVER="));
436 const auto major = NMEA::number<unsigned>(version.substr(0, version.find('.')));
437 if (!major || *major == 0) {
438 _board = Board::unknown;
439 return;
440 }
441 _proto_ver_27_or_higher = *major >= 27;
442 if (*major < 20) {
443 _timeModeUnsupported = true;
444 }
445 }
446
447 // "MOD=" Module identification. Set in production.
448 const char* mod_str = strstr((const char*) payload_rx_mon_ver_part2.extension, "MOD=");
449
450 if (mod_str != nullptr) {
451 if (strncmp(mod_str, "MOD=NEO-M8P-0", 13) == 0 || strcmp(mod_str, "MOD=NEO-M8N") == 0 ||
452 strcmp(mod_str, "MOD=NEO-M9N") == 0) {
453 _timeModeUnsupported = true;
454 }
455 strncpy(_model_name, mod_str + strlen("MOD="), sizeof(_model_name) - 1);
456 _model_name[sizeof(_model_name) - 1] = '\0';
457 _is_m8p = strstr(mod_str, "M8P") != nullptr;
458 // in case of u-blox9 family, check if it's an F9P
459 if (_board == Board::u_blox9) {
460 if (strstr(mod_str, "F9P")) {
461 _board = Board::u_blox9_F9P_L1L2;
462 }
463
464 } else if (_board == Board::u_blox10) {
465 if (strstr(mod_str, "DAN-F10N")) {
466 _board = Board::u_blox10_L1L5;
467 }
468 }
469
470 log(GPSProtocolLogLevel::Debug, "u-blox module: %s", mod_str + strlen("MOD="));
471 }
472 }
473}

475int // 0 = no message handled, 1 = message handled, 2 = sat info message handled
476GPSNativeUBX::payloadRxDone(uint16_t message, std::span<const uint8_t> payload, GPSNativePositionReport& position)
477{
478 int ret = 0;
479
480 // handle message
481 switch (message) {
482 case UBX_MSG_NAV_PVT: {
483 const auto decoded_payload_rx_nav_pvt = UBX::MessageCodec<ubx_payload_rx_nav_pvt_t>::decode(payload);
484 if (!decoded_payload_rx_nav_pvt) {
485 break;
486 }
487 const auto& payload_rx_nav_pvt = *decoded_payload_rx_nav_pvt;
488 position.fix_type = navigationFix(payload_rx_nav_pvt.fixType, payload_rx_nav_pvt.flags, true);
489 position.vel_ned_valid = velocityValid(position.fix_type);
490
491 position.satellites_used = payload_rx_nav_pvt.numSV;
492
493 if (_assembleEpochs ? !_epochHasHighPrecision : position.fix_type != GPSPositionReport::FixType::RTKFixed) {
494 // When RTK is active and solid (fix=6), these values will be filled by HPPOSLLH:
495 position.latitude_deg = payload_rx_nav_pvt.lat * UBX::DEGREES_PER_COORDINATE;
496 position.longitude_deg = payload_rx_nav_pvt.lon * UBX::DEGREES_PER_COORDINATE;
497 position.altitude_msl_m = payload_rx_nav_pvt.hMSL * 1e-3;
498 position.altitude_ellipsoid_m = payload_rx_nav_pvt.height * 1e-3;
499
500 position.eph = static_cast<float>(payload_rx_nav_pvt.hAcc) * 1e-3f;
501 position.accuracy_timestamp = nowUs();
502 position.epv = static_cast<float>(payload_rx_nav_pvt.vAcc) * 1e-3f;
503
504 _got_posllh = true;
505 }
506
507 position.speedAccuracyMetersPerSecond = static_cast<float>(payload_rx_nav_pvt.sAcc) * 1e-3f;
508
509 position.vel_m_s = static_cast<float>(payload_rx_nav_pvt.gSpeed) * 1e-3f;
510
511 position.vel_n_m_s = static_cast<float>(payload_rx_nav_pvt.velN) * 1e-3f;
512 position.vel_e_m_s = static_cast<float>(payload_rx_nav_pvt.velE) * 1e-3f;
513 position.vel_d_m_s = static_cast<float>(payload_rx_nav_pvt.velD) * 1e-3f;
514
515 position.cog_rad = static_cast<float>(payload_rx_nav_pvt.headMot) * GPS_DEG_TO_RAD * 1e-5f;
516 position.courseAccuracyRadians = static_cast<float>(payload_rx_nav_pvt.headAcc) * GPS_DEG_TO_RAD * 1e-5f;
517
518 // Check if time and date fix flags are good
519 if ((payload_rx_nav_pvt.valid & UBX_RX_NAV_PVT_VALID_VALIDDATE) &&
520 (payload_rx_nav_pvt.valid & UBX_RX_NAV_PVT_VALID_VALIDTIME) &&
521 (payload_rx_nav_pvt.valid & UBX_RX_NAV_PVT_VALID_FULLYRESOLVED)) {
522 tm timeinfo{};
523 timeinfo.tm_year = payload_rx_nav_pvt.year - 1900;
524 timeinfo.tm_mon = payload_rx_nav_pvt.month - 1;
525 timeinfo.tm_mday = payload_rx_nav_pvt.day;
526 timeinfo.tm_hour = payload_rx_nav_pvt.hour;
527 timeinfo.tm_min = payload_rx_nav_pvt.min;
528 timeinfo.tm_sec = payload_rx_nav_pvt.sec;
529 position.time_utc_usec = timeFromUtc(timeinfo, payload_rx_nav_pvt.nano);
530
531 } else {
532 // The struct is reused across messages, so without this a receiver
533 // that lost time in a reset keeps reporting the last time it knew.
534 // 0 is the defined "unavailable" value.
535 position.time_utc_usec = 0;
536 }
537
538 position.timestamp = nowUs();
539 _last_timestamp_time = position.timestamp;
540
541 _got_velned = true;
542
543 ret = 1;
544 break;
545 }
546 case UBX_MSG_INF_DEBUG:
547 case UBX_MSG_INF_NOTICE:
548 break;
549
550 case UBX_MSG_INF_ERROR:
551 case UBX_MSG_INF_WARNING: {
552 const std::string_view text(reinterpret_cast<const char*>(payload.data()), payload.size());
553 log(GPSProtocolLogLevel::Warning, "ubx msg: %.*s", int(text.size()), text.data());
554
555 if (text.starts_with("txbuf")) {
556 _comms_request_pending = true;
557 }
558 } break;
559
560 case UBX_MSG_MON_COMMS:
561 logCommsDiagnostics(payload);
562 break;
563 case UBX_MSG_CFG_TMODE3:
564 if (_timeModeReadbackPending && payload[0] == 0 && payload[1] == 0) {
565 _timeModeReadback = payload[2];
566 _timeModeReadbackReady = true;
567 }
568 break;
569 case UBX_MSG_CFG_VALGET:
570 if (const auto values = UBX::decodeConfigurationValues(payload)) {
571 _controller.accept(*values);
572 }
573 break;
574
575 case UBX_MSG_NAV_POSLLH: {
576 const auto decoded_payload_rx_nav_posllh = UBX::MessageCodec<ubx_payload_rx_nav_posllh_t>::decode(payload);
577 if (!decoded_payload_rx_nav_posllh) {
578 break;
579 }
580 const auto& payload_rx_nav_posllh = *decoded_payload_rx_nav_posllh;
581
582 position.latitude_deg = payload_rx_nav_posllh.lat * UBX::DEGREES_PER_COORDINATE;
583 position.longitude_deg = payload_rx_nav_posllh.lon * UBX::DEGREES_PER_COORDINATE;
584 position.altitude_msl_m = payload_rx_nav_posllh.hMSL * 1e-3;
585 position.altitude_ellipsoid_m = payload_rx_nav_posllh.height * 1e-3;
586 position.eph = static_cast<float>(payload_rx_nav_posllh.hAcc) * 1e-3f; // from mm to m
587 position.accuracy_timestamp = nowUs();
588 position.epv = static_cast<float>(payload_rx_nav_posllh.vAcc) * 1e-3f; // from mm to m
589
590 position.timestamp = nowUs();
591
592 _got_posllh = true;
593
594 ret = 1;
595 break;
596 }
597 case UBX_MSG_NAV_HPPOSLLH: {

1243int GPSNativeUBX::decodeValidatedPayload(uint16_t message, std::span<const uint8_t> payload)
1244{
1245 if (!UBX::validPayload(message, payload)) {
1246 return 0;
1247 }
1248 if (message == UBX::NAV_EOE && payload.size() == 4 && _assembleEpochs) {
1249 uint32_t tow = 0;
1250 for (unsigned index = 0; index < 4; ++index) {
1251 tow |= uint32_t(payload[index]) << (index * 8);
1252 }
1253 if (tow < UBXNavigationEpoch::WEEK_MS) {
1254 _navigationEpochs.end(tow, [this](const auto& report) { publishEpoch(report); });
1255 }
1256 return GPSDecodedBatch::PROTOCOL_ACTIVITY;
1257 }
1258 if (!payloadRxInit(message, payload)) {
1259 return 0;
1260 }
1261 UBXNavigationEpoch::Epoch* epoch = nullptr;
1262 const auto publish = [this](const auto& report) { publishEpoch(report); };
1263 const auto* schema = UBX::messageSchema(message);
1264 const bool timed = schema && schema->towOffset >= 0;
1265 if (_assembleEpochs && timed) {
1266 const size_t offset = schema->towOffset;
1267 const auto tow = LittleEndian::read<uint32_t>(payload, offset).value_or(0);
1268 epoch = _navigationEpochs.find(tow, nowUs(), publish);
1269 if (!epoch) {
1270 return GPSDecodedBatch::PROTOCOL_ACTIVITY;
1271 }
1272 _epochHasHighPrecision = epoch->highPrecision;
1273 }
1274 switch (message) {
1275 case UBX_MSG_NAV_SAT:
1276 *_satellite_info = {};
1277 decodeNavSat(payload);
1278 break;
1279 case UBX_MSG_NAV_SVINFO:
1280 *_satellite_info = {};
1281 decodeNavSvinfo(payload);
1282 break;
1283 case UBX_MSG_MON_VER:
1284 decodeMonVer(payload);
1285 break;
1286 default:
1287 break;
1288 }
1289 const int updates = payloadRxDone(message, payload, epoch ? epoch->position : *_gps_position);
1290 if (epoch) {
1291 if (message == UBX_MSG_NAV_PVT) {
1292 epoch->positionValid = epoch->velocityValid = true;
1293 } else if (message == UBX_MSG_NAV_POSLLH) {
1294 epoch->positionValid = true;
1295 } else if (message == UBX_MSG_NAV_VELNED) {
1296 epoch->velocityValid = true;
1297 } else if (message == UBX_MSG_NAV_HPPOSLLH && (updates & 1)) {
1298 epoch->highPrecision = true;
1299 }
1300 return GPSDecodedBatch::PROTOCOL_ACTIVITY;
1301 }
1302 // ACKs and ancillary metadata are useful protocol activity, not new position epochs.
1303 if ((updates & 1) && message != UBX_MSG_NAV_PVT && message != UBX_MSG_NAV_POSLLH &&
1304 message != UBX_MSG_NAV_HPPOSLLH && message != UBX_MSG_NAV_VELNED) {
1305 return (updates & ~1) | GPSDecodedBatch::PROTOCOL_ACTIVITY;
1306 }
1307 return updates;
1308}
