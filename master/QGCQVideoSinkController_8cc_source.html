24void QGCQVideoSinkController::followWindowVisibility(QQuickItem* videoOutput)
25{
26 (void) QObject::disconnect(_visibilityConnection);
27 _videoOutput = videoOutput;
28 // Qt defect, seen in Qt 6.12.0: QQuickVideoOutput::itemChange() never calls QQuickItem::itemChange()
29 // (qtmultimedia/src/multimediaquick/qquickvideooutput.cpp), so the output emits no windowChanged. The window is
30 // taken now, and VideoStreamOutput.qml registers an output again after it moved to another window. Once the minimum
31 // Qt has the fix, connect to QQuickItem::windowChanged here and drop that re-registration.
32 if (QQuickWindow* window = videoOutput ? videoOutput->window() : nullptr) {
33 _visibilityConnection = QObject::connect(window, &QWindow::visibilityChanged, this,
34 &QGCQVideoSinkController::_applyWindowVisibility);
35 }
36 _applyWindowVisibility();
37}
