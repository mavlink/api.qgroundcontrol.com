40std::string_view unquote(std::string_view field)
41{
42 if (field.size() < 2 || field.front() != '"' || field.back() != '"') {
43 return {};
44 }
45 field.remove_prefix(1);
46 field.remove_suffix(1);
47 return field.find('"') == std::string_view::npos ? field : std::string_view{};
48}

50bool supportedFirmware(std::string_view model, std::string_view firmware)
51{
52 constexpr std::string_view PREFIX = "R4.10Build";
53 if (!firmware.starts_with(PREFIX)) {
54 return false;
55 }
56 const auto build = NMEA::number<unsigned>(firmware.substr(PREFIX.size()));
57 // N4 R1.6 section 3.6 explicitly gives these minimum builds for the rover modes.
58 return build && ((model == "UM980" && *build >= 7923) || (model == "UM982" && *build >= 7650));
59}

63bool validXorChecksum(std::string_view line)
64{
65 const auto star = line.rfind('*');
66 return star != std::string_view::npos && star + 3 == line.size() &&
67 NMEA::hexNumber(line.substr(star + 1)) == NMEA::checksum(line.substr(0, star));

71bool modeMatches(std::string_view mode, Plan::Reply reply)
72{
73 const auto starts = [mode](std::string_view name) {
74 return mode == name || (mode.starts_with(name) && mode.size() > name.size() && mode[name.size()] == ' ');
75 };
76 switch (reply) {
77 case Plan::Reply::RoverMode:
78 return starts("MODE ROVER");
79 case Plan::Reply::AveragingMode:
80 return starts("MODE BASE TIME");
81 default:
82 return mode == "MODE BASE";
83 }

177bool Protocol::configure(GPSCommandChannel& channel, GPSConfig config, unsigned& baud)
178{
179 _startAttempt(channel.stream(), channel.context());
180 (void) channel.flush();
181 const auto* fixed = std::get_if<GPSBaseStationConfig::Fixed>(&config.base.mode);
182 const auto* averaging = std::get_if<GPSBaseStationConfig::ReceiverAveraging>(&config.base.mode);
183 if (!fixed && !averaging) {
184 channel.failControl(
185 QStringLiteral("Unicore requires receiver-managed averaging with a duration between 1 and 3600 seconds"));
186 return _failAttempt(channel);
187 }
188 _setBase(config.base.mode);
189 const auto scope = channel.deadlineScope(Plan::CONFIGURATION_TIMEOUT);
190 GPSCommandSequence::Result identity;
191 const auto detection = channel.detectBaud(Plan::BAUD_RATES, baud, [this, &channel, &identity]() -> GPSBaudProbe {
192 _nmea.reset(channel.stream());
193 identity = _identify(channel);
194 if (identity.succeeded()) {
195 return GPSBaudProbe::Found;
196 }
197 // A receiver that reported its model answered at this rate.
198 return _model.isEmpty() ? GPSBaudProbe::TryNext : GPSBaudProbe::Stop;
199 });
200 if (!detection.found) {
201 if (!_rejection.isEmpty()) {
202 channel.failControl(_rejection);
203 } else if (identity.outcome == GPSCommandOutcome::Rejected) {
204 channel.failSequence(identity);
205 } else {
206 channel.failNoAnswer(detection.probed);
207 }
208 return _failAttempt(channel);
209 }
210 baud = detection.baud;
211 qCDebug(UnicoreProtocolLog).noquote() << "Unicore" << _model << "firmware" << _firmware;
212 const auto plan = fixed ? Plan::fixedBase(_fixedECEF) : Plan::averagingBase(averaging->maximumDuration);
213 if (!channel.runRequired(_sequence(plan.role))) {
214 return _failAttempt(channel);
215 }
216 _monitorBaseStatus(channel.context());
217 if (!channel.runRequired(_sequence(plan.output))) {
218 return _failAttempt(channel);
219 }
220 // Corrections flow while the base is valid.
221 _ready = true;
222 _nmea.setRTCMEnabled(_baseValid);
223 return true;
224}

267GPSCommandOutcome Protocol::_match(const Plan::Command& command, std::string_view line)
268{
269 if (line.starts_with("$command,")) {
270 const auto response = line.find(",response: ", 9);
271 if (response == std::string_view::npos ||
272 QByteArrayView(line.substr(9, response - 9)).compare(command.text, Qt::CaseInsensitive) != 0) {
273 return GPSCommandOutcome::Pending;
274 }
275 _acknowledgementMatched = true;
276 if (line.substr(response + 11, line.rfind('*') - response - 11) != "OK") {
277 return GPSCommandOutcome::Rejected;
278 }
279 return command.reply == Plan::Reply::Acknowledgement ? GPSCommandOutcome::Acknowledged
280 : GPSCommandOutcome::Pending;
281 }
282 switch (command.reply) {
283 case Plan::Reply::Acknowledgement:
284 return GPSCommandOutcome::Pending;
285 case Plan::Reply::Version:
286 if (!line.starts_with("#VERSIONA,")) {
287 return GPSCommandOutcome::Pending;
288 }
289 if (supportedFirmware(_model.toStdString(), _firmware.toStdString())) {
290 return GPSCommandOutcome::ReadbackVerified;
291 }
292 _rejection = QStringLiteral(
293 "Unsupported Unicore receiver '%1' firmware '%2'; requires UM980 R4.10Build7923+ "
294 "or UM982 R4.10Build7650+")
295 .arg(QString::fromUtf8(_model), QString::fromUtf8(_firmware));
296 return GPSCommandOutcome::Rejected;
297 case Plan::Reply::RoverMode:
298 case Plan::Reply::AveragingMode:
299 case Plan::Reply::FixedMode:
300 if (!line.starts_with("#MODE,")) {
301 return GPSCommandOutcome::Pending;
302 }
303 return modeMatches(_reportedMode.toStdString(), command.reply) ? GPSCommandOutcome::ReadbackVerified
304 : GPSCommandOutcome::Rejected;
305 case Plan::Reply::FixedPosition:
306 if (!line.starts_with("#BESTNAVXYZA,") || !_reportedFixed) {
307 return GPSCommandOutcome::Pending;
308 }
309 return samePosition(_reportedPosition, _fixedECEF) ? GPSCommandOutcome::ReadbackVerified
310 : GPSCommandOutcome::Rejected;
311 }
312 return GPSCommandOutcome::Pending;
313}

334GPSReceiveUpdates Protocol::_decodeLine(std::string_view line, GPSDecodeContext& context)
335{
336 if (line.starts_with("$command,")) {
337 if (!context.replyPending() || !validXorChecksum(line)) {
338 return {};
339 }
340 _acknowledgementMatched = false;
341 context.offerReply(line);
342 return _acknowledgementMatched ? GPSReceiveUpdates(GPSReceiveUpdate::Activity) : GPSReceiveUpdates{};
343 }
344 if (!line.starts_with('#')) {
345 return {};
346 }
347 const auto comma = line.find(',');
348 const auto name = line.substr(1, comma - 1);
349 if (name != "VERSIONA" && name != "MODE" && name != "BESTNAVXYZA") {
350 return {};
351 }
352 // MODE carries an XOR checksum; the other logs a CRC-32.
353 if (name == "MODE" ? !validXorChecksum(line) : !gpsASCIILog(line)) {
354 return {};
355 }
356 const auto star = line.rfind('*');
357 const auto semicolon = line.find(';');
358 if (semicolon == std::string_view::npos || semicolon >= star) {
359 return {};
360 }
361 std::array<std::string_view, 10> header{};
362 if (NMEA::splitFields(line.substr(1, semicolon - 1), header) != header.size()) {
363 return {};
364 }
365 const auto body = line.substr(semicolon + 1, star - semicolon - 1);
366 bool valid = false;
367 if (name == "VERSIONA") {
368 valid = _handleVersion(body, context);
369 } else if (name == "MODE") {
370 valid = _handleMode(body, context);
371 } else {
372 if (!_monitorBase) {
373 return GPSReceiveUpdate::Activity;
374 }
375 const auto week = NMEA::number<uint16_t>(header[4]);
376 const auto milliseconds = NMEA::number<uint32_t>(header[5]);
377 if (header[2] != "GPS" || header[3] != "FINE" || !week || !milliseconds || *milliseconds >= GPSTime::WEEK_MS) {
378 return {};
379 }
380 const uint64_t epoch = GPSTime::epochMs(*week, *milliseconds);
381 if (_lastBaseEpoch && epoch < *_lastBaseEpoch) {
382 _invalidateBase(context, QStringLiteral("Unicore base status went back in time; the receiver may have "
383 "restarted"));
384 return GPSReceiveUpdate::Activity;
385 }
386 if (_lastBaseEpoch && epoch == *_lastBaseEpoch) {
387 return GPSReceiveUpdate::Activity;
388 }
389 _lastBaseEpoch = epoch;
390 valid = _handlePosition(body, context);
391 }
392 if (valid) {
393 context.offerReply(line);
394 }
395 return GPSReceiveUpdate::Activity;
396}

398bool Protocol::_handleVersion(std::string_view body, GPSDecodeContext& context)
399{
400 std::array<std::string_view, 6> fields{};
401 if (NMEA::splitFields(body, fields) != fields.size()) {
402 return false;
403 }
404 const auto model = unquote(fields[0]);
405 const auto firmware = unquote(fields[1]);
406 if (model.empty() || model.size() > 32 || firmware.empty() || firmware.size() > 32) {
407 return false;
408 }
409 if (context.replyPending()) {
410 _model = QByteArrayView(model).toByteArray();
411 _firmware = QByteArrayView(firmware).toByteArray();
412 } else if (_ready) {
413 // An unsolicited identity report can indicate a reboot; never retain old base validity.
414 _invalidateBase(context,
415 QStringLiteral("Unicore receiver reported its identity unasked; it may have restarted"));
416 }
417 return true;
418}

430bool Protocol::_handlePosition(std::string_view body, GPSDecodeContext& context)
431{
432 std::array<std::string_view, 28> fields{};
433 if (NMEA::splitFields(body, fields) != fields.size()) {
434 return false;
435 }
436 const auto x = NMEA::number<double>(fields[2]);
437 const auto y = NMEA::number<double>(fields[3]);
438 const auto z = NMEA::number<double>(fields[4]);
439 if (!x || !y || !z) {
440 return false;
441 }
442 const GPSProtocolMath::Ecef coordinates{*x, *y, *z};
443 const bool fixed =
444 fields[0] == "SOL_COMPUTED" && fields[1] == "FIXEDPOS" && GPSProtocolMath::nearEarthSurface(coordinates);
445 _reportedPosition = coordinates;
446 _reportedFixed = fixed;
447 const bool matches = _averaging || samePosition(coordinates, _fixedECEF);
448 if (_ready && _baseValid && (!fixed || !matches || !samePosition(coordinates, _baseECEF))) {
449 _invalidateBase(context, QStringLiteral("Unicore base position was lost or changed"));
450 return true;
451 }
452 _lastBaseStatus = context.nowUs();
453 _baseValid = fixed && matches;
454 if (_baseValid) {
455 _baseECEF = coordinates;
456 }
457 _nmea.setRTCMEnabled(_ready && _baseValid);
458 _publishBase(context, _averaging && !_baseValid, _baseValid);
459 return true;
460}

487QLatin1StringView signature(const GPSFrame& frame)
488{
489 const std::string_view text = frame.text();
490 if (frame.kind != GPSFrameKind::ASCIILine) {
491 return {};
492 }
493 if (text.starts_with("$command,") && text.find(",response: ") != std::string_view::npos && validXorChecksum(text)) {
494 return QLatin1StringView("Unicore command replies");
495 }
496 const auto log = gpsASCIILog(text);
497 const bool unicore = log && !log->source.empty() && std::ranges::all_of(log->source, NMEA::isAsciiDigit);
498 return unicore ? QLatin1StringView("Unicore ASCII logs") : QLatin1StringView();
