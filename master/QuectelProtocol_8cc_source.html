178bool Protocol::Attempt::run(GPSConfig config, unsigned& baud)
179{
180 _protocol._stop(_channel.context());
181 (void) _channel.flush();
182 _protocol._startAttempt(_channel.stream());
183 const auto* survey = std::get_if<GPSBaseStationConfig::SurveyIn>(&config.base.mode);
184 if (std::holds_alternative<GPSBaseStationConfig::ReceiverAveraging>(config.base.mode) ||
185 (survey && (survey->duration > Plan::MAXIMUM_SURVEY_DURATION ||
186 survey->accuracyMeters > Plan::MAXIMUM_SURVEY_ACCURACY_METERS))) {
187 return _fail(
188 QStringLiteral("Unsupported LG290P configuration: use native 1 Hz observation count (maximum 86400) and "
189 "3D position accuracy threshold (maximum 1000 m), not receiver-managed survey"));
190 }
191 const auto scope = _channel.deadlineScope(Plan::CONFIGURATION_TIMEOUT);
192 _protocol._setBase(config.base.mode);
193 const auto detection = _channel.detectBaud(Plan::BAUD_RATES, baud, [this]() -> GPSBaudProbe {
194 _protocol._restartStream(_channel.stream());
195 if (_identify()) {
196 return GPSBaudProbe::Found;
197 }
198 // Firmware that answered but is not an LG290P(03) answers the same at every rate.
199 return _protocol._firmware.isEmpty() ? GPSBaudProbe::TryNext : GPSBaudProbe::Stop;
200 });
201 if (!detection.found) {
202 if (_protocol._firmware.isEmpty()) {
203 _channel.failNoAnswer(detection.probed);
204 return _fail({});
205 }
206 return _fail(QStringLiteral("No verified LG290P(03) identity; receiver configuration was not changed"));
207 }
208 baud = detection.baud;
209 qCDebug(QuectelProtocolLog).noquote() << "Quectel receiver firmware:" << _protocol._firmware;
210 if (!_verifyRole(false)) {
211 return _fail(QStringLiteral("LG290P receiver role query failed; no role change was attempted"));
212 }
213
214 bool restarted = false;
215 if (config.allowPersistentChanges) {
216 // Work from saved settings, not another client's uncommitted changes. A role
217 // readback alone cannot distinguish a pending role from the active one.
218 if (!_restart(false)) {
219 return _fail(
220 QStringLiteral("LG290P saved role query after restart failed; no persistent change was attempted"));
221 }
222 restarted = true;
223 }
224 if (_role != Plan::BASE_ROLE) {
225 if (!config.allowPersistentChanges) {
226 return _fail(
227 QStringLiteral("LG290P role mismatch: save the requested base role externally, reboot and reconnect. "
228 "Alternatively, explicitly allow persistent changes for this connection"),
229 GPSProtocolError::ConsentRequired);
230 }
231 if (!_changeRole()) {
232 return _fail(QStringLiteral("LG290P role change was rejected or its readback did not match"));
233 }
234 if (!_save()) {
235 return _fail(QStringLiteral("LG290P role save failed; receiver activation is not verified"));
236 }
237 if (!_restart()) {
238 return _fail(QStringLiteral("LG290P saved role could not be verified after restart"));
239 }
240 restarted = true;
241 }
242
243 bool baseChanged = false;
244 if (!_verifyBase(false)) {
245 return _fail(QStringLiteral("LG290P base settings query failed; no base change was attempted"));
246 }
247 if (!_baseMatches && !config.allowPersistentChanges) {
248 return _fail(
249 QStringLiteral("LG290P base settings mismatch: provision/save the requested ECEF coordinates or survey "
250 "observation count/accuracy externally, or explicitly allow persistent changes for this "
251 "connection"),
252 GPSProtocolError::ConsentRequired);
253 }
254 if (!_baseMatches) {
255 const QByteArray base = Plan::writeBase(config.base.mode, _protocol._fixedECEF, _baseHasDistance);
256 if (!_changeBase(base, survey ? std::string_view{} : Plan::FIXED_BASE_LABEL)) {
257 return _fail(QStringLiteral("LG290P base change was rejected or its readback did not match"));
258 }
259 if (!_save()) {
260 return _fail(QStringLiteral("LG290P base save failed; receiver activation is not verified"));
261 }
262 if (!_restartWithBase()) {
263 return _fail(QStringLiteral("LG290P saved base settings could not be verified after restart"));
264 }
265 baseChanged = true;
266 }
267 if (survey && !baseChanged) {
268 qCDebug(QuectelProtocolLog)
269 << "LG290P survey uses accepted 1 Hz observations, not wall time; accuracy filters individual 3D fixes. "
270 "The receiver itself stores converged coordinates";
271 if (!_acknowledge(_surveyRestart, Plan::RESTART_SURVEY_LABEL)) {
272 return _fail(QStringLiteral("LG290P rejected restarting the unchanged, externally saved survey"));
273 }
274 restarted = false;
275 }
276 if (!_verifySaved(restarted, baseChanged)) {
277 return _fail(
278 QStringLiteral("LG290P restart/readback failed; saved role or base settings do not match the request"));
279 }
280
281 _protocol._monitorSurvey(_channel.context());
282 const auto baseOutput = Plan::messageRates(Plan::BASE_OUTPUT);
283 if (!_channel.runSequence(baseOutput).succeeded()) {
284 return _fail(QStringLiteral("LG290P base message output configuration/readback failed"));
285 }
286 const auto nmeaOutput = Plan::messageRates(Plan::NMEA_OUTPUT);
287 if (!_channel.runSequence(nmeaOutput).succeeded()) {
288 return _fail(QStringLiteral("LG290P NMEA output configuration/readback failed"));
289 }
290 _protocol._finishAttempt(_channel.context());
291 return true;
292}

304bool Protocol::Attempt::_identify(std::chrono::milliseconds timeout)
305{
306 const auto result = _channel.transact(Plan::command(
307 Plan::IDENTIFY,
308 [this](std::string_view body) {
309 const Fields reply(body);
310 if (rejected(reply, Plan::IDENTIFY)) {
311 return GPSCommandOutcome::Rejected;
312 }
313 if (reply.size() != 4 || reply[0] != Plan::IDENTIFY) {
314 return GPSCommandOutcome::Pending;
315 }
316 const std::string_view firmware = reply[1];
317 _protocol._setFirmware(firmware);
318 return firmware.starts_with(Plan::QUALIFIED_FIRMWARE) && firmware.size() > Plan::QUALIFIED_FIRMWARE.size()
319 ? GPSCommandOutcome::ReadbackVerified
320 : GPSCommandOutcome::Rejected;
321 },
322 timeout));
323 return result.outcome == GPSCommandOutcome::ReadbackVerified;

341bool Protocol::Attempt::_verifyBase(bool requireMatch)
342{
343 _baseMatches = false;
344 const auto rate = _channel.transact(Plan::command(Plan::READ_FIX_RATE, [](std::string_view body) {
345 const Fields reply(body);
346 unsigned interval = 0;
347 return readback(reply, "PQTMCFGFIXRATE",
348 reply.size() == 3 && number(reply[2], interval) && interval == Plan::FIX_INTERVAL_MS);
349 }));
350 if (rate.outcome != GPSCommandOutcome::ReadbackVerified) {
351 return false;
352 }
353 const auto base = _channel.transact(Plan::command(Plan::READ_BASE, [this, requireMatch](std::string_view body) {
354 const Fields reply(body);
355 unsigned mode = 0;
356 unsigned count = 0;
357 double accuracy = 0;
358 double distance = 0;
359 GPSProtocolMath::Ecef ecef;
360 const bool valid = (reply.size() == 8 || reply.size() == 9) && reply[0] == "PQTMCFGSVIN" && reply[1] == "OK" &&
361 number(reply[2], mode) && mode <= 2 && number(reply[3], count) &&
362 number(reply[4], accuracy) && number(reply[5], ecef.x) && number(reply[6], ecef.y) &&
363 number(reply[7], ecef.z) && (reply.size() == 8 || number(reply[8], distance)) &&
364 std::chrono::seconds(count) <= Plan::MAXIMUM_SURVEY_DURATION && accuracy >= 0 &&
365 accuracy <= Plan::MAXIMUM_SURVEY_ACCURACY_METERS && distance >= 0 && distance <= 10;
366 bool matches = valid;
367 if (valid) {
368 _baseHasDistance = reply.size() == 9;
369 }
370 const auto& requested = _protocol._baseMode;
371 if (matches && std::holds_alternative<GPSBaseStationConfig::Fixed>(requested)) {
372 const auto& fixed = _protocol._fixedECEF;
373 matches = mode == 2 && count == 0 && accuracy == 0 && GPSProtocolMath::distance(ecef, fixed) <= 0.00011;
374 } else if (matches) {
375 const auto& survey = std::get<GPSBaseStationConfig::SurveyIn>(requested);
376 matches = mode == 1 && count == survey.duration.count() &&
377 std::abs(accuracy - survey.accuracyMeters) <= 0.000000001 && distance == 0;
378 if (matches) {
379 // Re-execute precisely the read configuration, including otherwise ignored ECEF fields.
380 _surveyRestart = QByteArrayView(Plan::RESTART_SURVEY).toByteArray();
381 for (size_t index = 2; index < reply.size(); ++index) {
382 _surveyRestart += ',';
383 _surveyRestart.append(QByteArrayView(reply[index]));
384 }
385 }
386 }
387 _baseMatches = matches;
388 return readback(reply, "PQTMCFGSVIN", valid && (!requireMatch || matches));
389 }));
390 return base.outcome == GPSCommandOutcome::ReadbackVerified;
391}

432bool Protocol::Attempt::_save()
433{
434 const auto result =
435 _channel.transact(Plan::command(Plan::SAVE, Plan::acknowledgement(Plan::SAVE), Plan::SAVE_TIMEOUT));
436 const bool saved = result.outcome == GPSCommandOutcome::Acknowledged;
437 if (saved) {

447bool Protocol::Attempt::_restart(bool requireRoleMatch)
448{
449 _protocol._beginRestart(_channel.context());
450 qCDebug(QuectelProtocolLog) << "Restarting LG290P and verifying its saved configuration";
451 const auto scope = _channel.deadlineScope(Plan::RESTART_TIMEOUT);
452 const QByteArray restart = NMEAUtils::frame(Plan::RESTART);
453 if (!_channel.writeCommand({QByteArrayView(Plan::RESTART).toByteArray(), Plan::RESTART_TIMEOUT}, restart)) {
454 return false;
455 }
456 // PQTMSRR has no documented ACK. Do not inflate successful transport completion to acknowledgment.
457 _channel.finishCommand(GPSCommandOutcome::Written);
458 _protocol._watchBoot(_channel.stream());
459 do {
460 _channel.wait(Plan::RESTART_POLL);
461 if (_channel.failed()) {
462 break;
463 }
464 const bool identified = _identify(Plan::RESTART_IDENTIFY_TIMEOUT);
465 if (_protocol._restartRejected) {
466 _channel.setErrorDetail(QStringLiteral("LG290P rejected PQTMSRR."));
467 break;
468 }
469 if (identified && _protocol._sawBoot) {
470 _protocol._endBootWatch();
471 return _verifyRole(requireRoleMatch);
472 }
473 } while (!_channel.failed() && _channel.nowUs() < _channel.operationDeadline().untilUs);
474 _protocol._endBootWatch();
475 return false;
476}

478bool Protocol::Attempt::_fail(const QString& reason, GPSProtocolError failure)
479{
480 _protocol._stop(_channel.context());
481 (void) _channel.flush();
482 QString detail = reason;
483 if (!_channel.errorDetail().isEmpty()) {
484 detail += (detail.isEmpty() ? QString() : QStringLiteral(" ")) + _channel.errorDetail();
485 }
486 _channel.failControl(failure);
487 if (_saveAcknowledged || _saveUncertain) {
488 const QString persistence =
489 _saveAcknowledged ? QStringLiteral("An LG290P flash save was acknowledged. ") : QString();
490 const QString uncertain =
491 _saveUncertain ? QStringLiteral("The latest flash save may have taken effect. ") : QString();
492 _channel.setErrorDetail(persistence + uncertain + detail +
493 QStringLiteral(" Receiver settings may have changed; no rollback was attempted."));
494 } else if (_channel.error() != GPSProtocolError::Cancelled) {
495 _channel.setErrorDetail(detail);
496 }
497 return false;

561GPSReceiveUpdates Protocol::_decodeLine(std::string_view line, GPSDecodeContext& context)
562{
563 const auto body = QuectelCodec::checkedBody(line);
564 if (body.empty()) {
565 return {};
566 }
567 context.offerReply(body);
568 if (_expectingBoot && body.starts_with("PQTMSRR,") && QuectelCodec::rejected(Fields(body), "PQTMSRR")) {
569 _restartRejected = true;
570 }
571 if (body.starts_with("PQTMSVINSTATUS,")) {
572 return _handleSurvey(body, context) ? GPSReceiveUpdates(GPSReceiveUpdate::Activity) : GPSReceiveUpdates{};
573 } else if (body.starts_with("PQTMVER,")) {
574 const Fields reply(body);
575 const std::string_view firmware(_firmware.constData(), static_cast<size_t>(_firmware.size()));
576 if (reply.size() == 6 && reply[1] == "1" && reply[2] == "MODULE" && reply[3] == firmware) {
577 // §2.3.1: this is the first output upon each successful startup.
578 if (_expectingBoot && !_sawBoot) {
579 _sawBoot = true;
580 if (_phase == SurveyPhase::AwaitingBoot) {
581 _phase = SurveyPhase::Verifying;
582 }
583 } else if (!_expectingBoot && (_ready || _phase != SurveyPhase::Off)) {
584 _ready = false;
585 _phase = SurveyPhase::Off;
586 _revokeSurvey(context);
587 context.failControl(QStringLiteral("LG290P restarted; its base configuration is no longer verified"));
588 }
589 return GPSReceiveUpdate::Activity;
590 }
591 }
592 return {};
593}

622bool Protocol::_handleSurvey(std::string_view body, GPSDecodeContext& context)
623{
624 const Fields reply(body);
625 unsigned version = 0;
626 unsigned tow = 0;
627 unsigned validity = 0;
628 unsigned observations = 0;
629 unsigned configuredCount = 0;
630 double accuracy = 0;
631 GPSProtocolMath::Ecef ecef;
632 if (reply.size() != 12 || !number(reply[1], version) || version != 1 || !number(reply[2], tow) ||
633 tow >= GPSTime::WEEK_MS) {
634 return false;
635 }
636 // TOW is a modular measurement clock, not a receipt timestamp. A late packet from before Sunday's rollover must
637 // not look newer than the current week.
638 if (_lastTow && !GPSTime::towAdvances(tow, *_lastTow)) {
639 return false;
640 }
641 _lastTow = tow;
642 if (_phase != SurveyPhase::Verifying && _phase != SurveyPhase::Monitoring) {
643 return false;
644 }
645 if (!number(reply[3], validity) || validity > 2 || !reply[4].empty() || !number(reply[6], observations) ||
646 std::chrono::seconds(observations) > Plan::MAXIMUM_SURVEY_DURATION || !number(reply[7], configuredCount) ||
647 std::chrono::seconds(configuredCount) > Plan::MAXIMUM_SURVEY_DURATION || !number(reply[8], ecef.x) ||
648 !number(reply[9], ecef.y) || !number(reply[10], ecef.z) || !number(reply[11], accuracy) || accuracy < 0 ||
649 accuracy > std::numeric_limits<uint32_t>::max() / 1000.0) {
650 _revokeSurvey(context);
651 return false;
652 }
653 const bool fixed = std::holds_alternative<GPSBaseStationConfig::Fixed>(_baseMode);
654 const bool matches = fixed
655 ? configuredCount == 0 && observations == 0
656 : configuredCount == std::get<GPSBaseStationConfig::SurveyIn>(_baseMode).duration.count();
657 const bool coordinatesKnown = GPSProtocolMath::nearEarthSurface(ecef);
658 if (!matches || (!coordinatesKnown && (validity == 2 || (validity == 1 && observations != 0)))) {
659 _revokeSurvey(context);
660 return false;
661 }
662 if (fixed && validity == 2 && GPSProtocolMath::distance(ecef, _fixedECEF) > 0.001) {
663 _revokeSurvey(context);
664 return false;
665 }
666 // BOOT + identity + saved-role/base readback establish this survey's session.
667 // A one-observation survey can finish before any progress notification is sent.
668 const bool valid = validity == 2 && (fixed || observations >= configuredCount);
669 GPSSurveyReport report{};
670 // Fixed mode's MeanAcc=0 describes supplied coordinates, not measured position uncertainty.
671 if (!fixed && validity != 0 && observations != 0 && coordinatesKnown) {
672 report.meanAccuracyMeters = GPSProtocolMath::roundedToMillimeters(accuracy);
673 }
674 // Base positioning is fixed at 1 Hz. Gaps do not count as accepted observation seconds.
675 report.duration = std::chrono::seconds(observations);
676 report.valid = valid;
677 report.active = validity == 1 && !valid;
678 if (coordinatesKnown && validity != 0) {
679 report.position = GPSProtocolMath::fromEcef(ecef);
680 }

697{
