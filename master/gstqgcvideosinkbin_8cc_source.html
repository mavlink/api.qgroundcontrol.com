123// Probe: caller override → SoC-native (imxvideoconvert_g2d/nvvidconv) → videoconvert.
124static GstElement* gst_qgc_video_sink_bin_make_conversion_element(GstQgcVideoSinkBin* self)
125{
126 if (self->conversion_element && self->conversion_element[0] != '\0') {
127 GstElement* e = gst_element_factory_make(self->conversion_element, NULL);
128 if (e) {
129 GST_INFO_OBJECT(self, "Using conversion-element override '%s'", self->conversion_element);
130 return e;
131 }
132 GST_WARNING_OBJECT(self, "conversion-element='%s' factory missing — falling through to defaults",
133 self->conversion_element);
134 }
135 static const char* kSoCFactories[] = {"imxvideoconvert_g2d", "nvvidconv", NULL};
136 for (int i = 0; kSoCFactories[i] != NULL; ++i) {
137 if (GstElement* e = gst_element_factory_make(kSoCFactories[i], NULL)) {
138 GST_INFO_OBJECT(self, "Using SoC conversion element '%s'", kSoCFactories[i]);
139 return e;
140 }
141 }
142 return gst_element_factory_make("videoconvert", NULL);
143}

145// Link glupload -> glcolorconvert -> the adopted format-capsfilter -> qgcqvideosink and ghost glupload.
146static gboolean wireGlUploadPath(GstQgcVideoSinkBin* self, GstQgc::BinChain& chain, GstElement* videosink,
147 GstElement* capsf)
148{
149 GstElement* glupload = chain.adopt(gst_element_factory_make("glupload", NULL));
150 if (!glupload) {
151 GST_ERROR_OBJECT(self, "Failed to create glupload element");
152 return FALSE;
153 }
154 // glcolorconvert honours texture-target=2D (GstQgcCaps.cc); a passthrough when upstream is already 2D.
155 GstElement* glcolorconvert = chain.adopt(gst_element_factory_make("glcolorconvert", NULL));
156 if (!glcolorconvert) {
157 GST_ERROR_OBJECT(self, "Failed to create glcolorconvert element");
158 return FALSE;
159 }
160 if (!chain.linkChain({glupload, glcolorconvert, capsf, videosink}) || !chain.ghostSink(glupload)) {
161 GST_ERROR_OBJECT(self, "Failed to link/ghost glupload→glcolorconvert→capsfilter→qgcqvideosink GPU path");
162 return FALSE;
163 }
164 GST_INFO_OBJECT(self, "Using glupload→qgcqvideosink GPU path (DMABuf→GL EGLImage import)");
165 return TRUE;
166}

168// Wire the GPU branch of @p mode: the glupload chain, or the adopted format-capsfilter -> qgcqvideosink with the
169// capsfilter ghosted.
170static gboolean wireGpuPath(GstQgcVideoSinkBin* self, GstQgc::BinChain& chain, GstElement* videosink, GstElement* capsf,
171 HwBuffers::GpuBinMode mode)
172{
173 if (mode == HwBuffers::GpuBinMode::GlUpload) {
174 return wireGlUploadPath(self, chain, videosink, capsf);
175 }
176
177 // Keep advertised native GPU formats direct. Formats outside buildGpuCapsString() must
178 // renegotiate upstream; converting here would also upload software frames instead of
179 // letting qgcqvideosink use its CPU mapping fallback.
180 if (!chain.linkChain({capsf, videosink}) || !chain.ghostSink(capsf)) {
181 GST_ERROR_OBJECT(self, "Failed to link/ghost qgcqvideosink (GPU path)");
182 return FALSE;
183 }
184 if (mode == HwBuffers::GpuBinMode::DirectDmaBuf) {
185 GST_INFO_OBJECT(self, "Using qgcqvideosink GPU path (direct DMABuf import, no glupload)");
186 } else {
187 GST_INFO_OBJECT(self, "Using qgcqvideosink GPU path (native memory passthrough)");
188 }
189 return TRUE;
190}

192// Link videoconvert -> (optional PAR=1/1) -> the adopted format-capsfilter -> qgcqvideosink and ghost the head.
193static gboolean wireCpuPath(GstQgcVideoSinkBin* self, GstQgc::BinChain& chain, GstElement* videosink, GstElement* capsf)
194{
195 GstElement* videoconvert = chain.adopt(gst_qgc_video_sink_bin_make_conversion_element(self));
196 if (!videoconvert) {
197 GST_ERROR_OBJECT(self, "Failed to create video conversion element");
198 return FALSE;
199 }
200
201 // PAR=1/1 capsfilter normalizes non-square pixels; disable-par for v4l2 drivers without
202 // VIDIOC_CROPCAP that deadlock negotiation (GStreamer MR #6242).
203 GstElement* par = nullptr;
204 if (!self->disable_par) {
205 par = chain.adopt(gst_element_factory_make("capsfilter", "qgc-par-filter"));
206 if (!par) {
207 GST_WARNING_OBJECT(self, "capsfilter factory missing — PAR normalization disabled");
208 } else {
209 if (const GStreamer::GstCapsPtr parCaps =
210 GStreamer::adoptCaps(gst_caps_from_string("video/x-raw, pixel-aspect-ratio=(fraction)1/1"))) {
211 g_object_set(par, "caps", parCaps.get(), NULL);
212 }
213 }
214 }
215
216 const bool linked = par ? chain.linkChain({videoconvert, par, capsf, videosink})
217 : chain.linkChain({videoconvert, capsf, videosink});
218 if (!linked || !chain.ghostSink(videoconvert)) {
219 GST_ERROR_OBJECT(self, "Failed to link/ghost CPU path");
220 return FALSE;
221 }
222
223 GST_INFO_OBJECT(self, "Using qgcqvideosink CPU path (videoconvert%s → caps → qgcqvideosink → QVideoSink)",
224 par ? " → PAR=1/1" : "");
225 return TRUE;
226}

228static void gst_qgc_video_sink_bin_setup(GstQgcVideoSinkBin* self)
229{
230 // gpu-zerocopy is construct-only — pass it to construction; g_object_set after the fact is rejected.
231 GstElement* videosink = gst_element_factory_make_full("qgcqvideosink", "name", "qgcqvideosink", "gpu-zerocopy",
232 self->gpu_zerocopy, NULL);
233 if (!videosink) {
234 GST_ERROR_OBJECT(self, "Failed to create qgcqvideosink element");
235 return;
236 }
237 g_object_set(videosink, "active", (gboolean) TRUE, "sync", (gboolean) self->sync, "qos", (gboolean) self->qos,
238 "processing-deadline", self->processing_deadline, NULL);
239
240 GstElement* capsf = gst_element_factory_make("capsfilter", "qgc-format-filter");
241 if (!capsf) {
242 GST_ERROR_OBJECT(self, "Failed to create capsfilter for qgcqvideosink");
243 gst_clear_object(&videosink);
244 return;
245 }
246
247 // On failure the BinChain rolls the bin back and frees videosink/capsf; self->videosink stays NULL so change_state
248 // surfaces the construction error.
249 const char* const pathName = self->gpu_zerocopy ? "GPU" : "CPU";
250 GstQgc::BinChain chain(GST_BIN(self));
251 if (!chain.adopt(videosink)) {
252 gst_object_unref(capsf);
253 GST_ERROR_OBJECT(self, "Failed to add qgcqvideosink %s sink element to bin", pathName);
254 return;
255 }
256 if (!chain.adopt(capsf)) {
257 GST_ERROR_OBJECT(self, "Failed to add qgcqvideosink %s capsfilter element to bin", pathName);
258 return;
259 }
260
261 // The format capsfilter restricts negotiation to Qt-renderable formats: qgcqvideosink's pad template is CAPS_ANY,
262 // so without it upstream could pick a format that fails in set_caps. On the CPU path, listing the formats
263 // QVideoSink renders natively also keeps videoconvert from forcing BGRA.
264 HwBuffers::GpuBinMode gpuBin = HwBuffers::GpuBinMode::Passthrough;
265 std::string capsStr;
266 if (self->gpu_zerocopy) {
267 gpuBin = HwBuffers::gpuBinMode();
268 capsStr = GstQgc::buildGpuCapsString(gpuBin);
269 } else {
270 capsStr = GstQgc::buildCpuCapsString();
271 }
272 const GStreamer::GstCapsPtr caps = GStreamer::adoptCaps(gst_caps_from_string(capsStr.c_str()));
273 if (!caps) {
274 GST_ERROR_OBJECT(self, "gst_caps_from_string() returned NULL for %s caps", pathName);
275 return;
276 }
277 g_object_set(capsf, "caps", caps.get(), NULL);
278
279 if (self->gpu_zerocopy ? wireGpuPath(self, chain, videosink, capsf, gpuBin)
280 : wireCpuPath(self, chain, videosink, capsf)) {
281 chain.commit();
282 self->videosink = videosink;
283 }
284}
