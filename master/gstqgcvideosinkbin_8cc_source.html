121// A single-element converter: the conversion-element override, else a SoC converter, else videoconvertscale (GStreamer
122// 1.22+). Null when none is available.
123static GstElement* gst_qgc_video_sink_bin_make_conversion_element(GstQgcVideoSinkBin* self)
124{
125 if (self->conversion_element && self->conversion_element[0] != '\0') {
126 GstElement* e = gst_element_factory_make(self->conversion_element, NULL);
127 if (e) {
128 GST_INFO_OBJECT(self, "Using conversion-element override '%s'", self->conversion_element);
129 return e;
130 }
131 GST_WARNING_OBJECT(self, "conversion-element='%s' factory missing — falling through to defaults",
132 self->conversion_element);
133 }
134 // These SoCs' decoders output proprietary layouts under plain video/x-raw formats, which only their converter
135 // reads.
136 static const char* kSoCFactories[] = {"imxvideoconvert_g2d", "nvvidconv", NULL};
137 for (int i = 0; kSoCFactories[i] != NULL; ++i) {
138 if (GstElement* e = gst_element_factory_make(kSoCFactories[i], NULL)) {
139 GST_INFO_OBJECT(self, "Using SoC conversion element '%s'", kSoCFactories[i]);
140 return e;
141 }
142 }
143 return gst_element_factory_make("videoconvertscale", NULL);
144}

146// Adds the bin's head, which converts and scales system memory into what qgcqvideosink offers (scaling lets its
147// square-pixel caps take non-square-pixel sources) and passes GPU memory through. Before GStreamer 1.22 that's
148// videoconvert ! videoscale; @p head and @p tail are its ends.
149static gboolean add_converter(GstQgcVideoSinkBin* self, GstQgc::BinChain& chain, GstElement*& head, GstElement*& tail)
150{
151 if (GstElement* const converter = gst_qgc_video_sink_bin_make_conversion_element(self)) {
152 head = tail = chain.adopt(converter);
153 return head != NULL;
154 }
155 head = chain.adopt(gst_element_factory_make("videoconvert", NULL));
156 tail = chain.adopt(gst_element_factory_make("videoscale", NULL));
157 return head && tail && chain.linkChain({head, tail});
158}

160// The GPU variant for the render target (GstQgc::selectGpuBin). Any other renderer gets the passthrough variant, whose
161// sink offers only memory that renderer imports (a glupload bin under Vulkan would download every frame and upload it
162// again).
163static GstQgc::GpuBinMode select_gpu_bin(GstQgcVideoSinkBin* self)
164{
165 const GstQgc::GpuBinPreference preference = GstQgc::gpuBinPreference();
166 const GstQgc::RenderTarget target = GstQgc::currentRenderTarget();
167 const GstQgc::GpuBinMode mode = GstQgc::selectGpuBin(preference, target);
168 if (mode != preference.mode) {
169 GST_INFO_OBJECT(self, "Qt renders with %s (EGL %d): using the passthrough GPU bin",
170 GstQgc::toString(target.backend), target.egl);
171 } else if (preference.explicitChoice && (GstQgc::selectGpuBin({.mode = mode}, target) != mode)) {
172 GST_WARNING_OBJECT(self, "QGC_GST_GPU_BIN forces a GPU bin Qt's renderer doesn't suit (%s, EGL %d)",
173 GstQgc::toString(target.backend), target.egl);
174 }
175 return mode;
176}

178// Builds [converter ! (glupload ! glcolorconvert !) qgcqvideosink]. qgcqvideosink's get_caps decides what negotiates:
179// the formats Qt renders, the GPU memory the render target imports, square pixels unless disable-par.
180static void gst_qgc_video_sink_bin_setup(GstQgcVideoSinkBin* self)
181{
182 // gpu-zerocopy is construct-only — pass it to construction; g_object_set after the fact is rejected.
183 GstElement* videosink = gst_element_factory_make_full("qgcqvideosink", "name", "qgcqvideosink", "gpu-zerocopy",
184 self->gpu_zerocopy, NULL);
185 if (!videosink) {
186 GST_ERROR_OBJECT(self, "Failed to create qgcqvideosink element");
187 return;
188 }
189 g_object_set(videosink, "active", (gboolean) TRUE, "sync", (gboolean) self->sync, "qos", (gboolean) self->qos,
190 "processing-deadline", self->processing_deadline, NULL);
191 const GstQgc::GpuBinMode gpuBin = self->gpu_zerocopy ? select_gpu_bin(self) : GstQgc::GpuBinMode::Passthrough;
192 GstQgc::configureSinkCaps(videosink, gpuBin, !self->disable_par);
193
194 // On failure the BinChain rolls the bin back; self->videosink stays NULL so change_state surfaces the construction
195 // error.
196 GstQgc::BinChain chain(GST_BIN(self));
197 if (!chain.adopt(videosink)) {
198 GST_ERROR_OBJECT(self, "Failed to add qgcqvideosink to bin");
199 return;
200 }
201 GstElement* head = NULL;
202 GstElement* tail = NULL;
203 if (!add_converter(self, chain, head, tail)) {
204 GST_ERROR_OBJECT(self, "Failed to create the video conversion element");
205 return;
206 }
207 if (gpuBin == GstQgc::GpuBinMode::GlUpload) {
208 GstElement* const glupload = chain.adopt(gst_element_factory_make("glupload", NULL));
209 // glcolorconvert honours the sink's texture-target=2D; a passthrough when upstream is already 2D.
210 GstElement* const glcolorconvert = chain.adopt(gst_element_factory_make("glcolorconvert", NULL));
211 if (!glupload || !glcolorconvert) {
212 GST_ERROR_OBJECT(self, "Failed to create glupload/glcolorconvert");
213 return;
214 }
215 tail = chain.linkChain({tail, glupload, glcolorconvert}) ? glcolorconvert : NULL;
216 }
217 if (!tail || !chain.linkChain({tail, videosink}) || !chain.ghostSink(head)) {
218 GST_ERROR_OBJECT(self, "Failed to link/ghost the qgcqvideosink chain");
219 return;
220 }
221 chain.commit();
222 self->videosink = videosink;
223 GST_INFO_OBJECT(self, "qgcqvideosink %s bin%s", self->gpu_zerocopy ? GstQgc::gpuBinName(gpuBin) : "CPU",
224 self->disable_par ? " (non-square pixels allowed)" : "");
225}
