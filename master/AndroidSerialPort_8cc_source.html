100bool QSerialPort::open(OpenMode mode)
101{
102 if (isOpen()) {
103 _setError(OpenError);
104 return false;
105 }
106
107 if ((mode & (Append | Truncate | Text | Unbuffered)) || (mode == NotOpen)) {
108 _setError(UnsupportedOperationError, tr("Unsupported open mode"));
109 return false;
110 }
111
112 clearError();
113 if (!_openDevice(mode)) {
114 _closeDevice();
115 _discardBuffers();
116 return false;
117 }
118
119 // The port holds its input itself, so QIODevice is not to keep a copy of what the owner reads.
120 return QIODevice::open(mode | Unbuffered);
121}

123bool QSerialPort::_openDevice(OpenMode mode)
124{
125 qCDebug(AndroidSerialPortLog) << "Opening" << _systemLocation;
126
127 AndroidSerialBackend* const backend = _backendFor(_systemLocation);
128 if (!backend->open(mode)) {
129 return false;
130 }
131 _backend = backend;
132
133 if (!backend->setParameters(_baudRate, _dataBits, _stopBits, _parity, tr("Failed to set parameters"))) {
134 return false;
135 }
136
137 if (!backend->setFlowControl(_flowControl)) {
138 qCWarning(AndroidSerialPortLog) << "Failed to set serial port flow control for" << _systemLocation;
139 return false;
140 }
141
142 if ((mode & ReadOnly) && !_readOn()) {
143 qCWarning(AndroidSerialPortLog) << "Failed to start async read for" << _systemLocation;
144 return false;
145 }
146
147 if (!_clearDevice(backend, AllDirections)) {
148 qCWarning(AndroidSerialPortLog) << "Failed to clear buffers for" << _systemLocation;
149 return false;
150 }
151
152 return true;

438qint64 QSerialPort::readData(char* data, qint64 maxSize)
439{
440 qsizetype size = 0;
441 {
442 QMutexLocker locker(&_arrivalMutex);
443 size = static_cast<qsizetype>(qMin<qint64>(maxSize, _admitted));
444 if (size > 0) {
445 std::memcpy(data, _input.constData(), static_cast<size_t>(size));
446 _input.remove(0, size);
447 _admitted -= size;
448 }
449 }
450
451 // Every read comes here, also one that finds nothing. It may have made room for input that waited, and it
452 // makes sure that the backend reads.
453 (void) _readOn();
454
455 return size;

458bool QSerialPort::_readOn()
459{
460 if (!_backend) {
461 return false;
462 }
463
464 bool inputWaits = false;
465 bool hasRoom = false;
466 bool resumed = false;
467 {
468 QMutexLocker locker(&_arrivalMutex);
469 inputWaits = _hiddenSizeLocked() > 0;
470 hasRoom = _hasRoomForInputLocked();
471 if (hasRoom) {
472 resumed = std::exchange(_readingHeldBack, false);
473 }
474 }
475
476 if (resumed) {
477 qCDebug(AndroidSerialPortTraceLog).noquote() << _systemLocation << "RX resumed";
478 }
479 if (inputWaits) {
480 _scheduleAdmission();
481 }
482
483 // The backend reads only while the port has room. The admission that makes room asks it to read on.
484 return !hasRoom || _backend->startReading();
485}

541{

570bool QSerialPort::_newDataArrived(const char* bytes, int length)
571{
572 qCDebug(AndroidSerialPortTraceLog).noquote() << _systemLocation << tracedTransfer("RX", bytes, length);
573
574 bool hasRoom = false;
575 bool paused = false;
576 {
577 QMutexLocker locker(&_arrivalMutex);
578 // Never dropped, like the input of desktop QSerialPort, which leaves what its buffer has no room for with
579 // the kernel. Here the backend leaves it with the device once it hears that there is no room.
580 _input.append(bytes, length);
581 hasRoom = _hasRoomForInputLocked();
582 paused = !hasRoom && !_readingHeldBack;
583 _readingHeldBack = !hasRoom;
584 _arrivalCondition.wakeAll();
585 }
586 if (paused) {
587 qCDebug(AndroidSerialPortTraceLog).noquote() << _systemLocation << "RX paused";
588 }
589
590 _scheduleAdmission();
591 return hasRoom;
592}

645bool QSerialPort::waitForReadyRead(int msecs)
646{
647 const QDeadlineTimer deadline(msecs);
648
649 // The backend looks for input at least once, even if the time is already up.
650 bool expired = false;
651 for (;;) {
652 // Desktop QSerialPort also goes on sending pending writes while it waits for input, and gives the wait up
653 // when one of them fails.
654 if (!_pumpWrites()) {
655 return false;
656 }
657 // Input that the owner may read already was announced before: only new input ends the wait.
658 if (_admitArrivedInput()) {
659 _announceAdmittedInput();
660 return true;
661 }
662 if (expired) {
663 qCDebug(AndroidSerialPortLog) << "Timeout while waiting for ready read on" << _systemLocation;
664 _setError(TimeoutError, tr("Timeout while waiting for ready read"));
665 return false;
666 }
667
668 if (!_backend) {
669 // Like desktop QSerialPort, a port that is not open waits its time out.
670 _waitForArrivedInput(deadline);
671 } else if (!_backend->waitForInput(deadline)) {
672 return false;
673 }
674 expired = deadline.hasExpired();
675 }
676}

697qint64 QSerialPort::writeData(const char* data, qint64 maxSize)
698{
699 // QIODevice passes a write of nothing on, which is no error: one would keep close() from sending what waits.
700 if (maxSize == 0) {
701 return 0;
702 }
703
704 if (!data || (maxSize < 0)) {
705 qCWarning(AndroidSerialPortLog) << "Invalid data or size in writeData for" << _systemLocation;
706 _setError(WriteError, tr("Invalid data or size"));
707 return -1;
708 }
709
710 // Like desktop QSerialPort, writes are buffered and handed to the device from the owner's event loop,
711 // waitForBytesWritten(), waitForReadyRead(), flush() or close(), none of which waits for a device longer
712 // than it was told to.
713 _output.append(data, static_cast<qsizetype>(maxSize));
714 _scheduleWrite();
715 return maxSize;
716}

733bool QSerialPort::_pumpWrites()
734{
735 qint64 written = 0;
736 {
737 QMutexLocker locker(&_arrivalMutex);
738 written = std::exchange(_completedWriteBytes, 0);
739 _writeProgress = false;
740 }
741 const qint64 discarded = qMin(written, _discardedInFlight);
742 _discardedInFlight -= discarded;
743 written = qMin(written - discarded, _bytesInFlight);
744 _bytesInFlight -= written;
745
746 bool succeeded = true;
747 if (_backend && (_bytesInFlight == 0) && !_output.isEmpty()) {
748 const qint64 taken = _backend->startWrite(_output.constData(), _output.size());
749 if (taken < 0) {
750 succeeded = false;
751 } else {
752 if (taken > 0) {
753 qCDebug(AndroidSerialPortTraceLog).noquote()
754 << _systemLocation << tracedTransfer("TX", _output.constData(), taken);
755 }
756 _output.remove(0, static_cast<qsizetype>(taken));
757 _bytesInFlight += taken;
758 }
759 }
760
761 if (written > 0) {
762 emit bytesWritten(written);
763 }
764
765 return succeeded;
766}

768bool QSerialPort::_writeUntil(const QDeadlineTimer& deadline)
769{
770 for (;;) {
771 if (!_pumpWrites()) {
772 return false;
773 }
774 if (bytesToWrite() == 0) {
775 return true;
776 }
777
778 AndroidSerialBackend* const backend = _openBackend();
779 if (!backend) {
780 return false;
781 }
782
783 bool progressWaiting = false;
784 {
785 QMutexLocker locker(&_arrivalMutex);
786 progressWaiting = _writeProgress;
787 }
788 // Counted before the time is looked at: a device node takes output while it is handed over, and the
789 // caller tells by bytesToWrite() whether the time ran out.
790 if (progressWaiting) {
791 continue;
792 }
793 if (deadline.hasExpired()) {
794 return true;
795 }
796 if (!backend->waitForWritable(deadline)) {
797 return false;
798 }
799 }
800}

803{

808 }

832 }
833 if (directions & Output) {
834 _output.clear();
835 _discardedInFlight += std::exchange(_bytesInFlight, 0);
836 }
837
838 const bool cleared = _clearDevice(backend, directions);
839 // Only now, because a read of QIODevice has the backend read on.

841 (void) skip(held);
842 }
843
844 return cleared;
845}

847bool QSerialPort::_clearDevice(AndroidSerialBackend* backend, Directions directions)
848{
849 // Before the port reads on, so that it reads nothing that the device held from before the clear. Input that
850 // had already left a USB device by then still arrives: nothing can call it back.
851 const bool cleared = backend->clear(directions);
852 // The input that waited is discarded, and holds reading back no longer.
853 const bool readingOn = !(directions & Input) || _readOnIfRoom();
854 return cleared && readingOn;
855}
