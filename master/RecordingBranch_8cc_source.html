15// MP4/MOV robust muxing: qtmux reserves header space at the start of the file and rewrites the moov
16// there every update period, so a crash or power loss leaves a file playable up to the last update.
17// The reservation (550 B/s per track by default, ~8 MB here) outlasts the nominal duration since the
18// index grows ~140 B/s at 30 fps and ~260 B/s at 60 fps; overrunning it makes qtmux post an ERROR.
19// The period bounds crash data loss without rewriting the growing moov too often.
20constexpr guint64 kRecordingReservedMaxDurationNs = 2 * 3600 * GST_SECOND;
21constexpr guint64 kRecordingMoovUpdatePeriodNs = 5 * GST_SECOND;

36const char* recordingParserFactory(const GstCaps* caps)
37{
38 if (!caps || gst_caps_is_any(caps) || gst_caps_is_empty(caps)) {
39 return nullptr;
40 }
41
42 for (guint i = 0; i < gst_caps_get_size(caps); ++i) {
43 const GstStructure* structure = gst_caps_get_structure(caps, i);
44 if (gst_structure_has_name(structure, "video/x-h264")) {
45 return "h264parse";
46 }
47 if (gst_structure_has_name(structure, "video/x-h265")) {
48 return "h265parse";
49 }
50 }
51
52 return nullptr;
53}

58GstPadProbeReturn timestampForMuxer([[maybe_unused]] GstPad* pad, GstPadProbeInfo* info, [[maybe_unused]] gpointer data)
59{
60 GstBuffer* buffer = GST_PAD_PROBE_INFO_BUFFER(info);
61 if (!buffer || GST_BUFFER_PTS_IS_VALID(buffer)) {
62 return GST_PAD_PROBE_OK;
63 }
64 if (!GST_BUFFER_DTS_IS_VALID(buffer)) {
65 return GST_PAD_PROBE_DROP;
66 }
67 buffer = gst_buffer_make_writable(buffer);
68 GST_BUFFER_PTS(buffer) = GST_BUFFER_DTS(buffer);
69 GST_PAD_PROBE_INFO_DATA(info) = buffer;
70 return GST_PAD_PROBE_OK;

95GStreamer::GstElementPtr RecordingBranch::makeFileSink(const QString& file, VideoReceiver::FileFormat format,
96 const GstCaps* inputCaps)
97{
98 const char* const muxerName = muxerFactory(format);
99 if (!muxerName) {
100 qCCritical(GstVideoReceiverLog) << "Unsupported file format" << static_cast<int>(format);
101 return nullptr;
102 }
103
104 GstElement* const rawBin = gst_bin_new("sinkbin");
105 GStreamer::GstElementPtr bin = GStreamer::adoptElement(rawBin ? GST_ELEMENT(gst_object_ref_sink(rawBin)) : nullptr);
106 if (!bin) {
107 qCWarning(GstVideoReceiverLog) << "gst_bin_new('sinkbin') failed";
108 return nullptr;
109 }
110
111 GstElement* parser = nullptr;
112 if (const char* parserFactory = recordingParserFactory(inputCaps)) {
113 GStreamer::GstElementPtr parserOwner = GStreamer::makeElement(parserFactory, "parser");
114 if (!parserOwner) {
115 qCWarning(GstVideoReceiverLog) << "gst_element_factory_make() failed for" << parserFactory;
116 return nullptr;
117 }
118 g_object_set(parserOwner.get(), "config-interval", -1, nullptr);
119 parser = GStreamer::addToBin(bin.get(), std::move(parserOwner), "parser");
120 if (!parser) {
121 return nullptr;
122 }
123 }
124
125 GStreamer::GstElementPtr muxerOwner = GStreamer::makeElement(muxerName, "muxer");
126 GStreamer::GstElementPtr fileSinkOwner = GStreamer::makeElement("filesink", "filesink");
127 if (!muxerOwner || !fileSinkOwner) {
128 qCWarning(GstVideoReceiverLog) << "gst_element_factory_make() failed for" << muxerName << "or filesink";
129 return nullptr;
130 }
131
132 // faststart stays off: it would spool the recording to a temp file and write the real file only at EOS.
133 if ((format == VideoReceiver::FileFormat::Mp4) || (format == VideoReceiver::FileFormat::Mov)) {
134 g_object_set(muxerOwner.get(), "reserved-max-duration", kRecordingReservedMaxDurationNs,
135 "reserved-moov-update-period", kRecordingMoovUpdatePeriodNs, nullptr);
136 }
137 // Not async: a sink that waits to preroll would take the live pipeline out of PLAYING until the first keyframe.
138 g_object_set(fileSinkOwner.get(), "location", QFile::encodeName(file).constData(), "async", FALSE, nullptr);
139
140 GstElement* const muxer = GStreamer::addToBin(bin.get(), std::move(muxerOwner), "muxer");
141 GstElement* const fileSink = GStreamer::addToBin(bin.get(), std::move(fileSinkOwner), "filesink");
142 if (!muxer || !fileSink || !gst_element_link(muxer, fileSink)) {
143 qCWarning(GstVideoReceiverLog) << "Failed to link the muxer and filesink";
144 return nullptr;
145 }
146
147 // No release_request_pad on failure: disposing the bin reclaims the pad.
148 const GStreamer::GstPadPtr muxerPad = GStreamer::adoptPad(gst_element_request_pad_simple(muxer, "video_%u"));
149 if (!muxerPad) {
150 qCWarning(GstVideoReceiverLog) << "gst_element_request_pad_simple(" << muxerName << ", video_%u) failed";
151 return nullptr;
152 }
153 (void) gst_pad_add_probe(muxerPad.get(), GST_PAD_PROBE_TYPE_BUFFER, timestampForMuxer, nullptr, nullptr);
154
155 GStreamer::GstPadPtr parserSinkPad;
156 if (parser) {
157 const GStreamer::GstPadPtr parserSrcPad = GStreamer::adoptPad(gst_element_get_static_pad(parser, "src"));
158 parserSinkPad = GStreamer::adoptPad(gst_element_get_static_pad(parser, "sink"));
159 if (!parserSrcPad || !parserSinkPad) {
160 qCCritical(GstVideoReceiverLog) << "gst_element_get_static_pad(parser) failed";
161 return nullptr;
162 }
163
164 const GstPadLinkReturn linkResult = gst_pad_link(parserSrcPad.get(), muxerPad.get());
165 if (linkResult != GST_PAD_LINK_OK) {
166 qCWarning(GstVideoReceiverLog) << "Failed to link parser and muxer:" << gst_pad_link_get_name(linkResult);
167 return nullptr;
168 }
169 }
170
171 // Each element, not just the bin: a message can outlive the bin that held its source.
172 for (GstElement* const element : {bin.get(), parser, muxer, fileSink}) {
173 if (element) {
174 GStreamer::markBranch(element, GStreamer::PipelineBranch::Recording);
175 }
176 }
177
178 GstPad* const ghostpad = gst_ghost_pad_new("sink", parserSinkPad ? parserSinkPad.get() : muxerPad.get());
179 if (!ghostpad || !gst_element_add_pad(bin.get(), ghostpad)) {
180 qCWarning(GstVideoReceiverLog) << "Failed to add the recording sink's ghost pad";
181 return nullptr;
182 }
183
184 return bin;
185}

206bool RecordingBranch::build()
207{
208 const GStreamer::TeeBranch branch = GStreamer::addTeeBranch(_context.pipeline, _context.tee, "recording");
209 if (!branch.valve) {
210 return false;
211 }
212 _queue = branch.queue;
213 _valve = branch.valve;
214 GStreamer::markBranch(_queue, GStreamer::PipelineBranch::Recording);
215 GStreamer::markBranch(_valve, GStreamer::PipelineBranch::Recording);
216
217 // Every frame must reach the muxer, so no time or buffer limit; storage that falls this far behind ends the
218 // recording (_onQueueOverrun) instead of holding up the tee.
219 g_object_set(_queue, "max-size-time", G_GUINT64_CONSTANT(0), "max-size-buffers", 0, "max-size-bytes",
220 static_cast<guint>(kQueueMaxBytes), nullptr);
221 (void) _queueOverrun.connect(_queue, "overrun", G_CALLBACK(_onQueueOverrun), this);
222
223 // Preserve stream-start/caps/segment events while buffers are dropped. When recording
224 // starts later, the newly linked parser receives the negotiated codec configuration.
225 // drop-mode exists since GStreamer 1.20, the oldest version QGC supports.
226 gst_util_set_object_arg(G_OBJECT(_valve), "drop-mode", "forward-sticky-events");
227 return true;
228}

275void RecordingBranch::_start(const Request& request)
276{
277 _current = request;
278 (void) _recording.fetch_add(1, std::memory_order_relaxed);
279 _failure.clear();
280 _fileStarted.store(false, std::memory_order_relaxed);
281 _openedReported = false;
282 _state = State::Recording;
283
284 qCDebug(GstVideoReceiverLog) << "Starting recording" << request.file << _context.redactedUri;
285
286 // forward-sticky-events keeps the negotiated caps on the closed valve; the query covers the moment before.
287 const GStreamer::GstPadPtr valveSrc = GStreamer::adoptPad(gst_element_get_static_pad(_valve, "src"));
288 GStreamer::GstCapsPtr inputCaps = GStreamer::adoptCaps(gst_pad_get_current_caps(valveSrc.get()));
289 if (!inputCaps) {
290 inputCaps = GStreamer::adoptCaps(gst_pad_query_caps(valveSrc.get(), nullptr));
291 }
292
293 _fileSink = makeFileSink(request.file, request.format, inputCaps.get());
294 if (!_fileSink) {
295 _failure = QCoreApplication::translate("RecordingBranch", "Unable to create the recording sink.");
296 _finish(QString());
297 return;
298 }
299 // Before the sink can post, so its ERROR (such as a file that can't be opened) settles this recording.
300 _finalizer.arm(_fileSink.get());
301
302 gst_util_set_object_arg(G_OBJECT(_queue), "leaky", "no");
303 if (!gst_bin_add(GST_BIN(_context.pipeline), _fileSink.get()) || !gst_element_link(_valve, _fileSink.get())) {
304 qCWarning(GstVideoReceiverLog) << "Failed to add the recording sink" << _context.redactedUri;
305 _failure = QCoreApplication::translate("RecordingBranch", "Unable to add the recording sink.");
306 _finish(QString());
307 return;
308 }
309
310 if (!gst_element_sync_state_with_parent(_fileSink.get())) {
311 // The sink posted why (usually a file that can't be opened); that ERROR ends the recording.
312 GStreamer::flushAndDisposeChild(_fileSink.get());
313 _failure = QCoreApplication::translate("RecordingBranch", "Unable to start recording to %1.").arg(request.file);
314 _state = State::Finalizing;
315 _finalizeTimer.start();
316 return;
317 }
318
319 // The file starts with a keyframe at t=0, so playback can decode from its first frame.
320 if (!_keyframeWatch.add(valveSrc.get(), GST_PAD_PROBE_TYPE_BUFFER, _keyframeProbe,
321 new KeyframeWatch{.branch = this, .recording = _recording.load(std::memory_order_relaxed)},
322 [](gpointer watch) { delete static_cast<KeyframeWatch*>(watch); })) {
323 qCCritical(GstVideoReceiverLog) << "Failed to add the keyframe probe" << _context.redactedUri;
324 _failure = QCoreApplication::translate("RecordingBranch", "Unable to start recording.");
325 _finish(QString());
326 return;
327 }
328
329 g_object_set(_valve, "drop", FALSE, nullptr);
330 const GStreamer::GstPadPtr valveSink = GStreamer::adoptPad(gst_element_get_static_pad(_valve, "sink"));
331 GStreamer::requestKeyframe(valveSink.get());
332 _context.dotDumper->dump(_context.pipeline, "pipeline-with-filesink");
333}

335void RecordingBranch::_end(const QString& failure)
336{
337 if (_state != State::Recording) {
338 return;
339 }
340 if (!failure.isEmpty()) {
341 _failure = failure;
342 }
343
344 g_object_set(_valve, "drop", TRUE, nullptr);
345 _keyframeWatch.remove();
346
347 if (!_fileStarted.load(std::memory_order_acquire)) {
348 _finish(QString());
349 return;
350 }
351
352 // Unlinked, so the next recording can take the valve while this one drains parser -> muxer -> file. Not sent from
353 // here: the EOS waits for the recorder's streaming thread, which slow storage can hold up.
354 const GStreamer::GstPadPtr valveSrc = GStreamer::adoptPad(gst_element_get_static_pad(_valve, "src"));
355 const GStreamer::GstPadPtr recorderSink = GStreamer::adoptPad(gst_pad_get_peer(valveSrc.get()));
356 if (!recorderSink || !gst_pad_unlink(valveSrc.get(), recorderSink.get())) {
357 qCWarning(GstVideoReceiverLog) << "Unable to unlink the recording sink";
358 _finish(notFinalizedError());
359 return;
360 }
361 GStreamer::sendEventAsync(recorderSink.get(), gst_event_new_eos());
362 _state = State::Finalizing;
363 _finalizeTimer.start();
364}

366void RecordingBranch::_finish(const QString& error)
367{
368 _finalizeTimer.stop();
369 _keyframeWatch.remove();
370 if (_valve) {
371 g_object_set(_valve, "drop", TRUE, nullptr);
372 }
373
374 const RecordingFinalizer::Outcome outcome = _finalizer.disarm();
375 if (_fileSink) {
376 // Removed before going NULL, so nothing it posts while stopping reaches the bus. Flushed unless it closed its
377 // file: going NULL would otherwise wait for a streaming thread stuck inside it.
378 if (outcome == RecordingFinalizer::Outcome::Closed) {
379 GStreamer::disposeChild(_fileSink.get());
380 } else {
381 GStreamer::flushAndDisposeChild(_fileSink.get());
382 }
383 _fileSink.reset();
384 }
385 _state = State::Idle;
386 _context.dotDumper->dump(_context.pipeline, "pipeline-recording-stopped");
387
388 const QString file = _current.file;
389 const bool fileOpened = _fileStarted.load(std::memory_order_acquire);
390 if (fileOpened) {
391 _reportOpened();
392 const QString fileError = error.isEmpty() ? _failure : error;
393 qCDebug(GstVideoReceiverLog) << "Recording closed" << file << fileError;
394 _listener.onRecordingFileClosed(file, fileError);
395 } else if (QFile::exists(file) && !QFile::remove(file)) {
396 // The filesink created the file, but it never got a frame.
397 qCDebug(GstVideoReceiverLog) << "Unable to remove the empty recording" << file;
398 }
399
400 if (!_failure.isEmpty()) {
401 if (_requested && (_requested->file == file)) {
402 _requested.reset();
403 }
404 const QString failure = std::exchange(_failure, QString());
405 _listener.onRecordingFailed(file, failure, fileOpened);
406 }
407
408 _reconcile();
409}

411bool RecordingBranch::beginTeardown()
412{
413 _requested.reset();
414 if (_state != State::Recording) {
415 return finalizing();
416 }
417
418 // Its recorder is done already, so there is nothing left to drain.
419 if (_finalizer.outcome() != RecordingFinalizer::Outcome::Pending) {
420 _finish(QString());
421 return false;
422 }
423
424 if (!_fileStarted.load(std::memory_order_acquire)) {
425 _end();
426 return finalizing();
427 }
428
429 // EOS at the tee drains the recorder queue into the file. Not sent to the pipeline: a source handles that EOS under
430 // its streaming thread's stream lock, which a thread held downstream would never release.
431 const GStreamer::GstPadPtr teeSink = GStreamer::adoptPad(gst_element_get_static_pad(_context.tee, "sink"));
432 GStreamer::sendEventAsync(teeSink.get(), gst_event_new_eos());
433 _state = State::Finalizing;
434 return true;
435}

437void RecordingBranch::completeTeardown()
438{
439 if (_state != State::Finalizing) {
440 return;
441 }
442
443 if (_finalizer.outcome() == RecordingFinalizer::Outcome::Pending) {
444 // Before the flush, which a recorder stuck in the kernel can hold up.
445 warnNotFinalized();
446 // Not serialized, so it doesn't wait for the stream lock: wakes the tee's thread if the full queue holds it,

448 const GStreamer::GstPadPtr queueSink = GStreamer::adoptPad(gst_element_get_static_pad(_queue, "sink"));

456void RecordingBranch::onBusError(GstMessage* message, const QString& text, const char* debug)
457{
458 bool fromRecorder = false;
459 const bool settled =
460 _finalizer.settleFrom(GST_MESSAGE_SRC(message), RecordingFinalizer::Outcome::Failed, &fromRecorder);
461 if (!fromRecorder) {
462 qCDebug(GstVideoReceiverLog) << "Ignoring an error from an earlier recording:" << text;
463 return;
464 }
465
466 // The detail names the cause behind a generic text such as "Could not multiplex stream."
467 if (debug) {
468 qCWarning(GstVideoReceiverLog) << "Recording error:" << text << "-" << debug;
469 } else {
470 qCWarning(GstVideoReceiverLog) << "Recording error:" << text;
471 }
472 // An error after the file closed doesn't fail it; a recording still running ends either way.
473 if (!settled && (_state != State::Recording)) {
474 return;
475 }
476
477 _failure = text;
478 // The bus handler closed the valve already: a failing recorder ends the recording, not the stream.
479 _finish(QString());
480}

482void RecordingBranch::onBusElement(GstMessage* message)
483{
484 // The recorder's EOS as the pipeline forwards it (message-forward), ahead of the pipeline's own EOS. That one is
485 // ignored: without a display sink it follows every recorder's EOS, and may arrive once the next recording started.
486 const GstStructure* structure = gst_message_get_structure(message);
487 if (!structure || !gst_structure_has_name(structure, "GstBinForwarded")) {
488 return;
489 }
490 GstMessage* forwardedMessage = nullptr;
491 (void) gst_structure_get(structure, "message", GST_TYPE_MESSAGE, &forwardedMessage, nullptr);
492 const GStreamer::GstMessagePtr forwarded = GStreamer::adoptMessage(forwardedMessage);
493 if (!forwarded || (GST_MESSAGE_TYPE(forwarded.get()) != GST_MESSAGE_EOS)) {
494 return;
495 }
496 if (!_finalizer.settleFrom(GST_MESSAGE_SRC(forwarded.get()), RecordingFinalizer::Outcome::Closed)) {
497 return;
498 }
499 if (_state == State::Recording) {
500 // The stream's own EOS came through the open valve. Nothing more will stream through this pipeline, so the
501 // request waits for the next one rather than start over in the file just closed.
502 _streaming = false;
503 }
504 qCDebug(GstVideoReceiverLog) << "Recording finalized" << _current.file;
505 _finish(QString());
506}

536GstPadProbeReturn RecordingBranch::_keyframeProbe(GstPad* pad, GstPadProbeInfo* info, gpointer data)
537{
538 const auto* const watch = static_cast<const KeyframeWatch*>(data);
539 RecordingBranch* const self = watch->branch;
540 if (self->_fileStarted.load(std::memory_order_relaxed)) {
541 return GST_PAD_PROBE_OK;
542 }
543
544 // Upstream of the tee is always a parser (GstBaseParse), which pushes single buffers, never lists.
545 GstBuffer* buf = gst_pad_probe_info_get_buffer(info);
546 if (!buf || GST_BUFFER_FLAG_IS_SET(buf, GST_BUFFER_FLAG_DELTA_UNIT) || !GST_BUFFER_PTS_IS_VALID(buf)) {
547 return GST_PAD_PROBE_DROP;
548 }
549
550 // The offset applies to running time, which differs from the PTS when the segment doesn't start at 0.
551 GstClockTime runningTime = GST_CLOCK_TIME_NONE;
552 if (const GStreamer::GstEventPtr segmentEvent =
553 GStreamer::adoptEvent(gst_pad_get_sticky_event(pad, GST_EVENT_SEGMENT, 0))) {
554 const GstSegment* segment = nullptr;
555 gst_event_parse_segment(segmentEvent.get(), &segment);
556 if (segment && (segment->format == GST_FORMAT_TIME)) {
557 runningTime = gst_segment_to_running_time(segment, GST_FORMAT_TIME, GST_BUFFER_PTS(buf));
558 }
559 }
560 if (!GST_CLOCK_TIME_IS_VALID(runningTime)) {
561 return GST_PAD_PROBE_DROP;
562 }
563
564 // Starts the file at 0; only the elements downstream of the valve see the offset.
565 gst_pad_set_offset(pad, -static_cast<gint64>(runningTime));
566
567 self->_fileStarted.store(true, std::memory_order_release);
568 // Reported from the worker, which knows whether this recording is still the current one.
569 const quint64 recording = watch->recording;
570 self->_context.post([self, recording]() { self->_onFileOpened(recording); });
571 return GST_PAD_PROBE_OK;
572}
