14// MP4/MOV robust muxing: qtmux reserves header space at the start of the file and rewrites the moov
15// there every update period, so a crash or power loss leaves a file playable up to the last update.
16// The reservation (550 B/s per track by default, ~8 MB here) outlasts the nominal duration since the
17// index grows ~140 B/s at 30 fps and ~260 B/s at 60 fps; overrunning it makes qtmux post an ERROR.
18// The period bounds crash data loss without rewriting the growing moov too often.
19constexpr guint64 kRecordingReservedMaxDurationNs = 2 * 3600 * GST_SECOND;
20constexpr guint64 kRecordingMoovUpdatePeriodNs = 5 * GST_SECOND;

36const char* recordingParserFactory(const GstCaps* caps)
37{
38 if (!caps || gst_caps_is_any(caps) || gst_caps_is_empty(caps)) {
39 return nullptr;
40 }
41
42 for (guint i = 0; i < gst_caps_get_size(caps); ++i) {
43 const GstStructure* structure = gst_caps_get_structure(caps, i);
44 if (gst_structure_has_name(structure, "video/x-h264")) {
45 return "h264parse";
46 }
47 if (gst_structure_has_name(structure, "video/x-h265")) {
48 return "h265parse";
49 }
50 }
51
52 return nullptr;
53}

57GStreamer::GstElementPtr RecordingBranch::makeFileSink(const QString& videoFile, VideoReceiver::FILE_FORMAT format,
58 const GstCaps* inputCaps)
59{
60 if (!VideoReceiver::isValidFileFormat(format)) {
61 qCCritical(GstVideoReceiverLog) << "Unsupported file format";
62 return nullptr;
63 }
64
65 GStreamer::GstElementPtr bin = GStreamer::adoptElement(gst_bin_new("sinkbin"));
66 if (!bin) {
67 qCWarning(GstVideoReceiverLog) << "gst_bin_new('sinkbin') failed";
68 return nullptr;
69 }
70 (void) gst_object_ref_sink(bin.get());
71 // splitmuxsink completes its change to PAUSED only once the first fragment opens, at the next keyframe.
72 // Waited out inside the bin, that doesn't make the live pipeline lose PLAYING meanwhile, nor leave it
73 // stuck in PAUSED when the sink fails first.
74 g_object_set(bin.get(), "async-handling", TRUE, nullptr);
75
76 GstElement* parser = nullptr;
77 if (const char* parserFactory = recordingParserFactory(inputCaps)) {
78 GStreamer::GstElementPtr parserOwner =
79 GStreamer::adoptElement(gst_element_factory_make(parserFactory, nullptr));
80 if (!parserOwner) {
81 qCWarning(GstVideoReceiverLog) << "gst_element_factory_make() failed for" << parserFactory;
82 return nullptr;
83 }
84 g_object_set(parserOwner.get(), "config-interval", -1, nullptr);
85 parser = GStreamer::addToBin(bin.get(), std::move(parserOwner), "parser");
86 if (!parser) {
87 return nullptr;
88 }
89 }
90
91 GStreamer::GstElementPtr splitmuxOwner = GStreamer::adoptElement(gst_element_factory_make("splitmuxsink", nullptr));
92 if (!splitmuxOwner) {
93 qCWarning(GstVideoReceiverLog) << "gst_element_factory_make('splitmuxsink') failed";
94 return nullptr;
95 }
96 GstElement* const splitmux = splitmuxOwner.get();
97
98 // One file per recording: no size or time limit.
99 g_object_set(splitmux, "location", qPrintable(videoFile), "muxer-factory", kFileMux[format], "max-size-time",
100 G_GUINT64_CONSTANT(0), "max-size-bytes", G_GUINT64_CONSTANT(0), "async-finalize", TRUE,
101 // Forwards the inner file sink's EOS, which settles the recording like fragment-closed does.
102 "message-forward", TRUE, nullptr);
103
104 // faststart stays off: it would spool the recording to a temp file and write the real file only at EOS.
105 if (format == VideoReceiver::FILE_FORMAT_MP4 || format == VideoReceiver::FILE_FORMAT_MOV) {
106 const GStreamer::GstStructurePtr muxerProps = GStreamer::adoptStructure(
107 gst_structure_new("properties", "reserved-max-duration", G_TYPE_UINT64, kRecordingReservedMaxDurationNs,
108 "reserved-moov-update-period", G_TYPE_UINT64, kRecordingMoovUpdatePeriodNs, nullptr));
109 g_object_set(splitmux, "muxer-properties", muxerProps.get(), nullptr);
110 }
111
112 if (!GStreamer::addToBin(bin.get(), std::move(splitmuxOwner), "splitmuxsink")) {
113 return nullptr;
114 }
115
116 // No release_request_pad on failure: finalizing splitmuxsink with the bin reclaims the pad.
117 const GStreamer::GstPadPtr videopad = GStreamer::adoptPad(gst_element_request_pad_simple(splitmux, "video"));
118 if (!videopad) {
119 qCWarning(GstVideoReceiverLog) << "gst_element_request_pad_simple(splitmuxsink, video) failed";
120 return nullptr;
121 }
122
123 GStreamer::GstPadPtr parserSinkPad;
124 if (parser) {
125 const GStreamer::GstPadPtr parserSrcPad = GStreamer::adoptPad(gst_element_get_static_pad(parser, "src"));
126 parserSinkPad = GStreamer::adoptPad(gst_element_get_static_pad(parser, "sink"));
127 if (!parserSrcPad || !parserSinkPad) {
128 qCCritical(GstVideoReceiverLog) << "gst_element_get_static_pad(parser) failed";
129 return nullptr;
130 }
131
132 const GstPadLinkReturn linkResult = gst_pad_link(parserSrcPad.get(), videopad.get());
133 if (linkResult != GST_PAD_LINK_OK) {
134 qCWarning(GstVideoReceiverLog)
135 << "Failed to link parser and splitmuxsink:" << gst_pad_link_get_name(linkResult);
136 return nullptr;
137 }
138 }
139
140 GstPad* const ghostpad = gst_ghost_pad_new("sink", parserSinkPad ? parserSinkPad.get() : videopad.get());
141 if (!ghostpad) {
142 qCWarning(GstVideoReceiverLog) << "gst_ghost_pad_new() failed";
143 return nullptr;
144 }
145 if (!gst_element_add_pad(bin.get(), ghostpad)) {
146 qCWarning(GstVideoReceiverLog) << "gst_element_add_pad(ghost) failed";
147 return nullptr;
148 }
149
150 return bin;
151}

155GstPadProbeReturn keyframeWatch(GstPad* pad, GstPadProbeInfo* info, gpointer user_data)
156{
157 if (!info || !user_data) {
158 qCCritical(GstVideoReceiverLog) << "Invalid arguments";
159 return GST_PAD_PROBE_DROP;
160 }
161
162 auto* const watch = static_cast<KeyframeWatch*>(user_data);
163 if (watch->started) {
164 return GST_PAD_PROBE_OK;
165 }
166
167 // Upstream of the tee is always a parser (GstBaseParse), which pushes single buffers, never lists.
168 GstBuffer* buf = gst_pad_probe_info_get_buffer(info);
169 if (!buf || GST_BUFFER_FLAG_IS_SET(buf, GST_BUFFER_FLAG_DELTA_UNIT) || !GST_BUFFER_PTS_IS_VALID(buf)) {
170 // wait for a timestamped keyframe
171 return GST_PAD_PROBE_DROP;
172 }
173
174 // The offset applies to running time, which differs from the PTS when the segment doesn't start at 0.
175 GstClockTime runningTime = GST_CLOCK_TIME_NONE;
176 if (const GStreamer::GstEventPtr segmentEvent =
177 GStreamer::adoptEvent(gst_pad_get_sticky_event(pad, GST_EVENT_SEGMENT, 0))) {
178 const GstSegment* segment = nullptr;
179 gst_event_parse_segment(segmentEvent.get(), &segment);
180 if (segment && (segment->format == GST_FORMAT_TIME)) {
181 runningTime = gst_segment_to_running_time(segment, GST_FORMAT_TIME, GST_BUFFER_PTS(buf));
182 }
183 }
184 if (!GST_CLOCK_TIME_IS_VALID(runningTime)) {
185 return GST_PAD_PROBE_DROP;
186 }
187
188 // Starts the file at 0; only the elements downstream of the valve see the offset.
189 gst_pad_set_offset(pad, -static_cast<gint64>(runningTime));
190
191 qCDebug(GstVideoReceiverLog) << "Got keyframe, stop dropping buffers";
192
193 watch->started = true;
194 emit watch->receiver->recordingStarted(watch->file);
195 return GST_PAD_PROBE_OK;
196}

238void RecordingBranch::start(const QString& videoFile, VideoReceiver::FILE_FORMAT format)
239{
240 qCDebug(GstVideoReceiverLog) << "Starting recording" << _receiver._redactedUri();
241
242 if (!_pipeline) {
243 qCDebug(GstVideoReceiverLog) << "Streaming is not active!" << _receiver._redactedUri();
244 emit _receiver.onStartRecordingComplete(VideoReceiver::STATUS_INVALID_STATE);
245 return;
246 }
247
248 if (_state != State::Idle) {
249 qCDebug(GstVideoReceiverLog) << "Already recording!" << _receiver._redactedUri();
250 emit _receiver.onStartRecordingComplete(VideoReceiver::STATUS_INVALID_STATE);
251 return;
252 }
253
254 qCDebug(GstVideoReceiverLog) << "New video file:" << videoFile << _receiver._redactedUri();
255
256 const GStreamer::GstPadPtr probepad = GStreamer::adoptPad(gst_element_get_static_pad(_valve, "src"));
257 if (!probepad) {
258 qCCritical(GstVideoReceiverLog) << "gst_element_get_static_pad() failed" << _receiver._redactedUri();
259 emit _receiver.onStartRecordingComplete(VideoReceiver::STATUS_FAIL);
260 return;
261 }
262
263 const auto failRecordingStart = [this]() {
264 if (_fileSink) {
265 GStreamer::disposeChild(_fileSink.get());
266 _fileSink.reset();
267 }
268 // Only once the sink is out of the pipeline: until then its errors must still count as the recorder's.
269 (void) _finalizer.disarm();
270 emit _receiver.onStartRecordingComplete(VideoReceiver::STATUS_FAIL);
271 };
272
273 // forward-sticky-events preserves the exact negotiated caps while the valve is closed. Fall
274 // back to a proxied caps query during the narrow startup window before negotiation completes.
275 GStreamer::GstCapsPtr inputCaps = GStreamer::adoptCaps(gst_pad_get_current_caps(probepad.get()));
276 if (!inputCaps) {
277 inputCaps = GStreamer::adoptCaps(gst_pad_query_caps(probepad.get(), nullptr));
278 }
279 _fileSink = makeFileSink(videoFile, format, inputCaps.get());
280 if (!_fileSink) {
281 qCWarning(GstVideoReceiverLog) << "Failed to create the recording sink" << _receiver._redactedUri();
282 failRecordingStart();
283 return;
284 }
285
286 // Armed before the sink can post, so the bus handler attributes its messages (e.g. a failure to open the
287 // file) to this recording instead of to the pipeline.
288 _finalizer.arm(_fileSink.get());
289
290 // The bin takes its own ref; ours survives a failed add.
291 if (!gst_bin_add(GST_BIN(_pipeline), _fileSink.get())) {
292 qCWarning(GstVideoReceiverLog) << "Failed to add the recording sink" << _receiver._redactedUri();
293 failRecordingStart();
294 return;
295 }
296
297 if (!gst_element_link(_valve, _fileSink.get())) {
298 qCWarning(GstVideoReceiverLog) << "Failed to link valve and file sink" << _receiver._redactedUri();
299 failRecordingStart();
300 return;
301 }
302
303 if (!gst_element_sync_state_with_parent(_fileSink.get())) {
304 qCWarning(GstVideoReceiverLog) << "Recording sink failed to start" << _receiver._redactedUri();
305 failRecordingStart();
306 return;
307 }
308
309 _receiver._dotDumper.dump(_pipeline, "pipeline-with-filesink");
310
311 _receiver._recordingOutput = videoFile;
312 // The file starts with a keyframe at t=0, so playback can decode from its first frame.
313 if (!_keyframeWatch.add(probepad.get(), GST_PAD_PROBE_TYPE_BUFFER, keyframeWatch,
314 new KeyframeWatch{.receiver = &_receiver, .file = videoFile},
315 [](gpointer watch) { delete static_cast<KeyframeWatch*>(watch); })) {
316 qCCritical(GstVideoReceiverLog) << "Failed to add the keyframe probe" << _receiver._redactedUri();
317 failRecordingStart();
318 return;
319 }
320
321 g_object_set(_valve, "drop", FALSE, nullptr);
322
323 _state = State::Recording;
324 qCDebug(GstVideoReceiverLog) << "Recording started" << _receiver._redactedUri();
325 emit _receiver.onStartRecordingComplete(VideoReceiver::STATUS_OK);
326 emit _receiver.recordingChanged(true);
327}

329void RecordingBranch::stop()
330{
331 qCDebug(GstVideoReceiverLog) << "Stopping recording" << _receiver._redactedUri();
332
333 if (!_pipeline || (_state != State::Recording)) {
334 qCDebug(GstVideoReceiverLog) << "Not recording!" << _receiver._redactedUri();
335 emit _receiver.onStopRecordingComplete(VideoReceiver::STATUS_INVALID_STATE);
336 return;
337 }
338
339 g_object_set(_valve, "drop", TRUE, nullptr);
340
341 // The EOS drains parser -> muxer -> file; _onFinalized() completes the stop once the file is closed, the
342 // timer if that is never reported.
343 _state = State::Stopping;
344 _stopTimer->start();
345 if (!_unlinkBranch() || (_finalizer.outcome() != RecordingFinalizer::Outcome::Pending)) {
346 _shutdown();
347 }
348}

350void RecordingBranch::finishForTeardown()
351{
352 if (_state == State::Recording) {
353 // EOS at the tee drains the recorder queue into the file. Not sent to the pipeline: a source handles that
354 // EOS under its streaming thread's stream lock, which a thread held downstream would never release.
355 if (const GStreamer::GstPadPtr teeSink = GStreamer::adoptPad(gst_element_get_static_pad(_tee, "sink"))) {
356 (void) gst_pad_send_event(teeSink.get(), gst_event_new_eos());
357 }
358 }
359
360 // Also waits for a stop() still in flight; its EOS is already in the branch.
361 if (_state != State::Idle) {
362 switch (_finalizer.wait(QDeadlineTimer(kFinalizeTimeout))) {
363 case RecordingFinalizer::Outcome::Closed:
364 qCDebug(GstVideoReceiverLog) << "Recording finalized";
365 break;
366 case RecordingFinalizer::Outcome::Failed:
367 qCWarning(GstVideoReceiverLog) << "Recording failed while finalizing";
368 break;
369 case RecordingFinalizer::Outcome::Pending:
370 qCWarning(GstVideoReceiverLog)
371 << "Recording not finalized within" << std::chrono::milliseconds(kFinalizeTimeout).count()
372 << "ms — forcing pipeline NULL (recording may be truncated; "
373 << "MP4/MOV stay playable up to the last moov update)";
374 break;
375 }
376 }

387void RecordingBranch::onBusElement(GstMessage* message)
388{
389 const GstStructure* structure = gst_message_get_structure(message);
390 if (!structure) {
391 return;
392 }
393
394 // splitmuxsink posts fragment-closed once the muxer has written the file and passed EOS to its sink.
395 GstObject* closedSource = nullptr;
396 if (gst_structure_has_name(structure, "splitmuxsink-fragment-closed")) {
397 closedSource = GST_MESSAGE_SRC(message);
398 }
399
400 // Branch EOS (pipeline message-forward); only the recorder's needs handling.
401 GStreamer::GstMessagePtr forwarded;
402 if (gst_structure_has_name(structure, "GstBinForwarded")) {
403 GstMessage* forwardedMessage = nullptr;
404 (void) gst_structure_get(structure, "message", GST_TYPE_MESSAGE, &forwardedMessage, nullptr);
405 forwarded = GStreamer::adoptMessage(forwardedMessage);
406 if (forwarded && (GST_MESSAGE_TYPE(forwarded.get()) == GST_MESSAGE_EOS)) {
407 closedSource = GST_MESSAGE_SRC(forwarded.get());
408 }
409 }
410
411 if (closedSource && _finalizer.settleFrom(closedSource, RecordingFinalizer::Outcome::Closed)) {
412 _receiver._runOnWorkerForPipeline([this]() { _onFinalized(); });
413 }
414}

416bool RecordingBranch::onBusError(GstMessage* message, const QString& text)
417{
418 bool fromRecording = false;
419 const bool settled =
420 _finalizer.settleFrom(GST_MESSAGE_SRC(message), RecordingFinalizer::Outcome::Failed, &fromRecording);
421 if (!fromRecording) {
422 return false;
423 }
424
425 // A recorder failure (disk full, header reservation overrun) ends the recording, not the stream. Closing the
426 // valve now, before the failing element returns its flow error, keeps that error out of the tee: the valve
427 // reports OK for a push that fails after drop was set.
428 qCWarning(GstVideoReceiverLog) << "Recording error:" << text;
429 g_object_set(_valve, "drop", TRUE, nullptr);
430 if (settled) {
431 _receiver._runOnWorkerForPipeline([this]() { _onRecorderError(); });
432 }
433 return true;
434}

436bool RecordingBranch::_unlinkBranch()
437{
438 const GStreamer::GstPadPtr src = GStreamer::adoptPad(gst_element_get_static_pad(_valve, "src"));
439 if (!src) {
440 qCCritical(GstVideoReceiverLog) << "gst_element_get_static_pad() failed";
441 return false;
442 }
443
444 const GStreamer::GstPadPtr sink = GStreamer::adoptPad(gst_pad_get_peer(src.get()));
445 if (!sink) {
446 qCWarning(GstVideoReceiverLog) << "Branch is not linked";
447 return false;
448 }
449
450 if (!gst_pad_unlink(src.get(), sink.get())) {
451 qCWarning(GstVideoReceiverLog) << "gst_pad_unlink() failed";
452 return false;
453 }
454
455 // Send EOS at the beginning of the branch
456 const gboolean ret = gst_pad_send_event(sink.get(), gst_event_new_eos());
457
458 if (!ret) {
459 qCWarning(GstVideoReceiverLog) << "Branch EOS was NOT sent";
460 return false;
461 }
462
463 qCDebug(GstVideoReceiverLog) << "Branch EOS was sent";
464
465 return true;
466}

468void RecordingBranch::_shutdown()
469{
470 _stopTimer->stop();
471
472 _keyframeWatch.remove();
473
474 if (_fileSink) {
475 // Removed before going NULL, so nothing it posts while stopping reaches the bus as a pipeline error.
476 GStreamer::disposeChild(_fileSink.get());
477 _fileSink.reset();
478 }
479 const RecordingFinalizer::Outcome outcome = _finalizer.disarm();
480
481 const State state = std::exchange(_state, State::Idle);
482 if (state != State::Idle) {
483 qCDebug(GstVideoReceiverLog) << "Recording stopped";
484 emit _receiver.recordingChanged(false);
485 }
486 if (state == State::Stopping) {
487 emit _receiver.onStopRecordingComplete(
488 (outcome == RecordingFinalizer::Outcome::Closed) ? VideoReceiver::STATUS_OK : VideoReceiver::STATUS_FAIL);
489 }
490
491 _receiver._dotDumper.dump(_pipeline, "pipeline-recording-stopped");
492}
