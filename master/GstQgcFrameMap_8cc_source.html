97// Operates on the GstVideoOrientationMethod enum, which is in <gst/video/video.h>'s
98// always-present subset — independent of QGC_HAS_GST_VIDEO_ORIENTATION_META.
99void applyOrientationToFrame(QVideoFrame& frame, GstVideoOrientationMethod method)
100{
101 switch (method) {
102 case GST_VIDEO_ORIENTATION_IDENTITY:
103 frame.setRotation(QtVideo::Rotation::None);
104 frame.setMirrored(false);
105 break;
106 case GST_VIDEO_ORIENTATION_90R:
107 frame.setRotation(QtVideo::Rotation::Clockwise90);
108 frame.setMirrored(false);
109 break;
110 case GST_VIDEO_ORIENTATION_180:
111 frame.setRotation(QtVideo::Rotation::Clockwise180);
112 frame.setMirrored(false);
113 break;
114 case GST_VIDEO_ORIENTATION_90L:
115 frame.setRotation(QtVideo::Rotation::Clockwise270);
116 frame.setMirrored(false);
117 break;
118 case GST_VIDEO_ORIENTATION_HORIZ:
119 frame.setRotation(QtVideo::Rotation::None);
120 frame.setMirrored(true);
121 break;
122 case GST_VIDEO_ORIENTATION_VERT:
123 frame.setRotation(QtVideo::Rotation::Clockwise180);
124 frame.setMirrored(true);
125 break;
126 case GST_VIDEO_ORIENTATION_UL_LR:
127 frame.setRotation(QtVideo::Rotation::Clockwise90);
128 frame.setMirrored(true);
129 break;
130 case GST_VIDEO_ORIENTATION_UR_LL:
131 frame.setRotation(QtVideo::Rotation::Clockwise270);
132 frame.setMirrored(true);
133 break;
134 default:
135 static std::atomic<bool> s_warnedUnhandled{false};
136 if (!s_warnedUnhandled.exchange(true, std::memory_order_relaxed)) {
137 qCWarning(GstQgcFrameMapLog)
138 << "Unhandled GstVideoOrientationMethod" << method << "— treating as identity";
139 }
140 frame.setRotation(QtVideo::Rotation::None);
141 frame.setMirrored(false);
142 break;
143 }
144}

146void applyOrientationAndTiming(QVideoFrame& frame, [[maybe_unused]] GstBuffer* buffer, int streamOrientation)
147{
148 // Per-buffer meta wins (per-frame override) when gst-video exports it; stream-level fallback
149 // works on every install.
150#ifdef QGC_HAS_GST_VIDEO_ORIENTATION_META
151 if (GstVideoOrientationMeta* meta = gst_buffer_get_video_orientation_meta(buffer)) {
152 applyOrientationToFrame(frame, meta->orientation);
153 } else
154#endif
155 if (streamOrientation != static_cast<int>(GST_VIDEO_ORIENTATION_IDENTITY)) {
156 applyOrientationToFrame(frame, static_cast<GstVideoOrientationMethod>(streamOrientation));
157 }
158 if (GST_BUFFER_PTS_IS_VALID(buffer)) {
159 // GstClockTime is ns; QVideoFrame timestamps are µs.
160 frame.setStartTime(GST_BUFFER_PTS(buffer) / GST_USECOND);
161 if (GST_BUFFER_DURATION_IS_VALID(buffer)) {
162 frame.setEndTime((GST_BUFFER_PTS(buffer) + GST_BUFFER_DURATION(buffer)) / GST_USECOND);
163 }
164 }

167void applyColorimetry(QVideoFrameFormat& format, const GstVideoInfo& info, GstCaps* caps)
168{
169 const GstVideoColorimetry& colorimetry = GST_VIDEO_INFO_COLORIMETRY(&info);
170 QVideoFrameFormat::ColorSpace colorSpace = toQtColorSpace(colorimetry.matrix);
171 // Live RTSP sources often omit colorimetry caps; match Qt's renderer fallback
172 // (qvideotexturehelper.cpp): height > 576 is HD/BT.709, otherwise SD/BT.601.
173 if (colorSpace == QVideoFrameFormat::ColorSpace_Undefined) {
174 const int height = GST_VIDEO_INFO_HEIGHT(&info);
175 if (height > 0) {
176 colorSpace = (height > 576) ? QVideoFrameFormat::ColorSpace_BT709 : QVideoFrameFormat::ColorSpace_BT601;
177 }
178 }
179 format.setColorSpace(colorSpace);
180 format.setColorTransfer(toQtColorTransfer(colorimetry.transfer));
181 QVideoFrameFormat::ColorRange range = toQtColorRange(colorimetry.range);
182 // H.264/H.265 omit VUI range but encode limited per spec — else Qt skips its limited->full offset.
183 // Only infer for a known YUV matrix: an UNKNOWN matrix tells us nothing, and RGB is always full-range.
184 const bool knownYuvMatrix =
185 (colorimetry.matrix == GST_VIDEO_COLOR_MATRIX_BT601) || (colorimetry.matrix == GST_VIDEO_COLOR_MATRIX_BT709) ||
186 (colorimetry.matrix == GST_VIDEO_COLOR_MATRIX_BT2020) ||
187 (colorimetry.matrix == GST_VIDEO_COLOR_MATRIX_SMPTE240M) || (colorimetry.matrix == GST_VIDEO_COLOR_MATRIX_FCC);
188 if (range == QVideoFrameFormat::ColorRange_Unknown && knownYuvMatrix) {
189 range = QVideoFrameFormat::ColorRange_Video;
190 }
191 format.setColorRange(range);
192
193 // Prefer MaxCLL (tighter tone-mapping target) over mastering-display max-luminance.
194 GstVideoContentLightLevel cll;
195 bool clipApplied = false;
196 if (caps && gst_video_content_light_level_from_caps(&cll, caps) && cll.max_content_light_level > 0) {
197 format.setMaxLuminance(static_cast<float>(cll.max_content_light_level));
198 clipApplied = true;
199 }
200 if (!clipApplied) {
201 GstVideoMasteringDisplayInfo masteringInfo;
202 if (caps && gst_video_mastering_display_info_from_caps(&masteringInfo, caps)) {
203 // GstVideoMasteringDisplayColorVolume max_luma is in 0.0001 cd/m².
204 const double maxLuminance = static_cast<double>(masteringInfo.max_display_mastering_luminance) / 10000.0;
205 if (maxLuminance > 0.0) {
206 format.setMaxLuminance(static_cast<float>(maxLuminance));
207 }
208 }
209 }
210}

223 const QVideoFrameFormat& format, [[maybe_unused]] const HwVideoBufferContext& hwContext,

226{
