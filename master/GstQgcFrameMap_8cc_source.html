98void applyOrientationToFrame(QVideoFrame& frame, GstVideoOrientationMethod method)
99{
100 switch (method) {
101 case GST_VIDEO_ORIENTATION_IDENTITY:
102 frame.setRotation(QtVideo::Rotation::None);
103 frame.setMirrored(false);
104 break;
105 case GST_VIDEO_ORIENTATION_90R:
106 frame.setRotation(QtVideo::Rotation::Clockwise90);
107 frame.setMirrored(false);
108 break;
109 case GST_VIDEO_ORIENTATION_180:
110 frame.setRotation(QtVideo::Rotation::Clockwise180);
111 frame.setMirrored(false);
112 break;
113 case GST_VIDEO_ORIENTATION_90L:
114 frame.setRotation(QtVideo::Rotation::Clockwise270);
115 frame.setMirrored(false);
116 break;
117 case GST_VIDEO_ORIENTATION_HORIZ:
118 frame.setRotation(QtVideo::Rotation::None);
119 frame.setMirrored(true);
120 break;
121 case GST_VIDEO_ORIENTATION_VERT:
122 frame.setRotation(QtVideo::Rotation::Clockwise180);
123 frame.setMirrored(true);
124 break;
125 case GST_VIDEO_ORIENTATION_UL_LR:
126 frame.setRotation(QtVideo::Rotation::Clockwise90);
127 frame.setMirrored(true);
128 break;
129 case GST_VIDEO_ORIENTATION_UR_LL:
130 frame.setRotation(QtVideo::Rotation::Clockwise270);
131 frame.setMirrored(true);
132 break;
133 default:
134 static std::atomic<bool> s_warnedUnhandled{false};
135 if (!s_warnedUnhandled.exchange(true, std::memory_order_relaxed)) {
136 qCWarning(GstQgcFrameMapLog)
137 << "Unhandled GstVideoOrientationMethod" << method << "— treating as identity";
138 }
139 frame.setRotation(QtVideo::Rotation::None);
140 frame.setMirrored(false);
141 break;
142 }

145void applyOrientationAndTiming(QVideoFrame& frame, [[maybe_unused]] GstBuffer* buffer, int streamOrientation)
146{
147 // Per-buffer meta wins (per-frame override) when gst-video exports it; stream-level fallback
148 // works on every install.
149#ifdef QGC_HAS_GST_VIDEO_ORIENTATION_META
150 if (GstVideoOrientationMeta* meta = gst_buffer_get_video_orientation_meta(buffer)) {
151 applyOrientationToFrame(frame, meta->orientation);
152 } else
153#endif
154 if (streamOrientation != static_cast<int>(GST_VIDEO_ORIENTATION_IDENTITY)) {
155 applyOrientationToFrame(frame, static_cast<GstVideoOrientationMethod>(streamOrientation));
156 }
157 if (GST_BUFFER_PTS_IS_VALID(buffer)) {
158 // GstClockTime is ns; QVideoFrame timestamps are µs.
159 frame.setStartTime(GST_BUFFER_PTS(buffer) / GST_USECOND);
160 if (GST_BUFFER_DURATION_IS_VALID(buffer)) {
161 frame.setEndTime((GST_BUFFER_PTS(buffer) + GST_BUFFER_DURATION(buffer)) / GST_USECOND);
162 } else {
163 // No duration: collapse the interval so consumers never see a stale/zero endTime.
164 frame.setEndTime(GST_BUFFER_PTS(buffer) / GST_USECOND);
165 }
166 }

169void applyColorimetry(QVideoFrameFormat& format, const GstVideoInfo& info, GstCaps* caps)
170{
171 const GstVideoColorimetry& colorimetry = GST_VIDEO_INFO_COLORIMETRY(&info);
172 QVideoFrameFormat::ColorSpace colorSpace = toQtColorSpace(colorimetry.matrix);
173 // Live RTSP sources often omit colorimetry caps; match Qt's renderer fallback
174 // (qvideotexturehelper.cpp): height > 576 is HD/BT.709, otherwise SD/BT.601.
175 if (colorSpace == QVideoFrameFormat::ColorSpace_Undefined) {
176 const int height = GST_VIDEO_INFO_HEIGHT(&info);
177 if (height > 0) {
178 colorSpace = (height > 576) ? QVideoFrameFormat::ColorSpace_BT709 : QVideoFrameFormat::ColorSpace_BT601;
179 }
180 }
181 format.setColorSpace(colorSpace);
182 format.setColorTransfer(toQtColorTransfer(colorimetry.transfer));
183 QVideoFrameFormat::ColorRange range = toQtColorRange(colorimetry.range);
184 // H.264/H.265 omit VUI range but encode limited per spec — else Qt skips its limited->full offset.
185 // Only infer for a known YUV matrix: an UNKNOWN matrix tells us nothing, and RGB is always full-range.
186 const bool knownYuvMatrix =
187 (colorimetry.matrix == GST_VIDEO_COLOR_MATRIX_BT601) || (colorimetry.matrix == GST_VIDEO_COLOR_MATRIX_BT709) ||
188 (colorimetry.matrix == GST_VIDEO_COLOR_MATRIX_BT2020) ||
189 (colorimetry.matrix == GST_VIDEO_COLOR_MATRIX_SMPTE240M) || (colorimetry.matrix == GST_VIDEO_COLOR_MATRIX_FCC);
190 if (range == QVideoFrameFormat::ColorRange_Unknown && knownYuvMatrix) {
191 range = QVideoFrameFormat::ColorRange_Video;
192 }
193 format.setColorRange(range);
194
195 // Prefer MaxCLL (tighter tone-mapping target) over mastering-display max-luminance.
196 GstVideoContentLightLevel cll;
197 bool clipApplied = false;
198 if (caps && gst_video_content_light_level_from_caps(&cll, caps) && cll.max_content_light_level > 0) {
199 format.setMaxLuminance(static_cast<float>(cll.max_content_light_level));
200 clipApplied = true;
201 }
202 if (!clipApplied) {
203 GstVideoMasteringDisplayInfo masteringInfo;
204 if (caps && gst_video_mastering_display_info_from_caps(&masteringInfo, caps)) {
205 // GstVideoMasteringDisplayColorVolume max_luma is in 0.0001 cd/m².
206 const double maxLuminance = static_cast<double>(masteringInfo.max_display_mastering_luminance) / 10000.0;
207 if (maxLuminance > 0.0) {
208 format.setMaxLuminance(static_cast<float>(maxLuminance));
209 }
210 }
211 }
212}
