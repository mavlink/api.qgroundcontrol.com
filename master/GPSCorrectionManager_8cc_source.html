18bool reachesThisHost(const QHostAddress& target)
19{
20 const auto matches = [&target](const QHostAddress& address) {
21 return !address.isNull() && address.isEqual(target, QHostAddress::TolerantConversion);
22 };
23 if (target.isLoopback() || target.isBroadcast() || target.isMulticast() || matches(QHostAddress::Any)) {
24 return true;
25 }
26 return std::ranges::any_of(QNetworkInterface::allInterfaces(), [&matches](const QNetworkInterface& interface) {
27 return std::ranges::any_of(interface.addressEntries(), [&matches](const QNetworkAddressEntry& entry) {
28 return matches(entry.ip()) || matches(entry.broadcast());
29 });
30 });
31}

93void GPSCorrectionManager::setConfiguration(const Configuration& configuration)
94{
95 if (_shutdown || configuration == _configuration) {
96 return;
97 }
98 const std::optional<Configuration> previous = std::exchange(_configuration, configuration);
99 _selector.configure(configuration.source, _scheduler->nowMs());
100 const bool inputChanged = !previous || previous->udpInput != configuration.udpInput;
101 const bool enabledChanged = (previous && previous->udpInput.enabled) != configuration.udpInput.enabled;
102 const bool inputNotified = inputChanged && (_applyUdpInput() || enabledChanged);
103 // The output also depends on the input port, which it must not feed back into.
104 const bool outputNotified = (inputChanged || previous->udpOutput != configuration.udpOutput) && _applyUdpOutput();
105 _refresh();
106 if (inputNotified) {
107 emit udpInputChanged();
108 }
109 if (outputNotified) {
110 emit udpOutputChanged();
111 }

114bool GPSCorrectionManager::_applyUdpOutput()
115{
116 const UdpOutputConfiguration& output = _configuration->udpOutput;
117 const QString address = output.address.trimmed();
118 _udpOutput.stop();
119 QString error;
120 if (output.enabled) {
121 // Forwarding to this host's own UDP input would feed the selected stream back into itself.
122 if (_udpInputEnabled() && output.port == _configuration->udpInput.port &&
123 reachesThisHost(QHostAddress(address))) {
124 qCWarning(GPSCorrectionManagerLog)
125 << "Not forwarding corrections to this host's own UDP input port" << output.port;
126 //: %1 is an IP address, %2 a UDP port
127 error = tr("Not forwarding to %1 port %2: this computer's UDP input receives on that port.")
128 .arg(address)
129 .arg(output.port);
130 } else if (!_udpOutput.configure(address, output.port)) {
131 error = tr("Enter an IP address and a port for UDP output.");
132 }
133 }
134 return std::exchange(_udpOutputError, error) != error;
135}

222void GPSCorrectionManager::_refresh()
223{
224 const std::optional<GPSCorrectionStream> selected = _selector.selectedStream(_scheduler->nowMs());
225 State state = State::Inactive;
226 if (selected) {
227 state = State::Fresh;
228 } else {
229 _selectedRate.resetRate();
230 if (_selector.hasSources() || _udpInputEnabled()) {
231 state = State::Waiting;
232 }
233 }
234 const quint64 rate = static_cast<quint64>(qRound64(_selectedRate.bytesPerSec()));
235 const bool rateChanged = std::exchange(_selectedBytesPerSecond, rate) != rate;
236 const GPSCorrectionStream stream = selected.value_or(GPSCorrectionStream{});
237 const bool streamChanged = std::exchange(_selected, stream) != stream;
238 if (std::exchange(_state, state) != state || streamChanged) {
239 emit stateChanged();
240 }
241 if (rateChanged) {
242 emit selectedBytesPerSecondChanged();
243 }
244}
