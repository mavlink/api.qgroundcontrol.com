125void GPSCorrectionManager::_applyUdpInputSettings()
126{
127 if (!_settings || _shutdown) {
128 return;
129 }
130 const QPointer<GPSCorrectionManager> guard(this);
131 const quint64 revision = ++_udpConfigurationRevision;
132 const bool enabled = _settings->rtcmUdpInputEnabled()->rawValue().toBool();
133 const bool validate = _settings->rtcmUdpValidate()->rawValue().toBool();
134 const quint16 port = static_cast<quint16>(_settings->rtcmUdpInputPort()->rawValue().toUInt());
135 const auto current = [this, guard, revision]() {
136 return guard && !_shutdown && _udpConfigurationRevision == revision;
137 };
138 _udpRegistration.reset();
139 if (!current()) {
140 return;
141 }
142 _udpInput.stop();
143 if (!current()) {
144 return;
145 }
146 disconnect(&_udpInput, nullptr, this, nullptr);
147 _udpInput.configure(port, validate);
148 if (!current() || !enabled) {
149 return;
150 }
151 auto registration = registerSource(GPSCorrectionSource::Udp);
152 if (!current()) {
153 return;
154 }
155 _udpRegistration = std::move(registration);
156 const auto token = _udpRegistration.token();
157 connect(&_udpInput, &RTCMUdpInput::frameReceived, this,
158 [this, token](const GPSCorrectionFrame& frame) { acceptIngress(token.event(frame)); });
159 connect(&_udpInput, &RTCMUdpInput::frameRejected, this,
160 [this, token](const GPSCorrectionFrame& frame, GPSCorrectionReason reason) {
161 acceptIngress(token.event(frame, reason));
162 });
163 const bool started = _udpInput.start();
164 if (current() && !started) {
165 _udpRegistration.reset();
166 }
