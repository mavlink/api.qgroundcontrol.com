45void GstReceiverWorker::apply(const Request& request)
46{
47 const Request previous = std::exchange(_requested, request);
48
49 // A start, stop or restart takes the config and the video sink as they are now.
50 if (request.active != previous.active) {
51 _onActiveChanged();
52 } else if (request.restarts != previous.restarts) {
53 if (_requested.active) {
54 _restartStream();
55 }
56 } else {
57 if (request.config != previous.config) {
58 _onConfigChanged(previous.config);
59 }
60 if ((request.videoSink != previous.videoSink) && _session) {
61 _session->setVideoSink(_requested.videoSink.get());
62 }
63 }
64
65 const quint64 recordingId = request.recording ? request.recording->id : 0;
66 const quint64 previousRecordingId = previous.recording ? previous.recording->id : 0;
67 if (recordingId != previousRecordingId) {
68 if (request.recording) {
69 _startRecording(*request.recording);
70 } else {
71 _stopRecording();
72 }
73 }
74}

87void GstReceiverWorker::_onConfigChanged(const VideoReceiver::Config& previous)
88{
89 if (!_requested.active) {
90 return;
91 }
92 const VideoReceiver::Config& config = _requested.config;
93 VideoReceiver::Config sameStream = previous;
94 sameStream.autoReconnect = config.autoReconnect;
95 if (sameStream != config) {
96 _restartStream();
97 return;
98 }
99
100 // Only autoReconnect changed, which applies at once.
101 switch (_supervisor->autoReconnectChanged(config.autoReconnect)) {
102 case StreamSupervisor::AutoReconnectChange::Halted:
103 _onHalted(_supervisor->lastFailure());
104 break;
105 case StreamSupervisor::AutoReconnectChange::Resumable:
106 _reconcile();
107 break;
108 case StreamSupervisor::AutoReconnectChange::None:
109 break;
110 }
111}

180void GstReceiverWorker::_startSession()
181{
182 const PipelineSession::Settings settings{
183 .uri = _requested.config.uri,
184 .redactedUri = _redactedUri(),

260void GstReceiverWorker::_onWatchdogTick()
261{
262 if (!_session) {
263 return;
264 }
265
266 if (++_statsTicks >= kHwStatsLogIntervalTicks) {
267 _statsTicks = 0;
268 if (const GstQgc::PathStats hwStats = _session->sinkPathStats(); hwStats.totalDelivered > 0) {
269 qCDebug(GstVideoReceiverLog).noquote() << "GPU path live" << _redactedUri() << hwStats.line;
270 }
271 }
272 _sampleStats();
273
274 const StreamWatchdog::Result result = _watchdog.check(StreamWatchdog::now(), _session->displaying());
275 switch (result.stall) {
276 case StreamWatchdog::Stall::None:
277 return;
278 case StreamWatchdog::Stall::Decoder:
279 qCDebug(GstVideoReceiverLog) << "Video decoder timeout, no frames for" << result.silentMs << "ms"
280 << _redactedUri();
281 _session->snapshot(QStringLiteral("pipeline-decoder-timeout"));
282 _session->restartDecoder("decoder watchdog");
283 return;
284 case StreamWatchdog::Stall::Source:
285 qCDebug(GstVideoReceiverLog) << "Stream timeout, no frames for" << result.silentMs << "ms"
286 << _redactedUri();
287 _session->snapshot(QStringLiteral("pipeline-watchdog-timeout"));
288 _stopSession();
289 _recover({.reason = QStringLiteral("source watchdog"),
290 .message = QCoreApplication::translate("GstVideoReceiver", "No video received for %1 s.")
291 .arg(result.silentMs / 1000),
292 .permanent = false});
293 return;
294 }
