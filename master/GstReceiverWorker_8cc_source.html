23QString partFile(const QString& file, int part)
24{
25 const QString suffix = QStringLiteral("_part%1").arg(part);
26 const qsizetype dot = file.lastIndexOf(u'.');
27 const qsizetype separator = std::max(file.lastIndexOf(u'/'), file.lastIndexOf(u'\\'));
28 return (dot > separator) ? (file.left(dot) + suffix + file.mid(dot)) : (file + suffix);

184{

226{

268void GstReceiverWorker::_onWatchdogTick()
269{
270 if (!_session) {
271 return;
272 }
273
274 if (++_statsTicks >= kHwStatsLogIntervalTicks) {
275 _statsTicks = 0;
276 if (const GstQgc::PathStats hwStats = _session->sinkPathStats(); hwStats.totalDelivered > 0) {
277 qCDebug(GstVideoReceiverLog).noquote() << "GPU path live" << _redactedUri() << hwStats.line;
278 }
279 }
280
281 const StreamWatchdog::Result result = _watchdog.check(StreamWatchdog::now(), _session->displaying());
282 switch (result.stall) {
283 case StreamWatchdog::Stall::None:
284 return;
285 case StreamWatchdog::Stall::Decoder:
286 qCDebug(GstVideoReceiverLog) << "Video decoder timeout, no frames for" << result.silentMs << "ms"
287 << _redactedUri();
288 _session->snapshot(QStringLiteral("pipeline-decoder-timeout"));
289 _session->restartDecoder("decoder watchdog");
290 return;
291 case StreamWatchdog::Stall::Source:
292 qCDebug(GstVideoReceiverLog) << "Stream timeout, no frames for" << result.silentMs << "ms"
293 << _redactedUri();
294 _session->snapshot(QStringLiteral("pipeline-watchdog-timeout"));
295 _stopSession();
296 _recover({.reason = QStringLiteral("source watchdog"),
297 .message = QCoreApplication::translate("GstVideoReceiver", "No video received for %1 s.")
298 .arg(result.silentMs / 1000),
299 .permanent = false});
300 return;
301 }
302}
