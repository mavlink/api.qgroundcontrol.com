24QString listenUri(QLatin1StringView scheme, const QString& advertised)
25{
26 bool isNumber = false;
27 const int port = advertised.toInt(&isNumber);
28 if (isNumber) {
29 return VideoStreamUri::isValidPort(port) ? QStringLiteral("%1://0.0.0.0:%2").arg(scheme).arg(port) : advertised;
30 }
31
32 QUrl url(advertised);
33 const std::optional<VideoStreamUri::Transport> transport = VideoStreamUri::transportFor(url);
34 if (!transport || !VideoStreamUri::listens(*transport) || !VideoStreamUri::isValidPort(url.port())) {
35 return advertised;
36 }
37 url.setScheme(scheme);
38 if (url.host().isEmpty()) {
39 url.setHost(QStringLiteral("0.0.0.0"));
40 }
41 return url.toString();
42}

54QString autoStreamUri(const QGCVideoStreamInfo& streamInfo)
55{
56 const QString uri = streamInfo.uri();
57 switch (streamInfo.type()) {

141std::optional<VideoReceiver::Config> VideoStreamResolver::resolve(const Settings& settings,
142 const QGCVideoStreamInfo* streamInfo, Role role)
143{
144 QString uri;
145 if (isAutoStream(streamInfo)) {
146 uri = autoStreamUri(*streamInfo);
147 } else if (role == Role::Primary) {
148 uri = manualUri(settings);
149 }
150 if (uri.isEmpty()) {
151 return std::nullopt;
152 }
153
154 VideoReceiver::Config config;
155 config.uri = uri;
156 // Only RTSP's stall timeout is a setting: the other streams keep the receiver's default.
157 if (VideoStreamUri::transportFor(QUrl(uri)) == VideoStreamUri::Transport::Rtsp) {
158 config.timeout = std::chrono::seconds(settings.rtspTimeoutSec);
159 }
160 config.lowLatency = settings.lowLatency;
161 config.rtpJitterLatencyMs = settings.rtpJitterLatencyMs;
162 config.autoReconnect = settings.autoReconnect;
163 return config;
164}

176QString VideoStreamResolver::transportName(const QString& uri)
177{
178 using VideoStreamUri::Transport;
179
180 const QUrl url(uri);
181 const std::optional<Transport> transport = VideoStreamUri::transportFor(url);
182 if (!transport) {
183 return url.scheme().toUpper();
184 }
185 switch (*transport) {
186 case Transport::Rtsp:
187 return QStringLiteral("RTSP");
188 case Transport::RtpUdpH264:
189 return QStringLiteral("RTP/UDP (H.264)");
190 case Transport::RtpUdpH265:
191 return QStringLiteral("RTP/UDP (H.265)");
192 case Transport::MpegTsUdp:
193 return QStringLiteral("MPEG-TS/UDP");
194 case Transport::MpegTsTcp:
195 return QStringLiteral("MPEG-TS/TCP");
196 }
197 return QString();
198}
