29QString autoStreamUri(const QGCVideoStreamInfo& streamInfo)
30{
31 const QString uri = streamInfo.uri();
32 switch (streamInfo.type()) {
33 case VIDEO_STREAM_TYPE_RTPUDP:
34 return (streamInfo.encoding() == VIDEO_STREAM_ENCODING_H265)
35 ? listenUriWithScheme(QLatin1StringView("udp265://"), uri)
36 : listenUriWithScheme(QLatin1StringView("udp://"), uri);
37 case VIDEO_STREAM_TYPE_MPEG_TS:
38 return listenUriWithScheme(QLatin1StringView("mpegts://"), uri);
39 case VIDEO_STREAM_TYPE_RTSP:
40 case VIDEO_STREAM_TYPE_TCP_MPEG:
41 default:
42 return uri;
43 }

90QString manualUri(const VideoStreamResolver::Settings& settings)
91{
92 const ManualSource* const source = findManualSource(settings.videoSource);
93 if (!source) {
94 return QString();
95 }
96 const QLatin1StringView uri(source->uri);
97 return source->url ? addressWithScheme(uri, settings.*(source->url)) : QString(uri);

131std::optional<VideoReceiver::Config> VideoStreamResolver::resolve(const Settings& settings,
132 const QGCVideoStreamInfo* streamInfo, Role role)
133{
134 QString uri;
135 if (isAutoStream(streamInfo)) {
136 uri = autoStreamUri(*streamInfo);
137 } else if (role == Role::Primary) {
138 uri = manualUri(settings);
139 }
140 if (uri.isEmpty()) {
141 return std::nullopt;
142 }
143
144 VideoReceiver::Config config;
145 config.uri = uri;
146 config.timeout = uri.startsWith(QLatin1StringView("rtsp"), Qt::CaseInsensitive)
147 ? std::chrono::seconds(settings.rtspTimeoutSec)
148 : kNonRtspStreamTimeout;
149 config.lowLatency = settings.lowLatency;
150 config.rtpJitterLatencyMs = settings.rtpJitterLatencyMs;
151 config.autoReconnect = settings.autoReconnect;
152 return config;
153}

165QString VideoStreamResolver::transportName(const QString& uri)
166{
167 const QString scheme = QUrl(uri).scheme();
168 for (const TransportName& transport : kTransportNames) {
169 if (scheme.compare(QLatin1StringView(transport.scheme), Qt::CaseInsensitive) == 0) {
170 return QString::fromLatin1(transport.name);
171 }
172 }
173 return scheme.toUpper();
