26QString autoStreamUri(const QGCVideoStreamInfo& streamInfo)
27{
28 const QString uri = streamInfo.uri();
29 switch (streamInfo.type()) {
30 case VIDEO_STREAM_TYPE_RTPUDP:
31 return (streamInfo.encoding() == VIDEO_STREAM_ENCODING_H265)
32 ? listenUriWithScheme(QLatin1StringView("udp265://"), uri)
33 : listenUriWithScheme(QLatin1StringView("udp://"), uri);
34 case VIDEO_STREAM_TYPE_MPEG_TS:
35 return listenUriWithScheme(QLatin1StringView("mpegts://"), uri);
36 case VIDEO_STREAM_TYPE_RTSP:
37 case VIDEO_STREAM_TYPE_TCP_MPEG:
38 default:
39 return uri;

75QString manualUri(const VideoStreamResolver::Settings& settings)
76{
77 const ManualSource* const source = findManualSource(settings.videoSource);
78 if (!source) {
79 return QString();

82 return source->url ? addressWithScheme(uri, settings.*(source->url)) : QString(uri);
