10bool ntripSourceTableComplete(QByteArrayView body, qsizetype checkedSize)
11{
12 static constexpr QByteArrayView terminator("ENDSOURCETABLE");
13 // A terminator the earlier check missed ends past its data, so it starts no earlier than this.
14 const qsizetype from = (std::max) (qsizetype(0), checkedSize - terminator.size());
15 for (qsizetype offset = body.indexOf(terminator, from); offset >= 0;
16 offset = body.indexOf(terminator, offset + 1)) {
17 const qsizetype end = offset + terminator.size();
18 if ((offset == 0 || body.at(offset - 1) == '\n') &&
19 (end == body.size() || body.at(end) == '\r' || body.at(end) == '\n')) {
20 return true;
21 }
22 }
23 return false;

26std::optional<NTRIPMountpoint> NTRIPMountpoint::fromSourceTableLine(const QString& line)
27{
28 // Need >= 18 fields; split keeps empty fields, so a trailing ';' is harmless
29 // (the extra empty field parses as 0 / "" for its column).
30 const QStringList fields = line.split(';');
31 if (fields.size() < 18 || fields.at(0).trimmed().toUpper() != QStringLiteral("STR")) {
32 return std::nullopt;
33 }
34
35 NTRIPMountpoint mp;
36 mp.mountpoint = fields.at(1).trimmed();
37 mp.format = fields.at(3).trimmed();
38 mp.navSystem = fields.at(6).trimmed();
39 mp.country = fields.at(8).trimmed();
40 const auto parseCoord = [](const QString& s, double limit) -> double {
41 bool ok = false;
42 const double v = s.trimmed().toDouble(&ok);
43 return (ok && qIsFinite(v) && qAbs(v) <= limit) ? v : qQNaN();
44 };
45 mp.latitude = parseCoord(fields.at(9), 90.0);
46 mp.longitude = parseCoord(fields.at(10), 180.0);

51void NTRIPMountpoint::updateDistance(const QGeoCoordinate& from)
52{
53 const QGeoCoordinate mountCoord(latitude, longitude);
54 if (!from.isValid() || !mountCoord.isValid() || (latitude == 0.0 && longitude == 0.0)) {
55 distanceKm = -1.0;
56 return;
57 }
58 distanceKm = from.distanceTo(mountCoord) / 1000.0;

150bool NTRIPSourceTableSortModel::lessThan(const QModelIndex& left, const QModelIndex& right) const
151{
152 const double leftKm = left.data(sortRole()).toDouble();
153 const double rightKm = right.data(sortRole()).toDouble();
154 if ((leftKm < 0) != (rightKm < 0)) {
