49struct PrivState
50{
51 // Caller-owned and never unref'd; written by the controller on the GUI thread under GST_OBJECT_LOCK.
52 QVideoSink* qvideosink = nullptr;
53 // Read lock-free by show_frame.
54 std::atomic<bool> active{true};
55 // start() derives hw_context from it.
56 bool gpu_zerocopy = false;
57 // Negotiated state: set_caps/stop write it and show_frame reads it, all serialised by GstBaseSink.
58 std::atomic<bool> caps_valid{false};
59 GstVideoInfo video_info = {};
60 QVideoFrameFormat format;
61 // Written by start() under GST_OBJECT_LOCK and snapshotted by show_frame.
62 HwVideoBufferContext hw_context = {};

64 // in plain system memory.
65 bool gpu_caps = false;
66 std::atomic<int64_t> last_pts_ns{-1};
67 std::atomic<quint64> input_frames{0};
68 std::atomic<quint64> dropped_frames{0};

70 // Per-element render counter (read via `frames-delivered`) so multi-receiver setups
71 // don't see a shared process-global total.
72 std::atomic<quint64> delivered_frames{0};
73 // Delivered frames per path (None counts CPU frames), frames a GPU path was tried for but delivered through the
74 // CPU, and caps epochs that started on a GPU path and fell back to the CPU; reported through "frame-stats".
75 std::array<std::atomic<quint64>, GstHw::kPathCount> path_frames{};

77 std::atomic<quint64> stream_demotions{0};

118bool caps_carry_gpu_memory(GstCaps* caps)
119{
120 const GstCapsFeatures* const features = gst_caps_get_features(caps, 0);
121 if (!features || gst_caps_features_is_any(features)) {
122 return false;
123 }
124 for (guint i = 0; i < gst_caps_features_get_size(features); ++i) {
125 const gchar* const feature = gst_caps_features_get_nth(features, i);
126 if (g_str_has_prefix(feature, "memory:") && (g_strcmp0(feature, GST_CAPS_FEATURE_MEMORY_SYSTEM_MEMORY) != 0)) {
127 return true;
128 }
129 }
130 return false;
131}

173bool push_frame_queued(GstQgcQVideoSink* self, QVideoFrame&& frame)
174{
175 QVideoFrame superseded;
176 GST_OBJECT_LOCK(self);
177 QVideoSink* const sink = priv_of(self)->qvideosink;
178 if (!sink) {
179 GST_OBJECT_UNLOCK(self);
180 return false;
181 }
182 const std::shared_ptr<PendingFrameSlot> slot = priv_of(self)->pending;
183 bool needPost = false;
184 {
185 QMutexLocker locker(&slot->mutex);
186 superseded = std::exchange(slot->frame, std::move(frame));
187 needPost = !std::exchange(slot->posted, true);
188 }
189 if (needPost) {
190 QMetaObject::invokeMethod(
191 sink,
192 [sink, slot]() {
193 QVideoFrame latest;
194 {
195 QMutexLocker locker(&slot->mutex);
196 latest = std::exchange(slot->frame, QVideoFrame());
197 slot->posted = false;
198 }
199 if (latest.isValid()) {
200 sink->setVideoFrame(latest);
201 }
202 },
203 Qt::QueuedConnection);
204 }
205 GST_OBJECT_UNLOCK(self);
206 return true;
207}

260static void gst_qgc_q_video_sink_set_property(GObject* obj, guint id, const GValue* val, GParamSpec* pspec)
261{
262 GstQgcQVideoSink* self = GST_QGC_Q_VIDEO_SINK(obj);
263 PrivState* p = priv_of(self);
264 QVideoFrame undelivered;
265 GST_OBJECT_LOCK(self);
266 switch (id) {
267 case PROP_QVIDEOSINK: {
268 auto* const target = static_cast<QVideoSink*>(g_value_get_pointer(val));
269 if (target != p->qvideosink) {
270 // A frame bound for the previous sink must not reach the new one; a fresh slot also detaches any
271 // invoke still queued to the old sink.
272 undelivered = take_pending_frame(*p->pending);
273 p->pending = std::make_shared<PendingFrameSlot>();
274 }
275 p->qvideosink = target;
276 // Reset PTS guard on sink swap so a new sink doesn't see a stale last_pts_ns
277 // from the previous sink and erroneously drop the first frames.
278 p->last_pts_ns.store(-1, std::memory_order_relaxed);
279 break;
280 }
281 case PROP_ACTIVE:
282 p->active.store(g_value_get_boolean(val), std::memory_order_relaxed);
283 break;
284 case PROP_GPU_ZEROCOPY:
285 p->gpu_zerocopy = g_value_get_boolean(val);
286 break;
287 default:
288 G_OBJECT_WARN_INVALID_PROPERTY_ID(obj, id, pspec);
289 break;
290 }
291 GST_OBJECT_UNLOCK(self);
292}

327static gboolean gst_qgc_q_video_sink_set_caps(GstBaseSink* bsink, GstCaps* caps)
328{
329 GstQgcQVideoSink* self = GST_QGC_Q_VIDEO_SINK(bsink);
330 PrivState* p = priv_of(self);
331
332 GstVideoInfo parsedInfo = {};
333 if (!GstHw::dmaDrmAwareVideoInfo(caps, &parsedInfo)) {
334 qCWarning(GstQgcQVideoSinkLog) << "set_caps: failed to parse video info from caps";
335 return FALSE;
336 }
337
338 const QVideoFrameFormat::PixelFormat pixelFormat = toQtPixelFormat(GST_VIDEO_INFO_FORMAT(&parsedInfo));
339 if (pixelFormat == QVideoFrameFormat::Format_Invalid) {
340 qCWarning(GstQgcQVideoSinkLog) << "set_caps: unsupported video format"
341 << gst_video_format_to_string(GST_VIDEO_INFO_FORMAT(&parsedInfo));
342 return FALSE;
343 }
344
345 const int w = GST_VIDEO_INFO_WIDTH(&parsedInfo);
346 const int h = GST_VIDEO_INFO_HEIGHT(&parsedInfo);
347 if (w <= 0 || h <= 0) {
348 qCWarning(GstQgcQVideoSinkLog) << "set_caps: invalid dimensions" << w << "x" << h;
349 return FALSE;
350 }
351
352 QVideoFrameFormat fmt(QSize(w, h), pixelFormat);
353 applyColorimetry(fmt, parsedInfo, caps);
354 const int fpsN = GST_VIDEO_INFO_FPS_N(&parsedInfo);
355 const int fpsD = GST_VIDEO_INFO_FPS_D(&parsedInfo);
356 if (fpsN > 0 && fpsD > 0) {
357 fmt.setStreamFrameRate(static_cast<qreal>(fpsN) / static_cast<qreal>(fpsD));
358 }
359
360 p->video_info = parsedInfo;
361 p->format = std::move(fmt);
362 gst_clear_caps(&p->cached_caps);
363 // Hold the negotiated caps verbatim: rebuilding from video_info drops DRM modifiers and other
364 // negotiated fields the downstream frame mapping relies on.
365 p->cached_caps = gst_caps_ref(caps);
366 p->gpu_caps = caps_carry_gpu_memory(caps);
367 p->resolved_path_cache = HwResolvedPathCache{};
368 p->demotion_recorded = false;
369 // New caps = new segment; clear PTS history so a restart/format change that resumes at a
370 // lower PTS isn't wedged by the monotonic-PTS guard in show_frame.
371 p->last_pts_ns.store(-1, std::memory_order_relaxed);
372 p->caps_valid.store(true, std::memory_order_release);
373
374 // Posted on the bus so the controller can mirror negotiation state into Q_PROPERTY.
375 GstStructure* const capsInfo =
376 gst_structure_new(GstQgc::kCapsInfoMessage, "width", G_TYPE_INT, w, "height", G_TYPE_INT, h, "format",
377 G_TYPE_STRING, gst_video_format_to_string(GST_VIDEO_INFO_FORMAT(&parsedInfo)), nullptr);
378 gst_element_post_message(GST_ELEMENT(self), gst_message_new_element(GST_OBJECT(self), capsInfo));
379
380 qCInfo(GstQgcQVideoSinkLog).noquote()
381 << "set_caps: format=" << gst_video_format_to_string(GST_VIDEO_INFO_FORMAT(&parsedInfo)) << w << "x" << h
382 << "pixfmt=" << int(pixelFormat);
383 return TRUE;
384}

386// Sustained run of map failures (not a transient hiccup) means the import is broken — tear down + restart.
387constexpr quint64 kMaxConsecutiveMapFailures = 120;
388
389// A backward PTS jump larger than this is a new timeline (source restart, loop), not reordered output to drop.
390constexpr int64_t kPtsResyncThresholdNs = static_cast<int64_t>(GST_SECOND);
391
392static GstFlowReturn gst_qgc_q_video_sink_show_frame(GstVideoSink* vsink, GstBuffer* buf)
393{
394 GstQgcQVideoSink* self = GST_QGC_Q_VIDEO_SINK(vsink);
395 PrivState* p = priv_of(self);
396
397 p->input_frames.fetch_add(1, std::memory_order_relaxed);
398
399 if (!p->caps_valid.load(std::memory_order_acquire)) {
400 // Should never happen — GstBaseSink calls set_caps before show_frame.
401 return GST_FLOW_NOT_NEGOTIATED;
402 }
403 if (!p->active.load(std::memory_order_relaxed)) {
404 p->dropped_frames.fetch_add(1, std::memory_order_relaxed);
405 return GST_FLOW_OK; // drop silently — controller drives the active flag
406 }
407
408 HwVideoBufferContext hwCtx;
409 if (!snapshot_sink(self, hwCtx)) {
410 p->dropped_frames.fetch_add(1, std::memory_order_relaxed);
411 return GST_FLOW_OK; // no destination yet; drop
412 }
413 // Software-decoded frames skip GPU selection. Caps alone can't tell: GPU memory also arrives under plain
414 // video/x-raw (vtdec's CoreVideo/IOSurface buffers, V4L2 dmabuf export), so frames in other memory still try it.
415 hwCtx.gpuEnabled = hwCtx.gpuEnabled && (p->gpu_caps || !nothing_to_import(buf));
416
417 // PTS regression guard ahead of mapping: a regressed timestamp wedges QVideoOutput's internal
418 // advance, and checking first avoids a wasted full-frame map on a buffer we'd drop anyway.
419 // last_pts_ns is advanced only once the frame is actually delivered (below) so a transient map
420 // failure doesn't push it past a buffer we never rendered. DISCONT and large jumps resync instead.
421 const bool hasPts = GST_BUFFER_PTS_IS_VALID(buf);
422 const int64_t pts = hasPts ? static_cast<int64_t>(GST_BUFFER_PTS(buf)) : -1;
423 if (hasPts) {
424 const int64_t lastPts = p->last_pts_ns.load(std::memory_order_acquire);
425 const bool resync =
426 GST_BUFFER_FLAG_IS_SET(buf, GST_BUFFER_FLAG_DISCONT) || (lastPts - pts) > kPtsResyncThresholdNs;
427 if (lastPts >= 0 && pts < lastPts && !resync) {
428 p->dropped_frames.fetch_add(1, std::memory_order_relaxed);
429 return GST_FLOW_OK;
430 }
431 }
432
433 // Build a cropped format copy only when crop meta is present (rare); otherwise pass
434 // p->format by reference to avoid a per-frame QVideoFrameFormat refcount bump.
435 QVideoFrameFormat croppedFmt;
436 const bool hasCrop = (gst_buffer_get_video_crop_meta(buf) != nullptr);
437 if (hasCrop) {
438 croppedFmt = applyCropMeta(p->format, buf);
439 }
440 MappedFrame mapped = mapSampleToFrame(buf, p->cached_caps, p->video_info, hasCrop ? croppedFmt : p->format, hwCtx,
441 p->pinned_cpu_frames, &p->resolved_path_cache);
442 if (!mapped.frame.isValid()) {
443 p->dropped_frames.fetch_add(1, std::memory_order_relaxed);
444 const quint64 c = p->consecutive_map_failures.fetch_add(1, std::memory_order_relaxed) + 1;
445 if ((c & 0x3F) == 1) {
446 qCWarning(GstQgcQVideoSinkLog) << "mapping failed, consecutive=" << c;
447 }
448 if (c >= kMaxConsecutiveMapFailures) {
449 const char* const format = gst_video_format_to_string(GST_VIDEO_INFO_FORMAT(&p->video_info));
450 qCWarning(GstQgcQVideoSinkLog) << c << "consecutive map failures — erroring out";
451 // Without an element error upstream reports only a generic "Internal data stream error".
452 GST_ELEMENT_ERROR(self, STREAM, FAILED, ("Video frames could not be mapped for display"),
453 ("%" G_GUINT64_FORMAT " consecutive map failures (path %s, format %s)",
454 static_cast<guint64>(c), GstHw::toString(mapped.gpuPath), format ? format : "unknown"));
455 return GST_FLOW_ERROR;
456 }
457 return GST_FLOW_OK; // drop transient failure, keep the stream alive
458 }
459 p->consecutive_map_failures.store(0, std::memory_order_relaxed);
460
461 // Stream orientation is always IDENTITY here; tag-driven orientation lives in the
462 // controller (GST_TAG_IMAGE_ORIENTATION). Per-buffer GstVideoOrientationMeta still wins.
463 applyOrientationAndTiming(mapped.frame, buf, static_cast<int>(GST_VIDEO_ORIENTATION_IDENTITY));
464
465 const QSize frameSize = mapped.frame.size();
466 if (!push_frame_queued(self, std::move(mapped.frame))) {
467 p->dropped_frames.fetch_add(1, std::memory_order_relaxed);
468 return GST_FLOW_OK; // QVideoSink unbound while mapping
469 }
470
471 p->path_frames[static_cast<std::size_t>(mapped.gpuPath)].fetch_add(1, std::memory_order_relaxed);
472 if (mapped.demoted) {
473 p->gpu_fallbacks.fetch_add(1, std::memory_order_relaxed);
474 if (!std::exchange(p->demotion_recorded, true)) {
475 p->stream_demotions.fetch_add(1, std::memory_order_relaxed);
476 }
477 }
478
479 const quint64 delivered = p->delivered_frames.fetch_add(1, std::memory_order_relaxed) + 1;
480 if (delivered == 1) {
481 qCInfo(GstQgcQVideoSinkLog).noquote()
482 << "first frame delivered via" << GstHw::toString(mapped.gpuPath) << "path"
483 << QStringLiteral("%1x%2").arg(frameSize.width()).arg(frameSize.height());
484 } else if ((delivered % 300) == 0) {
485 const auto& cpuFrames = p->path_frames[static_cast<std::size_t>(HwVideoBufferPath::None)];
486 qCDebug(GstQgcQVideoSinkLog).noquote()
487 << "frame flow:" << GstHw::toString(mapped.gpuPath) << "delivered=" << delivered
488 << "input=" << p->input_frames.load(std::memory_order_relaxed)
489 << "dropped=" << p->dropped_frames.load(std::memory_order_relaxed)
490 << "cpuFrames=" << cpuFrames.load(std::memory_order_relaxed);
491 }
492 if (hasPts) {
493 p->last_pts_ns.store(pts, std::memory_order_release);
494 }
495 return GST_FLOW_OK;
496}

498// Resets the PTS guard on every timeline boundary GstBaseSink does not route through set_caps (it skips set_caps when

505 drop_pending_frame(self);

508 case GST_EVENT_SEGMENT:
509 case GST_EVENT_STREAM_START:
510 priv_of(self)->last_pts_ns.store(-1, std::memory_order_release);
511 break;
512 default:
513 break;
514 }
515 return GST_BASE_SINK_CLASS(gst_qgc_q_video_sink_parent_class)->event(bsink, event);
516}

531// READY->NULL: streaming has stopped, so drop all negotiated and per-stream state; a restart renegotiates from scratch
532// and no undelivered frame keeps a decoder buffer alive.
533static gboolean gst_qgc_q_video_sink_stop(GstBaseSink* bsink)
534{
535 GstQgcQVideoSink* self = GST_QGC_Q_VIDEO_SINK(bsink);
536 PrivState* p = priv_of(self);
537 p->caps_valid.store(false, std::memory_order_release);
538 gst_clear_caps(&p->cached_caps);
539 p->gpu_caps = false;

569 return TRUE;
570 }

692PathStats formatSinkStats(const GstStructure* stats, const GstStructure* since) noexcept
693{
694 PathStats out;
695 if (!stats) {
696 return out;
697 }
698 const auto countOf = [stats, since](const char* field) -> quint64 {
699 guint64 now = 0;
700 guint64 before = 0;
701 (void) gst_structure_get_uint64(stats, field, &now);
702 if (since) {
703 (void) gst_structure_get_uint64(since, field, &before);
704 }
705 return (now > before) ? (now - before) : 0;
706 };
707 for (std::size_t i = 0; i < GstHw::kPathCount; ++i) {
708 const auto path = HwVideoBufferPath(i);
709 if (const quint64 frames = countOf(sinkFramesField(path)); frames > 0) {
710 out.line += QStringLiteral(" %1:%2").arg(QLatin1String(GstHw::toString(path))).arg(frames);
711 if (path != HwVideoBufferPath::None) {
712 out.totalDelivered += frames;
713 }
714 }
715 }
716 for (const char* const field : {kGpuFallbacksField, kStreamDemotionsField}) {
717 if (const quint64 count = countOf(field); count > 0) {
718 out.line += QStringLiteral(" %1:%2").arg(QLatin1String(field)).arg(count);
719 }
720 }
721 return out;
722}
