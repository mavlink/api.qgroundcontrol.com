51struct PrivState
52{
53 // Caller-owned and never unref'd; written by the controller on the GUI thread under GST_OBJECT_LOCK.
54 QVideoSink* qvideosink = nullptr;
55 // Read lock-free by show_frame.
56 std::atomic<bool> active{true};
57 // start() derives hw_context from it.
58 bool gpu_zerocopy = false;
59 // Set by qgcvideosinkbin before linking (configureSinkCaps); with gpu_zerocopy, they decide what get_caps offers.
60 GstQgc::GpuBinMode gpu_bin = GstQgc::GpuBinMode::Passthrough;
61 bool square_pixels = true;
62 // What get_caps last offered, and for which policy; under GST_OBJECT_LOCK. Rebuilt when the policy changes, e.g.
63 // the render target or the DMABuf import state: building may query EGL.
64 GstCaps* offered_caps = nullptr;
65 GstQgc::SinkCapsPolicy offered_policy;
66 // Negotiated state: set_caps/stop write it and show_frame reads it, all serialised by GstBaseSink.
67 std::atomic<bool> caps_valid{false};
68 GstVideoInfo video_info = {};
69 QVideoFrameFormat format;
70 // Written by start() under GST_OBJECT_LOCK and snapshotted by show_frame, which adds the current DMABuf display.
71 HwVideoBufferContext hw_context = {};
72 // Whether the negotiated caps carry a GPU memory feature. Without one, show_frame skips path selection for frames
73 // in plain system memory.
74 bool gpu_caps = false;
75 // GstVideoOrientationMethod from the stream's image-orientation tag, for frames without an orientation meta.
76 std::atomic<int> stream_orientation{GST_VIDEO_ORIENTATION_IDENTITY};
77 std::atomic<quint64> input_frames{0};
78 std::atomic<quint64> dropped_frames{0};
79 std::atomic<quint64> consecutive_map_failures{0}; // sustained run escalates show_frame to error
80 // Per-element render counter (read via `frames-delivered`) so multi-receiver setups
81 // don't see a shared process-global total.
82 std::atomic<quint64> delivered_frames{0};
83 // Delivered frames per path (None counts CPU frames), frames a GPU path was tried for but delivered through the
84 // CPU, and caps epochs that started on a GPU path and fell back to the CPU; reported through "frame-stats".

145GstCaps* offered_caps(GstQgcQVideoSink* self)
146{
147 PrivState* const p = priv_of(self);
148 const GstQgc::RenderTarget target = GstQgc::currentRenderTarget();
149 void* const dmaBufDisplay = HwBuffers::dmaBufImportDisplay();
150 GST_OBJECT_LOCK(self);
151 const GstQgc::SinkCapsPolicy policy = caps_policy_locked(p, target, dmaBufDisplay);
152 if (p->offered_caps && (p->offered_policy == policy)) {
153 GstCaps* const cached = gst_caps_ref(p->offered_caps);
154 GST_OBJECT_UNLOCK(self);
155 return cached;
156 }
157 GST_OBJECT_UNLOCK(self);
158
159 GstCaps* caps = gst_caps_from_string(GstQgc::buildSinkCaps(policy).c_str());
160 if (!caps) {
161 GST_ERROR_OBJECT(self, "sink caps did not parse");
162 return gst_caps_new_empty();
163 }
164 GST_DEBUG_OBJECT(self, "offering %" GST_PTR_FORMAT " for render backend %s (EGL %d, DMABuf import %d)", caps,
165 GstQgc::toString(target.backend), target.egl, policy.dmaBufDisplay != nullptr);
166 GST_OBJECT_LOCK(self);
167 gst_caps_replace(&p->offered_caps, caps);
168 p->offered_policy = policy;
169 GST_OBJECT_UNLOCK(self);
170 return caps;
171}

176GstQgc::SinkCapsPolicy reconfigure_on_policy_change(GstQgcQVideoSink* self)
177{
178 PrivState* const p = priv_of(self);
179 const GstQgc::RenderTarget target = GstQgc::currentRenderTarget();
180 void* const dmaBufDisplay = HwBuffers::dmaBufImportDisplay();
181 GST_OBJECT_LOCK(self);
182 const GstQgc::SinkCapsPolicy policy = caps_policy_locked(p, target, dmaBufDisplay);
183 const bool changed = p->offered_caps && (p->offered_policy != policy);
184 if (changed) {
185 // Rebuilt by the next get_caps, which also re-arms this check.
186 gst_clear_caps(&p->offered_caps);
187 }
188 GST_OBJECT_UNLOCK(self);
189 if (changed) {
190 GST_INFO_OBJECT(self, "render backend now %s (EGL %d), DMABuf import %d; asking upstream to renegotiate",
191 GstQgc::toString(target.backend), target.egl, policy.dmaBufDisplay != nullptr);
192 (void) gst_pad_push_event(GST_BASE_SINK_PAD(self), gst_event_new_reconfigure());
193 }
194 return policy;
195}

198bool caps_carry_gpu_memory(GstCaps* caps)
199{
200 const GstCapsFeatures* const features = gst_caps_get_features(caps, 0);
201 if (!features || gst_caps_features_is_any(features)) {
202 return false;
203 }
204 for (guint i = 0; i < gst_caps_features_get_size(features); ++i) {
205 const gchar* const feature = gst_caps_features_get_nth(features, i);
206 if (g_str_has_prefix(feature, "memory:") && (g_strcmp0(feature, GST_CAPS_FEATURE_MEMORY_SYSTEM_MEMORY) != 0)) {
207 return true;
208 }
209 }
210 return false;
211}

256bool push_frame_queued(GstQgcQVideoSink* self, QVideoFrame&& frame)
257{
258 QVideoFrame superseded;
259 GST_OBJECT_LOCK(self);
260 QVideoSink* const sink = priv_of(self)->qvideosink;
261 if (!sink) {
262 GST_OBJECT_UNLOCK(self);
263 return false;
264 }
265 const std::shared_ptr<PendingFrameSlot> slot = priv_of(self)->pending;
266 bool needPost = false;
267 {
268 QMutexLocker locker(&slot->mutex);
269 superseded = std::exchange(slot->frame, std::move(frame));
270 needPost = !std::exchange(slot->posted, true);
271 }
272 if (needPost) {
273 QMetaObject::invokeMethod(
274 sink,
275 [sink, slot]() {
276 QVideoFrame latest;
277 {
278 QMutexLocker locker(&slot->mutex);
279 latest = std::exchange(slot->frame, QVideoFrame());
280 slot->posted = false;
281 }
282 if (latest.isValid()) {
283 sink->setVideoFrame(latest);
284 }
285 },
286 Qt::QueuedConnection);
287 }
288 GST_OBJECT_UNLOCK(self);
289 return true;
290}

344{
345 GstQgcQVideoSink* self = GST_QGC_Q_VIDEO_SINK(obj);
346 PrivState* p = priv_of(self);

348 GST_OBJECT_LOCK(self);
349 switch (id) {

362 p->active.store(g_value_get_boolean(val), std::memory_order_relaxed);
363 break;
364 case PROP_GPU_ZEROCOPY:
365 p->gpu_zerocopy = g_value_get_boolean(val);
366 break;
367 default:
368 G_OBJECT_WARN_INVALID_PROPERTY_ID(obj, id, pspec);
369 break;
370 }
371 GST_OBJECT_UNLOCK(self);
372}

418static gboolean gst_qgc_q_video_sink_set_caps(GstBaseSink* bsink, GstCaps* caps)
419{
420 GstQgcQVideoSink* self = GST_QGC_Q_VIDEO_SINK(bsink);
421 PrivState* p = priv_of(self);
422
423 GstVideoInfo parsedInfo = {};
424 if (!GstHw::dmaDrmAwareVideoInfo(caps, &parsedInfo)) {
425 qCWarning(GstQgcQVideoSinkLog) << "set_caps: failed to parse video info from caps";
426 return FALSE;
427 }
428
429 const QVideoFrameFormat::PixelFormat pixelFormat = toQtPixelFormat(GST_VIDEO_INFO_FORMAT(&parsedInfo));
430 if (pixelFormat == QVideoFrameFormat::Format_Invalid) {
431 qCWarning(GstQgcQVideoSinkLog) << "set_caps: unsupported video format"
432 << gst_video_format_to_string(GST_VIDEO_INFO_FORMAT(&parsedInfo));
433 return FALSE;
434 }
435
436 const int w = GST_VIDEO_INFO_WIDTH(&parsedInfo);
437 const int h = GST_VIDEO_INFO_HEIGHT(&parsedInfo);
438 if (w <= 0 || h <= 0) {
439 qCWarning(GstQgcQVideoSinkLog) << "set_caps: invalid dimensions" << w << "x" << h;
440 return FALSE;
441 }
442
443 QVideoFrameFormat fmt(QSize(w, h), pixelFormat);
444 applyColorimetry(fmt, parsedInfo, caps);
445 const int fpsN = GST_VIDEO_INFO_FPS_N(&parsedInfo);
446 const int fpsD = GST_VIDEO_INFO_FPS_D(&parsedInfo);
447 if (fpsN > 0 && fpsD > 0) {
448 fmt.setStreamFrameRate(static_cast<qreal>(fpsN) / static_cast<qreal>(fpsD));
449 }
450
451 p->video_info = parsedInfo;
452 p->format = std::move(fmt);
453 gst_clear_caps(&p->cached_caps);
454 // Hold the negotiated caps verbatim: rebuilding from video_info drops DRM modifiers and other
455 // negotiated fields the downstream frame mapping relies on.
456 p->cached_caps = gst_caps_ref(caps);
457 p->gpu_caps = caps_carry_gpu_memory(caps);
458 p->resolved_path_cache = HwResolvedPathCache{};
459 p->demotion_recorded = false;
460 p->tiled_drop_logged = false;
461 p->caps_valid.store(true, std::memory_order_release);
462
463 qCInfo(GstQgcQVideoSinkLog).noquote()
464 << "set_caps: format=" << gst_video_format_to_string(GST_VIDEO_INFO_FORMAT(&parsedInfo)) << w << "x" << h
465 << "pixfmt=" << int(pixelFormat);
466 return TRUE;
467}

469// Sustained run of map failures (not a transient hiccup) means the import is broken — tear down + restart.
470constexpr quint64 kMaxConsecutiveMapFailures = 120;

472static GstFlowReturn gst_qgc_q_video_sink_show_frame(GstVideoSink* vsink, GstBuffer* buf)
473{
474 GstQgcQVideoSink* self = GST_QGC_Q_VIDEO_SINK(vsink);
475 PrivState* p = priv_of(self);
476
477 p->input_frames.fetch_add(1, std::memory_order_relaxed);
478 const GstQgc::SinkCapsPolicy policy = reconfigure_on_policy_change(self);
479
480 if (!p->caps_valid.load(std::memory_order_acquire)) {
481 // Should never happen — GstBaseSink calls set_caps before show_frame.
482 return GST_FLOW_NOT_NEGOTIATED;
483 }
484 if (!p->active.load(std::memory_order_relaxed)) {
485 p->dropped_frames.fetch_add(1, std::memory_order_relaxed);
486 return GST_FLOW_OK; // drop silently — controller drives the active flag
487 }
488
489 HwVideoBufferContext hwCtx;
490 if (!snapshot_sink(self, hwCtx)) {
491 p->dropped_frames.fetch_add(1, std::memory_order_relaxed);
492 return GST_FLOW_OK; // no destination yet; drop
493 }
494 hwCtx.dmaBufEglDisplay = policy.dmaBufDisplay;
495 // Software-decoded frames skip GPU selection. Caps alone can't tell: GPU memory also arrives under plain
496 // video/x-raw (vtdec's CoreVideo/IOSurface buffers, V4L2 dmabuf export), so frames in other memory still try it.
497 hwCtx.gpuEnabled = hwCtx.gpuEnabled && (p->gpu_caps || !nothing_to_import(buf));
498
499 // Build a cropped format copy only when crop meta is present (rare); otherwise pass
500 // p->format by reference to avoid a per-frame QVideoFrameFormat refcount bump.
501 QVideoFrameFormat croppedFmt;
502 const bool hasCrop = (gst_buffer_get_video_crop_meta(buf) != nullptr);
503 if (hasCrop) {
504 croppedFmt = applyCropMeta(p->format, buf);
505 }
506 MappedFrame mapped = mapSampleToFrame(buf, p->cached_caps, p->video_info, hasCrop ? croppedFmt : p->format, hwCtx,
507 p->pinned_cpu_frames, &p->resolved_path_cache);
508 if (!mapped.frame.isValid()) {
509 p->dropped_frames.fetch_add(1, std::memory_order_relaxed);
510 const quint64 c = p->consecutive_map_failures.fetch_add(1, std::memory_order_relaxed) + 1;
511 if (mapped.noCpuLayout) {
512 // Until upstream renegotiates (the sink asked for it when the DMABuf import stopped), or the run escalates.
513 p->tiled_drops.fetch_add(1, std::memory_order_relaxed);
514 if (!std::exchange(p->tiled_drop_logged, true)) {
515 qCWarning(GstQgcQVideoSinkLog) << "tiled DMA_DRM frame not imported and has no CPU layout — dropped";
516 }
517 } else if ((c & 0x3F) == 1) {
518 qCWarning(GstQgcQVideoSinkLog) << "mapping failed, consecutive=" << c;
519 }
520 if (c >= kMaxConsecutiveMapFailures) {
521 const char* const format = gst_video_format_to_string(GST_VIDEO_INFO_FORMAT(&p->video_info));
522 qCWarning(GstQgcQVideoSinkLog) << c << "consecutive map failures — erroring out";
523 // Without an element error upstream reports only a generic "Internal data stream error".
524 GST_ELEMENT_ERROR(self, STREAM, FAILED, ("Video frames could not be mapped for display"),
525 ("%" G_GUINT64_FORMAT " consecutive map failures (path %s, format %s)",
526 static_cast<guint64>(c), GstHw::toString(mapped.gpuPath), format ? format : "unknown"));
527 return GST_FLOW_ERROR;
528 }
529 return GST_FLOW_OK; // drop transient failure, keep the stream alive
530 }
531 p->consecutive_map_failures.store(0, std::memory_order_relaxed);
532
533 applyOrientationAndTiming(mapped.frame, buf, p->stream_orientation.load(std::memory_order_relaxed));
534
535 const QSize frameSize = mapped.frame.size();
536 if (!push_frame_queued(self, std::move(mapped.frame))) {
537 p->dropped_frames.fetch_add(1, std::memory_order_relaxed);
538 return GST_FLOW_OK; // QVideoSink unbound while mapping
539 }
540
541 p->path_frames[static_cast<std::size_t>(mapped.gpuPath)].fetch_add(1, std::memory_order_relaxed);
542 if (mapped.demoted) {
543 p->gpu_fallbacks.fetch_add(1, std::memory_order_relaxed);
544 if (!std::exchange(p->demotion_recorded, true)) {
545 p->stream_demotions.fetch_add(1, std::memory_order_relaxed);
546 }
547 }
548
549 const quint64 delivered = p->delivered_frames.fetch_add(1, std::memory_order_relaxed) + 1;
550 if (delivered == 1) {

565// Tracks the stream's orientation tag and releases an undelivered frame on flush.
566static gboolean gst_qgc_q_video_sink_event(GstBaseSink* bsink, GstEvent* event)
567{
568 GstQgcQVideoSink* self = GST_QGC_Q_VIDEO_SINK(bsink);
569 switch (GST_EVENT_TYPE(event)) {
570 case GST_EVENT_TAG: {
571 // A camera mounted sideways tags the stream (e.g. "rotate-90") instead of rotating every frame.
572 GstTagList* tags = nullptr;
573 gst_event_parse_tag(event, &tags);
574 GstVideoOrientationMethod method = GST_VIDEO_ORIENTATION_IDENTITY;
575 if (tags && gst_video_orientation_from_tag(tags, &method)) {
576 priv_of(self)->stream_orientation.store(method, std::memory_order_relaxed);
577 }
578 break;
579 }
580 case GST_EVENT_FLUSH_START:
581 drop_pending_frame(self);
582 break;
583 case GST_EVENT_STREAM_START:
584 // A new stream starts untagged; its tags follow this event.
585 priv_of(self)->stream_orientation.store(GST_VIDEO_ORIENTATION_IDENTITY, std::memory_order_relaxed);
586 break;
587 default:
588 break;
589 }
590 return GST_BASE_SINK_CLASS(gst_qgc_q_video_sink_parent_class)->event(bsink, event);
591}

626// Add qgcqvideosink's consumed metas + min-buffer pool hint, then chain so GstBaseSink's default
627// allocation bookkeeping still runs.
628static gboolean gst_qgc_q_video_sink_propose_allocation(GstBaseSink* bsink, GstQuery* query)
629{
630 GstQgc::populateAllocationQuery(query);
631 // GstBaseSink/GstVideoSink install no default propose_allocation vmethod, so the parent pointer
632 // is NULL for a direct subclass; chaining unconditionally would call 0x0 on the first ALLOCATION query.
633 GstBaseSinkClass* parentClass = GST_BASE_SINK_CLASS(gst_qgc_q_video_sink_parent_class);
634 if (parentClass->propose_allocation) {
635 return parentClass->propose_allocation(bsink, query);
636 }
637 return TRUE;
638}
639
640// Answer CONTEXT (gst.gl.GLDisplay/app_context, Vulkan, D3D devices) synchronously so upstream GPU elements share Qt's
641// RHI device; the bus NEED_CONTEXT fallback races state changes. ALLOCATION reaches propose_allocation via the parent.
642static gboolean gst_qgc_q_video_sink_query(GstBaseSink* bsink, GstQuery* query)
643{
644 if ((GST_QUERY_TYPE(query) == GST_QUERY_CONTEXT) && HwBuffers::answerSinkBinContextQuery(query)) {
645 return TRUE;
646 }
647 if (GST_QUERY_TYPE(query) == GST_QUERY_ACCEPT_CAPS) {
648 // GstBaseSink wants a subset of get_caps, which rejects caps without a pixel-aspect-ratio (square pixels).
649 GstCaps* caps = nullptr;
650 gst_query_parse_accept_caps(query, &caps);
651 GstCaps* const offered = offered_caps(GST_QGC_Q_VIDEO_SINK(bsink));
652 gst_query_set_accept_caps_result(query, caps && gst_caps_can_intersect(caps, offered));
653 gst_caps_unref(offered);
654 return TRUE;
655 }
656 return GST_BASE_SINK_CLASS(gst_qgc_q_video_sink_parent_class)->query(bsink, query);
657}

781PathStats formatSinkStats(const GstStructure* stats, const GstStructure* since) noexcept
782{
783 PathStats out;
784 if (!stats) {
785 return out;
786 }
787 const auto countOf = [stats, since](const char* field) -> quint64 {
788 guint64 now = 0;
789 guint64 before = 0;
790 (void) gst_structure_get_uint64(stats, field, &now);
791 if (since) {
792 (void) gst_structure_get_uint64(since, field, &before);
793 }
794 return (now > before) ? (now - before) : 0;
795 };
796 for (std::size_t i = 0; i < GstHw::kPathCount; ++i) {
797 const auto path = HwVideoBufferPath(i);
798 if (const quint64 frames = countOf(sinkFramesField(path)); frames > 0) {
799 out.line += QStringLiteral(" %1:%2").arg(QLatin1String(GstHw::toString(path))).arg(frames);
800 if (path != HwVideoBufferPath::None) {
801 out.totalDelivered += frames;
802 }
803 }
804 }
805 for (const char* const field : {kGpuFallbacksField, kStreamDemotionsField, kTiledDropsField}) {
806 if (const quint64 count = countOf(field); count > 0) {
807 out.line += QStringLiteral(" %1:%2").arg(QLatin1String(field)).arg(count);
808 }
809 }
810 return out;
811}
