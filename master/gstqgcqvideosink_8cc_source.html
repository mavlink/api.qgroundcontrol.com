35struct PrivState
36{
37 // The QVideoSink frames go to, set by the controller on the GUI thread.
38 GstQgc::FrameDelivery delivery;
39 GstQgc::SinkCounters counters;
40 // Read lock-free by show_frame.

42 // start() derives hw_context from it.

100{

103 return {.gpuZeroCopy = p->gpu_zerocopy,

111GstCaps* offered_caps(GstQgcQVideoSink* self)
112{
113 PrivState* const p = priv_of(self);
114 const GstQgc::RenderTarget target = GstQgc::currentRenderTarget();
115 void* const dmaBufDisplay = HwBuffers::dmaBufImportDisplay();
116 GST_OBJECT_LOCK(self);
117 const GstQgc::SinkCapsPolicy policy = caps_policy_locked(p, target, dmaBufDisplay);
118 if (p->offered_caps && (p->offered_policy == policy)) {
119 GstCaps* const cached = gst_caps_ref(p->offered_caps);
120 GST_OBJECT_UNLOCK(self);
121 return cached;
122 }
123 GST_OBJECT_UNLOCK(self);
124
125 GstCaps* caps = gst_caps_from_string(GstQgc::buildSinkCaps(policy).c_str());
126 if (!caps) {
127 GST_ERROR_OBJECT(self, "sink caps did not parse");
128 return gst_caps_new_empty();
129 }
130 GST_DEBUG_OBJECT(self, "offering %" GST_PTR_FORMAT " for render backend %s (EGL %d, DMABuf import %d)", caps,
131 GstQgc::toString(target.backend), target.egl, policy.dmaBufDisplay != nullptr);
132 GST_OBJECT_LOCK(self);
133 gst_caps_replace(&p->offered_caps, caps);
134 p->offered_policy = policy;
135 GST_OBJECT_UNLOCK(self);
136 return caps;
137}

142GstQgc::SinkCapsPolicy reconfigure_on_policy_change(GstQgcQVideoSink* self)
143{
144 PrivState* const p = priv_of(self);
145 const GstQgc::RenderTarget target = GstQgc::currentRenderTarget();
146 void* const dmaBufDisplay = HwBuffers::dmaBufImportDisplay();
147 GST_OBJECT_LOCK(self);
148 const GstQgc::SinkCapsPolicy policy = caps_policy_locked(p, target, dmaBufDisplay);
149 const bool changed = p->offered_caps && (p->offered_policy != policy);
150 if (changed) {
151 // Rebuilt by the next get_caps, which also re-arms this check.
152 gst_clear_caps(&p->offered_caps);
153 }
154 GST_OBJECT_UNLOCK(self);
155 if (changed) {
156 GST_INFO_OBJECT(self, "render backend now %s (EGL %d), DMABuf import %d; asking upstream to renegotiate",
157 GstQgc::toString(target.backend), target.egl, policy.dmaBufDisplay != nullptr);
158 (void) gst_pad_push_event(GST_BASE_SINK_PAD(self), gst_event_new_reconfigure());
159 }
160 return policy;
161}

164bool caps_carry_gpu_memory(GstCaps* caps)
165{
166 const GstCapsFeatures* const features = gst_caps_get_features(caps, 0);
167 if (!features || gst_caps_features_is_any(features)) {
168 return false;
169 }
170 for (guint i = 0; i < gst_caps_features_get_size(features); ++i) {
171 const gchar* const feature = gst_caps_features_get_nth(features, i);
172 if (g_str_has_prefix(feature, "memory:") && (g_strcmp0(feature, GST_CAPS_FEATURE_MEMORY_SYSTEM_MEMORY) != 0)) {
173 return true;
174 }
175 }
176 return false;
177}

307static gboolean gst_qgc_q_video_sink_set_caps(GstBaseSink* bsink, GstCaps* caps)
308{
309 GstQgcQVideoSink* self = GST_QGC_Q_VIDEO_SINK(bsink);
310 PrivState* p = priv_of(self);
311
312 GstVideoInfo parsedInfo = {};
313 if (!GstHw::dmaDrmAwareVideoInfo(caps, &parsedInfo)) {
314 qCWarning(GstQgcQVideoSinkLog) << "set_caps: failed to parse video info from caps";
315 return FALSE;
316 }
317
318 const QVideoFrameFormat::PixelFormat pixelFormat = toQtPixelFormat(GST_VIDEO_INFO_FORMAT(&parsedInfo));
319 if (pixelFormat == QVideoFrameFormat::Format_Invalid) {
320 qCWarning(GstQgcQVideoSinkLog) << "set_caps: unsupported video format"
321 << gst_video_format_to_string(GST_VIDEO_INFO_FORMAT(&parsedInfo));
322 return FALSE;
323 }
324
325 const int w = GST_VIDEO_INFO_WIDTH(&parsedInfo);
326 const int h = GST_VIDEO_INFO_HEIGHT(&parsedInfo);
327 if (w <= 0 || h <= 0) {
328 qCWarning(GstQgcQVideoSinkLog) << "set_caps: invalid dimensions" << w << "x" << h;
329 return FALSE;
330 }
331
332 QVideoFrameFormat fmt(QSize(w, h), pixelFormat);
333 applyColorimetry(fmt, parsedInfo, caps);
334 const int fpsN = GST_VIDEO_INFO_FPS_N(&parsedInfo);
335 const int fpsD = GST_VIDEO_INFO_FPS_D(&parsedInfo);
336 if (fpsN > 0 && fpsD > 0) {
337 fmt.setStreamFrameRate(static_cast<qreal>(fpsN) / static_cast<qreal>(fpsD));
338 }
339
340 p->video_info = parsedInfo;
341 p->format = std::move(fmt);
342 gst_clear_caps(&p->cached_caps);
343 // Hold the negotiated caps verbatim: rebuilding from video_info drops DRM modifiers and other
344 // negotiated fields the downstream frame mapping relies on.
345 p->cached_caps = gst_caps_ref(caps);
346 p->gpu_caps = caps_carry_gpu_memory(caps);
347 p->path_session.startEpoch();
348 p->caps_valid.store(true, std::memory_order_release);
349
350 qCInfo(GstQgcQVideoSinkLog).noquote()
351 << "set_caps: format=" << gst_video_format_to_string(GST_VIDEO_INFO_FORMAT(&parsedInfo)) << w << "x" << h
352 << "pixfmt=" << int(pixelFormat);
353 return TRUE;
354}

356static GstFlowReturn gst_qgc_q_video_sink_show_frame(GstVideoSink* vsink, GstBuffer* buf)
357{
358 GstQgcQVideoSink* self = GST_QGC_Q_VIDEO_SINK(vsink);
359 PrivState* p = priv_of(self);
360
361 p->counters.noteInput();
362 const GstQgc::SinkCapsPolicy policy = reconfigure_on_policy_change(self);
363
364 if (!p->caps_valid.load(std::memory_order_acquire)) {
365 // Should never happen — GstBaseSink calls set_caps before show_frame.
366 return GST_FLOW_NOT_NEGOTIATED;
367 }
368 if (!p->active.load(std::memory_order_relaxed)) {
369 p->counters.noteDropped();
370 return GST_FLOW_OK; // drop silently — controller drives the active flag
371 }
372
373 HwVideoBufferContext hwCtx;
374 if (!snapshot_sink(self, hwCtx)) {
375 p->counters.noteDropped();
376 return GST_FLOW_OK; // no destination yet; drop
377 }
378 hwCtx.dmaBufEglDisplay = policy.dmaBufDisplay;
379 // Software-decoded frames skip GPU selection. Caps alone can't tell: GPU memory also arrives under plain
380 // video/x-raw (vtdec's CoreVideo/IOSurface buffers, V4L2 dmabuf export), so frames in other memory still try it.
381 hwCtx.gpuEnabled = hwCtx.gpuEnabled && (p->gpu_caps || !nothing_to_import(buf));
382
383 // Build a cropped format copy only when crop meta is present (rare); otherwise pass
384 // p->format by reference to avoid a per-frame QVideoFrameFormat refcount bump.
385 QVideoFrameFormat croppedFmt;
386 const bool hasCrop = (gst_buffer_get_video_crop_meta(buf) != nullptr);
387 if (hasCrop) {
388 croppedFmt = applyCropMeta(p->format, buf);
389 }
390 MappedFrame mapped = mapSampleToFrame(buf, p->cached_caps, p->video_info, hasCrop ? croppedFmt : p->format, hwCtx,
391 p->pinned_cpu_frames, p->path_session.pathCache());
392 if (!mapped.frame.isValid()) {
393 p->counters.noteDropped();
394 if (p->path_session.onMapFailed(mapped, p->counters) == GstQgc::PathSession::Verdict::Error) {
395 const quint64 failures = p->path_session.consecutiveMapFailures();
396 const char* const format = gst_video_format_to_string(GST_VIDEO_INFO_FORMAT(&p->video_info));
397 // Without an element error upstream reports only a generic "Internal data stream error".
398 GST_ELEMENT_ERROR(
399 self, STREAM, FAILED, ("Video frames could not be mapped for display"),
400 ("%" G_GUINT64_FORMAT " consecutive map failures (path %s, format %s)", static_cast<guint64>(failures),
401 GstHw::toString(mapped.gpuPath), format ? format : "unknown"));
402 return GST_FLOW_ERROR;
403 }
404 return GST_FLOW_OK; // drop transient failure, keep the stream alive
405 }
406 p->path_session.onMapped();
407
408 applyOrientationAndTiming(mapped.frame, buf, p->stream_orientation.load(std::memory_order_relaxed));
409
410 const QSize frameSize = mapped.frame.size();
411 const GstQgc::FrameDelivery::Result delivery = p->delivery.deliver(std::move(mapped.frame));
412 if (!delivery.delivered) {
413 p->counters.noteDropped();
414 return GST_FLOW_OK; // QVideoSink unbound while mapping
415 }
416 const quint64 delivered = p->counters.noteDelivered(mapped.gpuPath, delivery.superseded);
417 p->path_session.onDelivered(mapped, p->counters);
418
419 if (delivered == 1) {
420 qCInfo(GstQgcQVideoSinkLog).noquote()
421 << "first frame delivered via" << GstHw::toString(mapped.gpuPath) << "path"
422 << QStringLiteral("%1x%2").arg(frameSize.width()).arg(frameSize.height());
423 } else if ((delivered % 300) == 0) {
424 qCDebug(GstQgcQVideoSinkLog).noquote()
425 << "frame flow:" << GstHw::toString(mapped.gpuPath) << "delivered=" << delivered
426 << "input=" << p->counters.input() << "dropped=" << p->counters.dropped()
427 << "cpuFrames=" << p->counters.pathFrames(HwVideoBufferPath::None);
428 }
429 return GST_FLOW_OK;
430}

432// Tracks the stream's orientation tag and releases an undelivered frame on flush.
433static gboolean gst_qgc_q_video_sink_event(GstBaseSink* bsink, GstEvent* event)
434{
435 GstQgcQVideoSink* self = GST_QGC_Q_VIDEO_SINK(bsink);
436 switch (GST_EVENT_TYPE(event)) {
437 case GST_EVENT_TAG: {
438 // A camera mounted sideways tags the stream (e.g. "rotate-90") instead of rotating every frame.
439 GstTagList* tags = nullptr;
440 gst_event_parse_tag(event, &tags);
441 GstVideoOrientationMethod method = GST_VIDEO_ORIENTATION_IDENTITY;
442 if (tags && gst_video_orientation_from_tag(tags, &method)) {
443 priv_of(self)->stream_orientation.store(method, std::memory_order_relaxed);
444 }
445 break;
446 }
447 case GST_EVENT_FLUSH_START:
448 priv_of(self)->delivery.dropPending();
449 break;
450 case GST_EVENT_STREAM_START:
451 // A new stream starts untagged; its tags follow this event.
452 priv_of(self)->stream_orientation.store(GST_VIDEO_ORIENTATION_IDENTITY, std::memory_order_relaxed);
453 break;
454 default:
455 break;
456 }
457 return GST_BASE_SINK_CLASS(gst_qgc_q_video_sink_parent_class)->event(bsink, event);
458}

504// Answer CONTEXT (gst.gl.GLDisplay/app_context, Vulkan, D3D devices) synchronously so upstream GPU elements share Qt's
505// RHI device; the bus NEED_CONTEXT fallback races state changes. ALLOCATION reaches propose_allocation via the parent.
506static gboolean gst_qgc_q_video_sink_query(GstBaseSink* bsink, GstQuery* query)
507{
508 if ((GST_QUERY_TYPE(query) == GST_QUERY_CONTEXT) && HwBuffers::answerSinkBinContextQuery(query)) {
509 return TRUE;
510 }
511 if (GST_QUERY_TYPE(query) == GST_QUERY_ACCEPT_CAPS) {
512 // GstBaseSink wants a subset of get_caps, which rejects caps without a pixel-aspect-ratio (square pixels).
513 GstCaps* caps = nullptr;
514 gst_query_parse_accept_caps(query, &caps);
515 GstCaps* const offered = offered_caps(GST_QGC_Q_VIDEO_SINK(bsink));
516 gst_query_set_accept_caps_result(query, caps && gst_caps_can_intersect(caps, offered));
517 gst_caps_unref(offered);
518 return TRUE;
519 }
520 return GST_BASE_SINK_CLASS(gst_qgc_q_video_sink_parent_class)->query(bsink, query);
521}
