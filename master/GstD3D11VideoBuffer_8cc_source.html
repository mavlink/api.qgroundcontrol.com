161class D3D11VideoBuffer final : public GstD3DVideoBufferCommon::VideoBuffer<D3D11PlaneResolver>
162{
163public:
164 D3D11VideoBuffer(GstSample* sample, const GstVideoInfo& videoInfo, const QVideoFrameFormat& format)
165 : VideoBuffer(sample, videoInfo, format, HwVideoBufferPath::D3D11, s_diag)
166 {
167 D3D11PlaneResolver resolver;
168 if (!resolve(resolver)) {
169 return;
170 }
171 _slices = resolver.slices();
172 _copied = resolver.copied();
173
174 // Decoders pad their textures to their alignment (16 rows for H.264, 128 for HEVC/AV1) with no crop meta. The
175 // picture is copied out like an array slice's, so the frame keeps the picture's size and its CPU layout: Qt
176 // maps it to the CPU for a window on another device and for screenshots.
177 D3D11_TEXTURE2D_DESC desc = {};
178 _handles[0]->GetDesc(&desc);
179 if (GstHw::texturePadsPicture(QSize(int(desc.Width), int(desc.Height)), _format.frameSize(),
180 hasEvenSizes(desc.Format))) {
181 std::fill_n(_copied.begin(), _count, true);
182 }
183 }
184
185 QVideoFrameTexturesUPtr mapTextures(QRhi& rhi, QVideoFrameTexturesUPtr& old) override
186 {
187 GstBuffer* const buffer = beginMap(rhi, QRhi::D3D11);
188 if (!buffer) {
189 return GstHwPathTelemetry::fail(HwVideoBufferPath::D3D11);
190 }
191 // Also catches an unprimed bridge, which skips resolvePlanes()'s device guard: QRhi can't sample another
192 // device's texture.
193 const auto* native = static_cast<const QRhiD3D11NativeHandles*>(rhi.nativeHandles());
194 auto* device = native ? static_cast<ID3D11Device*>(native->dev) : nullptr;
195 auto* context = native ? static_cast<ID3D11DeviceContext*>(native->context) : nullptr;
196 if (!device || !context || !onDevice(_handles[0], device)) {
197 QGC_HW_WARN_ONCE(GstD3D11Log, s_loggedForeignDevice,
198 "mapTextures: decoder texture is not on QRhi's device — CPU fallback");
199 return GstHwPathTelemetry::fail(HwVideoBufferPath::D3D11);
200 }
201
202 std::array<ID3D11Texture2D*, kMaxPlanes> planes = _handles;
203 std::array<ID3D11Texture2D*, kMaxPlanes> copies{};
204 const auto releaseCopies = qScopeGuard([&copies] {
205 for (ID3D11Texture2D* copy : copies) {
206 if (copy) {
207 copy->Release();
208 }
209 }
210 });
211 for (int i = 0; i < _count; ++i) {
212 if (!_copied[std::size_t(i)]) {
213 continue;
214 }
215 copies[std::size_t(i)] = copyPicture(rhi, device, context, buffer, i, old);
216 if (!copies[std::size_t(i)]) {
217 return GstHwPathTelemetry::fail(HwVideoBufferPath::D3D11);
218 }
219 planes[std::size_t(i)] = copies[std::size_t(i)];
220 }
221
222 return finishMap(rhi, old, planes);
223 }

230 ID3D11Texture2D* copyPicture(QRhi& rhi, ID3D11Device* device, ID3D11DeviceContext* context, GstBuffer* buffer,
231 int plane, QVideoFrameTexturesUPtr& old)
232 {
233 ID3D11Texture2D* source = _handles[std::size_t(plane)];
234 D3D11_TEXTURE2D_DESC sourceDesc = {};
235 source->GetDesc(&sourceDesc);
236 // One texture holds every plane (NV12, P010), else this plane's texture holds only its own share of the frame.
237 const auto* textureDescription = QVideoTextureHelper::textureDescription(_format.pixelFormat());
238 const QSize picture = ((_count > 1) && textureDescription)
239 ? textureDescription->rhiPlaneSize(_format.frameSize(), plane, &rhi)
240 : _format.frameSize();
241 const UINT width = (std::min) (copyExtent(picture.width(), sourceDesc.Format), sourceDesc.Width);
242 const UINT height = (std::min) (copyExtent(picture.height(), sourceDesc.Format), sourceDesc.Height);
243
244 D3D11_TEXTURE2D_DESC targetDesc = {};
245 targetDesc.Width = width;
246 targetDesc.Height = height;
247 targetDesc.MipLevels = 1;
248 targetDesc.ArraySize = 1;
249 targetDesc.Format = sourceDesc.Format;
250 targetDesc.SampleDesc.Count = 1;
251 targetDesc.Usage = D3D11_USAGE_DEFAULT;
252 targetDesc.BindFlags = D3D11_BIND_SHADER_RESOURCE;
253
254 ID3D11Texture2D* target = reusableTarget(old, plane, targetDesc);
255 if (!target) {
256 if (FAILED(device->CreateTexture2D(&targetDesc, nullptr, &target)) || !target) {
257 QGC_HW_WARN_ONCE(GstD3D11Log, s_diag.loggedTextureCreateFail,
258 "mapTextures: CreateTexture2D for the slice copy failed (plane=" << plane << ")");
259 return nullptr;
260 }
261 (void) target->SetPrivateData(kCopyTargetTag, static_cast<UINT>(sizeof(kCopyTargetMark)), &kCopyTargetMark);
262 }
263
264 // Serializes with GStreamer's own use of this immediate context (the decoder, on the streaming thread).
265 auto* memory = reinterpret_cast<GstD3D11Memory*>(gst_buffer_peek_memory(buffer, guint(plane)));
266 const D3D11_BOX pictureBox = {0, 0, 0, width, height, 1};
267 gst_d3d11_device_lock(memory->device);
268 context->CopySubresourceRegion(target, 0, 0, 0, 0, source, _slices[std::size_t(plane)], &pictureBox);
269 gst_d3d11_device_unlock(memory->device);
270 return target;
271 }

274 static ID3D11Texture2D* reusableTarget(QVideoFrameTexturesUPtr& old, int plane, const D3D11_TEXTURE2D_DESC& desc)
275 {
276 auto* previous = dynamic_cast<D3D11FrameTextures*>(old.get());
277 ID3D11Texture2D* texture = previous ? previous->handle(plane) : nullptr;
278 if (!texture || !isCopyTarget(texture)) {
279 return nullptr;
280 }
281 D3D11_TEXTURE2D_DESC have = {};
282 texture->GetDesc(&have);
283 if ((have.Width != desc.Width) || (have.Height != desc.Height) || (have.Format != desc.Format)) {
284 return nullptr;
285 }
286 texture->AddRef();
287 return texture;
288 }
