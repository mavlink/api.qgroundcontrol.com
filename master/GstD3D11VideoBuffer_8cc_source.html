45 ID3D11Texture2D* acquire(ID3D11Device* dev, const StagingKey& key, const D3D11_TEXTURE2D_DESC& dstDesc,
46 int planeIdx, guint subIdx)
47 {
48 QMutexLocker lock(&_mutex);
49 StagingRing* ring = _entries.find(key).resource;
50 if (!ring) {
51 ring = _entries.insert(key, StagingRing{}).resource;
52 }
53 const std::size_t slot = ring->next;
54 ring->next = (ring->next + 1) % kRingSize;
55 if (ring->textures[slot]) {
56 GstHwPathTelemetry::record(HwVideoBufferPath::D3D11, GstHwPathTelemetry::Counter::ImageCacheHits);
57 ring->textures[slot]->AddRef();
58 return ring->textures[slot];
59 }
60 ID3D11Texture2D* tex = nullptr;
61 if (FAILED(dev->CreateTexture2D(&dstDesc, nullptr, &tex))) {
62 QGC_HW_WARN_ONCE(GstD3D11Log, s_diag.loggedTextureCreateFail,
63 "mapTextures: CreateTexture2D for slice copy failed (plane=" << planeIdx << "subresource="
64 << subIdx << ")");
65 return nullptr;
66 }
67 tex->AddRef();
68 ring->textures[slot] = tex; // pool keeps the create ref
69 GstHwPathTelemetry::record(HwVideoBufferPath::D3D11, GstHwPathTelemetry::Counter::ImageCacheMisses);
70 return tex;
71 }
72
73 void clear()
74 {
75 QMutexLocker lock(&_mutex);
76 _entries.clear();
77 }

106ID3D11Texture2D* copySliceToStaging(ID3D11Texture2D* tex, guint subIdx, int planeIdx,
107 const D3D11_TEXTURE2D_DESC& srcDesc, GstD3D11Memory* d3dmem)
108{
109 ID3D11Device* d3dDev = gst_d3d11_device_get_device_handle(d3dmem->device);
110 ID3D11DeviceContext* d3dCtx = gst_d3d11_device_get_device_context_handle(d3dmem->device);
111 D3D11_TEXTURE2D_DESC dstDesc = srcDesc;
112 dstDesc.ArraySize = 1;
113 dstDesc.BindFlags = D3D11_BIND_SHADER_RESOURCE;
114 dstDesc.MiscFlags = 0;
115 dstDesc.MipLevels = 1;
116 const StagingKey key{srcDesc.Width, srcDesc.Height, UINT(srcDesc.Format), planeIdx};
117 ID3D11Texture2D* stagingTex = StagingTexturePool::instance().acquire(d3dDev, key, dstDesc, planeIdx, subIdx);
118 if (!stagingTex) {
119 return nullptr;
120 }
121 // The immediate ID3D11DeviceContext is not free-threaded; gst-d3d11 device-lock contract requires this guard.
122 gst_d3d11_device_lock(d3dmem->device);
123 d3dCtx->CopySubresourceRegion(stagingTex, 0, 0, 0, 0, tex, subIdx, nullptr);
124 gst_d3d11_device_unlock(d3dmem->device);
125 return stagingTex;
126}
