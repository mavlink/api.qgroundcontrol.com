95 bool configure(GPSCommandChannel& channel, GPSConfig, unsigned& baud) override
96 {
97 _ready = false;
98 _reset();
99 channel.stream().reset();
100 if (!channel.setBaudrate(baud)) {
101 channel.failControl(QStringLiteral("The link cannot run at %1 baud").arg(baud));
102 return false;
103 }
104 _ready = true;
105 return true;
106 }
107
108 GPSReceiveUpdates onFrame(const GPSFrame& frame, GPSDecodeContext& context) override
109 {
110 _context = &context;
111 if (frame.kind == GPSFrameKind::RTCM3) {
112 context.publishRTCM(frame.bytes);
113 return {};
114 }
115 const auto input = _inputOf(frame);
116 if (!input) {
117 return {};
118 }
119 if (!_followed) {
120 _follow(*input);
121 }
122 (void) _inputs[*input].runtime->consumeFrame(frame);
123 return {};
124 }
125
126 void flush(GPSDecodeContext& context) override
127 {
128 _context = &context;
129 for (const auto& input : _inputs) {
130 (void) input.runtime->consume({});
131 }
132 }

149 static std::optional<size_t> _inputOf(const GPSFrame& frame)
150 {
151 for (size_t index = 0; index < INPUTS.size(); ++index) {
152 const auto& input = INPUTS[index];
153 if (input.frames.testFlag(frame.kind) && (!input.accepts || input.accepts(frame))) {
154 return index;
155 }
156 }
157 return std::nullopt;
158 }
159
160 void _reset()
161 {
162 _followed.reset();
163 for (size_t index = 0; index < INPUTS.size(); ++index) {
164 GPSRuntimeIO io;
165 io.clock.nowUs = [this] { return _context ? _context->nowUs() : 0; };
166 GPSRuntimeObserver observer;
167 observer.decoded = [this, index](const GPSEventBatch& batch) { _received(index, batch); };
168 _inputs[index] = {.runtime = std::make_unique<GPSProtocolRuntime>(*INPUTS[index].family, std::move(io),
169 std::move(observer)),
170 .lastPositionUs = 0};
171 _inputs[index].runtime->armNavigationDecode();
172 }
173 }
174
175 void _received(size_t index, const GPSEventBatch& batch)
176 {
177 if (batch.updates.testFlag(GPSReceiveUpdate::Position)) {
178 const uint64_t now = _context->nowUs();
179 _inputs[index].lastPositionUs = now;
180 if (!_followed || (*_followed != index &&
181 !MonotonicClock::fresh(_inputs[*_followed].lastPositionUs, now, SOURCE_TIMEOUT))) {
182 _follow(index);
183 }
184 }
185 if (_followed != index) {
186 return;
187 }
188 for (const auto& event : batch.events) {
189 _context->publishAsReceived(event);
190 }
191 _context->markUpdates(batch.updates);
192 }
193
194 void _follow(size_t index)
195 {
196 _followed = index;
197 // The NMEA decoder's family is passive, which reports standard NMEA.
198 const GPSType family = INPUTS[index].family->type;
199 qCDebug(PassiveProtocolLog) << "Passive input decodes" << family;
200 _context->publishAsReceived(GPSInputProtocol{family});
201 }
202
203 std::array<Input, INPUTS.size()> _inputs;
204 std::optional<size_t> _followed;
205 GPSDecodeContext* _context = nullptr;
206 bool _ready = false;
207};
