81 // An observer that stops or restarts the input retires this socket.
82 const auto current = [this, socket]() { return socket == _socket; };
83 UdpDrainBudget budget;
84 while (current() && socket->hasPendingDatagrams() && budget.available()) {
85 const QNetworkDatagram datagram = socket->receiveDatagram();
86 if (!datagram.isValid()) {

88 }

90 budget.consume(data.size());
91 if (data.isEmpty()) {
92 continue;
93 }
94 const qint64 receivedAtMs = _scheduler->nowMs();

96 // A stream is named by its sender's address, so selection stays on a sender that restarts on a new source
97 // port; each source port still frames independently.

99 const auto peer = _parserForPeer(sender, datagram.senderPort());
100 int framesFound = 0;

102 const bool delivered = peer->decoder.feed(data, receivedAtMs, [&](const RTCMDecodedFrame& decoded) {
103 if (!decoded.valid) {
104 ++framesDropped;
105 return current();
106 }
107 ++framesFound;

110 });

113 }
