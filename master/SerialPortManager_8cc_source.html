77QList<SerialPortManager::Port> SerialPortManager::availablePorts()
78{
79 if (_scanTimer.isValid() && _scanTimer.elapsed() < 1000) {
80 return _ports;
81 }
82 _ports = _enumerator();
83 _scanTimer.restart();
84 for (auto it = _reservations.begin(); it != _reservations.end(); ++it) {
85 if (const Port* const reserved = _portNamed(it.key())) {
86 it->listedLocation = reserved->systemLocation;

160SerialPortManager::ReservationPtr SerialPortManager::_reservationOf(const QString& name) const
161{
162 // Compared by what the names stand for now: a stable name stands for another system location once its
163 // device has been replugged. A second device of the same stable name leaves it standing for neither, and
164 // the reserved port is then the one at the location where the name last found it.
165 const QString location = _systemLocation(name);
166 for (auto it = _reservations.cbegin(); it != _reservations.cend(); ++it) {
167 ReservationPtr reservation = it->reservation.lock();
168 if (reservation && (_systemLocation(it.key()) == location || it->listedLocation == location)) {
169 return reservation;
170 }
171 }
172 return {};
173}

191QString SerialPortManager::persistentName(const QString& name) const
192{
193 QString trimmed = name.trimmed();
194 for (const Port& port : _ports) {
195 if (port.systemLocation != trimmed && port.portName != trimmed) {
196 continue;
197 }
198 const auto sharesStableName = [&port](const Port& other) { return other.stableName == port.stableName; };
199 const bool findsOnlyThisPort =
200 !port.stableName.isEmpty() && std::count_if(_ports.cbegin(), _ports.cend(), sharesStableName) == 1;
201 return findsOnlyThisPort ? port.stableName : port.systemLocation;
202 }
