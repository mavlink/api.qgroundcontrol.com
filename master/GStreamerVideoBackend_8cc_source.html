59bool GStreamerVideoBackend::initialize(const InitParams& params)
60{
61 switch (state()) {
62 case State::Initialized:
63 return true;
64 case State::Failed:
65 return false;
66 case State::Unprepared:
67 qCWarning(GStreamerVideoBackendLog) << "Initializing GStreamer without preparing its environment";
68 break;
69 case State::Prepared:
70 break;
71 }
72 const bool initialized = GStreamer::initialize(params);
73 _state.store(initialized ? State::Initialized : State::Failed, std::memory_order_release);
74 return initialized;
75}
