67bool GPSProtocol::validateConfiguration(const GPSConfig& config, bool allowReceiverAveraging,
68 bool supportsPersistentChanges) const
69{
70 if (config.allowPersistentChanges && !supportsPersistentChanges) {
71 log(GPSProtocolLogLevel::Warning, "Persistent configuration is not supported by this driver");
72 return false;
73 }
74 if (config.output_mode != OutputMode::GPS && config.output_mode != OutputMode::RTCM) {
75 log(GPSProtocolLogLevel::Warning, "Invalid receiver output mode");
76 return false;
77 }
78 if (config.output_mode == OutputMode::RTCM) {
79 if (!allowReceiverAveraging && config.base.surveyMode == GPSBaseStationConfig::SurveyMode::ReceiverManaged) {
80 log(GPSProtocolLogLevel::Warning, "Receiver-managed averaging is not supported by this driver");
81 return false;
82 }
83 const auto error = gpsValidateBaseStationConfig(config.base);
84 if (error != GPSReceiverConfigError::None) {
85 log(GPSProtocolLogLevel::Warning, "Invalid base station configuration (%d)", static_cast<int>(error));
86 return false;
87 }
88 }
89 return true;
90}

121int GPSProtocol::readAndDecode(unsigned timeout)
122{
123 uint8_t buffer[GPS_READ_BUFFER_SIZE];
124 const int count = read(buffer, sizeof(buffer), static_cast<int>(timeout));
125 if (count < 0) {
126 return count;
127 }
128 const int updates = consume({buffer, static_cast<size_t>(count)});
129 return ioError() ? ioError() : updates;
130}

145GPSCommandResult GPSProtocol::awaitCommand(GPSConfigurationStep step, const std::function<void()>& pump,
146 const std::function<GPSCommandOutcome()>& reply)
147{
148 if (_commandCompleted) {
149 return _commandWrite;
150 }
151 const auto timeout = static_cast<unsigned>(step.timeout.count());
152 const Operation operation(*this, timeout);
153 _operationDeadline.untilUs =
154 std::min(_operationDeadline.untilUs, _commandWrite.evidence.startedAtUs + uint64_t(timeout) * 1000);
155 _commandWrite.evidence.command = std::move(step.command);
156 _commandWrite.evidence.required = step.required;
157 _commandWrite.affectedSettings = step.affectedSettings;
158 const auto outcome = GPSCommandTransaction::await(
159 _operationDeadline.untilUs, [this] { return nowUs(); }, reply, pump,
160 [this] {
161 return ioError() == ReadCancelled ? GPSCommandOutcome::Cancelled
162 : ioError() ? GPSCommandOutcome::TransportError
163 : GPSCommandOutcome::Pending;
164 });
165 return completeCommand(outcome);
166}

168void GPSProtocol::beginCommandWrite(std::string command, GPSReceiverSettingSet settings, bool required)
169{
170 if (ioError()) {
171 return;
172 }
173 failCommandWrite(GPSCommandOutcome::Written);
174 _commandWrite = {};
175 _commandWrite.evidence.startedAtUs = nowUs();
176 _commandWrite.evidence.command = std::move(command);
177 _commandWrite.affectedSettings = settings;
178 _commandWrite.evidence.required = required;
179 _commandCompleted = false;
180}

182GPSCommandResult GPSProtocol::completeCommand(GPSCommandOutcome outcome)
183{
184 if (_commandCompleted) {
185 return _commandWrite;
186 }
187 _commandWrite.evidence.outcome = outcome;
188 _commandWrite.evidence.finishedAtUs = nowUs();
189 _commandCompleted = true;
190 // Completion observers may finish evidence again or begin another attempt.
191 const auto result = _commandWrite;
192 if (_io.commandFinished) {
193 _io.commandFinished(result);
194 }
195 return result;
196}

198int GPSProtocol::receiveDecoded(unsigned timeout)
199{
200 const Operation operation(*this, timeout);
201 const uint64_t deadline = _operationDeadline.untilUs;
202 do {
203 const int handled = readAndDecode(static_cast<unsigned>(remainingMilliseconds(deadline)));
204 if (handled) {
205 return handled;
206 }
207 } while (nowUs() < deadline);
208 return -1;
209}

226GPSDecodeResult GPSProtocol::decode(std::span<const uint8_t> bytes)
227{
228 flushDecoded();
229 size_t consumed = 0;
230 // One completed frame can publish relative/survey data plus position and satellites.
231 while (consumed < bytes.size() && _decoded.events.size() + 4 <= GPSDecodedBatch::MAX_EVENTS) {
232 const int updates = decodeByte(bytes[consumed++]);
233 if (updates > 0) {
234 _decoded.updates |= updates;
235 if ((updates & 1) && positionReport()) {
236 _decoded.events.emplace_back(*positionReport());
237 }
238 if ((updates & 2) && satelliteReport()) {
239 _decoded.events.emplace_back(*satelliteReport());
240 }
241 }
242 }
243 GPSDecodeResult result{consumed, std::move(_decoded)};
244 _decoded = {};
245 return result;
246}

248int GPSProtocol::consume(std::span<const uint8_t> bytes)
249{
250 int updates = 0;
251 do {
252 auto result = decode(bytes);
253 bytes = bytes.subspan(result.bytesConsumed);
254 updates |= result.batch.updates;
255 if (_io.decoded) {
256 _io.decoded(result.batch);
257 }
258 result.batch.events.clear();
259 // A nested callback may already have populated or replenished the decoder's storage.
260 if (_decoded.events.empty() && _decoded.events.capacity() < result.batch.events.capacity()) {
261 _decoded.events.swap(result.batch.events);
262 }
263 } while (!bytes.empty());
264 return updates;
265}
