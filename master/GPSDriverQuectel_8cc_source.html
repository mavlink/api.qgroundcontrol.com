208bool GPSNativeQuectel::_restart(bool requireRoleMatch, bool startSurveySession)
209{
210 _revokeSurvey();
211 _surveyPhase = startSurveySession ? SurveyPhase::AwaitingBoot : SurveyPhase::Off;
212 log(GPSProtocolLogLevel::Debug, "Restarting LG290P and verifying its saved configuration");
213 const Operation operation(*this, RESTART_TIMEOUT_MS);
214 const uint64_t deadline = nowUs() + uint64_t(RESTART_TIMEOUT_MS) * 1000;
215 const auto bytes = frame("PQTMSRR");
216 beginCommandWrite("PQTMSRR");
217 if (write(bytes.data(), static_cast<int>(bytes.size())) != static_cast<int>(bytes.size())) {
218 return false;
219 }
220 // PQTMSRR has no documented ACK. Do not inflate successful transport completion to acknowledgment.
221 failCommandWrite(GPSCommandOutcome::Written);
222 resetStream();
223 _expectingBoot = true;
224 _sawBoot = false;
225 _restartRejected = false;
226 do {
227 waitFor(std::chrono::milliseconds(200));
228 if (ioError()) {
229 break;
230 }
231 const bool identified = _identify(700);
232 if (_restartRejected) {
233 _ioErrorDetail = QStringLiteral("LG290P rejected PQTMSRR.");
234 break;
235 }
236 if (identified && _sawBoot) {
237 _expectingBoot = false;
238 return _verifyRole(requireRoleMatch);
239 }
240 } while (!ioError() && nowUs() < deadline);
241 _expectingBoot = false;
242 return false;
243}

315 }

319 }

352 }

374 }

383 }

386 // the saved role before claiming that the receiver actually operates in that role.
387 if ((!restarted && !_restart(true, _outputMode == OutputMode::RTCM)) ||

389 return _fail("LG290P restart/readback failed; saved role or base settings do not match the request");
390 }
391 if (_outputMode == OutputMode::RTCM) {

395 !_setMessageRate("RTCM3-107X", 1, "0")) {
396 return _fail("LG290P base message output configuration/readback failed");
397 }
398 }
399 for (const std::string_view name : {"GGA", "GST", "GSA", "GSV"}) {
400 if (!_setMessageRate(name, 1)) {
401 return _fail("LG290P NMEA output configuration/readback failed");
402 }
403 }
404 _configured = true;
405 _expireSurvey();
406 setRTCMEnabled(_surveyReport && (_surveyReport->flags & 1));
407 return 0;
