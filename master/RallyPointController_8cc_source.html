70bool RallyPointController::load(const QJsonObject& json, QString& errorString)
71{
72 removeAll();
73
74 errorString.clear();
75
76 if (json.contains(JsonParsing::jsonVersionKey) && json[JsonParsing::jsonVersionKey].toInt() == 1) {
77 // We just ignore old version 1 data
78 return true;
79 }
80
81 QList<JsonParsing::KeyValidateInfo> keyInfoList = {
82 { JsonParsing::jsonVersionKey, QJsonValue::Double, true },
83 { _jsonPointsKey, QJsonValue::Array, true },
84 };
85 if (!JsonParsing::validateKeys(json, keyInfoList, errorString)) {
86 return false;
87 }
88
89 QString errorStr;
90 QString errorMessage = tr("Rally: %1");
91
92 const int version = json[JsonParsing::jsonVersionKey].toInt();
93 if (version < _jsonOldestSupportedVersion || version > _jsonCurrentVersion) {
94 errorString =
95 tr("Rally Points supports versions %1 to %2").arg(_jsonOldestSupportedVersion).arg(_jsonCurrentVersion);
96 return false;
97 }
98
99 // Version 2 points are [lat, lon, alt] arrays with no frame, their altitudes are relative to home
100 auto loadPoint = [version](const QJsonValue& jsonPoint, QGeoCoordinate& coordinate,
101 QGroundControlQmlGlobal::AltitudeFrame& altitudeFrame, QString& error) {
102 altitudeFrame = QGroundControlQmlGlobal::AltitudeFrameRelative;
103 if (version < 3) {
104 return GeoJsonHelper::loadGeoCoordinate(jsonPoint, true /* altitudeRequired */, coordinate, error);
105 }
106
107 if (!jsonPoint.isObject()) {
108 error = tr("Each point must be an object");
109 return false;
110 }
111 const QJsonObject jsonPointObject = jsonPoint.toObject();
112 const QList<JsonParsing::KeyValidateInfo> pointKeyInfoList = {
113 {_jsonCoordinateKey, QJsonValue::Array, true},
114 {_jsonFrameKey, QJsonValue::Double, true},
115 };
116 if (!JsonParsing::validateKeys(jsonPointObject, pointKeyInfoList, error) ||
117 !GeoJsonHelper::loadGeoCoordinate(jsonPointObject[_jsonCoordinateKey], true /* altitudeRequired */,
118 coordinate, error)) {
119 return false;
120 }
121
122 const int frame = jsonPointObject[_jsonFrameKey].toInt(-1);
123 if (frame != MAV_FRAME_GLOBAL && frame != MAV_FRAME_GLOBAL_RELATIVE_ALT &&
124 frame != MAV_FRAME_GLOBAL_TERRAIN_ALT) {
125 error = tr("Unsupported frame %1").arg(frame);
126 return false;
127 }
128 altitudeFrame = RallyPoint::altitudeFrameFromMavFrame(static_cast<MAV_FRAME>(frame));
129 return true;
130 };
131
132 QObjectList pointList;
133 const QJsonArray jsonPoints = json[_jsonPointsKey].toArray();
134 for (const QJsonValue& jsonPoint : jsonPoints) {
135 QGeoCoordinate coordinate;
136 QGroundControlQmlGlobal::AltitudeFrame altitudeFrame;
137 if (!loadPoint(jsonPoint, coordinate, altitudeFrame, errorStr)) {
138 qDeleteAll(pointList);
139 errorString = errorMessage.arg(errorStr);
140 return false;
141 }
142 pointList.append(new RallyPoint(coordinate, altitudeFrame, this));
143 }
144 _points.swapObjectList(pointList);
145
146 setDirty(false);
147 _setFirstPointCurrent();
148
149 return true;
150}

152void RallyPointController::save(QJsonObject& json)
153{
154 json[JsonParsing::jsonVersionKey] = _jsonCurrentVersion;
155
156 QJsonArray rgPoints;
157 for (int i=0; i<_points.count(); i++) {

282void RallyPointController::addPoint(QGeoCoordinate point)
283{
284 Vehicle* const vehicle = _masterController->controllerVehicle();
285 const bool px4 = vehicle->px4Firmware();
286
287 double defaultAlt;
288 QGroundControlQmlGlobal::AltitudeFrame altitudeFrame = QGroundControlQmlGlobal::AltitudeFrameRelative;
289 if (px4) {
290 // The altitude is the ground height at the point, it is seeded from terrain below
291 defaultAlt = RallyPoint::getDefaultFactAltitude();
292 } else if (_points.count()) {
293 const RallyPoint* lastPoint = qobject_cast<RallyPoint*>(_points[_points.count() - 1]);
294 defaultAlt = lastPoint->coordinate().altitude();
295 altitudeFrame = lastPoint->altitudeFrame();
296 } else if (vehicle->apmFirmware() || vehicle->fixedWing()) {
297 defaultAlt = SettingsManager::instance()->appSettings()->defaultMissionItemAltitude()->rawValue().toDouble();
298 } else {
299 defaultAlt = RallyPoint::getDefaultFactAltitude();
300 }
301 point.setAltitude(defaultAlt);
302 RallyPoint* newPoint = new RallyPoint(point, altitudeFrame, this);
303 _points.append(newPoint);
304 setCurrentRallyPoint(newPoint);
305 setDirty(true);
306
307 if (px4) {
308 // Stays relative with the default altitude if terrain data is not available
309 newPoint->useTerrainAltitude();
310 }
311}

367bool RallyPointController::showPlanFromManagerVehicle (void)
368{
369 qCDebug(RallyPointControllerLog) << "showPlanFromManagerVehicle _flyView" << _flyView;
370 if (_masterController->offline()) {
371 qCCritical(RallyPointControllerLog) << "RallyPointController::showPlanFromManagerVehicle called while offline";
372 return true; // stops further propagation of showPlanFromManagerVehicle due to error
373 } else {
374 if (!_managerVehicle->initialPlanRequestComplete()) {
375 // The vehicle hasn't completed initial load, we can just wait for loadComplete to be signalled automatically
376 qCDebug(RallyPointControllerLog) << "showPlanFromManagerVehicle: !initialPlanRequestComplete, wait for signal";
377 return true;
378 } else if (syncInProgress()) {
379 // If the sync is already in progress, _loadComplete will be called automatically when it is done. So no need to do anything.
380 qCDebug(RallyPointControllerLog) << "showPlanFromManagerVehicle: syncInProgress wait for signal";
381 return true;
382 } else {
383 qCDebug(RallyPointControllerLog) << "showPlanFromManagerVehicle: sync complete";
384 _itemsRequested = true;
385 _managerLoadComplete();
386 return false;
387 }
