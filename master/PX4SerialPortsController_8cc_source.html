206void PX4SerialPortsController::_update()
207{
208 // The facts belong to the parameter manager, which may be gone if the vehicle disconnected
209 if (!_parameterManager) {
210 return;
211 }
212
213 const QList<PX4SerialPortConfig::Function> functions = makeFunctions(_functionFacts);
214 QList<QStringList> rowsParams; // Per row, parameters whose change marks it Reboot required
215
216 // Options can move to another row, so every row records the options it shows before any row is checked
217 for (int i = 0; i < _ports->count(); i++) {
218 PX4SerialPortRow* const row = _ports->value<PX4SerialPortRow*>(i);
219 const int portId = row->portId();
220 const QStringList onPort = PX4SerialPortConfig::functionsOnPort(functions, portId);
221 QStringList rowParams;
222
223 // Choosing "None" can't remove a function from its default port
224 const bool defaultFunctionHere =
225 std::any_of(functions.cbegin(), functions.cend(), [portId](const PX4SerialPortConfig::Function& f) {
226 return (f.defaultPortId == portId) && (f.currentPortId == portId);
227 });
228 row->setNoneAvailable(!defaultFunctionHere);
229 row->setFunctionIndex(onPort.isEmpty() ? 0 : row->functionIndexOf(onPort.first()));
230
231 QVariantList functionEnabled;
232 for (int functionIndex = 0; functionIndex < row->functionLabels().count(); functionIndex++) {
233 const QString paramName = row->functionParamName(functionIndex);
234 functionEnabled.append(paramName.isEmpty() ? onPort.isEmpty() : onPort.contains(paramName));
235 }
236 row->setFunctionEnabled(functionEnabled);
237
238 QStringList assignments;
239 for (const QString& paramName : onPort) {
240 const Fact* const fact = _fact(paramName);
241 const int value = fact ? fact->rawValue().toInt() : portId;
242 assignments.append((value == PX4SerialPortConfig::kDisabledPortId)
243 ? tr("%1 = %2 (default port)").arg(paramName).arg(value)
244 : QStringLiteral("%1 = %2").arg(paramName).arg(value));
245 }
246 row->setAssignments(assignments);
247
248 QString conflictText;
249 if ((onPort.count() > 1) && !row->shared()) {
250 const auto defaultsHere = std::find_if(
251 functions.cbegin(), functions.cend(), [&onPort, portId](const PX4SerialPortConfig::Function& f) {
252 return (f.defaultPortId == portId) && onPort.contains(f.paramName);
253 });
254 QStringList labels;
255 for (const QString& paramName : onPort) {
256 if ((defaultsHere == functions.cend()) || (paramName != defaultsHere->paramName)) {
257 labels.append(row->functionLabels().value(row->functionIndexOf(paramName)));
258 }
259 }
260 if (defaultsHere != functions.cend()) {
261 // Disabling the function doesn't stop it, so the usual "only one will start" doesn't apply
262 conflictText = tr("%1 always uses this port unless it is assigned to another port, so %2 will not work "
263 "here.")
264 .arg(defaultsHere->label, labels.join(", "));
265 } else {
266 conflictText =
267 (labels.count() == 2)
268 ? tr("%1 and %2 are both assigned to this port. Only one of them will start.")
269 .arg(labels[0], labels[1])
270 : tr("%1 are all assigned to this port. Only one of them will start.").arg(labels.join(", "));
271 }
272 }
273 row->setConflictText(conflictText);
274
275 Fact* baudFact = nullptr;
276 const QString baudParamName = PX4SerialPortConfig::baudParamName(portId);
277 if (!baudParamName.isEmpty()) {
278 baudFact = _fact(baudParamName);
279 if (baudFact) {
280 rowParams.append(baudParamName);
281 } else {
282 _requestParameter(baudParamName);
283 }
284 }
285 row->setBaudFact(baudFact);
286 row->setBaudLoading(!baudFact && _pendingParams.contains(baudParamName));
287
288 QVariantList optionFacts;
289 bool optionsLoading = false;
290 const QStringList optionFunctions = (row->shared() || (onPort.count() == 1)) ? onPort : QStringList();
291 for (const QString& functionParamName : optionFunctions) {
292 for (const QString& paramName : PX4SerialPortConfig::optionParamNames(functionParamName, portId)) {
293 if (Fact* const fact = _fact(paramName)) {
294 optionFacts.append(QVariant::fromValue(fact));
295 _optionParamPorts.insert(paramName, portId);
296 } else {
297 _requestParameter(paramName);
298 optionsLoading = optionsLoading || _pendingParams.contains(paramName);
299 }
300 }
301 }
302 row->setOptionFacts(optionFacts);
303 row->setOptionsLoading(optionsLoading);
304
305 rowsParams.append(rowParams);
306 }
307
308 int changedPortCount = 0;
309 for (int i = 0; i < _ports->count(); i++) {
310 PX4SerialPortRow* const row = _ports->value<PX4SerialPortRow*>(i);
311 const int portId = row->portId();
312 QStringList rowParams = rowsParams[i];
313
314 for (auto it = _optionParamPorts.cbegin(); it != _optionParamPorts.cend(); ++it) {
315 if (it.value() == portId) {
316 rowParams.append(it.key());
317 }
318 }
319
320 for (const PX4SerialPortConfig::Function& function : functions) {
321 const int initialValue = _initialValues.value(function.paramName).toInt();
322 const int initialPortId =
323 (initialValue == PX4SerialPortConfig::kDisabledPortId) ? function.defaultPortId : initialValue;
324 if ((function.currentPortId == portId) || (initialPortId == portId)) {
325 rowParams.append(function.paramName);
326 }
327 }
328 const bool rebootRequired = std::any_of(rowParams.cbegin(), rowParams.cend(), [this](const QString& paramName) {
329 return _differsFromInitial(paramName);
330 });
331 row->setRebootRequired(rebootRequired);
332 if (rebootRequired) {
333 changedPortCount++;
334 }
335 }
336
337 if (changedPortCount != _changedPortCount) {
338 _changedPortCount = changedPortCount;
339 emit changedPortCountChanged();
340 }

395bool PX4SerialPortsController::_differsFromInitial(const QString& paramName) const
396{
397 const auto initial = _initialValues.constFind(paramName);
398 if ((initial == _initialValues.cend()) || !_parameterManager ||
399 !_parameterManager->parameterExists(ParameterManager::defaultComponentId, paramName)) {
400 return false;
401 }
402
403 const Fact* const fact = _parameterManager->getParameter(ParameterManager::defaultComponentId, paramName);
404 return fact && (fact->rawValue() != *initial);
