27std::optional<quint64> unsignedNumber(QByteArrayView text, int base = 10)
28{
29 const auto digit = [base](char ch) {
30 return (ch >= '0' && ch <= '9') || (base == 16 && ((ch >= 'a' && ch <= 'f') || (ch >= 'A' && ch <= 'F')));
31 };
32 bool ok = false;

37std::chrono::milliseconds retryAfter(const QHttpHeaders& headers, const QDateTime& utcNow)
38{
39 const QByteArrayView value = headers.value(QHttpHeaders::WellKnownHeader::RetryAfter);
40 if (const auto seconds = unsignedNumber(value)) {
41 const auto maxSeconds = std::chrono::duration_cast<std::chrono::seconds>(NTRIPFailure::MAX_RETRY_AFTER).count();
42 return std::chrono::seconds{(std::min<quint64>) (*seconds, maxSeconds)};
43 }
44 auto deadline = headers.dateTimeValue(QHttpHeaders::WellKnownHeader::RetryAfter);
45 if (!deadline || !utcNow.isValid()) {
46 return {};
47 }
48 // Qt reads the two-digit RFC 850 year as 19yy; RFC 9110 wants the latest century not over 50 years ahead.
49 if (value.indexOf(',') > 3) {
50 *deadline = deadline->addYears(utcNow.date().year() / 100 * 100 - 1900);
51 if (*deadline > utcNow.addYears(50)) {
52 *deadline = deadline->addYears(-100);
53 }
54 }
55 const qint64 maxMs = NTRIPFailure::MAX_RETRY_AFTER.count();
56 return std::chrono::milliseconds{std::clamp(utcNow.msecsTo(*deadline), qint64(0), maxMs)};
57}

145NTRIPHttpDecoder::Result NTRIPHttpDecoder::feed(QByteArrayView bytes, const QDateTime& utcNow)
146{
147 Result result;
148 while (!bytes.isEmpty() && _state != State::Failed) {
149 if (_state == State::Complete) {
150 _fail(result, tr("Unexpected data after HTTP response"));
151 break;
152 }
153 if (_state == State::IcyHeaders) {
154 const auto first = static_cast<unsigned char>(bytes.front());
155 const bool binary = first >= 127 || (first < 32 && first != '\t' && first != '\r' && first != '\n');
156 // A legacy stream may start mid-frame, not at the RTCM preamble.
157 // Probe optional ASCII headers only until binary or non-header input arrives.
158 if (_headers.isEmpty() &&
159 (binary || (first == '\n' && !_line.endsWith('\r')) || (_line.endsWith('\r') && first != '\n'))) {
160 _beginIcyBody(result);
161 } else if (!_headers.isEmpty() && _line.isEmpty() && binary) {
162 // ICY casters may omit the blank separator even after optional headers.
163 _lineReceived(result, utcNow);
164 continue;
165 }
166 }
167 if (_state == State::Identity || _state == State::ChunkData) {
168 const bool bounded = _state == State::ChunkData || _contentLength.has_value();
169 const auto count =
170 bounded ? static_cast<qsizetype>((std::min<quint64>) (bytes.size(), _remaining)) : bytes.size();
171 if (_pendingFailure) {
172 _errorBody.append(bytes.data(), (std::min) (count, MAX_ERROR_BODY_BYTES - _errorBody.size()));
173 } else {
174 result.body.append(bytes.data(), count);
175 }
176 bytes = bytes.sliced(count);
177 if (bounded && (_remaining -= count) == 0) {
178 _state = _state == State::ChunkData ? State::ChunkEnd : State::Complete;
179 result.complete = _state == State::Complete;
180 }
181 if (_pendingFailure && (_state == State::Complete || _errorBody.size() == MAX_ERROR_BODY_BYTES)) {
182 _finishError(result);
183 }
184 continue;
185 }
186 if (_state == State::ChunkEnd) {
187 if (bytes.front() != (_chunkEndBytes == 0 ? '\r' : '\n')) {
188 _fail(result, tr("Invalid HTTP chunk terminator"));
189 break;
190 }
191 bytes = bytes.sliced(1);
192 if (++_chunkEndBytes == 2) {
193 _chunkEndBytes = 0;
194 _state = State::ChunkSize;
195 }
196 continue;
197 }
198 const char ch = bytes.front();
199 bytes = bytes.sliced(1);
200 if ((ch == '\n' && !_line.endsWith('\r')) || (_line.endsWith('\r') && ch != '\n')) {
201 _fail(result, tr("Invalid HTTP line ending"));
202 break;
203 }
204 _line.append(ch);
205 if ((_state != State::ChunkSize && ++_headerBytes > MAX_HEADER_BYTES) || _line.size() > MAX_LINE_BYTES) {
206 _fail(result, tr("HTTP response header too large"), NTRIPError::HeaderTooLarge);
207 break;
208 }
209 if (ch == '\n') {
210 _line.chop(2);
211 _lineReceived(result, utcNow);
212 _line.clear();
213 }
214 }
215 return result;
216}

218void NTRIPHttpDecoder::_lineReceived(Result& result, const QDateTime& utcNow)
219{
220 if (_state == State::Status) {
221 _status = parseStatusLine(_line);
222 if (!_status.valid) {
223 _fail(result, tr("Invalid HTTP status line"));
224 } else if (_line.left(11).compare("SOURCETABLE", Qt::CaseInsensitive) == 0) {
225 if (_purpose == NTRIPHttpPurpose::SourceTable && _status.code == 200) {
226 _state = State::IcyHeaders;
227 result.connected = true;
228 } else {
229 _fail(result, tr("Caster returned a source table; select a valid mountpoint"),
230 NTRIPError::InvalidMountpoint);
231 }
232 } else if (_line.first(3).compare("ICY", Qt::CaseInsensitive) == 0 && _status.code == 200) {
233 _state = State::IcyHeaders;
234 result.connected = true;
235 } else {
236 _http10 = _line.startsWith("HTTP/1.0");
237 _state = State::Headers;
238 }
239 return;
240 }
241 if (_state == State::ChunkSize) {
242 // NTRIP uses no chunk extensions, so everything after ';' (and the whitespace before it) is ignored.
243 QByteArrayView size(_line);
244 if (const qsizetype separator = size.indexOf(';'); separator >= 0) {
245 size = size.first(separator);
246 while (size.endsWith(' ') || size.endsWith('\t')) {
247 size.chop(1);
248 }
249 }
250 const auto length = unsignedNumber(size, 16);
251 if (!length || *length > MAX_CHUNK_BYTES) {
252 _fail(result, tr("Invalid or oversized HTTP chunk"));
253 } else {
254 _remaining = *length;
255 _state = _remaining == 0 ? State::Trailers : State::ChunkData;
256 }
257 return;
258 }
259 if (_line.isEmpty()) {
260 if (_state == State::Trailers) {
261 if (_pendingFailure) {
262 _finishError(result);
263 } else {
264 _state = State::Complete;
265 result.complete = true;
266 }
267 return;
268 }
269 if (_status.code >= 300) {
270 const int status = _status.code;
271 const auto code = status == 401 ? NTRIPError::AuthFailed
272 : status == 400 || status == 403 ? NTRIPError::RequestRejected
273 : NTRIPError::HttpError;
274 const QString detail = status == 401 ? tr("Authentication failed (401): check username and password")
275 : tr("HTTP %1: %2").arg(status).arg(_status.reason);
276 _pendingFailure = NTRIPFailure{code, detail};
277 if (_headers.values(QHttpHeaders::WellKnownHeader::RetryAfter).size() == 1) {
278 _pendingFailure->retryAfter = retryAfter(_headers, utcNow);
279 }
280 if (code == NTRIPError::AuthFailed) {
281 _finishError(result);
282 return;
283 }
284 }
285 const auto lengths = _headers.values(QHttpHeaders::WellKnownHeader::ContentLength);
286 for (const auto& field : lengths) {
287 for (const auto& value : field.split(',')) {
288 const auto length = unsignedNumber(QByteArrayView(value).trimmed());
289 if (!length || (_contentLength && *_contentLength != *length)) {
290 _fail(result, tr("Invalid or conflicting HTTP content length"));
291 return;
292 }
293 _contentLength = *length;
294 }
295 }
296 const auto transfers = _headers.values(QHttpHeaders::WellKnownHeader::TransferEncoding);
297 _chunked = !transfers.isEmpty();
298 if (_chunked && (_http10 || transfers.size() != 1 ||
299 transfers.first().compare("chunked", Qt::CaseInsensitive) != 0 || _contentLength)) {
300 _fail(result, tr("Unsupported or conflicting HTTP transfer encoding"));
301 return;
302 }
303 if (_status.code >= 100 && _status.code < 200) {
304 if (_status.code == 101 || _contentLength || _chunked || ++_informationalResponses > 4) {
305 _fail(result, tr("Unsupported HTTP informational response"));
306 } else {
307 _headers.clear();
308 _state = State::Status;
309 }
310 return;
311 }
312 const auto encodings = _headers.values(QHttpHeaders::WellKnownHeader::ContentEncoding);
313 if (!encodings.isEmpty() &&
314 (encodings.size() != 1 || encodings.first().compare("identity", Qt::CaseInsensitive) != 0)) {
315 _fail(result, tr("Unsupported HTTP content encoding"));
316 return;
317 }
318 if (_pendingFailure) {
319 _beginBody(result);
320 return;
321 }
322 for (const auto& value : _headers.values(QHttpHeaders::WellKnownHeader::ContentType)) {
323 if (_purpose == NTRIPHttpPurpose::Corrections &&
324 value.split(';').first().trimmed().compare("gnss/sourcetable", Qt::CaseInsensitive) == 0) {
325 _fail(result, tr("Caster returned a source table; select a valid mountpoint"),
326 NTRIPError::InvalidMountpoint);
327 return;
328 }
329 }
330 if (_status.code == 204 || _status.code == 205) {
331 if (_chunked || (_contentLength && *_contentLength != 0)) {
332 _fail(result, tr("Unexpected framing for an empty HTTP response"));
333 return;
334 }
335 _contentLength = 0;
336 }
337 result.connected = result.connected || _state != State::IcyHeaders;
338 _beginBody(result);
339 return;
340 }
341
342 const qsizetype colon = _line.indexOf(':');
343 const QByteArrayView line(_line);
344 if (colon <= 0 || !std::all_of(line.begin(), line.begin() + colon, isToken) ||
345 !std::all_of(line.begin() + colon + 1, line.end(), isFieldValue)) {
346 if (_state == State::IcyHeaders && _headers.isEmpty()) {
347 _line += "\r\n";
348 _beginIcyBody(result);
349 return;
350 }
351 _fail(result, tr("Invalid HTTP response header"));
352 return;
353 }
354 if (_headers.size() >= 128) {
355 _fail(result, tr("Too many HTTP response headers"), NTRIPError::HeaderTooLarge);
356 return;
357 }
358 const auto name = line.first(colon);
359 const auto value = line.sliced(colon + 1);
360 if (_state == State::Trailers &&
361 (name.compare("content-length", Qt::CaseInsensitive) == 0 ||
362 name.compare("transfer-encoding", Qt::CaseInsensitive) == 0 ||
363 name.compare("content-encoding", Qt::CaseInsensitive) == 0 ||
364 name.compare("content-type", Qt::CaseInsensitive) == 0 ||
365 name.compare("retry-after", Qt::CaseInsensitive) == 0 || name.compare("host", Qt::CaseInsensitive) == 0)) {
366 _fail(result, tr("Forbidden HTTP trailer"));
367 return;
368 }
369 // Validated above, so append() cannot reject the field.
370 (void) _headers.append(QLatin1StringView(name.data(), name.size()), QLatin1StringView(value.data(), value.size()));
371}

387NTRIPHttpDecoder::Result NTRIPHttpDecoder::finish()
388{
389 Result result;
390 if (_pendingFailure) {
391 _finishError(result);
392 } else if ((_state == State::Identity && !_contentLength) ||
393 (_state == State::IcyHeaders && _line.isEmpty() && _headers.isEmpty()) || _state == State::Complete) {
394 _state = State::Complete;
395 result.complete = true;
396 } else if (_state == State::Identity || _state == State::ChunkSize || _state == State::ChunkData ||
397 _state == State::ChunkEnd || _state == State::Trailers) {
398 _fail(result, tr("Caster closed the connection mid-transfer"));
399 } else if (_state != State::Failed) {
400 _fail(result, tr("Caster disconnected before completing the HTTP response"));
401 }
402 return result;
