125thread_local std::vector<RetiredImage> t_retiredImages;

128void destroyRetired(const RetiredImage& retired)
129{
130 const OwnedImage& o = retired.owned;
131 o.fns.destroyImage(o.dev, o.image, nullptr);
132 o.fns.freeMemory(o.dev, o.memory, nullptr);
133 if (retired.fence != VK_NULL_HANDLE) {
134 o.fns.destroyFence(o.dev, retired.fence, nullptr);
135 }
136}

138// Destroys @p dev's retired images whose fence signalled; with @p waitAll (QRhi teardown) waits for every fence first.
139void reapRetired(VkDevice dev, bool waitAll)
140{
141 for (auto it = t_retiredImages.begin(); it != t_retiredImages.end();) {
142 const OwnedImage& o = it->owned;
143 if (o.dev != dev) {
144 ++it;
145 continue;
146 }
147 if (waitAll) {
148 (void) o.fns.waitForFences(dev, 1, &it->fence, VK_TRUE, UINT64_MAX);
149 } else if (o.fns.getFenceStatus(dev, it->fence) != VK_SUCCESS) {
150 ++it;
151 continue;
152 }
153 destroyRetired(*it);
154 it = t_retiredImages.erase(it);
155 }
156}
157
158void retire(VkQueue queue, const OwnedImage& owned)
159{
160 RetiredImage retired{owned};
161 const VkFenceCreateInfo fenceInfo = {VK_STRUCTURE_TYPE_FENCE_CREATE_INFO, nullptr, 0};
162 // A submit without batches signals its fence once all work already submitted to the queue has completed.
163 if (owned.fns.createFence(owned.dev, &fenceInfo, nullptr, &retired.fence) != VK_SUCCESS ||
164 owned.fns.queueSubmit(queue, 0, nullptr, retired.fence) != VK_SUCCESS) {
165 (void) owned.fns.queueWaitIdle(queue);
166 destroyRetired(retired);
167 return;
168 }
169 t_retiredImages.push_back(retired);
170 reapRetired(owned.dev, /*waitAll=*/false);
171}

177std::function<void()> deferredRelease(QRhi& rhi, OwnedImage owned)
178{
179 const auto* nh = static_cast<const QRhiVulkanNativeHandles*>(rhi.nativeHandles());
180 const VkQueue queue = nh ? nh->gfxQueue : VK_NULL_HANDLE;
181 const VkDevice dev = owned.dev;
182 // Keyed: one teardown hook per QRhi, run while its device is still alive.

187// VkFormat for single-plane RGB layouts. Multi-planar YUV (NV12/P010/I420) maps to VK_FORMAT_UNDEFINED: QRhi would wrap
188// a multi-planar VkImage as one single-plane texture with a COLOR-aspect view (invalid usage, no chroma), so those
189// layouts take the CPU path until a per-plane import exists.
190VkFormat vkFormatForVideoInfo(const GstVideoInfo& videoInfo)
191{
192 if (GST_VIDEO_INFO_N_PLANES(&videoInfo) != 1) {
193 return VK_FORMAT_UNDEFINED;
194 }
195 // Bytes in memory order, which Qt's RGB shaders swizzle per pixel format.
196 switch (GST_VIDEO_INFO_FORMAT(&videoInfo)) {
197 case GST_VIDEO_FORMAT_RGBA:
198 case GST_VIDEO_FORMAT_RGBx:
199 case GST_VIDEO_FORMAT_ARGB:
200 case GST_VIDEO_FORMAT_xRGB:
201 case GST_VIDEO_FORMAT_ABGR:
202 case GST_VIDEO_FORMAT_xBGR:
203 case GST_VIDEO_FORMAT_AYUV:
204 return VK_FORMAT_R8G8B8A8_UNORM;
205 case GST_VIDEO_FORMAT_BGRA:
206 case GST_VIDEO_FORMAT_BGRx:
207 return VK_FORMAT_B8G8R8A8_UNORM;
208 default:
209 return VK_FORMAT_UNDEFINED;
210 }
211}

223bool validateVulkanPlaneLayout(const GstVideoInfo& videoInfo, GstVideoMeta* vmeta, quint64 memoryBound,
224 VkSubresourceLayout& planeLayout)
225{
226 const int width = GST_VIDEO_INFO_WIDTH(&videoInfo);
227 const int height = GST_VIDEO_INFO_HEIGHT(&videoInfo);
228 if (width <= 0 || height <= 0) {
229 qCWarning(GstDmaBufVulkanLog) << "Vulkan import: invalid DMABuf dimensions" << width << "x" << height;
230 return false;
231 }
232
233 const auto [offset, stride] = GstHw::planeLayout(videoInfo, vmeta, 0);
234 const int pixelStride = GST_VIDEO_INFO_COMP_PSTRIDE(&videoInfo, 0);
235 if (stride <= 0 || pixelStride <= 0) {
236 qCWarning(GstDmaBufVulkanLog) << "Vulkan import: invalid DMABuf plane layout"
237 << "stride" << stride << "pixelStride" << pixelStride;
238 return false;
239 }
240
241 constexpr quint64 kMaxVkDeviceSize = static_cast<quint64>((std::numeric_limits<VkDeviceSize>::max)());
242 const quint64 rowPitch = static_cast<quint64>(stride);
243 const quint64 minRowBytes = static_cast<quint64>(width) * static_cast<quint64>(pixelStride);
244 if (rowPitch < minRowBytes || offset > kMaxVkDeviceSize || rowPitch > kMaxVkDeviceSize) {
245 qCWarning(GstDmaBufVulkanLog) << "Vulkan import: implausible DMABuf plane layout"
246 << "offset" << static_cast<quint64>(offset) << "stride" << stride << "minRowBytes"
247 << minRowBytes;
248 return false;
249 }
250
251 if (rowPitch > std::numeric_limits<quint64>::max() / static_cast<quint64>(height)) {
252 qCWarning(GstDmaBufVulkanLog) << "Vulkan import: DMABuf plane size overflow"
253 << "stride" << stride << "height" << height;
254 return false;
255 }
256 const quint64 planeSize = rowPitch * static_cast<quint64>(height);
257 if (static_cast<quint64>(offset) > std::numeric_limits<quint64>::max() - planeSize) {
258 qCWarning(GstDmaBufVulkanLog) << "Vulkan import: DMABuf plane end overflow"
259 << "offset" << static_cast<quint64>(offset) << "planeSize" << planeSize;
260 return false;
261 }
262 const quint64 planeEnd = static_cast<quint64>(offset) + planeSize;
263 if (memoryBound > 0 && planeEnd > memoryBound) {
264 qCWarning(GstDmaBufVulkanLog) << "Vulkan import: DMABuf plane exceeds buffer bounds"
265 << "end" << planeEnd << "bound" << memoryBound;
266 return false;
267 }
268
269 planeLayout.offset = static_cast<VkDeviceSize>(offset);
270 planeLayout.rowPitch = static_cast<VkDeviceSize>(rowPitch);
271 return true;
272}

291QVideoFrameTexturesUPtr GstDmaBufVideoBuffer::importVulkan(QRhi& rhi, QVideoFrameTexturesUPtr& old)
292{
293 // dmabuf-fd -> VkImage via VK_EXT_external_memory_dma_buf + VkImageDrmFormatModifierExplicitCreateInfoEXT, wrapped
294 // with QRhiTexture::createFrom. Every step is capability-gated; any miss returns fail(). Single-plane RGB layouts
295 // only (see vkFormatForVideoInfo); validatePlaneHandles() already routed other layouts to the CPU path.
296 const auto* nh = static_cast<const QRhiVulkanNativeHandles*>(rhi.nativeHandles());
297 if (!nh || nh->dev == VK_NULL_HANDLE || nh->physDev == VK_NULL_HANDLE || nh->gfxQueue == VK_NULL_HANDLE ||
298 !nh->inst) {
299 QGC_HW_WARN_ONCE(GstDmaBufVulkanLog, s_loggedVulkanUnimpl, "Vulkan import: native device handles unavailable");
300 return GstHwPathTelemetry::fail(HwVideoBufferPath::DmaBuf);
301 }
302 const VkDevice dev = nh->dev;
303 const DeviceFns& fns = deviceFns(*nh->inst, dev);
304 if (!fns.ok()) {
305 QGC_HW_WARN_ONCE(GstDmaBufVulkanLog, s_loggedVulkanUnimpl,
306 "Vulkan import: required device functions (VK_EXT_external_memory_dma_buf) unavailable");
307 return GstHwPathTelemetry::fail(HwVideoBufferPath::DmaBuf);
308 }
309 GstBuffer* buffer = gst_sample_get_buffer(_sample);
310 if (!buffer || gst_buffer_n_memory(buffer) != 1) {
311 // Multi-fd disjoint planes: out of scope for this conservative single-VkImage import.
312 QGC_HW_WARN_ONCE(GstDmaBufVulkanLog, s_loggedVulkanUnimpl,
313 "Vulkan import: multi-fd DMABuf not supported — CPU upload");
314 return GstHwPathTelemetry::fail(HwVideoBufferPath::DmaBuf);
315 }
316 const VkFormat vkFormat = vkFormatForVideoInfo(_videoInfo);
317 if (vkFormat == VK_FORMAT_UNDEFINED) {
318 QGC_HW_WARN_ONCE(GstDmaBufVulkanLog, s_loggedVulkanUnimpl,
319 "Vulkan import: no VkFormat for DMABuf layout — CPU upload");
320 return GstHwPathTelemetry::fail(HwVideoBufferPath::DmaBuf);
321 }
322
323 // Pre-flight RHI format/size support before building the VkImage so an unsupported import demotes to CPU on a
324 // query.
325 if (!GstHwImportPreflight::preflightOrRecord(&rhi, HwVideoBufferPath::DmaBuf, _format.pixelFormat(),
326 _format.frameSize())) {
327 return GstHwPathTelemetry::fail(HwVideoBufferPath::DmaBuf);
328 }
329
330 GstMemory* gstMemory = gst_buffer_peek_memory(buffer, 0);
331 const int fd = gst_dmabuf_memory_get_fd(gstMemory);
332 if (fd < 0) {
333 return GstHwPathTelemetry::fail(HwVideoBufferPath::DmaBuf);
334 }
335 VkSubresourceLayout planeLayout = {};
336 if (!validateVulkanPlaneLayout(_videoInfo, gst_buffer_get_video_meta(buffer), dmaMemoryBound(gstMemory, _videoInfo),
337 planeLayout)) {
338 return GstHwPathTelemetry::fail(HwVideoBufferPath::DmaBuf);
339 }
340
341 VkImageDrmFormatModifierExplicitCreateInfoEXT modInfo = {};
342 modInfo.sType = VK_STRUCTURE_TYPE_IMAGE_DRM_FORMAT_MODIFIER_EXPLICIT_CREATE_INFO_EXT;
343 // Legacy DMABuf caps carry no modifier; Vulkan needs an explicit one, and those producers export LINEAR.
344 modInfo.drmFormatModifier = (_drmModifier == DRM_FORMAT_MOD_INVALID) ? DRM_FORMAT_MOD_LINEAR : _drmModifier;
345 modInfo.drmFormatModifierPlaneCount = 1;
346 modInfo.pPlaneLayouts = &planeLayout;
347
348 VkExternalMemoryImageCreateInfo extImg = {};
349 extImg.sType = VK_STRUCTURE_TYPE_EXTERNAL_MEMORY_IMAGE_CREATE_INFO;
350 extImg.pNext = &modInfo;
351 extImg.handleTypes = VK_EXTERNAL_MEMORY_HANDLE_TYPE_DMA_BUF_BIT_EXT;
352
353 VkImageCreateInfo imgInfo = {};
354 imgInfo.sType = VK_STRUCTURE_TYPE_IMAGE_CREATE_INFO;
355 imgInfo.pNext = &extImg;
356 imgInfo.imageType = VK_IMAGE_TYPE_2D;
357 imgInfo.format = vkFormat;
358 imgInfo.extent = {static_cast<uint32_t>(GST_VIDEO_INFO_WIDTH(&_videoInfo)),
359 static_cast<uint32_t>(GST_VIDEO_INFO_HEIGHT(&_videoInfo)), 1};
360 imgInfo.mipLevels = 1;
361 imgInfo.arrayLayers = 1;
362 imgInfo.samples = VK_SAMPLE_COUNT_1_BIT;
363 imgInfo.tiling = VK_IMAGE_TILING_DRM_FORMAT_MODIFIER_EXT;
364 imgInfo.usage = VK_IMAGE_USAGE_SAMPLED_BIT;
365 imgInfo.sharingMode = VK_SHARING_MODE_EXCLUSIVE;
366 imgInfo.initialLayout = VK_IMAGE_LAYOUT_UNDEFINED;
367
368 VkImage image = VK_NULL_HANDLE;
369 if (fns.createImage(dev, &imgInfo, nullptr, &image) != VK_SUCCESS || image == VK_NULL_HANDLE) {
370 QGC_HW_WARN_ONCE(GstDmaBufVulkanLog, s_loggedVulkanUnimpl, "Vulkan import: vkCreateImage failed — CPU upload");
371 return GstHwPathTelemetry::fail(HwVideoBufferPath::DmaBuf);
372 }
373 const auto destroyImage = qScopeGuard([&] {
374 if (image != VK_NULL_HANDLE)
375 fns.destroyImage(dev, image, nullptr);
376 });
377
378 // dup the fd: Vulkan takes ownership of the imported fd on successful allocate.
379 const int dupFd = ::fcntl(fd, F_DUPFD_CLOEXEC, 0);
380 if (dupFd < 0) {
381 return GstHwPathTelemetry::fail(HwVideoBufferPath::DmaBuf);
382 }
383 bool fdConsumed = false;
384 const auto closeFd = qScopeGuard([&] {
385 if (!fdConsumed && dupFd >= 0)
386 ::close(dupFd);
387 });
388
389 VkMemoryFdPropertiesKHR fdProps = {};
390 fdProps.sType = VK_STRUCTURE_TYPE_MEMORY_FD_PROPERTIES_KHR;
391 if (fns.getMemoryFdProperties(dev, VK_EXTERNAL_MEMORY_HANDLE_TYPE_DMA_BUF_BIT_EXT, dupFd, &fdProps) != VK_SUCCESS) {
392 return GstHwPathTelemetry::fail(HwVideoBufferPath::DmaBuf);
393 }
394
395 VkMemoryRequirements memReq = {};
396 fns.getImageMemoryRequirements(dev, image, &memReq);
397 VkPhysicalDeviceMemoryProperties physMem = {};
398 fns.getPhysicalDeviceMemoryProperties(nh->physDev, &physMem);
399 uint32_t memTypeIndex = UINT32_MAX;
400 const uint32_t allowed = memReq.memoryTypeBits & fdProps.memoryTypeBits;
401 for (uint32_t i = 0; i < physMem.memoryTypeCount; ++i) {
402 if (allowed & (1u << i)) {
403 memTypeIndex = i;
404 break;
405 }
406 }
407 if (memTypeIndex == UINT32_MAX) {
408 return GstHwPathTelemetry::fail(HwVideoBufferPath::DmaBuf);
409 }
410
411 VkImportMemoryFdInfoKHR importInfo = {};
412 importInfo.sType = VK_STRUCTURE_TYPE_IMPORT_MEMORY_FD_INFO_KHR;
413 importInfo.handleType = VK_EXTERNAL_MEMORY_HANDLE_TYPE_DMA_BUF_BIT_EXT;
414 importInfo.fd = dupFd;
415
416 VkMemoryDedicatedAllocateInfo dedicated = {};
417 dedicated.sType = VK_STRUCTURE_TYPE_MEMORY_DEDICATED_ALLOCATE_INFO;
418 dedicated.image = image;
419 dedicated.pNext = &importInfo;
420
421 VkMemoryAllocateInfo allocInfo = {};
422 allocInfo.sType = VK_STRUCTURE_TYPE_MEMORY_ALLOCATE_INFO;
423 allocInfo.pNext = &dedicated;
424 allocInfo.allocationSize = memReq.size;
425 allocInfo.memoryTypeIndex = memTypeIndex;
426
427 // VK_KHR_external_memory_fd transfers fd ownership to the implementation only on a SUCCESSFUL import; on failure
428 // the application still owns the fd and must close it. Disarm the close guard only after VK_SUCCESS.
429 VkDeviceMemory memory = VK_NULL_HANDLE;
430 if (fns.allocateMemory(dev, &allocInfo, nullptr, &memory) != VK_SUCCESS || memory == VK_NULL_HANDLE) {
431 QGC_HW_WARN_ONCE(GstDmaBufVulkanLog, s_loggedVulkanUnimpl,
432 "Vulkan import: vkAllocateMemory failed — CPU upload");
433 return GstHwPathTelemetry::fail(HwVideoBufferPath::DmaBuf);
434 }
435 fdConsumed = true;
436 const auto freeMemory = qScopeGuard([&] {
437 if (memory != VK_NULL_HANDLE)
438 fns.freeMemory(dev, memory, nullptr);
439 });
440
441 if (const VkResult bindRes = fns.bindImageMemory(dev, image, memory, 0); bindRes != VK_SUCCESS) {
442 QGC_HW_WARN_ONCE(GstDmaBufVulkanLog, s_loggedVulkanUnimpl,
443 "Vulkan import: vkBindImageMemory failed (VkResult=" << bindRes << ") — CPU upload");
444 return GstHwPathTelemetry::fail(HwVideoBufferPath::DmaBuf);
445 }
446
447 const GstHwMapTail tail{.path = HwVideoBufferPath::DmaBuf,
448 .frameSize = _format.frameSize(),
449 .pixelFormat = _format.pixelFormat(),
450 .log = &GstDmaBufVulkanLog(),
451 .diag = &s_diag,
452 .preflight = false,
453 .successTag = "DMABuf->Vulkan"};
454 return GstHwFrameTexturesBase::finishMap<GstVulkanOwnedFrameTextures>(*this, rhi, old, tail, nullptr, [&] {
455 auto bundle = std::make_unique<GstVulkanOwnedFrameTextures>(&rhi, tail.frameSize, tail.pixelFormat, image);
456 if (bundle->allPlanesValid()) {
457 // Ownership of image/memory passes to the bundle; disarm the scope guards.
458 bundle->adoptVulkanResources(deferredRelease(rhi, {dev, image, memory, fns}));
459 image = VK_NULL_HANDLE;
460 memory = VK_NULL_HANDLE;
461 }
462 return bundle;
463 });
