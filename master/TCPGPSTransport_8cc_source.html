60GPSReadResult TCPGPSTransport::read(uint8_t* buffer, int length, int timeoutMs)
61{
62 if (isCancelled()) {
63 return {GPSReadStatus::Cancelled};
64 }
65 if (!buffer || length < 0) {
66 return {GPSReadStatus::InvalidData};
67 }
68 if (!_socket) {
69 return {GPSReadStatus::Closed};
70 }
71 if (length == 0) {
72 return {fatalError() ? GPSReadStatus::Closed : GPSReadStatus::Data};
73 }
74 if (timeoutMs <= 0 && _socket->bytesAvailable() == 0 && !fatalError()) {
75 // bytesAvailable() only sees Qt's buffer; immediate polls must also service kernel input.
76 _socket->waitForReadyRead(0);
77 }
78 if (_socket->bytesAvailable() == 0 && !gpsWaitForSocket(
79 *this, _socket.get(), [this]() { return _socket->bytesAvailable() > 0; },
80 QDeadlineTimer((std::max) (timeoutMs, 0)), kCancellationPollMs)) {
81 return {isCancelled() ? GPSReadStatus::Cancelled
82 : fatalError() ? GPSReadStatus::Closed
83 : GPSReadStatus::TimedOut,
84 0, fatalError() ? _socket->errorString() : QString()};
85 }
86 const auto count = _socket->read(reinterpret_cast<char*>(buffer), length);
87 return {count < 0 ? GPSReadStatus::Error : GPSReadStatus::Data, static_cast<int>((std::max) (count, qint64(0))),
88 count < 0 ? _socket->errorString() : QString()};
89}
