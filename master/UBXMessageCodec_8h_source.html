54template <typename T>
55struct MessageCodec
56{
57 static constexpr uint16_t MESSAGE = MESSAGE_ID<T>;
58
59 static std::optional<T> decode(std::span<const uint8_t> bytes)
60 {
61 if constexpr (MESSAGE != 0) {
62 if (!validPayload(MESSAGE, bytes)) {
63 return std::nullopt;
64 }
65 if constexpr (std::is_same_v<T, ubx_payload_rx_mon_hw_ubx6_t> ||
66 std::is_same_v<T, ubx_payload_rx_mon_hw_ubx7_t>) {
67 if (bytes.size() != WIRE_SIZE<T>) {
68 return std::nullopt;
69 }
70 }
71 } else if (bytes.size() != WIRE_SIZE<T>) {
72 return std::nullopt;
73 }
74 return detail::decodeFields<T>(bytes);
75 }
76
77 static std::optional<T> block(std::span<const uint8_t> bytes, size_t offset = 0)
78 {
79 if (offset > bytes.size() || WIRE_SIZE<T> > bytes.size() - offset) {
80 return std::nullopt;
81 }
82 return decode(bytes.subspan(offset, WIRE_SIZE<T>));
83 }
84};
