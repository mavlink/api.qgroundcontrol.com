38CorrectionsConfig correctionsFromSettings(GPSCorrectionSettings* settings, FactReader& read)
39{
40 CorrectionsConfig configuration;
41 const int source = read(settings->correctionSource()).toInt();
42 if (source >= GPSCorrectionSettings::HighestPriority && source < GPS_CORRECTION_SOURCE_COUNT) {
43 configuration.source = static_cast<GPSCorrectionSettings::CorrectionSource>(source);
44 } else {
45 qCWarning(GPSSettingsBindingsLog) << "Invalid correction source; using automatic selection";
46 }
47 configuration.udpInput.enabled = read(settings->rtcmUdpInputEnabled()).toBool();
48 configuration.udpInput.port = read(settings->rtcmUdpInputPort()).value<quint16>();
49 configuration.udpOutput.enabled = read(settings->rtcmUdpOutputEnabled()).toBool();
50 configuration.udpOutput.address = read(settings->rtcmUdpOutputAddress()).toString().trimmed();
51 configuration.udpOutput.port = read(settings->rtcmUdpOutputPort()).value<quint16>();
52 return configuration;
53}

76GPSBaseStationConfig baseStationFromSettings(RTKSettings* settings, FactReader& read)
77{
78 using Mode = BaseModeDefinition::Mode;
79 const auto mode = static_cast<Mode>(read(settings->useFixedBasePosition()).toInt());
80 const GPSBaseStationConfig::Fixed fixed{
81 .position = {.latitudeDegrees = read(settings->fixedBasePositionLatitude()).toDouble(),
82 .longitudeDegrees = read(settings->fixedBasePositionLongitude()).toDouble(),
83 .altitudeMeters = read(settings->fixedBasePositionAltitude()).toDouble()},
84 .accuracyMeters = read(settings->fixedBasePositionAccuracy()).toFloat()};
85 const GPSBaseStationConfig::SurveyIn surveyIn{
86 .accuracyMeters = read(settings->surveyInAccuracyLimit()).toDouble(),
87 .duration = std::chrono::seconds(read(settings->surveyInMinObservationDuration()).value<int64_t>())};
88 const GPSBaseStationConfig::ReceiverAveraging averaging{
89 .maximumDuration = std::chrono::seconds(read(settings->receiverAveragingDuration()).value<uint32_t>())};
90
91 GPSBaseStationConfig base{.compactObservations = read(settings->compactRtcmCorrections()).toBool()};
92 switch (mode) {
93 case Mode::BaseFixed:
94 base.mode = fixed;
95 break;
96 case Mode::BaseReceiverAveraging:
97 base.mode = averaging;
98 break;
99 case Mode::BaseSurveyIn:
100 base.mode = surveyIn;
101 break;
102 default:
103 qCWarning(GPSSettingsBindingsLog) << "Invalid base mode; using survey-in";
104 base.mode = surveyIn;
105 break;
106 }
107 return base;
108}

161template <typename Settings, typename Configuration, typename Target>
162void bind(Settings* settings, Configuration (*build)(Settings*, FactReader&), Target* target,
163 void (Target::*setter)(const Configuration&))
164{
165 const auto apply = [settings, build, target, setter]() {
166 FactReader reader;
167 (target->*setter)(build(settings, reader));
168 return reader.facts();
169 };
170 for (Fact* const fact : apply()) {
171 logSetting(fact);
172 QObject::connect(fact, &Fact::rawValueChanged, target, [apply, fact]() {
173 logSetting(fact);
174 (void) apply();
175 });
176 }
177}
