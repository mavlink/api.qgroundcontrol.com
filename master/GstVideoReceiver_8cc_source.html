52GstVideoReceiver::~GstVideoReceiver()
53{
54 _postToWorker([](GstReceiverWorker& worker) { worker.shutdown(); });
55 if (_workerThread->wait(QDeadlineTimer(kWorkerShutdownTimeout))) {
56 qCDebug(GstVideoReceiverLog) << this;
57 return;
58 }
59
60 // Only a recorder stuck in the kernel (a hung network mount) holds the worker this long. It reaches this receiver
61 // only through its signals, so it can finish on its own and clean up after itself.
62 qCWarning(GstVideoReceiverLog) << "Worker still busy after"
63 << std::chrono::milliseconds(kWorkerShutdownTimeout).count()
64 << "ms; leaving it to finish on its own";
65 _workerThread->setParent(nullptr);
66 (void) connect(_workerThread, &QThread::finished, _workerThread, &QObject::deleteLater);
67 if (_workerThread->isFinished()) {
68 delete _workerThread;

165void GstVideoReceiver::_bindVideoSink(QVideoSink* videoSink)
166{
167 (void) disconnect(_videoSizeConnection);
168 if (_displaySink) {
169 _sinkController->setVideoSink(QPointer<QVideoSink>(videoSink));
170 } else if (!_buildDisplaySink(videoSink)) {
171 return;
172 }
173 _sinkController->followWindowVisibility(videoOutput());
174 // Per frame, not on QVideoSink::videoSizeChanged: that follows the frame size, which for a decoder's buffer cropped
175 // by crop meta is larger than the picture, and misses a viewport that changes on its own. Frames reach the sink on
176 // this thread.
177 _videoSizeConnection = connect(videoSink, &QVideoSink::videoFrameChanged, this,
178 [this](const QVideoFrame& frame) { setVideoSize(presentedSize(frame)); });
179 if (const QSize size = presentedSize(videoSink->videoFrame()); size.isValid()) {
180 setVideoSize(size);
