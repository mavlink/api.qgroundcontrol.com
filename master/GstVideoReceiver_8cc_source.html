22// The destructor's wait for the worker: a recording's finalization plus time for the pipeline to stop.
23constexpr std::chrono::seconds kWorkerShutdownTimeout = RecordingBranch::kFinalizeTimeout + std::chrono::seconds{2};
24static_assert(GstVideoReceiver::kHeartbeatInterval > kWorkerShutdownTimeout);
25
26// Checks in a row, 60 s, that find the worker has run none of its tasks since the check before, until it counts as
27// stuck: a name lookup for a tcp:// or udp:// host, which runs on the worker, can take half a minute to fail. Each
28// check comes a full interval after the one before, so one that is late (a suspended machine, a stopped process, a
29// stalled GUI thread) can't count twice against a worker that got no time to run a task.
30constexpr int kStuckWorkerChecks = 6;

61GstVideoReceiver::~GstVideoReceiver()
62{
63 _postToWorker([](GstReceiverWorker& worker) { worker.shutdown(); });
64 if (_workerThread->wait(QDeadlineTimer(kWorkerShutdownTimeout))) {
65 qCDebug(GstVideoReceiverLog) << this;
66 return;
67 }
68
69 // Only a streaming thread that doesn't return (a recorder stuck in the kernel on a hung network mount, a decoder
70 // deadlocked in its library) holds the worker this long. It reaches this receiver only through its signals, so it
71 // can finish on its own and clean up after itself.
72 qCWarning(GstVideoReceiverLog) << "Worker still busy after"
73 << std::chrono::milliseconds(kWorkerShutdownTimeout).count()
74 << "ms; leaving it to finish on its own";
75 _workerThread->setParent(nullptr);
76 (void) connect(_workerThread, &QThread::finished, _workerThread, &QObject::deleteLater);
77 if (_workerThread->isFinished()) {
78 delete _workerThread;
79 }

158void GstVideoReceiver::_checkWorker()
159{
160 const bool frameShown = std::exchange(_frameShown, false);
161 const bool ranTask = _tasksDone != std::exchange(_tasksDoneAtCheck, _tasksDone);
162 if (_tasksDone == _tasksPosted) {
163 _stalledChecks = 0;
164 if (!active()) {
165 // Through with all it was posted, and no stream is wanted of it: it has nothing to get stuck in.
166 return;
167 }
168 // The heartbeat of a worker that is posted nothing else: what it does of its own, as replacing a decoder that
169 // stalled, can hold it too.
170 _postToWorker([](GstReceiverWorker& /*worker*/) {});
171 } else if (ranTask) {
172 // Slow, or behind with a row of requests, but not stuck.
173 _stalledChecks = 0;
174 } else if (_workerStuck) {
175 _applyStuckStatus(frameShown);
176 } else if (++_stalledChecks >= kStuckWorkerChecks) {
177 qCWarning(GstVideoReceiverLog) << "The worker of" << name() << "has run none of its tasks for"
178 << (kStuckWorkerChecks * _heartbeatTimer->intervalAsDuration()).count()
179 << "ms: a streaming thread it waits for doesn't return";
180 _workerStuck = true;
181 _applyStuckStatus(frameShown);
182 emit errorOccurred(
183 tr("The video decoder or recorder stopped responding. Restart %1 to get video working again.")
184 .arg(QCoreApplication::applicationName()));
185 }
186 // A full interval from now, not from when this check was due: one that comes late puts the next ones off.
187 _heartbeatTimer->start();
188}

200void GstVideoReceiver::_applyStuckStatus(bool frameShown)
201{
202 // The pipeline runs on without its worker, which a recorder that doesn't return holds as well as a decoder: only
203 // the frames tell which. While they reach the output the stream is as the worker reported it last, though nothing
204 // measures it. Without them it is as one that gave up, so that no frozen picture passes for the live one.
205 GstReceiverStatus status = _workerStatus;
206 status.stats = {};
207 if (!frameShown) {
208 status.state = State::Stopped;
209 status.decoding = false;
210 status.decoderName.clear();
211 }
212 _applyStatus(status);
213}

281void GstVideoReceiver::_bindVideoSink(QVideoSink* videoSink)
282{
283 (void) disconnect(_videoSizeConnection);
284 if (_displaySink) {
285 _sinkController->setVideoSink(QPointer<QVideoSink>(videoSink));
286 } else if (!_buildDisplaySink(videoSink)) {
287 return;
288 }
289 _sinkController->followWindowVisibility(videoOutput());
290 // Per frame, not on QVideoSink::videoSizeChanged: that follows the frame size, which for a decoder's buffer cropped
291 // by crop meta is larger than the picture, and misses a viewport that changes on its own. Frames reach the sink on
292 // this thread. The picture's size is the display sink's: a viewport that leaves out the padding of a squeezed
293 // texture is smaller.
294 const auto sizeOf = [this](const QVideoFrame& frame) {
295 return presentedSize(frame, _sinkController ? _sinkController->pictureSize() : QSize());
296 };
297 _videoSizeConnection =
298 connect(videoSink, &QVideoSink::videoFrameChanged, this, [this, sizeOf](const QVideoFrame& frame) {
299 _frameShown = _frameShown || frame.isValid();

301 });
