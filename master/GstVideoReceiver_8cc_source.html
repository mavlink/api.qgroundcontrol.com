76bool isRecoverableH265PaciError(GstMessage* msg, const GError* error, const gchar* debug)
77{
78 if (!msg || !error || (error->domain != GST_STREAM_ERROR) || (error->code != GST_STREAM_ERROR_FORMAT) || !debug ||

80 return false;
81 }

89 return factory && (g_strcmp0(GST_OBJECT_NAME(factory), "rtph265depay") == 0);

129GstVideoReceiver::~GstVideoReceiver()
130{
131 // Tear down synchronously: queued tasks and pipeline callbacks capture `this`, so the worker
132 // must be idle and the pipeline gone before members are destroyed. No onStopComplete here;
133 // listeners must not be handed a receiver that is being destroyed.
134 _runOnWorker([this]() {
135 _reconnectTimer->stop();
136 (void) _stopPipeline();
137 _workerThread->quit();
138 });
139
140 if (!_workerThread->wait(QDeadlineTimer(kWorkerShutdownTimeout))) {
141 // Destroying a running QThread aborts the process, and queued tasks still reference this receiver.
142 qCWarning(GstVideoReceiverLog) << "Worker still busy after"
143 << std::chrono::milliseconds(kWorkerShutdownTimeout).count()
144 << "ms; waiting for it to finish";
145 (void) _workerThread->wait();
146 }
147
148 qCDebug(GstVideoReceiverLog) << this;
149}

174void GstVideoReceiver::_startPipeline(const StartConfig& config)
175{
176 if (_pipeline) {
177 qCDebug(GstVideoReceiverLog) << "Already running!" << _redactedUri();
178 emit onStartComplete(STATUS_INVALID_STATE);
179 return;
180 }
181
182 if (config.uri.isEmpty()) {
183 qCDebug(GstVideoReceiverLog) << "Failed because URI is not specified";
184 emit onStartComplete(STATUS_INVALID_URL);
185 return;
186 }
187
188 _config = config;
189 _watchdog.start(_config.timeoutSec);
190
191 qCDebug(GstVideoReceiverLog) << "Starting" << _redactedUri() << ", lowLatency" << _config.lowLatency << ", timeout"
192 << _config.timeoutSec;
193
194 bool running = false;
195
196 do {
197 _dotDumper.setReceiverName(_config.name);
198 _pipeline = GStreamer::adoptElement(gst_pipeline_new(_dotDumper.pipelineName().constData()));
199 if (!_pipeline) {
200 qCWarning(GstVideoReceiverLog) << "gst_pipeline_new() failed";
201 break;
202 }
203 // Nothing else ever takes a ref on the pipeline, so ours is a plain one.
204 (void) gst_object_ref_sink(_pipeline.get());
205 _attachBus();
206
207 g_object_set(_pipeline.get(), "message-forward", TRUE, nullptr);
208
209 _tee = gst_element_factory_make("tee", nullptr);
210 if (!_tee) {
211 qCWarning(GstVideoReceiverLog) << "gst_element_factory_make('tee') failed";
212 break;
213 }
214 (void) gst_bin_add(GST_BIN(_pipeline.get()), _tee);
215
216 const GStreamer::GstPadPtr teeSink = GStreamer::adoptPad(gst_element_get_static_pad(_tee, "sink"));
217 if (!teeSink) {
218 qCCritical(GstVideoReceiverLog) << "gst_element_get_static_pad() failed";
219 break;
220 }
221
222 if (!_teeFrameProbe.add(teeSink.get(), GST_PAD_PROBE_TYPE_BUFFER, _teeProbe, this)) {
223 // _teeProbe feeds the watchdog; without it the watchdog fires spuriously instead of reporting a real
224 // failure.
225 qCCritical(GstVideoReceiverLog) << "Failed to add the tee frame probe";
226 break;
227 }
228
229 if (!_decodingBranch.build(_pipeline.get(), _tee)) {
230 break;
231 }
232
233 if (!_recordingBranch.build(_pipeline.get(), _tee)) {
234 break;
235 }
236
237 GStreamer::SourceFactory::Config sourceConfig;
238 sourceConfig.jitterBuffer = _config.lowLatency ? GStreamer::SourceFactory::JitterBuffer::None
239 : GStreamer::SourceFactory::JitterBuffer::DropOnLatency;
240 sourceConfig.latencyMs = _config.rtpJitterLatencyMs;
241 sourceConfig.doRetransmission = (_config.rtpJitterLatencyMs >= kMinRetransmissionLatencyMs) &&
242 (sourceConfig.jitterBuffer != GStreamer::SourceFactory::JitterBuffer::None);
243 _source = GStreamer::SourceFactory::create(_config.uri, sourceConfig);
244 if (!_source) {
245 qCWarning(GstVideoReceiverLog) << "Failed to create the source for" << _redactedUri();
246 break;
247 }
248 (void) gst_bin_add(GST_BIN(_pipeline.get()), _source);
249
250 if (const GStreamer::GstPadPtr srcPad = GStreamer::firstSrcPad(_source)) {
251 _onNewSourcePad(srcPad.get());
252 } else {
253 (void) g_signal_connect(_source, "pad-added", G_CALLBACK(_onNewPad), this);
254 }
255
256 _dotDumper.dump(_pipeline.get(), "pipeline-initial");
257 running = (gst_element_set_state(_pipeline.get(), GST_STATE_PLAYING) != GST_STATE_CHANGE_FAILURE);
258 } while(0);
259
260 if (!running) {
261 qCWarning(GstVideoReceiverLog) << "Failed to start" << _redactedUri();
262 _resetPipelineState();
263
264 emit onStartComplete(STATUS_FAIL);
265 _scheduleReconnect("start failure");
266 } else {
267 _dotDumper.dump(_pipeline.get(), "pipeline-started");
268 qCDebug(GstVideoReceiverLog) << "Started" << _redactedUri();
269
270 _watchdogTimer->start();
271 emit onStartComplete(STATUS_OK);
272 }

285bool GstVideoReceiver::_stopPipeline()
286{
287 if (!_pipeline) {
288 qCDebug(GstVideoReceiverLog) << "Not running" << _redactedUri();
289 // Still drop any state left behind.
290 _resetPipelineState();
291 return false;
292 }
293
294 qCDebug(GstVideoReceiverLog) << "Stopping" << _redactedUri();
295
296 _recordingBranch.finishForTeardown();
297
298 // Before the teardown releases the video sink.
299 const GstQgc::PathStats hwStats = _decodingBranch.sinkPathStats();
300
301 _resetPipelineState();
302
303 qCDebug(GstVideoReceiverLog) << "Stopped" << _redactedUri();
304
305 if (hwStats.totalDelivered > 0) {
306 qCDebug(GstVideoReceiverLog).noquote()
307 << "GPU path stats" << _redactedUri() << hwStats.line + HwBuffers::takeExtraPathStats();
308 }
309
310 return true;

313void GstVideoReceiver::_resetPipelineState()
314{
315 _watchdogTimer->stop();
316
317 if (_pipeline) {
318 // Synchronous: the streaming threads have stopped when this returns.
319 (void) gst_element_set_state(_pipeline.get(), GST_STATE_NULL);
320 }
321
322 // Callbacks have stopped: streaming threads are joined and no thread is left in the bus handler. Work they
323 // queued for this pipeline is now stale.
324 _detachBus();
325 _pipelineGeneration.fetch_add(1, std::memory_order_relaxed);
326
327 _teeFrameProbe.remove();
328 _sourceEosProbe.remove();
329
330 _recordingBranch.reset();
331 _decodingBranch.reset();
332
333 _dotDumper.dump(_pipeline.get(), "pipeline-stopped");
334
335 _pipeline.reset();
336
337 // Owned by the (now released) pipeline.
338 _tee = nullptr;
339 _source = nullptr;
340
341 if (std::exchange(_streaming, false)) {
342 qCDebug(GstVideoReceiverLog) << "Streaming stopped" << _redactedUri();
343 emit streamingChanged(false);
344 }

373void GstVideoReceiver::_onWatchdogTick()
374{
375 if (!_pipeline) {
376 return;
377 }
378
379 if (++_statsTickCounter >= kHwStatsLogIntervalTicks) {
380 _statsTickCounter = 0;
381 if (const GstQgc::PathStats hwStats = _decodingBranch.sinkPathStats(); hwStats.totalDelivered > 0) {
382 qCDebug(GstVideoReceiverLog).noquote() << "GPU path live" << _redactedUri() << hwStats.line;
383 }
384 }
385
386 const StreamWatchdog::Result result = _watchdog.check(StreamWatchdog::now(), _decodingBranch.isDecoding());
387 if (result.stall == StreamWatchdog::Stall::None) {
388 return;
389 }
390
391 const bool sourceStalled = (result.stall == StreamWatchdog::Stall::Source);
392 qCDebug(GstVideoReceiverLog) << (sourceStalled ? "Stream timeout, no frames for"
393 : "Video decoder timeout, no frames for")
394 << result.silentMs << "ms" << _redactedUri();
395 _dotDumper.snapshot(_pipeline.get(), QStringLiteral("pipeline-watchdog-timeout"));
396 emit timeout();
397 _onStreamFailure(sourceStalled ? "source watchdog" : "decoder watchdog");
398}

522GstBusSyncReply GstVideoReceiver::_onBusSyncMessage(GstBus* /*bus*/, GstMessage* message, gpointer user_data)
523{
524 // The context bridges answer NEED_CONTEXT synchronously and consume (unref) the requests they handle.
525 if (HwBuffers::onBusSyncMessage(message) == GST_BUS_DROP) {
526 return GST_BUS_DROP;
527 }
528
529 auto* self = static_cast<GstVideoReceiver*>(user_data);
530 switch (GST_MESSAGE_TYPE(message)) {
531 case GST_MESSAGE_ERROR:
532 self->_onBusError(message);
533 break;
534 case GST_MESSAGE_WARNING:
535 logBusWarning(message);
536 break;
537 case GST_MESSAGE_EOS:
538 self->_recordingBranch.onBusEos();
539 break;
540 case GST_MESSAGE_ELEMENT:
541 if (gst_message_has_name(message, GstQgc::kCapsInfoMessage)) {
542 self->_decodingBranch.onBusCapsInfo(message);
543 } else {
544 self->_recordingBranch.onBusElement(message);
545 }
546 break;
547 case GST_MESSAGE_STREAM_COLLECTION:
548 selectVideoStreams(message);
549 break;
550 case GST_MESSAGE_STATE_CHANGED:
551 if (pipelineEnteredPlaying(message)) {
552 self->_runOnWorkerForPipeline([self]() { self->_onPipelinePlaying(); });
553 }
554 break;
555 case GST_MESSAGE_LATENCY:
556 // Recalculation hands the sinks their new latency in a LATENCY event.
557 self->_runOnWorkerForPipeline(
558 [self]() { (void) gst_bin_recalculate_latency(GST_BIN(self->_pipeline.get())); });
559 break;
560 default:
561 break;
562 }
563
564 // Every message is handled above and nothing pops this bus; queueing it would grow the bus for the
565 // pipeline's lifetime. A sync handler that drops a message owns it.
566 gst_message_unref(message);
567 return GST_BUS_DROP;
568}

570void GstVideoReceiver::_onBusError(GstMessage* message)
571{
572 GError* rawError = nullptr;
573 gchar* rawDebug = nullptr;
574 gst_message_parse_error(message, &rawError, &rawDebug);
575 const GStreamer::GErrorPtr error = GStreamer::adoptError(rawError);
576 const GStreamer::GCharPtr debug = GStreamer::adoptGChar(rawDebug);
577 const bool recoverableH265PaciError = isRecoverableH265PaciError(message, error.get(), debug.get());
578 const QString text = QString::fromUtf8(error ? error->message : "(no message)");
579 if (debug) {
580 qCDebug(GstVideoReceiverLog) << "GStreamer debug:" << debug.get();
581 }
582
583 if (recoverableH265PaciError) {
584 qCWarning(GstVideoReceiverLog) << "Ignoring unsupported H.265 RTP PACI packet from rtph265depay:" << text;
585 return;
586 }
587
588 if (_recordingBranch.onBusError(message, text, debug.get())) {
589 return;
590 }
591
592 const GStreamer::GCharPtr sourcePath = GStreamer::adoptGChar(
593 GST_MESSAGE_SRC(message) ? gst_object_get_path_string(GST_MESSAGE_SRC(message)) : nullptr);
594 qCWarning(GstVideoReceiverLog) << "GStreamer error from" << (sourcePath ? sourcePath.get() : "(unknown)") << ":"
595 << text;
596 HwBuffers::onPipelineError(error.get(), debug.get());
597 // A failing pipeline usually posts several ERRORs; the first one's teardown makes the rest stale.
598 _runOnWorkerForPipeline([this]() { _onPipelineError(); });
599}

601void GstVideoReceiver::_onNewPad(GstElement* element, GstPad* pad, gpointer data)
602{
603 auto* self = static_cast<GstVideoReceiver*>(data);
604
605 // Runs on a streaming thread; the pipeline is only changed on the worker. Data waits at the pad until the
606 // worker has linked it, since an unlinked push would fail the stream.
607 const gulong blockId = gst_pad_add_probe(pad, GST_PAD_PROBE_TYPE_BLOCK_DOWNSTREAM, holdPad, nullptr, nullptr);
608 self->_runOnWorkerForPipeline([self, blockId,
609 padElement = GStreamer::adoptElement(GST_ELEMENT(gst_object_ref(element))),
610 newPad = GStreamer::adoptPad(GST_PAD(gst_object_ref(pad)))]() {
611 self->_onPadAdded(padElement.get(), newPad.get());
612 gst_pad_remove_probe(newPad.get(), blockId);
613 });
614}
