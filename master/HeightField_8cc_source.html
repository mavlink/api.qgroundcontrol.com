22bool HeightField::insertTile(const TileMath::TileKey& key, ElevationTilePyramid::Grid grid)
23{
24 // Capture before the move: on rejection the grid has been consumed
25 const int gridWidth = grid.width;
26 const int gridHeight = grid.height;
27 TileMath::TileKey evicted{0, 0, -1};
28 if (!_pyramid.insertTile(key, std::move(grid), &evicted)) {
29 qCWarning(GeoMapHeightFieldLog) << "insertTile rejected: key" << key << "grid" << gridWidth << "x"
30 << gridHeight;
31 return false;
32 }
33 _memoView = ElevationTilePyramid::View{}; // grid pointers die on insert
34 qCDebug(GeoMapHeightFieldVerboseLog) << "inserted tile" << key;
35
36 const auto tileExtent = [](const TileMath::TileKey& k) {
37 const QPointF corner = TileMath::tileMinCorner(k);
38 const double span = TileMath::tileSpanAtZoom(k.zoom);
39 return QRectF(corner.x(), corner.y(), span, span);
40 };
41 if (TileMath::isValidKey(evicted)) {
42 qCDebug(GeoMapHeightFieldVerboseLog) << "evicted tile" << evicted;
43 emit regionChanged(tileExtent(evicted));
44 }
45 emit regionChanged(tileExtent(key));
46 return true;
47}

49double HeightField::heightAt(const QPointF& world) const
50{
51 const TileMath::TileKey query = TileMath::tileForWorld(world, TileMath::kMaxZoom);
52
53 // Memoized fast path: the last resolved tile answers when the position
54 // resolves to it (the memo is only populated when no stored descendant
55 // could override it, and every insert invalidates it). The hit test uses
56 // the same key arithmetic as the full lookup — a world-coordinate bounds
57 // check can disagree with tileForWorld by an ulp exactly on a tile
58 // boundary, silently answering with the wrong neighbor's data there.
59 if (_memoView.isValid()) {
60 const int shift = TileMath::kMaxZoom - _memoView.key.zoom;
61 if (((query.x >> shift) == _memoView.key.x) && ((query.y >> shift) == _memoView.key.y)) {
62 const double u = (world.x() - _memoMinX) / _memoSpan;
63 const double v = (_memoMaxY - world.y()) / _memoSpan; // grid origin is the NW corner
64 return PatchSampler::heightAtUV(*_memoView.grid, u, v);
65 }
66 }
67
68 // The pyramid resolves the finest stored cover of the deepest-zoom query
69 const ElevationTilePyramid::View view = _pyramid.bestTileFor(query);
70 if (!view.isValid()) {
71 return 0.0;
72 }
73
74 const QPointF corner = TileMath::tileMinCorner(view.key);
75 const double span = TileMath::tileSpanAtZoom(view.key.zoom);
76 if (!_pyramid.hasDescendant(view.key)) {
77 // Nothing finer exists anywhere inside this tile, so it answers for
78 // every position within its bounds until the next insert
79 _memoView = view;
80 _memoMinX = corner.x();
81 _memoMaxY = corner.y() + span;
82 _memoSpan = span;
83 }
84
85 const double u = (world.x() - corner.x()) / span;
86 const double v = ((corner.y() + span) - world.y()) / span; // grid origin is the NW corner
87 return PatchSampler::heightAtUV(*view.grid, u, v);
88}

90QList<float> HeightField::samplePatch(const TileMath::TileKey& key, int gridSize) const
91{
92 if ((gridSize < 1) || (gridSize > kMaxGridSize) || !TileMath::isValidKey(key)) {
93 qCWarning(GeoMapHeightFieldLog) << "samplePatch rejected: key" << key << "gridSize" << gridSize;
94 return QList<float>();
95 }
96
97 // Interior vertices resolve the patch's backing view once, by the
98 // patch's own key, and interpolate within that one grid. Boundary
99 // vertices are shared with neighbor patches whose backing views can
100 // differ (adjacent exact tiles, fine tile next to an ancestor-backed
101 // neighbor), so they resolve canonically by position instead: the
102 // deepest-zoom cells touching the vertex, tried east/south first, pick
103 // the same stored tile no matter which patch asks. Positions are exact
104 // dyadic values, so coincident vertices compute bit-identical UVs and
105 // sample bit-identical heights: meshes never crack where data exists.
106 return PatchSampler(_pyramid, key, gridSize).sample();
107}
