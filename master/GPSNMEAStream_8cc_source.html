35GPSDecodedSatellites satelliteReport(const NMEA::SatelliteSystem& system)
36{
37 GPSDecodedSatellites report;

75GPSReceiveUpdates GPSNMEAStream::_decodeStandard(std::string_view line, GPSDecodeContext& context)
76{
77 const auto sentence = NMEA::sentence(line);
78 if (!sentence) {
79 return {};
80 }
81 const uint64_t now = context.nowUs();
82 GPSReceiveUpdates updates = GPSReceiveUpdate::Activity;
83 auto satelliteUpdate = _satelliteAssembler.ingest(*sentence, now);
84 _queueSatellites(satelliteUpdate.completed);
85 if (_navigation == Navigation::ReceiverSpecific) {
86 return updates;
87 }
88 const auto navigation = NMEA::navigationStatus(*sentence);
89 const auto update = sentence->type() == "GSA" && navigation && navigation->valid && !satelliteUpdate.accepted
90 ? std::optional<NMEA::NavigationUpdate>()
91 : _navigationAssembler.ingest(*sentence, now);
92 if (update && update->type == NMEA::NavigationUpdate::Type::FixLoss) {
93 _position = {};
94 _position.navigation.timestampUs = now;
95 _position.navigation.fixType = GPSPositionReport::FixType::NoFix;
96 // Without a fix no satellite is used, unless the sentence reports a count of its own.
97 const uint8_t used = gpsSatellitesUsed(update->epoch.satellitesUsed).value_or(0);
98 _position.navigation.satellitesUsed = used;
99 context.publishSatelliteUsage(used);
100 updates |= GPSReceiveUpdate::Position;
101 } else if (update && update->type == NMEA::NavigationUpdate::Type::Epoch &&
102 update->trigger == NMEA::NavigationUpdate::Trigger::Position && sentence->type() == "GGA") {
103 applyNavigationEpoch(_position, update->epoch);
104 context.publishSatelliteUsage(gpsSatellitesUsed(update->epoch.satellitesUsed));
105 updates |= GPSReceiveUpdate::Position;
106 } else if (update && update->type == NMEA::NavigationUpdate::Type::Epoch &&
107 update->trigger == NMEA::NavigationUpdate::Trigger::TimedMetadata) {
108 applyNavigationEpoch(_position, update->epoch);
109 updates |= GPSReceiveUpdate::Position;
110 }
111 return updates;
112}
