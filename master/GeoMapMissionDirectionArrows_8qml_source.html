11import QtPositioning

13import QGroundControl.Controls

15/// General flight-path direction indicators for the GeoMap: one arrow per
16/// entry in MissionController.directionArrows (added to the second flight
17/// path segment and every 5 segments thereafter, regardless of simple vs
18/// complex item — see MissionController::_recalcFlightPathSegments),
19/// mirroring the MapLineArrow Repeater in src/FlightMap/MapItems/PlanMapItems.qml.
20Item {
21 id: root
22
23 property var missionController
24 property var scene
25 property var surfaceModel
26 property real homeTerrainBias: 0 ///< See GeoMapWaypointItem.homeTerrainBias
27
28 Repeater {
29 model: root.missionController ? root.missionController.directionArrows : 0
30
31 GeoMapSegmentArrow {
32 id: arrow
33 scene: root.scene
34 surfaceModel: root.surfaceModel
35 fromCoord: {
36 _terrainRev
37 return _terrainLeg ? _drapedPoint(0.75 - _halfChordFraction)
38 : QtPositioning.coordinate(object.coordinate1.latitude, object.coordinate1.longitude,
39 object.coord1AMSLAlt + root.homeTerrainBias)
40 }
41 toCoord: {
42 _terrainRev
43 return _terrainLeg ? _drapedPoint(0.75 + _halfChordFraction)
44 : QtPositioning.coordinate(object.coordinate2.latitude, object.coordinate2.longitude,
45 object.coord2AMSLAlt + root.homeTerrainBias)
46 }
47 fraction: _terrainLeg ? 0.5 : 0.75
48
49 // Terrain-frame legs are draped over the DEM (see
50 // GeoMapMissionPath), so anchor the arrow to a short draped chord
51 // around the 0.75 mark rather than the straight AMSL chord: the
52 // chord midpoint puts the arrow on the ribbon and the chord ends
53 // give it the local tangent heading
54 readonly property bool _terrainLeg: !!root.surfaceModel && object.segmentType === FlightPathSegment.SegmentTypeTerrainFrame
55
56 // Cap the chord at 50 m so it stays local to the ribbon on long
57 // legs (a 10% chord averages away terrain curvature), while short
58 // legs keep enough screen separation for a stable heading
59 readonly property real _halfChordFraction: {
60 const len = object.coordinate1.distanceTo(object.coordinate2)
61 return len > 0 ? Math.min(0.05, 25 / len) : 0.05
62 }
63
64 // terrainHeightAt is not NOTIFY-reactive; read in the coordinate
65 // bindings above so they re-sample as DEM patches stream in
66 property int _terrainRev: 0
67
68 // Same AGL-ramp draping as GeoMapMissionPath._appendTerrainLeg
69 function _drapedPoint(f) {
70 const c1 = object.coordinate1
71 const c2 = object.coordinate2
72 const point = c1.atDistanceAndAzimuth(c1.distanceTo(c2) * f, c1.azimuthTo(c2))
73 const agl1 = object.coord1AMSLAlt - root.surfaceModel.terrainHeightAt(c1)
74 const agl2 = object.coord2AMSLAlt - root.surfaceModel.terrainHeightAt(c2)
75 return QtPositioning.coordinate(point.latitude, point.longitude,
76 root.surfaceModel.terrainHeightAt(point) + agl1 + ((agl2 - agl1) * f) + root.homeTerrainBias)
77 }
78
79 Connections {
80 target: root.surfaceModel
81 enabled: arrow._terrainLeg
82 function onTerrainHeightsChanged() { arrow._terrainRev++ }
83 }
84 }
85 }
