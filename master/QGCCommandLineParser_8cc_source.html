65#ifdef QGC_UNITTEST_BUILD
66QStringList normalizeArgs(const QStringList &args)
67#else
68static QStringList normalizeArgs(const QStringList &args)
69#endif
70{
71 QStringList out;
72 out.reserve(args.size() + 4);
73
74 for (qsizetype i = 0; i < args.size(); ++i) {
75 const QString &arg = args.at(i);
76#ifdef QGC_UNITTEST_BUILD
77 if (arg == QLatin1String("--unittest")) {
78 const bool valueFollows = ((i + 1) < args.size()) && !args.at(i + 1).startsWith(QLatin1Char('-'));
79 out << arg;
80 if (!valueFollows) {
81 out << QString();
82 }
83 continue;
84 }
85#endif
86
87 // Convert --option:value to --option value
88 if (arg.startsWith(QLatin1String("--")) && arg.contains(QLatin1Char(':'))) {
89 const qsizetype idx = arg.indexOf(QLatin1Char(':'));
90 const QString opt = arg.left(idx);
91 const QString val = arg.mid(idx + 1);
92 out << opt;
93 if (!val.isEmpty()) {
94 out << val;
95 }
96#ifdef QGC_UNITTEST_BUILD
97 else if (opt == QLatin1String("--unittest")) {
98 out << QString();
99 }
100#endif
101 } else {
102 out << arg;
103 }
104 }
105
106 qCDebug(QGCCommandLineParserLog) << "Normalized arguments:" << out;
107 return out;
108}

465std::optional<int> handleParseResult(const CommandLineParseResult& result)
466{
467 switch (result.statusCode) {
468 case CommandLineParseResult::Status::Error: {
469 const QString msg = result.errorString.value_or(QStringLiteral("Unknown error"));
470 QCommandLineParser::showMessageAndExit(QCommandLineParser::MessageType::Error, msg, 1);
471 Q_UNREACHABLE();
472 }
473 case CommandLineParseResult::Status::HelpRequested:
474 case CommandLineParseResult::Status::VersionRequested:
475 return 0;
476 case CommandLineParseResult::Status::Ok:
477 return std::nullopt;
478 }
479 Q_UNREACHABLE();
480}

498std::optional<quint32> parseSystemId(const QString& value)
499{
500 bool ok = false;
501 const qulonglong systemId = value.toULongLong(&ok);
502 if (!ok || (systemId < 1) || (systemId > std::numeric_limits<quint32>::max())) {
503 return std::nullopt;
504 }
505 return static_cast<quint32>(systemId);
506}

508std::optional<std::pair<uint, uint>> parseUnitTestShard(const QString& value)
509{
510 const QStringList parts = value.split(QLatin1Char('/'));
511 if (parts.size() != 2) {
512 return std::nullopt;
513 }
514 bool indexOk = false;
515 bool countOk = false;
516 const uint index = parts.at(0).toUInt(&indexOk);
517 const uint count = parts.at(1).toUInt(&countOk);
518 if (!indexOk || !countOk || (index == 0) || (index > count)) {
519 return std::nullopt;
520 }
521 return std::make_pair(index, count);
522}
