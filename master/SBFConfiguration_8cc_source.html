36int GPSNativeSBF::configure(unsigned& baudrate, const GPSConfig& config)
37{
38 _configured = false;
39 resetIOError();
40 _survey_duration = 0;
41 _survey_active = false;
42 _survey_activation_date = 0;
43 if (!validateConfiguration(config)) {
44 return -1;
45 }
46 _baseConfig = config.base;
47 char buf[GPS_READ_BUFFER_SIZE];
48 char msg[MSG_SIZE];
49
50 _epochs = {};
51 _lastPublishedEpoch.reset();
52 _rtcm_parsing.reset();
53 decodeInit();
54
55 setBaudrate(SBF_TX_CFG_PRT_BAUDRATE);
56 baudrate = SBF_TX_CFG_PRT_BAUDRATE;
57 _output_mode = config.output_mode;
58
59 // Make sure we can send commands to the receiver
60 sendMessage(SBF_CONFIG_FORCE_INPUT);
61
62 // Disable previous output for now so we can detect the COM port
63 for (int i = 1; i <= 2; i++) {
64 snprintf(msg, sizeof(msg), SBF_CONFIG_DISABLE_OUTPUT, "COM", i);
65 sendMessageAndWaitForAck(msg, SBF_CONFIG_TIMEOUT, {}, false);
66 }
67
68 for (int i = 1; i <= 4; i++) {
69 snprintf(msg, sizeof(msg), SBF_CONFIG_DISABLE_OUTPUT, "USB", i);
70 sendMessageAndWaitForAck(msg, SBF_CONFIG_TIMEOUT, {}, false);
71 }
72
73 char com_port[5]{};
74 size_t offset = 1;
75 bool response_detected = false;
76 uint64_t time_started = nowUs();
77 sendMessage("\n\r");
78
79 // Read buffer to get the COM port
80 do {
81 --offset; // overwrite the null-char
82 int ret = read(reinterpret_cast<uint8_t*>(buf) + offset, sizeof(buf) - offset - 1, SBF_CONFIG_TIMEOUT);
83
84 if (ret < 0) {
85 return ret;
86 }
87
88 offset += ret;
89 buf[offset++] = '\0';
90
91 char* p = strstr(buf, ">");
92
93 if (p) { // check if the length of the com port == 4 and contains a > sign
94 for (int i = 0; i < 4; i++) {
95 com_port[i] = buf[i];
96 }
97
98 response_detected = true;
99 }
100
101 if (offset >= sizeof(buf)) {
102 offset = 1;
103 }
104
105 } while (time_started + 1000 * SBF_CONFIG_TIMEOUT > nowUs() && !response_detected);
106
107 if (response_detected) {
108 log(GPSProtocolLogLevel::Debug, "Septentrio GNSS receiver COM port: %s", com_port);
109 response_detected = false; // for future use
110
111 } else {
112 log(GPSProtocolLogLevel::Warning, "No COM port detected");
113 return -1;
114 }
115
116 // Delete all sbf outputs on current COM port to remove clutter data
117 snprintf(msg, sizeof(msg), SBF_CONFIG_RESET, com_port);
118
119 if (!sendMessageAndWaitForAck(msg, SBF_CONFIG_TIMEOUT)) {
120 return -1; // connection and/or baudrate detection failed
121 }
122
123 // Set baudrate, unless we're connected over USB
124 if (strncmp(com_port, "USB1", 4) != 0 && strncmp(com_port, "USB2", 4) != 0) {
125 snprintf(msg, sizeof(msg), SBF_CONFIG_BAUDRATE, com_port, baudrate);
126
127 if (!sendMessageAndWaitForAck(msg, SBF_CONFIG_TIMEOUT)) {
128 return -1; // connection and/or baudrate detection failed
129 }
130 }
131
132 // At this point we have correct baudrate on both ends
133
134 // Define/inquire the type of data that the receiver should accept/send on a given connection descriptor
135 snprintf(msg, sizeof(msg), SBF_DATA_IO, com_port);
136
137 if (!sendMessageAndWaitForAck(msg, SBF_CONFIG_TIMEOUT)) {
138 return -1;
139 }
140
141 // Set the type of dynamics the GNSS antenna is subjected to.
142 if (_output_mode != OutputMode::RTCM) {
143 // Release a previously configured static base position before starting navigation.
144 if (!sendMessageAndWaitForAck("setPVTMode, Rover, All, auto\n", SBF_CONFIG_TIMEOUT)) {
145 return -1;
146 }
147
148 // Specify the offsets that the receiver applies to the computed attitude angles.
149 snprintf(msg, sizeof(msg), SBF_CONFIG_ATTITUDE_OFFSET, (double) (_heading_offset * 180 / GPS_PI),
150 (double) _pitch_offset);
151
152 if (!sendMessageAndWaitForAck(msg, SBF_CONFIG_TIMEOUT, {GPSReceiverSetting::HeadingOffsetDeg})) {
153 return -1;
154 }
155
156 snprintf(msg, sizeof(msg), SBF_CONFIG_RECEIVER_DYNAMICS, "high");
157
158 if (!sendMessageAndWaitForAck(msg, SBF_CONFIG_TIMEOUT)) {
159 return -1;
160 }
161
162 snprintf(msg, sizeof(msg), SBF_CONFIG, com_port);
163
164 if (!sendMessageAndWaitForAck(msg, SBF_CONFIG_TIMEOUT)) {
165 return -1;
166 }
167 }
168
169 int i = 0;
170
171 do {
172 ++i;
173
174 if (!sendMessageAndWaitForAck(msg, SBF_CONFIG_TIMEOUT)) {
175 if (i >= 5) {
176 return -1; // connection and/or baudrate detection failed
177 }
178
179 } else {
180 response_detected = true;
181 }
182 } while (i < 5 && !response_detected);
183
184 if (_output_mode == OutputMode::RTCM) {
185 if (!_rtcm_parsing) {
186 _rtcm_parsing.emplace();
187 }
188
189 _rtcm_parsing->reset();
190 }
191
192 if (_output_mode != OutputMode::GPS) {
193 if (!sendMessageAndWaitForAck(SBF_CONFIG_OUTPUT_RTCM3, SBF_CONFIG_TIMEOUT)) {
194 return -1;
195 }
196 }
197
198 if (_output_mode == OutputMode::RTCM) {
199 if (_baseConfig.useFixedBase) {
200 snprintf(msg, sizeof(msg), SBF_CONFIG_RTCM_STATIC_COORDINATES, _baseConfig.fixedPosition.latitudeDegrees,
201 _baseConfig.fixedPosition.longitudeDegrees,
202 static_cast<double>(_baseConfig.fixedPosition.altitudeMeters));
203 if (!sendMessageAndWaitForAck(msg, SBF_CONFIG_TIMEOUT)) {
204 return -1;
205 }
206
207 snprintf(msg, sizeof(msg), SBF_CONFIG_RTCM_STATIC_OFFSET, 0.0, 0.0, 0.0);
208 if (!sendMessageAndWaitForAck(msg, SBF_CONFIG_TIMEOUT)) {
209 return -1;
210 }
211
212 if (!sendMessageAndWaitForAck(SBF_CONFIG_RTCM_STATIC1, SBF_CONFIG_TIMEOUT)) {
213 return -1;
214 }
215 if (!sendMessageAndWaitForAck(SBF_CONFIG_RTCM_STATIC2, SBF_CONFIG_TIMEOUT)) {
216 return -1;
217 }
218 } else {
219 if (!sendMessageAndWaitForAck(SBF_CONFIG_RTCM_SURVEY_IN, SBF_CONFIG_TIMEOUT)) {
220 return -1;
221 }
222 }
223
224 if (!sendMessageAndWaitForAck(SBF_CONFIG_RTCM_STATUS, SBF_CONFIG_TIMEOUT)) {
225 return -1;
226 }
227 _survey_activation_date = _baseConfig.useFixedBase ? 0 : nowUs();
228 }
229
230 _configured = true;
231 return ioError();
232}

243bool GPSNativeSBF::sendMessageAndWaitForAck(const char* msg, int timeout, GPSReceiverSettingSet settings, bool required)
244{
245 const GPSConfigurationStep step{msg, std::chrono::milliseconds(timeout), settings, required};
246 if (!writeCommand(step, {reinterpret_cast<const uint8_t*>(msg), strlen(msg)})) {
247 return false;
248 }
249 const std::string expected = "$R: " + std::string(msg);
250 std::string received;
251 GPSCommandOutcome response = GPSCommandOutcome::Pending;
252 const auto result = awaitCommand(
253 step,
254 [&] {
255 uint8_t bytes[GPS_READ_BUFFER_SIZE];
256 const int count = read(bytes, sizeof(bytes), timeout);
257 if (count <= 0) {
258 return;
259 }
260 received.append(reinterpret_cast<const char*>(bytes), count);
261 if (received.find(expected) != std::string::npos) {
262 response = GPSCommandOutcome::Acknowledged;
263 } else if (received.find("$R?") != std::string::npos) {
264 response = GPSCommandOutcome::Rejected;
265 }
266 if (received.size() > expected.size()) {
267 received.erase(0, received.size() - expected.size());
268 }
269 },
270 [&] { return response; });
271 return result.evidence.outcome == GPSCommandOutcome::Acknowledged;
272}
