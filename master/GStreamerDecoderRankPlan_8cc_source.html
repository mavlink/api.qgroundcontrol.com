12// What forcing a decoder class or family gives it. GStreamer already ranks some decoders PRIMARY + 1 (the d3d11 and va
13// ones), so a forced family goes above the highest other decoder (see prioritizeFamily).
14constexpr uint16_t kPrioritizedRank = GST_RANK_PRIMARY + 1;
15// Above kPrioritizedRank, so the GPU-memory decoders the platform prefers win autoplug.
16constexpr uint16_t kZeroCopyRank = GST_RANK_PRIMARY + 2;

49 // @p family's decoders, at least kPrioritizedRank, above every other decoder; those stay as fallbacks for the
50 // codecs the family can't decode.
51 void prioritizeFamily(VideoDecoder::Options family)
52 {
53 unsigned highestOtherRank = GST_RANK_NONE;
54 for (const DecoderInfo& decoder : _decoders) {
55 if (decoder.family != family) {
56 highestOtherRank = std::max(highestOtherRank, decoder.rank);
57 }
58 }
59 const unsigned rank = std::max<unsigned>(kPrioritizedRank, std::min<unsigned>(highestOtherRank, 0xFFFE) + 1);
60 rankFamily(family, static_cast<uint16_t>(rank));
61 }
62
63 // Every decoder of the class at kPrioritizedRank, and every autoplugged one of the other class out of autoplug.
64 void prioritizeClass(bool hardware, QStringList& warnings)
65 {
66 const qsizetype matched =
67 rank([hardware](const DecoderInfo& decoder) { return decoder.hardware == hardware; }, kPrioritizedRank);
68 if (!_decoders.isEmpty() && (matched == 0)) {
69 warnings.append(QStringLiteral("No %1 video decoder factories found to reprioritize.")
70 .arg(hardware ? QStringLiteral("hardware") : QStringLiteral("software")));
71 }
72 (void) rank(
73 [hardware](const DecoderInfo& decoder) {
74 return (decoder.hardware != hardware) && (decoder.rank >= GST_RANK_MARGINAL);
75 },
76 GST_RANK_NONE);
77 }

97RankPlan planRanks(VideoDecoder::Options option, const QList<DecoderInfo>& decoders, const RankTraits& traits)
98{
99 RankPlan plan;
100 Planner planner(decoders);
101
102 switch (option) {
103 case ForceVideoDecoderDefault:
104 if (traits.preferZeroCopyDecoders) {
105 planner.rankPrefix("vaapi", GST_RANK_NONE);
106 planner.rankFamily(ForceVideoDecoderVAAPI, kZeroCopyRank);
107 planner.rankPrefix("v4l2sl", kZeroCopyRank);
108 }
109 break;
110 case ForceVideoDecoderSoftware:
111 planner.prioritizeClass(false, plan.warnings);
112 break;
113 case ForceVideoDecoderHardware:
114 planner.prioritizeClass(true, plan.warnings);
115 break;
116 case ForceVideoDecoderVulkan:
117 // gst-vulkan decodes on its own VkDevice, which fails the GPU path's device-match guard: CPU copy.
118 plan.warnings.append(
119 QStringLiteral("Forcing Vulkan video decoders: zero-copy import is dormant "
120 "(foreign VkDevice → CPU fallback), so decode will not be zero-copy."));
121 [[fallthrough]];
122 case ForceVideoDecoderVAAPI:
123 case ForceVideoDecoderNVIDIA:
124 case ForceVideoDecoderDirectX3D:
125 case ForceVideoDecoderVideoToolbox:
126 case ForceVideoDecoderIntel:
127 planner.prioritizeFamily(option);
128 break;
129 }
130
131 if (traits.rhiD3DApi) {
132 const bool d3d12 = (*traits.rhiD3DApi == D3DApi::D3D12);
133 // Forcing the D3D family already ranks it above every other decoder, so only the other API is demoted then.
134 if ((option == ForceVideoDecoderDefault) || (option == ForceVideoDecoderHardware)) {
135 planner.rankPrefix(d3d12 ? "d3d12" : "d3d11", kZeroCopyRank);
136 }
137 planner.rankPrefix(d3d12 ? "d3d11" : "d3d12", GST_RANK_NONE);
138 }
139
140 plan.changes = planner.changes();
141 return plan;
142}
