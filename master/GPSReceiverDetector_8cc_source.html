83std::vector<unsigned> GPSReceiverDetector::baudCandidates(std::span<const GPSReceiverFamily* const> families)
84{
85 std::vector<unsigned> listed;
86 for (const auto* family : families) {
87 listed.insert(listed.end(), family->baudCandidates.begin(), family->baudCandidates.end());
88 }
89 std::ranges::sort(listed);
90 listed.erase(std::ranges::unique(listed).begin(), listed.end());
91 std::vector<unsigned> rates;
92 for (const unsigned rate : BAUD_PRIORITY) {
93 if (std::ranges::binary_search(listed, rate)) {
94 rates.push_back(rate);
95 }
96 }
97 for (const unsigned rate : listed) {
98 if (std::ranges::find(BAUD_PRIORITY, rate) == BAUD_PRIORITY.end()) {
99 rates.push_back(rate);
100 }
101 }
102 return rates;
103}

105GPSReceiverDetection GPSReceiverDetector::detect(unsigned baud)
106{
107 const std::vector<unsigned> rates = baud ? std::vector<unsigned>{baud} : baudCandidates(_families);
108 const uint64_t untilUs = GPSDeadline::after(_io.clock.nowUs(), TIMEOUT).untilUs;
109 GPSReceiverDetection result;
110 std::vector<unsigned> tried;
111 for (const unsigned rate : rates) {
112 if (_stopped(result)) {
113 return result;
114 }
115 if (_io.clock.nowUs() >= untilUs) {
116 break;
117 }
118 const GPSBaudStatus status = _io.setBaudrate ? _io.setBaudrate(rate) : GPSBaudStatus::Unsupported;
119 if (status == GPSBaudStatus::Unsupported) {
120 qCDebug(GPSReceiverDetectorLog) << "Link cannot run at" << rate << "baud; skipped";
121 continue;
122 }
123 if (status != GPSBaudStatus::Configured) {
124 result.error =
125 status == GPSBaudStatus::Cancelled ? GPSProtocolError::Cancelled : GPSProtocolError::Transport;
126 result.errorDetail = status == GPSBaudStatus::Cancelled
127 ? QString()
128 : QStringLiteral("Cannot set the receiver link to %1 baud").arg(rate);
129 return result;
130 }
131 tried.push_back(rate);
132 _signatures.reset();
133 switch (_listen(rate, untilUs, result)) {
134 case Listen::Failed:
135 return result;
136 case Listen::Identified:
137 return _identified(*_signatures.match()->family, rate, _signatures.match()->evidence);
138 case Listen::Quiet:
139 break;
140 }
141 for (const auto* family : _families) {
142 if (family->baudCandidates.empty() || (!baud && !supportsRate(*family, rate))) {
143 continue;
144 }
145 const uint64_t now = _io.clock.nowUs();
146 if (now >= untilUs) {
147 break;
148 }
149 qCDebug(GPSReceiverDetectorLog)
150 << "Probing" << gpsReceiverName(family->type) << "identity at" << rate << "baud";
151 GPSProtocolRuntime runtime(*family, _io, {.commandFinished = _commandFinished});
152 const bool answered = runtime.probe(GPSDeadline{untilUs}.remaining(now));
153 if (runtime.error() == GPSProtocolError::Cancelled || runtime.error() == GPSProtocolError::Transport) {
154 result.error = runtime.error();
155 result.errorDetail = runtime.errorDetail();
156 return result;
157 }
158 if (answered) {
159 return _identified(*family, rate, QStringLiteral("identity query"));
160 }
161 if (const auto& match = _signatures.match()) {
162 return _identified(*match->family, rate, match->evidence);
163 }
164 }
165 if (_signatures.framedTraffic()) {
166 result.error = GPSProtocolError::Protocol;
167 result.errorDetail = QStringLiteral(
168 "The receiver sends standard NMEA or RTCM at %1 baud, but no supported "
169 "receiver family answered its identification query")
170 .arg(rate);
171 qCDebug(GPSReceiverDetectorLog).noquote() << result.errorDetail;
172 return result;
173 }
174 }
175 if (_stopped(result)) {
176 return result;
177 }
178 result.error = GPSProtocolError::Protocol;
179 if (tried.empty()) {
180 result.errorDetail = QStringLiteral("The receiver link cannot run at any rate a supported receiver uses");
181 } else if (_io.clock.nowUs() >= untilUs) {
182 result.errorDetail = QStringLiteral("Receiver detection timed out after %1 s")
183 .arg(std::chrono::duration_cast<std::chrono::seconds>(TIMEOUT).count());
184 } else {
185 result.errorDetail = QStringLiteral("No supported receiver answered at %1 baud").arg(gpsRateList(tried));
186 }
187 qCDebug(GPSReceiverDetectorLog).noquote() << result.errorDetail;
188 return result;
189}

191GPSReceiverDetector::Listen GPSReceiverDetector::_listen(unsigned baud, uint64_t untilUs, GPSReceiverDetection& result)
192{
193 qCDebug(GPSReceiverDetectorLog) << "Listening at" << baud << "baud";
194 const uint64_t listenUntilUs = std::min(untilUs, GPSDeadline::after(_io.clock.nowUs(), LISTEN_WINDOW).untilUs);
195 std::array<uint8_t, GPSCommandChannel::READ_CHUNK_SIZE> buffer{};
196 Listen outcome = Listen::Quiet;
197 while (outcome == Listen::Quiet && !_signatures.match() && _io.clock.nowUs() < listenUntilUs) {
198 if (_stopped(result)) {
199 outcome = Listen::Failed;
200 break;
201 }
202 const GPSReadResult read = _io.read(buffer, GPSDeadline{listenUntilUs});
203 switch (read.status) {
204 case GPSReadStatus::Data:
205 case GPSReadStatus::TimedOut:
206 break;
207 case GPSReadStatus::Cancelled:
208 result.error = GPSProtocolError::Cancelled;
209 outcome = Listen::Failed;
210 break;
211 default:
212 result.error = GPSProtocolError::Transport;
213 result.errorDetail = read.detail;
214 outcome = Listen::Failed;
215 break;
216 }
217 }
218 if (outcome == Listen::Quiet && _signatures.match()) {
219 outcome = Listen::Identified;
220 }
221 return outcome;
222}
