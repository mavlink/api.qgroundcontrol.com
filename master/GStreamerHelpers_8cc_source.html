55GstElement* addToBin(GstElement* bin, GstElementPtr element, const char* what)
56{
57 if (!element || g_object_is_floating(element.get())) {
58 qCCritical(GStreamerHelpersLog) << "Missing or floating element for" << what;
59 return nullptr;
60 }
61 GstElement* const added = element.release();
62 const bool ok = gst_bin_add(GST_BIN(bin), added);
63 // The bin took its own reference; drop ours.
64 gst_object_unref(added);
