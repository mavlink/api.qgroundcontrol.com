23bool isValidRtspUri(const gchar* uri_str)
24{
25 if (!uri_str) {
26 return false;
27 }
28
29 if (!gst_uri_is_valid(uri_str)) {
30 return false;
31 }
32
33 GstRTSPUrl* url = nullptr;
34 const GstRTSPResult res = gst_rtsp_url_parse(uri_str, &url);
35 if ((res != GST_RTSP_OK) || !url) {
36 if (url) {
37 gst_rtsp_url_free(url);
38 }
39 return false;
40 }
41
42 const bool hasHost = (url->host && url->host[0] != '\0');
43 gst_rtsp_url_free(url);
44 return hasHost;
45}

49void removeAndStop(GstElement* element, bool flush)
50{
51 const GstObjectPtr parent = element ? adoptObject(gst_object_get_parent(GST_OBJECT(element))) : nullptr;
52 if (!parent) {
53 return;
54 }
55
56 // gst_bin_remove drops the bin's ref, which may be the last one, before the element reaches NULL.
57 const GstElementPtr keepAlive = adoptElement(GST_ELEMENT(gst_object_ref(element)));
58 (void) gst_bin_remove(GST_BIN(parent.get()), element);
59 if (flush) {
60 (void) gst_element_foreach_sink_pad(
61 element,
62 [](GstElement* /*element*/, GstPad* pad, gpointer /*userData*/) -> gboolean {
63 (void) gst_pad_send_event(pad, gst_event_new_flush_start());
64 return TRUE;
65 },
66 nullptr);
67 }
68 // Synchronous: gst_element_change_state() commits an ASYNC return at once when the target is READY or NULL, so
69 // the streaming threads have stopped when this returns.
70 (void) gst_element_set_state(element, GST_STATE_NULL);
71}

95TeeBranch addTeeBranch(GstElement* pipeline, GstElement* tee, const char* what)
96{
97 GstElementPtr queue = adoptElement(gst_element_factory_make("queue", nullptr));
98 GstElementPtr valve = adoptElement(gst_element_factory_make("valve", nullptr));
99 if (!queue || !valve) {
100 qCWarning(GStreamerHelpersLog) << "Failed to create the queue and valve of the" << what << "branch";
101 return {};
102 }
103 g_object_set(valve.get(), "drop", TRUE, nullptr);
104
105 const TeeBranch branch{.queue = addToBin(pipeline, std::move(queue), "queue"),
106 .valve = addToBin(pipeline, std::move(valve), "valve")};
107 if (!branch.queue || !branch.valve || !gst_element_link_many(tee, branch.queue, branch.valve, nullptr)) {
108 qCWarning(GStreamerHelpersLog) << "Unable to link the" << what << "branch";
109 return {};
110 }
111 return branch;
112}

161// Single name-based decoder classification for the hardware/software split, the ForceVideoDecoder*
162// rank overrides and availableDecoderFamilies(). Prefix (not substring) matching, so e.g. avdec_wnv1
163// isn't taken for an NVIDIA decoder. The first match wins: specific prefixes precede general ones.
164constexpr DecoderFamilyPrefix kDecoderFamilyPrefixes[] = {
165 // Android MediaCodec: Google/Codec2 software wrappers; every other amcviddec is a hardware codec.
166 {"amcviddec-omxgoogle", ForceVideoDecoderDefault, false},
167 {"amcviddec-c2android", ForceVideoDecoderDefault, false},
168 {"amcviddec-", ForceVideoDecoderDefault, true},
169 // Legacy gstreamer-vaapi is demoted on Linux and never bumped by ForceVideoDecoderVAAPI (modern va* only).
170 {"vaapi", ForceVideoDecoderDefault, true},
171 {"va", ForceVideoDecoderVAAPI, true},
172 {"nv", ForceVideoDecoderNVIDIA, true},
173 {"d3d11", ForceVideoDecoderDirectX3D, true},
174 {"d3d12", ForceVideoDecoderDirectX3D, true},
175 {"dxva", ForceVideoDecoderDirectX3D, true},
176 {"vtdec", ForceVideoDecoderVideoToolbox, true},
177 {"qsv", ForceVideoDecoderIntel, true},
178 {"msdk", ForceVideoDecoderIntel, true},
179 {"vulkan", ForceVideoDecoderVulkan, true},
180};

184 const GCharPtr lowerName = adoptGChar(g_ascii_strdown(factoryName, -1));

205bool isHardwareDecoderFactory(GstElementFactory* factory)
206{
207 if (!factory) {
208 return false;
209 }
210
211 const gchar* factoryName = gst_plugin_feature_get_name(GST_PLUGIN_FEATURE(factory));
212 if (!factoryName) {
213 return false;
214 }
215
216 if (const DecoderFamilyPrefix* entry = findDecoderFamilyPrefix(factoryName)) {
217 return entry->isHardware;
218 }
219
220 const auto containsHardware = [](const gchar* value) {
221 if (!value) {
222 return false;
223 }
224 const GCharPtr lower = adoptGChar(g_ascii_strdown(value, -1));
225 return g_strrstr(lower.get(), "hardware") != nullptr;
226 };
227
228 return containsHardware(gst_element_factory_get_metadata(factory, GST_ELEMENT_METADATA_KLASS)) ||
229 containsHardware(gst_element_factory_get_metadata(factory, GST_ELEMENT_METADATA_DESCRIPTION));
230}

296void prioritizeByHardwareClass(uint16_t prioritizedRank, bool requireHardware)
297{
298 qCDebug(GStreamerHelpersLog) << "Prioritizing" << (requireHardware ? "hardware" : "software")
299 << "video decoders with rank:" << prioritizedRank;
300 int decoderFactories = 0;
301 int matchedFactories = 0;
302 forEachVideoDecoderFactory(GST_RANK_NONE, [&](GstElementFactory* factory) {
303 ++decoderFactories;
304 if (isHardwareDecoderFactory(factory) == requireHardware) {
305 setFeatureRank(GST_PLUGIN_FEATURE(factory), prioritizedRank);
306 ++matchedFactories;
307 }
308 });
309
310 if (decoderFactories == 0) {
311 qCDebug(GStreamerHelpersLog) << "No decoder factories available while prioritizing"
312 << (requireHardware ? "hardware" : "software") << "decoders";
313 return;
314 }
315
316 if (matchedFactories == 0) {
317 qCWarning(GStreamerHelpersLog) << "No" << (requireHardware ? "hardware" : "software")
318 << "video decoder factories found to reprioritize.";
319 }
320
321 qCDebug(GStreamerHelpersLog) << "Lowering" << (requireHardware ? "software" : "hardware") << "decoder ranks.";
322 lowerDecoderRanksByClass(!requireHardware);
323}

342// A decoder whose D3D family mismatches the active QRhi backend can't be sampled zero-copy and drops to a CPU copy.
343void alignD3DDecoderRanksToRhi(bool promoteMatchedFamily)
344{
345 // Prefer the resolved QRhi backend; ranks usually apply before the scene graph resolves, so cachedRhi() is often
346 // null here and graphicsApi() Unknown — fall back to QSG_RHI_BACKEND (Qt's D3D11 default) in that Windows case.
347 // QGCRhiCapture/qrhi.h link only when a GPU path is compiled (GuiPrivate), so the resolved-backend lookup is gated;
348 // a no-GPU Windows build relies on graphicsApi()/QSG_RHI_BACKEND alone.
349 bool wantD3D12 = false;
350 bool resolved = false;
351#if defined(QGC_HAS_ANY_GPU_PATH)
352 if (QRhi* rhi = QGCRhiCapture::cachedRhi()) {
353 wantD3D12 = (rhi->backend() == QRhi::D3D12);
354 resolved = true;
355 }
356#endif
357 if (!resolved) {
358 switch (QQuickWindow::graphicsApi()) {
359 case QSGRendererInterface::Direct3D12:
360 wantD3D12 = true;
361 break;
362 case QSGRendererInterface::Direct3D11:
363 wantD3D12 = false;
364 break;
365 default:
366 wantD3D12 =
367 qEnvironmentVariable("QSG_RHI_BACKEND").compare(QLatin1String("d3d12"), Qt::CaseInsensitive) == 0;
368 break;
369 }
370 }
371 const char* const matched = wantD3D12 ? "d3d12" : "d3d11";
372 const char* const mismatched = wantD3D12 ? "d3d11" : "d3d12";
373 qCDebug(GStreamerHelpersLog) << "Aligning D3D decoder ranks to" << (wantD3D12 ? "D3D12" : "D3D11")
374 << "RHI - demoting the mismatched decoder family";
375 if (promoteMatchedFamily) {
376 applyPrefixRank(matched, kZeroCopyRank);
377 }
378 applyPrefixRank(mismatched, GST_RANK_NONE);
379}
380#endif // Q_OS_WIN
381
382#ifdef Q_OS_LINUX
383// Steer autoplug toward zero-copy GPU-memory decoders (DMABuf/DMA_DRM) over system-memory ones. Rank-only and
384// additive: absent factories no-op, va/v4l2 self-demote at runtime without a device, software fallback untouched.
385void preferZeroCopyDecoders()
386{
387 // Legacy gstreamer-vaapi negotiates GstVaapiMemory the qgcqvideosink/HwBuffers paths can't import
388 // zero-copy; demote it so the modern va plugin (clean DMABuf/DMA_DRM output) wins when both exist.
389 applyPrefixRank("vaapi", GST_RANK_NONE);
390
391 // Modern va decoders (gst-va, supersedes gstreamer-vaapi) emit DMABuf/DMA_DRM, and V4L2 stateless
392 // decoders expose DMA_DRM caps on 1.26+; bump both above software so the zero-copy allocator is picked.
393 applyFamilyRank(ForceVideoDecoderVAAPI, kZeroCopyRank);
394 applyPrefixRank("v4l2sl", kZeroCopyRank);
395}
396#endif // Q_OS_LINUX

410void setCodecPriorities(VideoDecoderOptions option)
411{
412 if (!gst_registry_get()) {
413 qCCritical(GStreamerHelpersLog) << "Failed to get gstreamer registry.";
414 return;
415 }
416
417 static constexpr uint16_t PrioritizedRank = GST_RANK_PRIMARY + 1;
418
419 switch (option) {
420 case ForceVideoDecoderDefault:
421 // HW-decoder GPU caps (GLMemory/DMABuf/VAMemory) auto-plug to system memory via gldownload/vapostproc.
422#ifdef Q_OS_LINUX
423 preferZeroCopyDecoders();
424#endif
425 break;
426 case ForceVideoDecoderSoftware:
427 prioritizeByHardwareClass(PrioritizedRank, false);
428 break;
429 case ForceVideoDecoderHardware:
430 prioritizeByHardwareClass(PrioritizedRank, true);
431 break;
432 case ForceVideoDecoderVulkan:
433 // Vulkan zero-copy import is dormant: gst-vulkan's own VkDevice fails QGC's device-match guard → CPU copy.
434 qCWarning(GStreamerHelpersLog) << "Forcing Vulkan video decoders: zero-copy import is dormant "
435 "(foreign VkDevice → CPU fallback), so decode will not be zero-copy.";
436 [[fallthrough]];
437 case ForceVideoDecoderVAAPI:
438 case ForceVideoDecoderNVIDIA:
439 case ForceVideoDecoderDirectX3D:
440 case ForceVideoDecoderVideoToolbox:
441 case ForceVideoDecoderIntel:
442 applyFamilyRank(option, PrioritizedRank);
443 break;
444 }
445
446#ifdef Q_OS_WIN
447 // After any option-driven rank changes, force the D3D decoder API to match the active QRhi
448 // backend so a D3D12 decoder never wins over D3D11 on Qt's default Windows RHI (and vice-versa).
449 const bool promoteMatchingD3D = (option == ForceVideoDecoderDefault) || (option == ForceVideoDecoderHardware) ||
450 (option == ForceVideoDecoderDirectX3D);
451 alignD3DDecoderRanksToRhi(promoteMatchingD3D);
452#endif
453}
