20void removeAndStop(GstElement* element, bool flush)
21{
22 const GstObjectPtr parent = element ? adoptObject(gst_object_get_parent(GST_OBJECT(element))) : nullptr;
23 if (!parent) {
24 return;
25 }
26
27 // gst_bin_remove drops the bin's ref, which may be the last one, before the element reaches NULL.
28 const GstElementPtr keepAlive = adoptElement(GST_ELEMENT(gst_object_ref(element)));
29 // Under the bin's state lock, which a state change of the bin holds while it sets its children's states: one that
30 // took the element from the bin's children before the removal (as the continuation GStreamer's thread pool runs
31 // once an async sink is gone) would otherwise set it to the bin's state after it reached NULL here.
32 GST_STATE_LOCK(parent.get());
33 (void) gst_bin_remove(GST_BIN(parent.get()), element);
34 if (flush) {
35 (void) gst_element_foreach_sink_pad(
36 element,
37 [](GstElement* /*element*/, GstPad* pad, gpointer /*userData*/) -> gboolean {
38 (void) gst_pad_send_event(pad, gst_event_new_flush_start());
39 return TRUE;
40 },
41 nullptr);
42 }
43 // Synchronous: gst_element_change_state() commits an ASYNC return at once when the target is READY or NULL, so
44 // the streaming threads have stopped when this returns.
45 (void) gst_element_set_state(element, GST_STATE_NULL);
46 GST_STATE_UNLOCK(parent.get());
47}

77GstElement* addToBin(GstElement* bin, GstElementPtr element, const char* what)
78{
79 if (!element || g_object_is_floating(element.get())) {
80 qCCritical(GStreamerHelpersLog) << "Missing or floating element for" << what;
81 return nullptr;
82 }
83 GstElement* const added = element.release();
84 const bool ok = gst_bin_add(GST_BIN(bin), added);
85 // The bin took its own reference; drop ours.
86 gst_object_unref(added);
87 if (!ok) {
88 qCWarning(GStreamerHelpersLog) << "gst_bin_add() failed for" << what;
89 return nullptr;
90 }
91 return added;
92}

112{
113 struct PendingEvent
114 {
115 GstPadPtr pad;
116 GstEventPtr event;
117 };
118
119 GstElementPtr parent = pad ? adoptElement(gst_pad_get_parent_element(pad)) : nullptr;
120 if (!parent) {
121 gst_event_unref(event);
122 return;
123 }
124 auto* const pending = new PendingEvent{.pad = adoptPad(GST_PAD(gst_object_ref(pad))), .event = adoptEvent(event)};
125#if GST_CHECK_VERSION(1, 28, 0)
126 // Replaces gst_element_call_async(), deprecated in 1.28. It takes no destroy notify: the call frees its data.
127 gst_object_call_async(
128 GST_OBJECT_CAST(parent.get()),
129 [](GstObject* /*object*/, gpointer data) {
130 const std::unique_ptr<PendingEvent> owned(static_cast<PendingEvent*>(data));
131 (void) gst_pad_send_event(owned->pad.get(), owned->event.release());
132 },
133 pending);
134#else
135 gst_element_call_async(
136 parent.get(),
137 [](GstElement* /*element*/, gpointer data) {
138 auto* const queued = static_cast<PendingEvent*>(data);
139 (void) gst_pad_send_event(queued->pad.get(), queued->event.release());
140 },
141 pending, [](gpointer data) { delete static_cast<PendingEvent*>(data); });
142#endif

166std::function<void()> holdPadUntilLinked(GstPad* pad, std::function<void(GstPad* pad)> link)
167{
168 // Returning OK from a blocking probe keeps the pad blocked until the probe is removed.
169 const gulong blockId = gst_pad_add_probe(
170 pad, GST_PAD_PROBE_TYPE_BLOCK_DOWNSTREAM,
171 [](GstPad* /*pad*/, GstPadProbeInfo* /*info*/, gpointer /*data*/) { return GST_PAD_PROBE_OK; }, nullptr,
172 nullptr);
173 return [blockId, heldPad = shareObject(pad), link = std::move(link)]() {
174 link(heldPad.get());
175 gst_pad_remove_probe(heldPad.get(), blockId);
176 };
177}

179void forEachPlugin(GstRegistry* registry, const std::function<void(GstPlugin*)>& visitor)
180{
181 if (!registry || !visitor)
182 return;
183 GList* plugins = gst_registry_get_plugin_list(registry);
184 for (GList* node = plugins; node != nullptr; node = node->next) {
185 GstPlugin* plugin = static_cast<GstPlugin*>(node->data);
186 if (plugin)
187 visitor(plugin);
188 }
189 gst_plugin_list_free(plugins);
190}
