57// Orders Qt's context after gst-gl's fence. gst_gl_sync_meta_wait() would wait inside gst-gl's own context and leave
58// Qt's unordered; GL sync objects are shared across the share group, so Qt's context waits on the GLsync server-side.
59// CPU wait when Qt's context has no sync objects or the meta carries no GLsync (a custom meta, or gst-gl without
60// fences).
61void waitInQtContext(GstGLSyncMeta* syncMeta, GstGLContext* glCtx)
62{
63 QOpenGLContext* qtCtx = QOpenGLContext::currentContext();
64 const auto sync = static_cast<GLsync>(syncMeta->data);
65 if (qtCtx && sync) {
66 const QSurfaceFormat format = qtCtx->format();
67 const bool hasSyncObjects = qtCtx->isOpenGLES() ? format.majorVersion() >= 3
68 : (format.version() >= qMakePair(3, 2) ||
69 qtCtx->hasExtension(QByteArrayLiteral("GL_ARB_sync")));
70 QOpenGLExtraFunctions* gl = hasSyncObjects ? qtCtx->extraFunctions() : nullptr;
71 if (gl && gl->glIsSync(sync)) {
72 gl->glWaitSync(sync, 0, GL_TIMEOUT_IGNORED);
73 GstHwPathTelemetry::record(HwVideoBufferPath::GlMemory, GstHwPathTelemetry::Counter::GpuWaits);
74 return;
75 }
76 }
77 gst_gl_sync_meta_wait_cpu(syncMeta, glCtx);
78 GstHwPathTelemetry::record(HwVideoBufferPath::GlMemory, GstHwPathTelemetry::Counter::CpuWaits);
79}

81// mapTextures() binds GLMemory only under Qt's OpenGL(ES2) QRhi, but the glupload bin still emits GLMemory under a
82// non-GL RHI (forced Vulkan): those frames take the CPU path. Streaming thread, so read the published backend rather
83// than the render-thread-owned QRhi. Records each rejected frame.
84bool admitGlMemory()
85{
86 const int backend = QGCRhiCapture::deviceSnapshot().backend.load(std::memory_order_acquire);
87 if ((backend < 0) || (backend == static_cast<int>(QRhi::OpenGLES2))) {
88 return true;
89 }
90 static std::atomic<bool> s_warned{false};
91 if (!s_warned.exchange(true, std::memory_order_relaxed)) {
92 qCWarning(GstGlBufLog) << "GLMemory frame but QRhi backend is not OpenGL — routing to CPU pool";
93 }
94 GstHwPathTelemetry::recordFallbackReason(HwVideoBufferPath::GlMemory,
95 GstHwPathTelemetry::HwFallbackReason::ImportUnsupported);
96 return false;
97}

112{
113 const GstHwPathTelemetry::ScopedMapTimer mapTimer(HwVideoBufferPath::GlMemory);
114 // QRhi::OpenGLES2 covers both desktop GL and GLES — Qt collapses both; there is no separate QRhi::OpenGL.
115 GstBuffer* buffer = nullptr;
116 if (!checkMapPreconditions(rhi, static_cast<int>(QRhi::OpenGLES2), GstGlBufLog(), s_diag, buffer)) {
117 return GstHwPathTelemetry::fail(HwVideoBufferPath::GlMemory);
118 }
119
120 // Defensive: custom RHI integrations may not have Qt's GL context current on QSGRenderThread entry.
121 rhi.makeThreadLocalNativeContextCurrent();
122
123 GstMemory* mem0 = gst_buffer_peek_memory(buffer, 0);
124 if (!mem0 || !gst_is_gl_memory(mem0)) {
125 return GstHwPathTelemetry::fail(HwVideoBufferPath::GlMemory);
126 }
127
128 CpuVideoFramePool::FrameMapping mapping;
129 // GST_MAP_GL (a gst-gl extension flag) OR'd with GST_MAP_READ triggers GstGLMemory upload.
130 if (!mapping.mapFrame(buffer, _videoInfo, static_cast<GstMapFlags>(GST_MAP_READ | GST_MAP_GL))) {
131 qCWarning(GstGlBufLog) << "gst_video_frame_map(GST_MAP_READ | GST_MAP_GL) failed";
132 GstHwPathTelemetry::recordFallbackReason(HwVideoBufferPath::GlMemory,
133 GstHwPathTelemetry::HwFallbackReason::MapFailed);
134 return GstHwPathTelemetry::fail(HwVideoBufferPath::GlMemory);
135 }

137 // Always set a fresh sync point after the GST_MAP_GL map above: glupload attaches its meta without one and uploads
138 // lazily on that map, so an upstream sync point would predate the upload. Synthesize a meta when upstream omits
139 // one.
140 GstGLContext* glCtx = GST_GL_BASE_MEMORY_CAST(mem0)->context;
141 if (glCtx) {
142 GstGLSyncMeta* syncMeta = gst_buffer_get_gl_sync_meta(buffer);
143 GstBuffer* throwaway = nullptr;
144 if (!syncMeta) {
145 throwaway = gst_buffer_new();
146 if (!throwaway) {
147 qCWarning(GstGlBufLog) << "gst_buffer_new() failed while creating GL sync meta holder";
148 GstHwPathTelemetry::recordFallbackReason(HwVideoBufferPath::GlMemory,
149 GstHwPathTelemetry::HwFallbackReason::MapFailed);
150 return GstHwPathTelemetry::fail(HwVideoBufferPath::GlMemory);
151 }
152 syncMeta = gst_buffer_add_gl_sync_meta(glCtx, throwaway);
153 }
154 if (syncMeta) {
155 gst_gl_sync_meta_set_sync_point(syncMeta, glCtx);
156 waitInQtContext(syncMeta, glCtx);
157 }
158 if (throwaway) {
159 gst_buffer_unref(throwaway);
160 }
161 }

163 const GstVideoFrame& frame = mapping.frame();
164 const int planeCount = std::clamp(int(GST_VIDEO_FRAME_N_PLANES(&frame)), 1, kMaxPlanes);
165 std::array<GLuint, kMaxPlanes> names{};
166 for (int i = 0; i < planeCount; ++i) {
167 // GST_MAP_GL maps return a *pointer to* the GLuint texture id, not the id itself.
168 const GLuint* texIdPtr = static_cast<const GLuint*>(GST_VIDEO_FRAME_PLANE_DATA(&frame, i));
169 names[i] = texIdPtr ? *texIdPtr : 0;
170 }
171

173 const std::span<const GLuint> planeNames(names.data(), std::size_t(planeCount));

177 .pixelFormat = _format.pixelFormat(),
178 .log = &GstGlBufLog(),
179 .diag = &s_diag};

181 return GstHwFrameTexturesBase::finishMap<FrameTextures>(
182 *this, rhi, old, tail,
183 [&](const FrameTextures& prev) {
184 return prev.matchesImport(&rhi, tail.frameSize, tail.pixelFormat, planeNames);
