54QImage VideoScreenshot::displayedImage(const QVideoFrame& frame, QtVideo::Rotation outputRotation, bool outputMirrored)
55{
56 if (!frame.isValid()) {
57 return {};
58 }
59
60 // toImage() applies only the surface format's own transformation: not the frame's rotation or mirroring, nor the
61 // viewport that crops decoder padding.
62 QImage image = frame.toImage();
63 if (image.isNull()) {
64 return image;
65 }
66
67 const QRect viewport = frame.surfaceFormat().viewport();
68 if (viewport.isValid() && (image.size() == frame.size()) && (viewport != image.rect())) {
69 image = image.copy(viewport.intersected(image.rect()));
70 }
71
72 // Same order as Qt's presentation (qNormalizedFrameTransformation): the frame's rotation and mirroring, then the
73 // output's.
74 const auto rotateAndMirror = [&image](QtVideo::Rotation rotation, bool mirrored) {
75 if (const int degrees = static_cast<int>(rotation); degrees != 0) {
76 image = image.transformed(QTransform().rotate(degrees));
77 }
78 if (mirrored) {
79 image = image.flipped(Qt::Horizontal);
80 }
81 };
82 rotateAndMirror(frame.rotation(), frame.mirrored());
83 rotateAndMirror(outputRotation, outputMirrored);
84 return image;
85}

103 }
104

106 QBuffer buffer(&jpeg);
107 if (!buffer.open(QIODevice::WriteOnly) || !image.save(&buffer, "JPG")) {
108 qCWarning(VideoScreenshotLog) << "Failed to encode screenshot" << file;
109 return false;
110 }
111 buffer.close();
112 // The picture matters more than its tags.
113 if (!addExif(jpeg, exif)) {
114 qCWarning(VideoScreenshotLog) << "Failed to add EXIF data to screenshot" << file;
115 }
116 QSaveFile out(file);
117 if (!out.open(QIODevice::WriteOnly) || (out.write(jpeg) != jpeg.size()) || !out.commit()) {
118 qCWarning(VideoScreenshotLog) << "Failed to save screenshot" << file << out.errorString();
119 return false;
120 }
121
122 qCDebug(VideoScreenshotLog) << "Screenshot saved:" << file;
