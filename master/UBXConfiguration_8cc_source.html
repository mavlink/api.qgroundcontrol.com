42uint32_t fixedAccuracyWireUnits(float accuracyMeters)
43{
44 // Match shared validation: multiplying directly by 10000 can round an accepted value past UINT32_MAX.
45 const float accuracyMillimeters = accuracyMeters * 1000.0f;
46 return static_cast<uint32_t>(accuracyMillimeters * 10.0f);

50GPSNativeUBX::BaseStationCapability GPSNativeUBX::baseStationCapability() const
51{
52 if (_board == Board::u_blox8) {
53 return _is_m8p ? BaseStationCapability::Supported
54 : (_model_name[0] ? BaseStationCapability::Unsupported : BaseStationCapability::Unknown);
55 }
56 const auto profile = UBX::receiverProfile(_board);
57 return profile.rtcmOutput ? BaseStationCapability::Supported
58 : profile.baseCapabilityKnown ? BaseStationCapability::Unsupported
59 : BaseStationCapability::Unknown;

62int GPSNativeUBX::configure(unsigned& baudrate, const GPSConfig& config)
63{
64 _baseConfig = config.base;
65 _dyn_model = config.dynamicModel;
66 _survey_duration = 0;
67 resetIOError();
68 _timeModeUnsupported = false;
69 _valsetAckAmbiguous = false;
70 _configured = false;
71 _decodeNavigation = false;
72 _assembleEpochs = false;
73 _navigationEpochs = {};
74 _controller = {};
75 _pendingDisableMessage = 0;
76 _comms_request_pending = false;
77 _rtcmActivationPending = false;
78 _rtcm_parsing.reset();
79 if (!validateConfiguration(config)) {
80 return -1;
81 }
82 _output_mode = config.output_mode;
83
84 const bool auto_baudrate = baudrate == 0;
85 const auto identify = [this] {
86 // Slow factory NMEA output can delay a MON-VER response by over a second.
87 const Operation operation(*this, 2000);
88 _board = Board::unknown;
89 return sendMessage(UBX_MSG_MON_VER, nullptr, 0) && waitForAck(UBX_MSG_MON_VER, 2000, true) == 0;
90 };
91 constexpr unsigned BAUD_RATES[] = {38400, 57600, 9600, 115200, 230400, 460800, 921600};
92 unsigned detectedBaud = 0;
93 for (const unsigned candidate : BAUD_RATES) {
94 const unsigned selected = auto_baudrate ? candidate : baudrate;
95 if (setBaudrate(selected) < 0) {
96 return -1;
97 }
98 decodeInit();
99 bool readError = false;
100 receiveInternal(20, readError);
101 decodeInit();
102 if (readError) {
103 return -1;
104 }
105 if (identify()) {
106 detectedBaud = selected;
107 break;
108 }
109 if (ioError() || !auto_baudrate) {
110 return -1;
111 }
112 }
113 // Discovery only polls identity: silence or an unsupported identity must not change receiver settings.
114 if (!detectedBaud || _board == Board::unknown) {
115 return -1;
116 }
117 if (_output_mode == OutputMode::RTCM && baseStationCapability() == BaseStationCapability::Unsupported) {
118 return -1;
119 }
120
121 const unsigned desiredBaud = !auto_baudrate ? baudrate
122 : _proto_ver_27_or_higher || _board == Board::u_blox8 ? UBX_BAUDRATE_M8_AND_NEWER
123 : UBX_TX_CFG_PRT_BAUDRATE;
124 ubx_payload_tx_cfg_prt_t ports[2]{};
125 if (_proto_ver_27_or_higher) {
126 static constexpr CfgValsetItem UART1_UBX[] = {
127 {UBX_CFG_KEY_CFG_UART1_STOPBITS, 1}, {UBX_CFG_KEY_CFG_UART1_DATABITS, 0},
128 {UBX_CFG_KEY_CFG_UART1_PARITY, 0}, {UBX_CFG_KEY_CFG_UART1INPROT_UBX, 1},
129 {UBX_CFG_KEY_CFG_UART1INPROT_NMEA, 0}, {UBX_CFG_KEY_CFG_UART1OUTPROT_UBX, 1},
130 {UBX_CFG_KEY_CFG_UART1OUTPROT_NMEA, 0},
131 };
132 initCfgValset();
133 cfgValset(UART1_UBX);
134 if (!sendCfgValset() || waitForAck(UBX_MSG_CFG_VALSET, 2000, true) < 0) {
135 return -1;
136 }
137 } else {
138 for (auto& port : ports) {
139 port.mode = UBX_TX_CFG_PRT_MODE;
140 port.baudRate = detectedBaud;
141 port.inProtoMask = _output_mode == OutputMode::GPS ? UBX_TX_CFG_PRT_PROTO_UBX | UBX_TX_CFG_PRT_PROTO_RTCM
142 : UBX_TX_CFG_PRT_PROTO_UBX;
143 port.outProtoMask = _output_mode == OutputMode::GPS ? UBX_TX_CFG_PRT_PROTO_UBX
144 : UBX_TX_CFG_PRT_PROTO_UBX | UBX_TX_CFG_PRT_PROTO_RTCM;
145 }
146 ports[0].portID = UBX_TX_CFG_PRT_PORTID;
147 ports[1].portID = UBX_TX_CFG_PRT_PORTID_USB;
148 if (!sendMessage(UBX_MSG_CFG_PRT, UBX::encode(ports)) ||
149 waitForAck(UBX_MSG_CFG_PRT, UBX_CONFIG_TIMEOUT, true) < 0) {
150 return -1;
151 }
152 }
153 if (desiredBaud != detectedBaud) {
154 const bool modern = _proto_ver_27_or_higher;
155 const Board identifiedBoard = _board;
156 if (modern) {
157 initCfgValset();
158 cfgValset<uint32_t>(UBX_CFG_KEY_CFG_UART1_BAUDRATE, desiredBaud);
159 if (!sendCfgValset()) {
160 return -1;
161 }
162 } else {
163 for (auto& port : ports) {
164 port.baudRate = desiredBaud;
165 }
166 if (!sendMessage(UBX_MSG_CFG_PRT, UBX::encode(ports))) {
167 return -1;
168 }
169 }
170 const uint16_t command = modern ? UBX_MSG_CFG_VALSET : UBX_MSG_CFG_PRT;
171 const bool acknowledged = waitForAck(command, UBX_CONFIG_TIMEOUT, false) == 0;
172 if (ioError() || _last_ack_rejected || setBaudrate(desiredBaud) < 0) {
173 return -1;
174 }
175 if (modern && !acknowledged) {
176 // Do not clear ACK ambiguity after UART handoff: every later VALSET needs matching readback.
177 _controller.requireConfigurationReadback(command);
178 }
179 decodeInit();
180 if (!identify() || _board != identifiedBoard || _proto_ver_27_or_higher != modern ||
181 _controller.lateRejection()) {
182 return -1;
183 }
184 if (modern && !acknowledged && waitForAck(command, UBX_CONFIG_TIMEOUT, true) < 0) {
185 return -1;
186 }
187 }
188 baudrate = desiredBaud;
189
190 if (_output_mode == OutputMode::RTCM) {
191 if (!_rtcm_parsing) {
192 _rtcm_parsing.emplace();
193 }
194
195 _rtcm_parsing->reset();
196 }
197
198 if (_output_mode == OutputMode::RTCM) {
199 // RTCM mode force stationary dynamic model
200 _dyn_model = 2;
201 }
202
203 int ret;
204
205 /* Configure the device, use config commands depending on protocol version */
206 if (_proto_ver_27_or_higher) {
207 ret = configureDevice(config);
208
209 } else {
210 ret = configureDevicePreV27(config.gnss_systems);
211 }
212
213 if (ret != 0) {
214 return ret;
215 }
216
217 // A position source must leave any previous base-station time mode.
218 if (_output_mode != OutputMode::RTCM) {
219 if (disableTimeMode() < 0) {
220 return -1;
221 }
222 if (!_timeModeUnsupported) {
223 if (_proto_ver_27_or_higher) {
224 initCfgValset();
225 cfgValsetPort(UBX_CFG_KEY_MSGOUT_UBX_NAV_SVIN_I2C, 0);
226 if (sendCfgValsetAcked() < 0) {
227 return -1;
228 }
229 } else if (!configureMessageRateAndAck(UBX_MSG_NAV_SVIN, 0, true)) {
230 return -1;
231 }
232 }
233 }
234
235 if (_output_mode == OutputMode::RTCM) {
236 if (restartSurveyIn() < 0) {
237 return -1;
238 }
239 }
240
241 _configured = true;
242 _decodeNavigation = true;
243 _assembleEpochs = true;
244 return 0;
245}

247int GPSNativeUBX::configureDevicePreV27(const GNSSSystemsMask& gnssSystems)
248{
249 ubx_payload_tx_cfg_gnss_t payload_tx_cfg_gnss{};
250 ubx_payload_tx_cfg_nav5_t payload_tx_cfg_nav5{};
251 ubx_payload_tx_cfg_rate_t payload_tx_cfg_rate{};
252
253 /* Send a CFG-RATE message to define update rate */
254 payload_tx_cfg_rate = {};
255 payload_tx_cfg_rate.measRate = UBX_TX_CFG_RATE_MEASINTERVAL;
256 payload_tx_cfg_rate.navRate = UBX_TX_CFG_RATE_NAVRATE;
257 payload_tx_cfg_rate.timeRef = UBX_TX_CFG_RATE_TIMEREF;
258
259 if (!sendMessage(UBX_MSG_CFG_RATE, UBX::encode(payload_tx_cfg_rate))) {
260 return -1;
261 }
262
263 if (waitForAck(UBX_MSG_CFG_RATE, UBX_CONFIG_TIMEOUT, true) < 0) {
264 return -1;
265 }
266
267 /* send a NAV5 message to set the options for the internal filter */
268 payload_tx_cfg_nav5 = {};
269 payload_tx_cfg_nav5.mask = UBX_TX_CFG_NAV5_MASK;
270 payload_tx_cfg_nav5.dynModel = _dyn_model;
271 payload_tx_cfg_nav5.fixMode = UBX_TX_CFG_NAV5_FIXMODE;
272
273 if (!sendMessage(UBX_MSG_CFG_NAV5, UBX::encode(payload_tx_cfg_nav5))) {
274 return -1;
275 }
276
277 if (waitForAck(UBX_MSG_CFG_NAV5, UBX_CONFIG_TIMEOUT, true) < 0) {
278 return -1;
279 }
280
281 /* configure active GNSS systems (number of channels and used signals taken from U-Center default) */
282 if (static_cast<int32_t>(gnssSystems) != 0) {
283 payload_tx_cfg_gnss = {};
284 payload_tx_cfg_gnss.msgVer = 0x00;
285 payload_tx_cfg_gnss.numTrkChHw = 0x00; // read only
286 payload_tx_cfg_gnss.numTrkChUse = 0xFF; // use max number of HW channels
287 payload_tx_cfg_gnss.numConfigBlocks = 7; // always configure all systems
288
289 // GPS and QZSS should always be enabled and disabled together, according to uBlox
290 payload_tx_cfg_gnss.block[0].gnssId = UBX_TX_CFG_GNSS_GNSSID_GPS;
291 payload_tx_cfg_gnss.block[1].gnssId = UBX_TX_CFG_GNSS_GNSSID_QZSS;
292
293 if (gnssSystems & GNSSSystemsMask::ENABLE_GPS) {
294 payload_tx_cfg_gnss.block[0].resTrkCh = 8;
295 payload_tx_cfg_gnss.block[0].maxTrkCh = 16;
296 payload_tx_cfg_gnss.block[0].flags = UBX_TX_CFG_GNSS_FLAGS_GPS_L1CA | UBX_TX_CFG_GNSS_FLAGS_ENABLE;
297 payload_tx_cfg_gnss.block[1].resTrkCh = 0;
298 payload_tx_cfg_gnss.block[1].maxTrkCh = 3;
299 payload_tx_cfg_gnss.block[1].flags = UBX_TX_CFG_GNSS_FLAGS_QZSS_L1CA | UBX_TX_CFG_GNSS_FLAGS_ENABLE;
300 }
301
302 payload_tx_cfg_gnss.block[2].gnssId = UBX_TX_CFG_GNSS_GNSSID_SBAS;
303
304 if (gnssSystems & GNSSSystemsMask::ENABLE_SBAS) {
305 payload_tx_cfg_gnss.block[2].resTrkCh = 1;
306 payload_tx_cfg_gnss.block[2].maxTrkCh = 3;
307 payload_tx_cfg_gnss.block[2].flags = UBX_TX_CFG_GNSS_FLAGS_SBAS_L1CA | UBX_TX_CFG_GNSS_FLAGS_ENABLE;
308 }
309
310 payload_tx_cfg_gnss.block[3].gnssId = UBX_TX_CFG_GNSS_GNSSID_GALILEO;
311
312 if (gnssSystems & GNSSSystemsMask::ENABLE_GALILEO) {
313 payload_tx_cfg_gnss.block[3].resTrkCh = 4;
314 payload_tx_cfg_gnss.block[3].maxTrkCh = 8;
315 payload_tx_cfg_gnss.block[3].flags = UBX_TX_CFG_GNSS_FLAGS_GALILEO_E1 | UBX_TX_CFG_GNSS_FLAGS_ENABLE;
316 }
317
318 payload_tx_cfg_gnss.block[4].gnssId = UBX_TX_CFG_GNSS_GNSSID_BEIDOU;
319
320 if (gnssSystems & GNSSSystemsMask::ENABLE_BEIDOU) {
321 payload_tx_cfg_gnss.block[4].resTrkCh = 8;
322 payload_tx_cfg_gnss.block[4].maxTrkCh = 16;
323 payload_tx_cfg_gnss.block[4].flags = UBX_TX_CFG_GNSS_FLAGS_BEIDOU_B1I | UBX_TX_CFG_GNSS_FLAGS_ENABLE;
324 }
325
326 payload_tx_cfg_gnss.block[5].gnssId = UBX_TX_CFG_GNSS_GNSSID_GLONASS;
327
328 if (gnssSystems & GNSSSystemsMask::ENABLE_GLONASS) {
329 payload_tx_cfg_gnss.block[5].resTrkCh = 8;
330 payload_tx_cfg_gnss.block[5].maxTrkCh = 14;
331 payload_tx_cfg_gnss.block[5].flags = UBX_TX_CFG_GNSS_FLAGS_GLONASS_L1 | UBX_TX_CFG_GNSS_FLAGS_ENABLE;
332 }
333
334 // IMES always disabled
335 payload_tx_cfg_gnss.block[6].gnssId = UBX_TX_CFG_GNSS_GNSSID_IMES;
336 payload_tx_cfg_gnss.block[6].flags = 0;
337
338 // send message
339 if (!sendMessage(UBX_MSG_CFG_GNSS, UBX::encode(payload_tx_cfg_gnss))) {
340 return -1;
341 }
342
343 if (waitForAck(UBX_MSG_CFG_GNSS, UBX_CONFIG_TIMEOUT, true) < 0 && !ioError()) {
344 // The receiver rejects the configuration as a whole if it names a constellation it
345 // cannot receive, e.g. BeiDou on a SAM-M8Q, or more of them than it can track at
346 // once. Keep the receiver's own selection rather than losing the fix over it.
347 log(GPSProtocolLogLevel::Warning, "GNSS constellation config rejected, keeping receiver config");
348 }
349
350 // On u-blox 8 the Galileo change only takes effect once the configuration has been
351 // saved and the receiver hardware reset, which we cannot do without dropping the
352 // rest of this session's configuration
353 if ((gnssSystems & GNSSSystemsMask::ENABLE_GALILEO) && !ioError()) {
354 log(GPSProtocolLogLevel::Warning, "Galileo needs a receiver power cycle to take effect");
355 }
356
357 waitForGnssReset();
358 }
359
360 /* configure message rates */
361 /* the last argument is divisor for measurement rate (set by CFG RATE), i.e. 1 means 5Hz */
362
363 /* try to set rate for NAV-PVT */
364 /* (implemented for ubx7+ modules only, use NAV-SOL, NAV-POSLLH, NAV-VELNED and NAV-TIMEUTC for ubx6) */
365 if (!configureMessageRate(UBX_MSG_NAV_PVT, 1)) {
366 return -1;
367 }
368
369 if (waitForAck(UBX_MSG_CFG_MSG, UBX_CONFIG_TIMEOUT, true) < 0) {
370 _use_nav_pvt = false;
371
372 } else {
373 _use_nav_pvt = true;
374 }
375
376 if (!_use_nav_pvt) {
377 if (!configureMessageRateAndAck(UBX_MSG_NAV_TIMEUTC, 5, true)) {
378 return -1;
379 }
380
381 if (!configureMessageRateAndAck(UBX_MSG_NAV_POSLLH, 1, true)) {
382 return -1;
383 }
384
385 if (!configureMessageRateAndAck(UBX_MSG_NAV_SOL, 1, true)) {
386 return -1;
387 }
388
389 if (!configureMessageRateAndAck(UBX_MSG_NAV_VELNED, 1, true)) {
390 return -1;
391 }
392 }
393
394 if (!configureMessageRateAndAck(UBX_MSG_NAV_STATUS, 1, true)) {
395 return -1;
396 }
397
398 if (!configureMessageRateAndAck(UBX_MSG_NAV_DOP, 1, true)) {
399 return -1;
400 }
401
402 if (!configureMessageRateAndAck(UBX_MSG_NAV_SVINFO, (_satellite_info != nullptr) ? 5 : 0, true)) {
403 return -1;
404 }
405
406 if (!configureMessageRateAndAck(UBX_MSG_MON_HW, 1, true)) {
407 return -1;
408 }
409
410 return 0;
411}

413int GPSNativeUBX::configureDevice(const GPSConfig& config)
414{
415 // There is no RTCM or USB interface on M10
416 if (UBX::receiverProfile(_board).usb) {
417 initCfgValset();
418
419 const uint8_t enable_corrections_in = (_output_mode == OutputMode::RTCM) ? 0 : 1;
420
421 cfgValset<uint8_t>(UBX_CFG_KEY_CFG_UART1INPROT_RTCM3X, enable_corrections_in);
422
423 // USB
424 cfgValset<uint8_t>(UBX_CFG_KEY_CFG_USBINPROT_UBX, 1);
425 cfgValset<uint8_t>(UBX_CFG_KEY_CFG_USBINPROT_RTCM3X, enable_corrections_in);
426 cfgValset<uint8_t>(UBX_CFG_KEY_CFG_USBINPROT_NMEA, 0);
427 cfgValset<uint8_t>(UBX_CFG_KEY_CFG_USBOUTPROT_UBX, 1);
428
429 cfgValset<uint8_t>(UBX_CFG_KEY_CFG_USBOUTPROT_NMEA, 0);
430
431 // Only RTCM-output-capable receivers expose these keys. M9 SPG rejects
432 // the entire VALSET if they are included, even with a value of zero.
433 if (UBX::receiverProfile(_board).rtcmOutput) {
434 cfgValset<uint8_t>(UBX_CFG_KEY_CFG_UART1OUTPROT_RTCM3X, _output_mode != OutputMode::GPS);
435 cfgValset<uint8_t>(UBX_CFG_KEY_CFG_USBOUTPROT_RTCM3X, _output_mode != OutputMode::GPS);
436 }
437
438 if (sendCfgValsetAcked() < 0) {
439 return -1;
440 }
441
442 // Optional SPARTN input (PointPerfect). Sent separately so modules without
443 // SPARTN support can NACK without failing the rest of configuration.
444 initCfgValset();
445 cfgValset<uint8_t>(UBX_CFG_KEY_CFG_UART1INPROT_SPARTN, enable_corrections_in);
446 cfgValset<uint8_t>(UBX_CFG_KEY_CFG_USBINPROT_SPARTN, enable_corrections_in);
447
448 sendCfgValsetAcked(false);
449 }
450
451 /* set configuration parameters */
452 initCfgValset();
453 cfgValset<uint8_t>(UBX_CFG_KEY_NAVSPG_FIXMODE, 3 /* Auto 2d/3d */);
454 cfgValset<uint8_t>(UBX_CFG_KEY_NAVSPG_UTCSTANDARD, 3 /* USNO (U.S. Naval Observatory derived from GPS) */);
455 cfgValset<uint8_t>(UBX_CFG_KEY_NAVSPG_DYNMODEL, _dyn_model);
456
457 // measurement rate
458 // M9N max rate is 8Hz for all satellites, above 8Hz the number of used satellites is restricted to 16.
459 // F9P L1L2 in firmware <1.50 the max update rate with 4 constellations is 9Hz without RTK and 7Hz with RTK
460 // F9P L1L2 in firmware >=1.50 the max update rate with 4 constellations is 7Hz without RTK and 5Hz with RTK
461 // F9P L1L5 the max update rate with 4 constellations is 8Hz without RTK and 7Hz with RTK
462 // DAN-F10N the max update rate is 10Hz with GPS+GAL+BDS(Default)
463 // X20 max update rate is 25Hz, but 25Hz at 115200 baud causes high dropouts, especially with RTK. So default 10Hz
464 // is selected. Receivers such as M9N and DAN-F10N can go higher than 10Hz, but the number of used satellites will
465 // be restricted to 16. (Not mentioned in datasheet)
466 int rate_meas = 100; // 10Hz
467
468 switch (_board) {
469 case Board::u_blox9:
470 rate_meas = 125; // 8Hz
471 break;
472
473 case Board::u_blox9_F9P_L1L2:
474 case Board::u_blox9_F9P_L1L5:
475 rate_meas = 200; // 5Hz
476 break;
477
478 default:
479 break;
480 }
481
482 cfgValset<uint16_t>(UBX_CFG_KEY_RATE_MEAS, rate_meas);
483 cfgValset<uint16_t>(UBX_CFG_KEY_RATE_NAV, 1);
484 cfgValset<uint8_t>(UBX_CFG_KEY_RATE_TIMEREF, 0);
485
486 if (sendCfgValsetAcked() < 0) {
487 return -1;
488 }
489
490 // Disable odometer. Separate, non-fatal VALSET: CFG-ODO-* was removed on the
491 // F20 platform (ZED-X20P HPG 2.10+), so a NAK here must not abort config.
492 initCfgValset();
493 cfgValset<uint8_t>(UBX_CFG_KEY_ODO_USE_ODO, 0);
494
495 // M9 (SPG) only has USE_ODO and PROFILE in CFG-ODO
496 if (_board != Board::u_blox9) {
497 static constexpr uint32_t odo_keys[] = {UBX_CFG_KEY_ODO_USE_COG, UBX_CFG_KEY_ODO_OUTLPVEL,
498 UBX_CFG_KEY_ODO_OUTLPCOG};
499 cfgValset(odo_keys, 0);
500 }
501
502 sendCfgValsetAcked(false);
503
504 // RTK (optional, as only RTK devices like F9P support it)
505 initCfgValset();
506 cfgValset<uint8_t>(UBX_CFG_KEY_NAVHPG_DGNSSMODE, 3 /* RTK Fixed */);
507
508 if (!sendCfgValset()) {
509 return -1;
510 }
511
512 waitForAck(UBX_MSG_CFG_VALSET, UBX_CONFIG_TIMEOUT, false);
513
514 // Jamming detection. Firmware with CFG-SEC-JAMDET (F9 HPG 1.50+, F9 L1L5, F20/X20) has
515 // detection always on and no CFG-ITFM; everything older has CFG-ITFM and no JAMDET key.
516 // A NAK on the sensitivity key is therefore the signal that the monitor still needs enabling.
517 initCfgValset();
518 cfgValset<uint8_t>(UBX_CFG_KEY_SEC_JAMDET_SENSITIVITY_HI, 0);
519
520 if (sendCfgValsetAcked(false) < 0) {
521 if (_valsetAckAmbiguous) {
522 if (!ioError()) {
523 log(GPSProtocolLogLevel::Warning, "CFG-SEC-JAMDET_SENSITIVITY_HI not supported");
524 }
525 return -1;
526 }
527 initCfgValset();
528 cfgValset<uint8_t>(UBX_CFG_KEY_ITFM_ENABLE, 1);
529
530 if (sendCfgValsetAcked(false) < 0 && !ioError()) {
531 log(GPSProtocolLogLevel::Warning, "Jamming monitor not supported by this receiver");
532 }
533 }
534
535 // configure active GNSS systems (leave signal bands as is)
536 // Note: For M10 configuration if changing from default. As per the
537 // MAX-M10S integration guide UBX-20053088 - R03, see section
538 // 2.1.1.3 GNSS signal configuration for details on some restrictions.
539 // Implementing these restrictions are a TODO item for M10.
540 if (static_cast<int32_t>(config.gnss_systems) != 0) {
541 initCfgValset();
542
543 // GPS and QZSS should always be enabled and disabled together, according to uBlox
544 if (config.gnss_systems & GNSSSystemsMask::ENABLE_GPS) {
545 cfgValset<uint8_t>(UBX_CFG_KEY_SIGNAL_GPS_ENA, 1);
546 cfgValset<uint8_t>(UBX_CFG_KEY_SIGNAL_QZSS_ENA, 1);
547
548 cfgValset<uint8_t>(UBX_CFG_KEY_SIGNAL_QZSS_L1CA_ENA, 1);
549
550 // M9 (SPG) has no CFG-SIGNAL key for QZSS L1S
551 if (_board != Board::u_blox9) {
552 cfgValset<uint8_t>(UBX_CFG_KEY_SIGNAL_QZSS_L1S_ENA, 1);
553 }
554

885bool GPSNativeUBX::cfgValsetRaw(uint32_t key_id, uint32_t value)
886{
887 if (key_id == UBX_CFG_KEY_NAVSPG_DYNMODEL) {
888 _valsetSettings.add(GPSReceiverSetting::DynamicModel);
889 }
890 if (key_id == UBX_CFG_KEY_RATE_MEAS || key_id == UBX_CFG_KEY_RATE_NAV) {
891 _valsetSettings.add(GPSReceiverSetting::OutputRateHz);
892 }
893 if ((key_id & 0xffff0000u) == 0x10310000u) {
894 _valsetSettings.add(GPSReceiverSetting::ConstellationMask);
895 }
896
897 const unsigned value_size = UBX::configurationValueBytes(key_id);
898 if (!value_size || (value_size < 4 && value >= (1u << (value_size * 8))) || ((key_id >> 28) == 1 && value > 1)) {
899 return false;
900 }
901 if ((key_id == UBX_CFG_KEY_CFG_UART1OUTPROT_RTCM3X || key_id == UBX_CFG_KEY_CFG_USBOUTPROT_RTCM3X) &&
902 !UBX::receiverProfile(_board).rtcmOutput) {
903 return false;
904 }
905
906 if (_tx_cfg_valset_size + sizeof(key_id) + value_size > sizeof(_tx_cfg_valset_buf)) {
907 // If this ever fires, either bump UBX_CFG_VALSET_BUF_SIZE or split the
908 // batch into multiple CFG-VALSET messages at the call site.
909 log(GPSProtocolLogLevel::Warning, "buf for CFG_VALSET too small");
910 return false;
911 }
912
913 const std::span<uint8_t> output(_tx_cfg_valset_buf);
914 if (!LittleEndian::write(output, _tx_cfg_valset_size, key_id)) {
915 return false;
916 }
917 const auto valueOffset = _tx_cfg_valset_size + sizeof(key_id);
918 const bool written = value_size == 1 ? LittleEndian::write(output, valueOffset, static_cast<uint8_t>(value))
919 : value_size == 2 ? LittleEndian::write(output, valueOffset, static_cast<uint16_t>(value))
920 : LittleEndian::write(output, valueOffset, value);
921 if (!written) {
922 return false;
923 }
924 _tx_cfg_valset_size = valueOffset + value_size;
925 return true;
926}

973int GPSNativeUBX::disableTimeMode()
974{
975 if (_timeModeUnsupported) {
976 return 0;
977 }
978 if (_proto_ver_27_or_higher) {
979 initCfgValset();
980 cfgValset<uint8_t>(UBX_CFG_KEY_TMODE_MODE, 0);
981 if (sendCfgValsetAcked() < 0) {
982 return -1;
983 }
984 return verifyConfigValue(UBX_CFG_KEY_TMODE_MODE, 0);
985 }
986
987 const ubx_payload_tx_cfg_tmode3_t disabled{};
988 if (!sendMessage(UBX_MSG_CFG_TMODE3, UBX::encode(disabled)) ||
989 waitForAck(UBX_MSG_CFG_TMODE3, UBX_CONFIG_TIMEOUT, true) < 0) {
990 return -1;
991 }
992 _timeModeReadbackPending = true;
993 _timeModeReadbackReady = false;
994 const auto clearReadback = qScopeGuard([this] { _timeModeReadbackPending = false; });
995 if (!sendMessage(UBX_MSG_CFG_TMODE3, nullptr, 0)) {
996 return -1;
997 }
998 const auto result =
999 awaitCommand({"UBX-CFG-TMODE3 disabled readback", std::chrono::milliseconds(UBX_CONFIG_TIMEOUT)}, [this] {
1000 if (_controller.lateRejection()) {
1001 return GPSCommandOutcome::Rejected;
1002 }
1003 return !_timeModeReadbackReady ? GPSCommandOutcome::Pending
1004 : _timeModeReadback == 0 ? GPSCommandOutcome::ReadbackVerified
1005 : GPSCommandOutcome::Rejected;
1006 });
1007 return result.evidence.outcome == GPSCommandOutcome::ReadbackVerified ? 0 : -1;
1008}

1010int GPSNativeUBX::verifyConfigValue(uint32_t key, uint8_t value)
1011{
1012 const std::array keys{key};
1013 _controller.beginReadback(keys);
1014 const auto clearReadback = qScopeGuard([this] { _controller.finishReadback(); });
1015 std::array<uint8_t, 8> request{};
1016 (void) LittleEndian::write(request, 4, key);
1017 if (!sendMessage(UBX_MSG_CFG_VALGET, request)) {
1018 return -1;
1019 }
1020 const auto result = awaitCommand(
1021 {"UBX-CFG-VALGET " + std::to_string(key), std::chrono::milliseconds(UBX_CONFIG_TIMEOUT)}, [this, value] {
1022 if (_controller.lateRejection()) {
1023 return GPSCommandOutcome::Rejected;
1024 }
1025 return !_controller.readbackReady() ? GPSCommandOutcome::Pending
1026 : _controller.readback().values[0] == value ? GPSCommandOutcome::ReadbackVerified
1027 : GPSCommandOutcome::Rejected;
1028 });
1029 return result.evidence.outcome == GPSCommandOutcome::ReadbackVerified ? 0 : -1;
1030}

1032int GPSNativeUBX::waitForSurveyStop()
1033{
1034 _survey_in_stopped = false;
1035 const uint64_t stop_deadline = nowUs() + 3000000;
1036
1037 while (!_survey_in_stopped && nowUs() < stop_deadline) {
1038 if (!sendMessage(UBX_MSG_NAV_SVIN, nullptr, 0)) {
1039 return -1;
1040 }
1041
1042 const uint64_t poll_deadline = nowUs() + 100000;
1043
1044 while (!_survey_in_stopped && nowUs() < poll_deadline) {
1045 bool read_error;
1046 receiveInternal(100, read_error);
1047
1048 if (read_error) {
1049 return -1;
1050 }
1051 }
1052 }

1192GPSNativeUBX::waitForAck(const uint16_t msg, const unsigned timeout, const bool report)
1193{
1194 const Operation operation(*this, timeout);
1195 _last_ack_rejected = false;
1196 _controller.beginAcknowledgement(msg);
1197 if (msg == UBX_MSG_CFG_VALSET && _controller.configurationReadbackRequired()) {
1198 _controller.finishAcknowledgement();
1199 const auto settings = _pendingCommandSettings;
1200 const std::span<const uint8_t> bytes(_tx_cfg_valset_buf, static_cast<size_t>(_tx_cfg_valset_size));
1201 for (size_t offset = 4; offset < bytes.size();) {
1202 UBX::ConfigurationValues expected;
1203 std::array<uint8_t, 40> request{};
1204 while (offset < bytes.size() && expected.count < expected.keys.size()) {
1205 const auto key = LittleEndian::read<uint32_t>(bytes, offset);
1206 if (!key) {
1207 return -1;
1208 }
1209 offset += 4;
1210 const unsigned width = UBX::configurationValueBytes(*key);
1211 if (!width || width > bytes.size() - offset) {
1212 return -1;
1213 }
1214 uint32_t value = 0;
1215 for (unsigned index = 0; index < width; ++index) {
1216 value |= uint32_t(bytes[offset++]) << (8 * index);
1217 }
1218 (void) LittleEndian::write(request, 4 + expected.count * 4, *key);
1219 expected.keys[expected.count] = *key;
1220 expected.values[expected.count++] = value;
1221 }
1222 _controller.beginReadback(std::span(expected.keys).first(expected.count));
1223 const auto clearReadback = qScopeGuard([this] { _controller.finishReadback(); });
1224 if (!sendMessage(UBX_MSG_CFG_VALGET, std::span(request).first(4 + expected.count * 4))) {
1225 return -1;
1226 }
1227 const auto result = awaitCommand(
1228 {"UBX-CFG-VALSET readback", std::chrono::milliseconds(timeout), settings, report}, [this, &expected] {
1229 if (_controller.lateRejection()) {
1230 return GPSCommandOutcome::Rejected;
1231 }
1232 if (!_controller.readbackReady()) {
1233 return GPSCommandOutcome::Pending;
1234 }
1235 return _controller.readback().values == expected.values ? GPSCommandOutcome::ReadbackVerified
1236 : GPSCommandOutcome::Rejected;
1237 });
1238 _last_ack_rejected = result.evidence.outcome == GPSCommandOutcome::Rejected;
1239 if (result.evidence.outcome != GPSCommandOutcome::ReadbackVerified) {
1240 return -1;
1241 }
1242 }
1243 _pendingCommandSettings = {};
1244 return 0;
1245 }
1246 const auto clearAcknowledgement = qScopeGuard([this] { _controller.finishAcknowledgement(); });
1247 const auto result =
1248 awaitCommand({std::to_string(msg), std::chrono::milliseconds(timeout), _pendingCommandSettings, report},
1249 [this] { return _controller.acknowledgement(); });
1250 _pendingCommandSettings = {};
1251 _last_ack_rejected = result.evidence.outcome == GPSCommandOutcome::Rejected;
1252 if (msg == UBX_MSG_CFG_VALSET && result.evidence.outcome == GPSCommandOutcome::TimedOut) {
