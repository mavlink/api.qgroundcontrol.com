41EGLDisplay qtEglDisplay(QOpenGLContext* qtCtx)
42{
43 // QEGLContext::display() is the only handle that guarantees EGLImage import/sample compatibility; fall back to
44 // EGL_DEFAULT_DISPLAY when unavailable.
45 if (qtCtx) {
46 if (auto* egl = qtCtx->nativeInterface<QNativeInterface::QEGLContext>()) {
47 EGLDisplay d = egl->display();
48 if (d != EGL_NO_DISPLAY)
49 return d;
50 }
51 }
52 return eglGetDisplay(EGL_DEFAULT_DISPLAY);

88bool GlContextBridge::primeLocked()
89{
90 QOpenGLContext* qtCtx = QOpenGLContext::globalShareContext();
91 if (!qtCtx) {
92 if (retryWithinBudget()) {
93 qCInfo(GstGlBridgeLog) << "globalShareContext() is null — Qt GL not initialized yet"
94 << "(attempt" << failedPrimes() << "/" << kMaxPrimeRetries << ")";
95 } else if (retriesJustExhausted()) {
96 qCWarning(GstGlBridgeLog) << "globalShareContext() still null after" << kMaxPrimeRetries
97 << "retries; GL bridge giving up";
98 }
99 return false;
100 }
101
102#if defined(QGC_GST_BRIDGE_HAS_EGL)
103 // Try EGL first (Linux: Wayland/eglfs/xcb_egl; Windows: ANGLE-EGL); fall back to GLX (Linux xcb default on Qt 6).
104 EGLContext eglCtx = qtEglContext(qtCtx);
105 EGLDisplay eglDisp = (eglCtx != EGL_NO_CONTEXT) ? qtEglDisplay(qtCtx) : EGL_NO_DISPLAY;
106 if (eglCtx != EGL_NO_CONTEXT && eglDisp != EGL_NO_DISPLAY) {
107#if defined(QGC_GST_BRIDGE_HAS_WAYLAND)
108 // On Wayland, primary display must be GstGLDisplayWayland; derived EGL display is marked foreign so unref
109 // doesn't tear down Qt's EGLDisplay. Before 1.26 that derived display can't be marked foreign and its finalize
110 // eglTerminate()s Qt's display, so older GStreamer takes the foreign EGL wrap below instead.
111 const QString platformName = QGuiApplication::platformName();
112 if (platformName == QLatin1String("wayland") || platformName == QLatin1String("wayland-egl")) {
113 struct wl_display* wlDisp = nullptr;
114 if (auto* wl = qGuiApp->nativeInterface<QNativeInterface::QWaylandApplication>()) {
115 wlDisp = wl->display();
116 }
117 if (wlDisp) {
118 GstGLDisplayWayland* displayWl = gst_gl_display_wayland_new_with_display(wlDisp);
119 if (displayWl) {
120 _display = GST_GL_DISPLAY(displayWl);
121 if (const std::unique_ptr<GstGLDisplayEGL, GStreamer::GstObjectDeleter> derived{
122 gst_gl_display_egl_from_gl_display(_display)}) {
123 gst_gl_display_egl_set_foreign(derived.get(), TRUE);
124 }
125 }
126 }
127 }
128#endif
129 if (!_display) {
130 // Marked foreign: finalize never eglTerminate()s Qt's EGLDisplay, and gst_gl_display_egl_from_gl_display()
131 // returns this same object rather than deriving a non-foreign one.
132 GstGLDisplayEGL* displayEgl = gst_gl_display_egl_new_with_egl_display(eglDisp);
133 if (!displayEgl) {
134 qCWarning(GstGlBridgeLog) << "gst_gl_display_egl_new_with_egl_display failed";
135 return retryLater();
136 }
137 _display = GST_GL_DISPLAY(displayEgl);
138 }
139
140 _context = gst_gl_context_new_wrapped(_display, reinterpret_cast<guintptr>(eglCtx), GST_GL_PLATFORM_EGL,
141 static_cast<GstGLAPI>(GST_GL_API_GLES2 | GST_GL_API_OPENGL));
142 if (!_context) {
143 qCWarning(GstGlBridgeLog) << "gst_gl_context_new_wrapped (EGL) failed";
144 gst_clear_object(&_display);
145 return retryLater();
146 }
147#if defined(QGC_GST_BRIDGE_HAS_WAYLAND)
148 const bool isWayland = GST_IS_GL_DISPLAY_WAYLAND(_display);
149 qCInfo(GstGlBridgeLog) << (isWayland ? "GL bridge primed (Wayland+EGL)" : "GL bridge primed (EGL)");
150#else
151 qCInfo(GstGlBridgeLog) << "GL bridge primed (EGL)";
152#endif
153 } else {
154#if defined(QGC_GST_BRIDGE_HAS_GLX)
155 // Qt's QGLXContext exposes the X11 Display* + GLXContext we need to wrap.
156 auto* glx = qtCtx->nativeInterface<QNativeInterface::QGLXContext>();
157 if (!glx) {
158 qCWarning(GstGlBridgeLog) << "Qt GL context exposes neither EGL nor GLX; GL bridge disabled";
159 return retryLater();
160 }
161 Display* xdisp = nullptr;
162 if (auto* x11 = qGuiApp->nativeInterface<QNativeInterface::QX11Application>()) {
163 xdisp = x11->display();
164 }
165 if (!xdisp) {
166 qCWarning(GstGlBridgeLog) << "X11 Display unresolvable; GL bridge disabled";
167 return retryLater();
168 }
169 GstGLDisplayX11* displayX11 = gst_gl_display_x11_new_with_display(xdisp);
170 if (!displayX11) {
171 qCWarning(GstGlBridgeLog) << "gst_gl_display_x11_new_with_display failed";
172 return retryLater();
173 }
174 _display = GST_GL_DISPLAY(displayX11);
175 _context = gst_gl_context_new_wrapped(_display, reinterpret_cast<guintptr>(glx->nativeContext()),
176 GST_GL_PLATFORM_GLX, static_cast<GstGLAPI>(GST_GL_API_OPENGL));
177 if (!_context) {
178 qCWarning(GstGlBridgeLog) << "gst_gl_context_new_wrapped (GLX) failed";
179 gst_clear_object(&_display);
180 return retryLater();
181 }
182 qCInfo(GstGlBridgeLog) << "GL bridge primed (GLX)";
183#else
184 qCWarning(GstGlBridgeLog) << "Qt EGLContext unresolvable and GLX bridge not built; GL bridge disabled";
185 return retryLater();
186#endif
187 }
188
189 // Don't call gst_gl_context_fill_info(): a freshly wrapped context has no active thread yet, and GStreamer fills
190 // info lazily on first activation.
191 qCDebug(GstGlBridgeLog) << "GL bridge primed: display=" << _display << "context=" << _context;
192 return true;
193#else
194 // EGL-only by design; Windows uses the D3D11/D3D12 bridge, macOS/iOS use CVPixelBuffer. No WGL/CGL wrap: forcing
195 // QSG_RHI_BACKEND=opengl on Windows (rare; RHI defaults to D3D11) just falls back to the CPU path here.
196 qCInfo(GstGlBridgeLog) << "GL bridge inactive on this platform (non-EGL)";
197 return false;
198#endif
199}

209GstContext* GlContextBridge::buildContext(const char* contextType, GstObject* object)
210{
211 if (g_strcmp0(contextType, GST_GL_DISPLAY_CONTEXT_TYPE) == 0) {
212 GstContext* ctx = gst_context_new(GST_GL_DISPLAY_CONTEXT_TYPE, TRUE);
213 gst_context_set_gl_display(ctx, GST_GL_DISPLAY(object));
214 return ctx;
215 }
216 GstContext* ctx = gst_context_new(kGlAppContextType, TRUE);
217 GstStructure* s = gst_context_writable_structure(ctx);
218 gst_structure_set(s, "context", GST_TYPE_GL_CONTEXT, GST_GL_CONTEXT(object), NULL);
219 return ctx;
220}
