64void VehicleGPSFactGroup::handleMessage(Vehicle* vehicle, const mavlink_message_t& message)
65{
66 Q_UNUSED(vehicle);
67
68 switch (message.msgid) {
69 case MAVLINK_MSG_ID_GPS_RAW_INT:
70 if (_receiver == ReceiverIndex::Primary) {
71 _handleGpsRaw(message);
72 }
73 break;
74 case MAVLINK_MSG_ID_GPS2_RAW:
75 if (_receiver == ReceiverIndex::Secondary) {
76 _handleGpsRaw(message);
77 }
78 break;
79 case MAVLINK_MSG_ID_HIGH_LATENCY:
80 if (_receiver == ReceiverIndex::Primary) {
81 _handleHighLatency(message);
82 }
83 break;
84 case MAVLINK_MSG_ID_HIGH_LATENCY2:
85 if (_receiver == ReceiverIndex::Primary) {
86 _handleHighLatency2(message);
87 }
88 break;
89 case MAVLINK_MSG_ID_GNSS_INTEGRITY:
90 _handleGnssIntegrity(message);
91 break;
92 case MAVLINK_MSG_ID_GPS_RTK:
93 if (_receiver == ReceiverIndex::Primary) {
94 _handleGpsRtk(message);
95 }
96 break;
97 case MAVLINK_MSG_ID_GPS2_RTK:
98 if (_receiver == ReceiverIndex::Secondary) {
99 _handleGpsRtk(message);
100 }
101 break;
102 default:
103 break;
104 }

131void VehicleGPSFactGroup::_handleGpsRaw(const mavlink_message_t& message)
132{
133 const auto update = [this](const auto& raw) {
134 QGeoPositionInfo position;
135 position.setCoordinate(QGeoCoordinate(raw.lat * 1e-7, raw.lon * 1e-7, raw.alt / 1000.0));
136 if (raw.h_acc != 0) {
137 position.setAttribute(QGeoPositionInfo::HorizontalAccuracy, raw.h_acc / 1000.0);
138 }
139 if (raw.v_acc != 0) {
140 position.setAttribute(QGeoPositionInfo::VerticalAccuracy, raw.v_acc / 1000.0);
141 }
142 if (raw.cog < 36000) {
143 position.setAttribute(QGeoPositionInfo::Direction, raw.cog / 100.0);
144 }
145 const double hdopValue = raw.eph == UINT16_MAX ? qQNaN() : raw.eph / 100.0;
146 const double vdopValue = raw.epv == UINT16_MAX ? qQNaN() : raw.epv / 100.0;
147 const double yawValue = raw.yaw > 0 && raw.yaw <= 36000 ? (raw.yaw % 36000) / 100.0 : qQNaN();
148 _updateFix(raw.fix_type, raw.satellites_visible, hdopValue, vdopValue,
149 position.attribute(QGeoPositionInfo::Direction), yawValue);
150 _reportPosition(position, raw.fix_type);
151 };
152 if (message.msgid == MAVLINK_MSG_ID_GPS_RAW_INT) {
153 mavlink_gps_raw_int_t raw{};
154 mavlink_msg_gps_raw_int_decode(&message, &raw);
155 update(raw);
156 } else {
157 mavlink_gps2_raw_t raw{};
158 mavlink_msg_gps2_raw_decode(&message, &raw);
159 update(raw);
160 }
161}

176{

178 mavlink_msg_high_latency2_decode(&message, &highLatency2);
179
180 QGeoPositionInfo position;
181 position.setCoordinate(
182 QGeoCoordinate(highLatency2.latitude * 1e-7, highLatency2.longitude * 1e-7, highLatency2.altitude));
183 if (highLatency2.eph != UINT8_MAX) {
184 position.setAttribute(QGeoPositionInfo::HorizontalAccuracy, highLatency2.eph / 10.0);
185 }
186 if (highLatency2.epv != UINT8_MAX) {
187 position.setAttribute(QGeoPositionInfo::VerticalAccuracy, highLatency2.epv / 10.0);
188 }
189 // HIGH_LATENCY2 reports an estimated global position and no receiver state, so the fix facts keep their values
190 // and the report does not count as a GPS fix.
191 _reportPosition(position, GPS_FIX_TYPE_NO_GPS);

223void VehicleGPSFactGroup::_handleGpsRtk(const mavlink_message_t& message)
224{
225 const auto apply = [this](const auto& rtk) {
226 // Baseline length is independent of whether the components are ECEF or NED.
227 const double a = rtk.baseline_a_mm;
228 const double b = rtk.baseline_b_mm;
229 const double c = rtk.baseline_c_mm;
230 rtkBaseline()->setRawValue(std::sqrt(a * a + b * b + c * c) / 1000.0);
231 rtkRate()->setRawValue(rtk.rtk_rate);
232 rtkSatellites()->setRawValue(rtk.nsats);
233 };
234 if (message.msgid == MAVLINK_MSG_ID_GPS2_RTK) {
235 mavlink_gps2_rtk_t rtk;
236 mavlink_msg_gps2_rtk_decode(&message, &rtk);
237 apply(rtk);
238 } else {
239 mavlink_gps_rtk_t rtk;
240 mavlink_msg_gps_rtk_decode(&message, &rtk);
241 apply(rtk);
242 }
243 (void) _rtkStatusExpiry.schedule(RTK_STATUS_TIMEOUT, [this]() { _clearRtkStatus(); });
244}

253QString VehicleGPSFactGroup::systemErrorText() const
254{
255 // MAVLink GPS_SYSTEM_ERROR_FLAGS.
256 static constexpr const char* ERROR_NAMES[] = {
257 QT_TR_NOOP("Incoming correction"),
258 QT_TR_NOOP("Configuration"),
259 QT_TR_NOOP("Software"),
260 QT_TR_NOOP("Antenna"),
261 QT_TR_NOOP("Event congestion"),
262 QT_TR_NOOP("CPU overload"),
263 QT_TR_NOOP("Output congestion"),
264 };
265 const quint32 errors = _systemErrorsFact.rawValue().toUInt();
266 QStringList names;
267 quint32 known = 0;
268 for (quint32 bit = 0; bit < std::size(ERROR_NAMES); ++bit) {
269 known |= 1u << bit;
270 if (errors & (1u << bit)) {
271 names.append(tr(ERROR_NAMES[bit]));
272 }
273 }
274 if (errors & ~known) {
275 names.append(tr("Other (0x%1)").arg(errors & ~known, 0, 16));
276 }
277 return names.join(QStringLiteral(", "));
278}
