53 Claim claim(uint8_t byte) const override
54 {

56 return Claim::Locked;
57 }

61 void feed(uint8_t byte, GPSStreamDemux::Handler& handler) override
62 {
63 // A completed candidate can leave further frames in its recovery suffix.
64 for (bool complete = _framer.addByte(byte); complete; complete = _framer.nextFrame()) {
65 if (!_framer.valid()) {

67 }
68 const auto frame = _framer.frame();

73 .messageId = RTCMFramer::frameMessageId(frame)});

117class SBFFramer final : public GPSStreamDemux::Framer
118{
119public:
120 explicit SBFFramer(bool shareSync)
121 : Framer(GPSFrameKind::SBF)
122 , _shareSync(shareSync)
123 {}
124
125 Claim claim(uint8_t byte) const override
126 {
127 switch (_state) {
128 case State::Sync1:
129 return byte == SBF::SYNC1 ? (_shareSync ? Claim::Watch : Claim::Owns) : Claim::None;
130 case State::Sync2:
131 return _shareSync ? Claim::Watch : Claim::Locked;
132 case State::Payload:
133 case State::Skip:
134 return Claim::Locked;
135 }
136 return Claim::None;
137 }
138
139 void feed(uint8_t byte, GPSStreamDemux::Handler& handler) override
140 {
141 switch (_state) {
142 case State::Sync1:
143 if (byte == SBF::SYNC1) {
144 (void) _add(byte);
145 _state = State::Sync2;
146 }
147 break;
148 case State::Sync2:
149 if (byte == SBF::SYNC2) {
150 (void) _add(byte);
151 _state = State::Payload;
152 } else {
153 reset();
154 }
155 break;
156 case State::Payload:
157 if (_add(byte)) {
158 // A complete candidate holds at least the header.
159 const std::span<const uint8_t> bytes(_wire.data(), _size);
160 const auto header = _header();
161 const uint16_t remaining =
162 header.length > _size && header.length % 4 == 0 && header.length <= MAX_SKIPPED_LENGTH
163 ? header.length - _size
164 : 0;
165 handler.frame(
166 {.kind = GPSFrameKind::SBF, .bytes = bytes, .payload = bytes, .messageId = header.number()});
167 reset();
168 if (remaining > 0) {
169 _state = State::Skip;
170 _remaining = remaining;
171 }
172 }
173 break;
174 case State::Skip:
175 if (--_remaining == 0) {
176 reset();
177 }
178 break;
179 }
180 }
181
182 void reset() override
183 {
184 _state = State::Sync1;
185 _size = 0;
186 _remaining = 0;
187 }
188
189private:
190 enum class State : uint8_t
191 {
192 Sync1,
193 Sync2,
194 Payload,
196 Skip,
197 };
198
200 static constexpr uint16_t MAX_SKIPPED_LENGTH = 4096;
201
203 SBF::BlockHeader _header() const { return Wire::decode<SBF::BlockHeader>(_wire); }
204
206 bool _add(uint8_t byte)
207 {
208 _wire[_size++] = byte;
209 return (_size >= Wire::SIZE<SBF::BlockHeader> && _size >= _header().length) || _size >= _wire.size();
210 }
211
212 std::array<uint8_t, 110> _wire{};
213 uint16_t _size = 0;
214 uint16_t _remaining = 0;
215 State _state = State::Sync1;
216 bool _shareSync = false;

303void GPSStreamDemux::push(uint8_t byte, Handler& handler)
304{
305 std::array<Claim, FRAMER_COUNT> claims{};
306 Framer* exclusive = nullptr;
307 for (size_t index = 0; index < _framers.size(); ++index) {
308 const auto& framer = _framers[index];
309 if (!framer || !framer->enabled) {
310 continue;
311 }
312 claims[index] = framer->claim(byte);
313 if (claims[index] == Claim::Locked) {
314 exclusive = framer.get();
315 break;
316 }
317 if (claims[index] == Claim::Owns && !exclusive) {
318 exclusive = framer.get();
319 }
320 }
321 if (exclusive) {
322 for (const auto& framer : _framers) {
323 if (framer && framer->enabled && framer.get() != exclusive) {
324 framer->reset();
325 }
326 }
327 exclusive->feed(byte, handler);
328 return;
329 }
330 bool defaultFed = false;
331 for (size_t index = 0; index < _framers.size(); ++index) {
332 if (claims[index] == Claim::Watch || (claims[index] == Claim::Default && !defaultFed)) {
333 defaultFed |= claims[index] == Claim::Default;
334 _framers[index]->feed(byte, handler);
335 }
336 }
337}
