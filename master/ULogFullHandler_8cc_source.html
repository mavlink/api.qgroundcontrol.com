112void ULogFullHandler::data(const ulog_cpp::Data &data)
113{
114 if (!_headerComplete) {
115 return;
116 }
117
118 const auto it = _subscriptions.find(data.msgId());
119 if (it == _subscriptions.cend()) {
120 return;
121 }
122
123 const SubscriptionInfo &sub = it->second;
124 if (!sub.format) {
125 return;
126 }
127
128 try {
129 const ulog_cpp::TypedDataView view(data, *sub.format);
130
131 // Extract timestamp (ULog convention: field named "timestamp", unit µs)
132 double timestampSecs = -1.0;
133 if (sub.format->fieldMap().count("timestamp") > 0) {
134 const uint64_t tsUs = view.at("timestamp").as<uint64_t>();
135 timestampSecs = static_cast<double>(tsUs) / 1e6;
136 _lastTimestampSecs = timestampSecs;
137 }
138
139 // Field name: "topic_name.field" or "topic_name[N].field" for multi-instance
140 const QString prefix = (sub.multiId > 0)
141 ? QStringLiteral("%1[%2].").arg(QString::fromStdString(sub.topicName)).arg(sub.multiId)
142 : QString::fromStdString(sub.topicName) + QLatin1Char('.');
143
144 for (const auto &field : sub.format->fields()) {
145 // Skip padding fields and the timestamp itself
146 if (field->name().rfind("_padding", 0) == 0) {
147 continue;
148 }
149 if (field->name() == "timestamp") {
150 continue;
151 }
152 if (!field->definitionResolved()) {
153 continue;
154 }
155
156 const QString fieldName = prefix + QString::fromStdString(field->name());
157 _fieldSet.insert(fieldName);
158
159 if (!_isNumericScalarField(*field) || timestampSecs < 0.0) {
160 continue;
161 }
162
163 const double value = view.at(field).as<double>();
164 _result.fieldSamples[fieldName].append(QPointF(timestampSecs, value));
165 _plottableFieldSet.insert(fieldName);
166 }
167
168 _result.sampleCount++;
169
170 if (timestampSecs >= 0.0) {
171 if (_result.minTimestamp < 0.0 || timestampSecs < _result.minTimestamp) {
172 _result.minTimestamp = timestampSecs;
173 }
174 _result.maxTimestamp = std::max(_result.maxTimestamp, timestampSecs);
175 }
176 } catch (const std::exception &e) {
177 qCWarning(ULogFullHandlerLog) << "Failed to decode data message:" << e.what();
178 }
179}

272void ULogFullHandler::finalize()
273{
274 // Detect vehicle type from vehicle_status.vehicle_type
275 // PX4 vehicle_type enum: 0=Unknown, 1=Rotary Wing, 2=Fixed Wing, 3=Rover, 4=Airship
276 const auto vehicleTypeIt = _result.fieldSamples.constFind(QStringLiteral("vehicle_status.vehicle_type"));
277 if (vehicleTypeIt != _result.fieldSamples.cend() && !vehicleTypeIt->isEmpty()) {
278 const int vtype = static_cast<int>(vehicleTypeIt->first().y());
279 switch (vtype) {
280 case 1: _result.detectedVehicleType = QStringLiteral("Multirotor/Helicopter"); break;
281 case 2: _result.detectedVehicleType = QStringLiteral("Fixed Wing"); break;
282 case 3: _result.detectedVehicleType = QStringLiteral("Rover"); break;
283 case 4: _result.detectedVehicleType = QStringLiteral("Airship"); break;
284 default: break; // 0 = Unknown, leave empty so UI shows "Unknown"
285 }
286 }
287
288 // Derive mode segments from vehicle_status.nav_state samples.
289 // nav_state is a uint8_t mapped to the PX4 navigation_state enum.
290 const auto navStateIt = _result.fieldSamples.constFind(QStringLiteral("vehicle_status.nav_state"));
291 if (navStateIt != _result.fieldSamples.cend()) {
292 const QVector<QPointF> &samples = navStateIt.value();
293 int lastNavState = -1;
294 double segmentStart = -1.0;
295 QString segmentMode;
296
297 for (const QPointF &pt : samples) {
298 const int navState = static_cast<int>(pt.y());
299 if (navState != lastNavState) {
300 // Close the previous segment
301 if (lastNavState >= 0 && segmentStart >= 0.0) {
302 QVariantMap seg;
303 seg[QStringLiteral("mode")] = segmentMode;
304 seg[QStringLiteral("start")] = segmentStart;
305 seg[QStringLiteral("end")] = pt.x();
306 _result.modeSegments.append(seg);
307 }
308 lastNavState = navState;
309 segmentStart = pt.x();
310 segmentMode = _px4NavStateName(navState);
311 }
312 }
313
314 // Close the final open segment
315 if (lastNavState >= 0 && segmentStart >= 0.0 && _result.maxTimestamp >= segmentStart) {
316 QVariantMap seg;
317 seg[QStringLiteral("mode")] = segmentMode;
318 seg[QStringLiteral("start")] = segmentStart;
319 seg[QStringLiteral("end")] = _result.maxTimestamp;
320 _result.modeSegments.append(seg);
321 }
322 }
323
324 // Sort field lists for consistent display
325 _result.availableFields = _fieldSet.values();
326 std::sort(_result.availableFields.begin(), _result.availableFields.end());
327 _result.plottableFields = _plottableFieldSet.values();
328 std::sort(_result.plottableFields.begin(), _result.plottableFields.end());
329
330 // Annotate parameter rows with default value info now that all ParameterDefault
331 // messages have been collected.
332 for (int i = 0; i < _result.parameters.size(); i++) {
333 QVariantMap row = _result.parameters[i].toMap();
334 const QString name = row.value(QStringLiteral("name")).toString();
335 const auto it = _paramDefaults.constFind(name);
336 if (it != _paramDefaults.constEnd()) {
337 const double defaultVal = it.value();
338 const QVariant valueVariant = row.value(QStringLiteral("value"));
339 row[QStringLiteral("hasDefault")] = true;
340 row[QStringLiteral("defaultValue")] = defaultVal;
341 // Treat parameters whose value couldn't be read (invalid QVariant from
342 // the catch block in parameter()) as non-default so they are never
343 // hidden by the "Changed only" filter.
344 bool isDefault = false;
345 if (valueVariant.isValid() && valueVariant.canConvert<double>()) {
346 const double currentVal = valueVariant.toDouble();
347 // qFuzzyCompare is undefined when either value is 0.0, so use exact
348 // equality for the zero case and relative comparison otherwise.
349 isDefault = (currentVal == defaultVal)
350 || (!qFuzzyIsNull(defaultVal) && qFuzzyCompare(currentVal, defaultVal));
351 }
352 row[QStringLiteral("isDefault")] = isDefault;
353 } else {
354 row[QStringLiteral("hasDefault")] = false;
355 row[QStringLiteral("defaultValue")] = QVariant();
356 row[QStringLiteral("isDefault")] = false;
357 }
358 _result.parameters[i] = row;
359 }
360
361 _result.sourceType = LogParseResult::SourceType::PX4ULog;
362 _result.ok = true;
