16[[nodiscard]] constexpr size_t cfgKeyValueBytes(uint32_t keyId)
17{
18 switch ((keyId >> 28) & 0x7) {
19 case 1: // A one-bit value still occupies a byte.
20 case 2:
21 return 1;
22 case 3:
23 return 2;
24 case 4:
25 return 4;
26 case 5:
27 return 8;
28 default:
29 return 0;
30 }
31}

191[[nodiscard]] constexpr unsigned configurationValueBytes(uint32_t key)
192{
193 const size_t width = cfgKeyValueBytes(key);
194 return width <= sizeof(uint32_t) ? static_cast<unsigned>(width) : 0;
195}

197struct ConfigurationValue
198{
199 uint32_t key;
200 uint32_t value;
201
202 bool operator==(const ConfigurationValue&) const = default;
203};
204
206class ConfigurationValueCursor
207{
208public:
209 explicit ConfigurationValueCursor(std::span<const uint8_t> entries)
210 : _remaining(entries)
211 {}
212
213 bool empty() const { return _remaining.empty(); }
214
215 std::optional<ConfigurationValue> next()
216 {
217 if (!_valid || empty()) {
218 return std::nullopt;
219 }
220 const auto key = LittleEndian::read<uint32_t>(_remaining, 0);
221 const unsigned width = key ? configurationValueBytes(*key) : 0;
222 if (!width || _remaining.size() < sizeof(uint32_t) + width) {
223 _valid = false;
224 return std::nullopt;
225 }
226 uint32_t value = 0;
227 for (unsigned byte = 0; byte < width; ++byte) {
228 value |= uint32_t(_remaining[sizeof(uint32_t) + byte]) << (8 * byte);
229 }
230 if ((*key >> 28) == 1 && value > 1) {
231 _valid = false;
232 return std::nullopt;
233 }
234 _remaining = _remaining.subspan(sizeof(uint32_t) + width);
235 return ConfigurationValue{*key, value};
236 }
237
238private:
239 std::span<const uint8_t> _remaining;
240 bool _valid = true;
241};
242
244struct ConfigurationValues
245{
247 static constexpr size_t MAX_KEYS = 9;
248
249 std::array<ConfigurationValue, MAX_KEYS> values{};
250 size_t count = 0;
251};
252
254[[nodiscard]] inline std::optional<ConfigurationValues> decodeConfigurationValues(std::span<const uint8_t> payload)
255{
256 if (payload.size() < 4 || payload[0] != 1 || payload[1] || payload[2] || payload[3]) {
257 return std::nullopt;
258 }
259 ConfigurationValues result;
260 ConfigurationValueCursor cursor(payload.subspan(4));
261 while (!cursor.empty()) {
262 const auto entry = cursor.next();
263 if (!entry || result.count == result.values.size()) {
264 return std::nullopt;
265 }
266 result.values[result.count++] = *entry;
267 }
268 return result;
269}

273template <size_t Capacity>
274struct CheckedValsetBatch
275{
276 static_assert(Capacity >= 4);
277 std::array<uint8_t, Capacity> bytes{0, 1}; // RAM layer; no persistent writes.
278 size_t size = 4;
279 bool valid = true;
280
281 std::span<const uint8_t> payload() const
282 {
283 return valid && size > 4 ? std::span<const uint8_t>(bytes).first(size) : std::span<const uint8_t>{};
284 }
285
287 std::span<const uint8_t> entries() const { return std::span<const uint8_t>(bytes).subspan(4, size - 4); }
288
289 bool append(uint32_t key, uint32_t value)
290 {
291 const unsigned width = configurationValueBytes(key);
292 if (!valid || !width || (width < 4 && value >= (1u << (width * 8))) || ((key >> 28) == 1 && value > 1) ||
293 sizeof(key) + width > bytes.size() - size) {
294 valid = false;
295 return false;
296 }
297 (void) LittleEndian::write(bytes, size, key);
298 for (unsigned byte = 0; byte < width; ++byte) {
299 bytes[size + sizeof(key) + byte] = static_cast<uint8_t>(value >> (8 * byte));
300 }
301 size += sizeof(key) + width;
302 return true;
303 }
