39std::optional<NavigationUpdate> NavigationEpochAssembler::ingest(const Sentence& sentence, uint64_t receivedAtUs)
40{
41 const auto sequence = ++_sequence;
42
43 if (const auto navigation = navigationStatus(sentence)) {
44 if (!navigation->valid) {
45 _navigationValid = false;
46 _invalidThroughSequence = sequence;
47 NavigationEpoch loss;
48 loss.timeMs = navigation->utcMilliseconds;
49 loss.receivedAtUs = receivedAtUs;
50 loss.positionReceivedAtUs = receivedAtUs;
51 loss.sequence = sequence;
52 loss.fixQuality = GPSFixQuality::NoFix;
53 if (sentence.type() == "GGA" && sentence.count > Field::GGA_SATELLITES_USED) {
54 loss.satellitesUsed = number<unsigned>(sentence.fields[Field::GGA_SATELLITES_USED]);
55 }
56 return NavigationUpdate{
57 .type = NavigationUpdate::Type::FixLoss, .trigger = NavigationUpdate::Trigger::Position, .epoch = loss};
58 }
59 _navigationValid = true;
60 }
61
62 if (auto update = _handlePositionSentence(sentence, receivedAtUs)) {
63 return update;
64 }
65 if (auto update = _handleAccuracy(sentence, receivedAtUs)) {
66 return update;
67 }
68 return _handleUntimedMetadata(sentence, receivedAtUs);
69}

83NavigationEpochAssembler::StoredEpoch& NavigationEpochAssembler::_find(std::optional<int> timeMs)
84{
85 if (auto* existing = _findExisting(timeMs)) {
86 return *existing;
87 }
88
89 auto oldest = _epochs.begin();
90 for (auto epoch = _epochs.begin(); epoch != _epochs.end(); ++epoch) {
91 if (!epoch->active) {
92 oldest = epoch;
93 break;
94 }
95 if (epoch->epoch.receivedAtUs < oldest->epoch.receivedAtUs) {
96 oldest = epoch;
97 }
98 }
99 _resetEpoch(*oldest, timeMs);
100 return *oldest;

157std::optional<NavigationUpdate> NavigationEpochAssembler::_handlePositionSentence(const Sentence& sentence,
158 uint64_t receivedAtUs)
159{
160 // Position sentences reach here only with a receiver-declared valid fix; ingest() reports fix loss first.
161 NavigationEpoch* epoch = nullptr;
162 if (const auto rmcFix = rmc(sentence)) {
163 if (!rmcFix->utcMilliseconds) {
164 return std::nullopt;
165 }
166 epoch = &_positionEpoch(rmcFix->utcMilliseconds, receivedAtUs);
167 epoch->latitude = rmcFix->latitude;
168 epoch->longitude = rmcFix->longitude;
169 epoch->speedMetersPerSecond = finiteValue(rmcFix->speedMetersPerSecond);
170 epoch->courseDegrees = finiteValue(rmcFix->courseDegrees);
171 } else if (const auto ggaFix = gga(sentence)) {
172 epoch = &_positionEpoch(utcMilliseconds(sentence.fields[Field::UTC_TIME]), receivedAtUs);
173 epoch->ggaQuality = ggaFix->quality;
174 epoch->latitude = ggaFix->latitude;
175 epoch->longitude = ggaFix->longitude;
176 epoch->altitudeMslMeters = finiteValue(ggaFix->altitude);
177 epoch->geoidSeparationMeters = finiteValue(ggaFix->geoidSeparation);
178 epoch->satellitesUsed = ggaFix->satellitesUsed;
179 epoch->horizontalDop = finiteValue(ggaFix->hdop);
180 if (!epoch->accuracyReceivedAtUs ||
181 !MonotonicClock::withinAge(epoch->accuracyReceivedAtUs, receivedAtUs, METADATA_MAX_AGE)) {
182 epoch->horizontalAccuracyMeters.reset();
183 epoch->verticalAccuracyMeters.reset();
184 epoch->accuracyReceivedAtUs = 0;
185 }
186 } else if (const auto gllFix = gll(sentence)) {
187 if (!gllFix->utcMilliseconds) {
188 return std::nullopt;
189 }
190 epoch = &_positionEpoch(gllFix->utcMilliseconds, receivedAtUs);
191 epoch->latitude = gllFix->latitude;
192 epoch->longitude = gllFix->longitude;
193 } else {
194 return std::nullopt;
195 }
196
197 epoch->fixQuality = fixQuality(epoch->ggaQuality.value_or(GgaQuality::GPS), GPSFixQuality::Fix3D);
198 markReceipt(*epoch, receivedAtUs);
199 epoch->positionReceivedAtUs = receivedAtUs;
200 epoch->sequence = _sequence;
201 return NavigationUpdate{
202 .type = NavigationUpdate::Type::Epoch, .trigger = NavigationUpdate::Trigger::Position, .epoch = *epoch};
203}

205std::optional<NavigationUpdate> NavigationEpochAssembler::_handleAccuracy(const Sentence& sentence,
206 uint64_t receivedAtUs)
207{
208 const auto accuracy = gst(sentence);
209 if (!accuracy) {
210 return std::nullopt;
211 }
212 const auto timeMs = utcMilliseconds(sentence.fields[Field::UTC_TIME]);
213 if (!timeMs) {
214 return std::nullopt;
215 }
216 auto& stored = _find(timeMs);
217 _resetExpiredEpoch(stored, receivedAtUs);
218 auto& epoch = stored.epoch;
219 epoch.horizontalAccuracyMeters = finiteValue(accuracy->horizontalAccuracy);
220 epoch.verticalAccuracyMeters = finiteValue(accuracy->verticalAccuracy);
221 epoch.accuracyReceivedAtUs = receivedAtUs;
222 markReceipt(epoch, receivedAtUs);
223 if (!_hasCurrent || _currentKey != stored.key || _hiddenByFixLoss(epoch)) {
224 return std::nullopt;
225 }
226 return NavigationUpdate{
227 .type = NavigationUpdate::Type::Epoch, .trigger = NavigationUpdate::Trigger::TimedMetadata, .epoch = epoch};
228}

230std::optional<NavigationUpdate> NavigationEpochAssembler::_handleUntimedMetadata(const Sentence& sentence,
231 uint64_t receivedAtUs)
232{
233 auto* stored = _current();
234 if (!stored || !stored->epoch.positionReceivedAtUs ||
235 !MonotonicClock::withinAge(stored->epoch.positionReceivedAtUs, receivedAtUs, METADATA_MAX_AGE)) {
236 return std::nullopt;
237 }
238
239 auto& epoch = stored->epoch;
240 if (sentence.type() == "GSA") {
241 if (sentence.count < Field::GSA_MIN_FIELDS) {
242 return std::nullopt;
243 }
244 if (!epoch.horizontalDop) {
245 epoch.horizontalDop = nonnegativeNumber(sentence.fields[Field::GSA_HDOP]);
246 }
247 const auto vdop = nonnegativeNumber(sentence.fields[Field::GSA_VDOP]);
248 epoch.verticalDop = vdop;
249 epoch.verticalDopReceivedAtUs = vdop ? receivedAtUs : 0;
250 } else if (const auto velocity = vtg(sentence)) {
251 epoch.speedMetersPerSecond = finiteValue(velocity->speedMetersPerSecond);
252 epoch.courseDegrees = finiteValue(velocity->courseDegrees);
253 } else {
254 return std::nullopt;
255 }
256 if (_hiddenByFixLoss(epoch)) {
257 return std::nullopt;
258 }
259 return NavigationUpdate{
260 .type = NavigationUpdate::Type::Epoch, .trigger = NavigationUpdate::Trigger::UntimedMetadata, .epoch = epoch};
261}
