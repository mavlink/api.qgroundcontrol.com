201 bool connect(gpointer object, const char* signal, GCallback callback, gpointer data)
202 {
203 disconnect();
204 _id = g_signal_connect(object, signal, callback, data);
205 if (_id != 0) {
206 g_weak_ref_init(&_object, object);
207 }
208 return _id != 0;
209 }
210
211 void disconnect() noexcept
212 {
213 if (_id == 0) {
214 return;
215 }
216 if (gpointer object = g_weak_ref_get(&_object)) {
217 g_signal_handler_disconnect(object, _id);
218 g_object_unref(object);
219 }
220 g_weak_ref_clear(&_object);
221 _id = 0;
222 }
223
224private:
225 GWeakRef _object{};
226 gulong _id = 0;
227};
