176 bool connect(gpointer object, const char* signal, GCallback callback, gpointer data)
177 {
178 disconnect();
179 _id = g_signal_connect(object, signal, callback, data);
180 if (_id != 0) {
181 g_weak_ref_init(&_object, object);
182 }
183 return _id != 0;
184 }
185
186 void disconnect() noexcept
187 {
188 if (_id == 0) {
189 return;
190 }
191 if (gpointer object = g_weak_ref_get(&_object)) {
192 g_signal_handler_disconnect(object, _id);
193 g_object_unref(object);
194 }
195 g_weak_ref_clear(&_object);
196 _id = 0;
197 }

199private:
200 GWeakRef _object{};
201 gulong _id = 0;
