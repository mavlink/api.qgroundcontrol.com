102void onGraphicsDeviceRemoved()
103{
104#if defined(QGC_HAS_ANY_GPU_PATH)
105 // Before the reset, which forgets them.
106 std::vector<GStreamer::GstElementPtr> pipelines;
107 for (const HwPathDescriptor* path : GstHwPathRegistry::paths()) {
108 if (!path->bridge) {
109 continue;
110 }
111 for (GStreamer::GstElementPtr& pipeline : path->bridge().takeServed()) {
112 const bool known = std::ranges::any_of(pipelines, [&pipeline](const GStreamer::GstElementPtr& other) {
113 return other.get() == pipeline.get();
114 });
115 if (!known) {
116 pipelines.push_back(std::move(pipeline));
117 }
118 }
119 }
120 resetCachedGpuResources();
121 for (const GStreamer::GstElementPtr& pipeline : pipelines) {
122 (void) gst_element_post_message(
123 pipeline.get(),
124 gst_message_new_element(GST_OBJECT(pipeline.get()), gst_structure_new_empty(kGraphicsDeviceLostMessage)));
125 }

141void configureQtGraphics(bool softwareRendering)
142{
143 s_softwareRendering.store(softwareRendering, std::memory_order_release);
144#if defined(Q_OS_LINUX) && !defined(Q_OS_ANDROID) && \
145 (defined(QGC_HAS_GST_GLMEMORY_GPU_PATH) || defined(QGC_HAS_GST_DMABUF_GPU_PATH))
146 // GL takes every desktop-Linux zero-copy path (GLMemory and DMABuf/EGLImage both import into a GL RHI), Vulkan
147 // only DMABuf, in the layouts its device imports: pin GL unless the user chose a backend. GL is always present on
148 // Linux, so no QRhi::probe.
149 if (!softwareRendering && !qEnvironmentVariableIsSet("QSG_RHI_BACKEND")) {
150 QQuickWindow::setGraphicsApi(QSGRendererInterface::OpenGL);
151 }
152 // GStreamer's zero-copy imports need Qt on EGL, but Qt's xcb integration picks GLX first (the AppImage runs xcb
153 // even in Wayland sessions). Qt falls back to GLX when EGL fails to initialize; the user's value wins.
154 if (!qEnvironmentVariableIsSet("QT_XCB_GL_INTEGRATION")) {
155 (void) qputenv("QT_XCB_GL_INTEGRATION", "xcb_egl");
156 }
157#endif
158 // The GL and DMABuf paths both need QOpenGLContext::globalShareContext(), which this attribute enables.
159#if defined(QGC_HAS_GST_GLMEMORY_GPU_PATH) || defined(QGC_HAS_GST_DMABUF_GPU_PATH)
160 QCoreApplication::setAttribute(Qt::AA_ShareOpenGLContexts);
161#endif
162 // Read only when Qt creates a Vulkan device, as it does with QSG_RHI_BACKEND=vulkan.
163#if defined(QGC_HAS_GST_DMABUF_VULKAN_IMPORT)
164 GstDmaBufVulkan::requestDeviceExtensions();

294QString extraPathStatsSince(const GstHwPathTelemetry::Snapshot& baseline)
295{
296 QString out;
297#if defined(QGC_HAS_ANY_GPU_PATH)
298 using GstHwPathTelemetry::Counter;
299 const GstHwPathTelemetry::Snapshot now = GstHwPathTelemetry::snapshot();
300 const auto delta = [&now, &baseline](HwVideoBufferPath path, auto key) {
301 return now.count(path, key) - baseline.count(path, key);
302 };
303
304 // "<path>-<label>:<count>", or "<count>/<count>" for a pair, when any count is nonzero.
305 struct CounterField
306 {
307 const char* label;
308 Counter first;
309 std::optional<Counter> second = std::nullopt;
310 };
311
312 static constexpr CounterField kCounterFields[] = {
313 {"failures", Counter::MapFailures},
314 {"reuse", Counter::TextureReuse},
315 {"wait[gpu/cpu]", Counter::GpuWaits, Counter::CpuWaits},
316 {"fence-timeouts", Counter::ProducerFenceTimeouts},
317 {"image-cache[hit/miss]", Counter::ImageCacheHits, Counter::ImageCacheMisses},
318 };
319 static constexpr std::pair<GstHwPathTelemetry::HwFallbackReason, const char*> kReasons[] = {
320 {GstHwPathTelemetry::HwFallbackReason::NoExt, "no-ext"},
321 {GstHwPathTelemetry::HwFallbackReason::EglBadMatch, "egl-bad-match"},
322 {GstHwPathTelemetry::HwFallbackReason::ValidateFailed, "validate"},
323 {GstHwPathTelemetry::HwFallbackReason::UnknownMemType, "unknown-mem"},
324 {GstHwPathTelemetry::HwFallbackReason::MapFailed, "map-failed"},
325 {GstHwPathTelemetry::HwFallbackReason::VulkanNoSync, "vulkan-no-sync"},
326 {GstHwPathTelemetry::HwFallbackReason::ImportUnsupported, "import-unsupported"},
327 };
328 forEachCompiledPath([&out, &delta](HwVideoBufferPath path) {
329 const QLatin1String pathLabel(GstHw::toString(path));
330 for (const CounterField& field : kCounterFields) {
331 const quint64 first = delta(path, field.first);
332 bool nonzero = (first > 0);
333 QString counts = QString::number(first);
334 if (field.second) {
335 const quint64 second = delta(path, *field.second);
336 nonzero = nonzero || (second > 0);
337 counts += u'/' + QString::number(second);
338 }
339 if (nonzero) {
340 out += QStringLiteral(" %1-%2:%3").arg(pathLabel, QLatin1String(field.label), counts);
341 }
342 }
343 const quint64 ewmaUs = GstHwPathTelemetry::peekMapDurationUsEwma(path);
344 if (ewmaUs > 0) {
345 out += QStringLiteral(" %1-map-us:%2").arg(pathLabel).arg(ewmaUs);
346 }
347 QString reasonBreakdown;
348 for (const auto& [reason, label] : kReasons) {
349 const quint64 n = delta(path, reason);
350 if (n > 0) {
351 reasonBreakdown += QStringLiteral("%1=%2,").arg(QLatin1String(label)).arg(n);
352 }
353 }
354 if (!reasonBreakdown.isEmpty()) {
355 reasonBreakdown.chop(1);
356 out += QStringLiteral(" %1-fallback[%2]").arg(pathLabel).arg(reasonBreakdown);
357 }
358 });
359 // None-path accumulates UnknownMemType (no allocator matched any compiled path).
360 const quint64 unknownMem = delta(HwVideoBufferPath::None, GstHwPathTelemetry::HwFallbackReason::UnknownMemType);
361 if (unknownMem > 0) {
362 out += QStringLiteral(" None-fallback[unknown-mem=%1]").arg(unknownMem);
363 }
364#else
365 Q_UNUSED(baseline);
366#endif
367 return out;
368}
