41void onPipelineError(const GError* error, const gchar* debug) noexcept
42{
43 // Only a real device loss should drop every GPU cache; ordinary stream errors must not. There is no structured
44 // device-lost event, so classify by the D3D/DXGI device-removed/reset text surfaced in the error/debug strings.
45 const auto mentionsDeviceLoss = [](const gchar* text) {
46 return text && (g_strstr_len(text, -1, "DEVICE_REMOVED") || g_strstr_len(text, -1, "DEVICE_RESET") ||
47 g_strstr_len(text, -1, "device removed") || g_strstr_len(text, -1, "device reset"));
48 };
49 if (mentionsDeviceLoss(error ? error->message : nullptr) || mentionsDeviceLoss(debug)) {
50 resetCachedGpuResources();
51 }
52}

106GpuBinChoice chooseGpuBin() noexcept
107{
108 const QByteArray value = qgetenv("QGC_GST_GPU_BIN").trimmed().toLower();
109#if defined(QGC_GST_BIN_HAS_GLUPLOAD) && defined(QGC_GST_BIN_HAS_DIRECT_DMABUF)
110 if (value == "dmabuf") {
111 return {GpuBinMode::DirectDmaBuf, "from QGC_GST_GPU_BIN"};
112 }
113 if (value == "glupload") {
114 return {GpuBinMode::GlUpload, "from QGC_GST_GPU_BIN"};
115 }
116 if (!value.isEmpty()) {
117 qCWarning(HwBuffersConfigLog) << "Unknown QGC_GST_GPU_BIN value" << value
118 << "(expected glupload or dmabuf); using the default glupload bin";
119 }
120 return {GpuBinMode::GlUpload, "default"};
121#else
122#if defined(QGC_GST_BIN_HAS_GLUPLOAD)
123 constexpr GpuBinMode kOnlyMode = GpuBinMode::GlUpload;
124#elif defined(QGC_GST_BIN_HAS_DIRECT_DMABUF)
125 constexpr GpuBinMode kOnlyMode = GpuBinMode::DirectDmaBuf;
126#else
127 constexpr GpuBinMode kOnlyMode = GpuBinMode::Passthrough;
128#endif
129 return {kOnlyMode, value.isEmpty() ? "only variant compiled" : "only variant compiled; QGC_GST_GPU_BIN ignored"};
130#endif
131}

280QString takeExtraPathStats() noexcept
281{
282 QString out;
283#if defined(QGC_HAS_ANY_GPU_PATH)
284 using GstHwPathTelemetry::Counter;
285
286 // "<path>-<label>:<count>", or "<count>/<count>" for a pair, when any count is nonzero.
287 struct CounterField
288 {
289 const char* label;
290 Counter first;
291 std::optional<Counter> second = std::nullopt;
292 };
293
294 static constexpr CounterField kCounterFields[] = {
295 {"failures", Counter::MapFailures},
296 {"reuse", Counter::TextureReuse},
297 {"wait[gpu/cpu]", Counter::GpuWaits, Counter::CpuWaits},
298 {"fence-timeouts", Counter::ProducerFenceTimeouts},
299 {"image-cache[hit/miss]", Counter::ImageCacheHits, Counter::ImageCacheMisses},
300 };
301 static constexpr std::pair<GstHwPathTelemetry::HwFallbackReason, const char*> kReasons[] = {
302 {GstHwPathTelemetry::HwFallbackReason::NoExt, "no-ext"},
303 {GstHwPathTelemetry::HwFallbackReason::ModifierRejected, "modifier"},
304 {GstHwPathTelemetry::HwFallbackReason::EglBadMatch, "egl-bad-match"},
305 {GstHwPathTelemetry::HwFallbackReason::ValidateFailed, "validate"},
306 {GstHwPathTelemetry::HwFallbackReason::UnknownMemType, "unknown-mem"},
307 {GstHwPathTelemetry::HwFallbackReason::MapFailed, "map-failed"},
308 {GstHwPathTelemetry::HwFallbackReason::VulkanNoSync, "vulkan-no-sync"},
309 {GstHwPathTelemetry::HwFallbackReason::ImportUnsupported, "import-unsupported"},
310 };
311 forEachCompiledPath([&out](HwVideoBufferPath path) {
312 const QLatin1String pathLabel(GstHw::toString(path));
313 for (const CounterField& field : kCounterFields) {
314 const quint64 first = GstHwPathTelemetry::take(path, field.first);
315 bool nonzero = (first > 0);
316 QString counts = QString::number(first);
317 if (field.second) {
318 const quint64 second = GstHwPathTelemetry::take(path, *field.second);
319 nonzero = nonzero || (second > 0);
320 counts += u'/' + QString::number(second);
321 }
322 if (nonzero) {
323 out += QStringLiteral(" %1-%2:%3").arg(pathLabel, QLatin1String(field.label), counts);
324 }
325 }
326 const quint64 ewmaUs = GstHwPathTelemetry::peekMapDurationUsEwma(path);
327 if (ewmaUs > 0) {
328 out += QStringLiteral(" %1-map-us:%2").arg(pathLabel).arg(ewmaUs);
329 }
330 QString reasonBreakdown;
331 for (const auto& [reason, label] : kReasons) {
332 const quint64 n = GstHwPathTelemetry::takeFallbackReason(path, reason);
333 if (n > 0) {
334 reasonBreakdown += QStringLiteral("%1=%2,").arg(QLatin1String(label)).arg(n);
335 }
336 }
337 if (!reasonBreakdown.isEmpty()) {
338 reasonBreakdown.chop(1);
339 out += QStringLiteral(" %1-fallback[%2]").arg(pathLabel).arg(reasonBreakdown);
340 }
341 });
342 // None-path accumulates UnknownMemType (no allocator matched any compiled path).
343 const quint64 unknownMem = GstHwPathTelemetry::takeFallbackReason(
344 HwVideoBufferPath::None, GstHwPathTelemetry::HwFallbackReason::UnknownMemType);
345 if (unknownMem > 0) {
346 out += QStringLiteral(" None-fallback[unknown-mem=%1]").arg(unknownMem);
347 }
348#endif
349 return out;
350}
