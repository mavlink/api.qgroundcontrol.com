65template <class T>
66std::optional<T> number(std::string_view field, int base = DECIMAL_BASE)
67{
68 if (field.empty())
69 return std::nullopt;
70 if (field.front() == '+') {
71 field.remove_prefix(1);
72 if (!field.empty() && (field.front() == '-' || field.front() == '+'))
73 return std::nullopt;
74 }
75 if (field.empty())
76 return std::nullopt;
77 T value{};
78 if constexpr (std::is_integral_v<T>) {
79 const auto parsed = std::from_chars(field.data(), field.data() + field.size(), value, base);
80 if (parsed.ec != std::errc{} || parsed.ptr != field.data() + field.size())
81 return std::nullopt;
82 } else if constexpr (requires { std::from_chars(field.data(), field.data() + field.size(), value); }) {
83 const auto parsed = std::from_chars(field.data(), field.data() + field.size(), value);
84 if (parsed.ec != std::errc{} || parsed.ptr != field.data() + field.size())
85 return std::nullopt;
86 } else {
87 // Older libc++ versions lack floating-point from_chars. Keep parsing locale independent.
88 std::istringstream stream{std::string(field)};
89 stream.imbue(std::locale::classic());
90 stream >> std::noskipws >> value;
91 if (stream.fail() || stream.peek() != std::char_traits<char>::eof())
92 return std::nullopt;
93 }
94 if constexpr (std::is_floating_point_v<T>) {
95 if (!std::isfinite(value))
96 return std::nullopt;
97 }
98 return value;
99}
