94template <class T>
95[[nodiscard]] std::optional<T> number(std::string_view field, int base = DECIMAL_BASE)
96{
97 if (field.empty()) {
98 return std::nullopt;
99 }
100 if (field.front() == '+') {
101 field.remove_prefix(1);
102 if (!field.empty() && (field.front() == '-' || field.front() == '+')) {
103 return std::nullopt;
104 }
105 }
106 if (field.empty()) {
107 return std::nullopt;
108 }
109 T value{};
110 if constexpr (std::is_integral_v<T>) {
111 const auto parsed = std::from_chars(field.data(), field.data() + field.size(), value, base);
112 if (parsed.ec != std::errc{} || parsed.ptr != field.data() + field.size()) {
113 return std::nullopt;
114 }
115 } else {
116 static_assert(std::is_same_v<T, double> || std::is_same_v<T, float>);
117 // Qt parses floating point in the C locale on every platform; libc++ before 20 has no floating-point
118 // from_chars. Qt skips surrounding whitespace, which a field must not carry.
119 constexpr std::string_view WHITESPACE = " \t\n\v\f\r";
120 if (WHITESPACE.find(field.front()) != std::string_view::npos ||
121 WHITESPACE.find(field.back()) != std::string_view::npos) {
122 return std::nullopt;
123 }
124 bool ok = false;
125 if constexpr (std::is_same_v<T, float>) {
126 value = QByteArrayView(field).toFloat(&ok);
127 } else {
128 value = QByteArrayView(field).toDouble(&ok);
129 }
130 if (!ok || !std::isfinite(value)) {
131 return std::nullopt;
132 }
133 }
134 return value;
135}
